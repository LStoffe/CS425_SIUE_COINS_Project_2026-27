package edu.calpoly.its.mas.big.reports;

//JDK imports
import java.util.Hashtable;

//BIG imports
import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.users.User;
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.users.admin.Admin;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Period;

//BIG DAO imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

///Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;


//Servlet imports
import javax.servlet.ServletContext;
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles all four Consulting reports: Labor Availability Index
 * Report, Material Cost Index Report, Weather Forecast Report, and
 * Market Analysis of Future Construction Demand Report.
 *
 * Each consulting report has an associated cost of viewing. If a user
 * requests a report for which she has not paid, she will be prompted
 * with the amount and asked to choose whether or not to purchase the
 * report.  Once a user has paid for a report, she may then view it
 * as many times as she wishes without paying again.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.14 $ */

public class ConsultingReportsAction extends Action
{
	public static final int COMPLETE_LIST_OF_BIDS = 0;
	public static final int CONSTRUCTION_DEMAND = 1;
	public static final int LABOR_AVAILABILITY = 2;
	public static final int MATERIAL_COST_INDEX = 3;
	public static final int WEATHER_FORECAST = 4;
	public static final int CAM_REPORT = 5;

	public ActionForward execute(ActionMapping mapping,
								 ActionForm form,
								 HttpServletRequest request,
                                 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		// The ActionForward to which we will ultimately redirect
		ActionForward destination = mapping.findForward("success");

	    ActionMessages errors = new ActionMessages();

	    // Check for existance of Game object in user's session
        ServletContext context = servlet.getServletContext();

        HttpSession session = request.getSession();
        if(session == null)
        {
		   return (mapping.findForward("loginFailure"));
		}

	    Game game = (Game) session.getAttribute("game");
        if(game == null)
        {
		   return (mapping.findForward("loginFailure"));
		}

		Company company = null;

        String periodNumber = (String) request.getParameter("period");
        User user = (User) session.getAttribute("user");

		if (user instanceof Admin)
		{
			company = (Company) session.getAttribute("company");
		}
		else if (user instanceof Company)
		{
              LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
              // If for any reason the company should not be allowed to login, route the
              // company to an error message page.
	          if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	          {
		        return (mapping.findForward("loginFailure"));
		      }
			  company = (Company) user;
	    }

		// purchase query parameter is "yes" if the user
		// has chosen to purchase a report.  Otherwise it is null/blank
		String purchaseFlag = request.getParameter("purchase");

		String periodString = request.getParameter("period");

		String reportIDString = request.getParameter("report");
		int reportID = -1;


		// Validate the reportID
		if (reportIDString == null)
		{
			if(log.isWarnEnabled())
			{
				log.warn(getClass().getName() + ": No report ID");
			}
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}
		else
		{
			try
			{
				reportID = Integer.parseInt(reportIDString);
			}
			catch (NumberFormatException e)
			{
				if(log.isWarnEnabled())
				{
					log.warn(getClass().getName(), e);
				}
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
		}

		int period = game.getCurrentPeriodNumber();

		// The maximum allowable period, usually the current period, but this
		// changes in the case of the Complete List of Bids Report
		int maxPeriod = period;

		// Deal with Complete list of Bids/CAM Report cases
		// Unless a period is specified, the Complete List of Bids
		// report is for the period prior to  one currently being played.
		if ((reportID == COMPLETE_LIST_OF_BIDS) || (reportID == CAM_REPORT) )
		{
			if (null != periodString && !"".equals(periodString))
			{
				period = Integer.parseInt(periodString);
			}
			else
			{
				period--;

			}
			maxPeriod--;  // The range of acceptable periods becomes smaller...
		}


		// No reports are available until a game actually begins
		if (period < 1 || period > maxPeriod)
			return (mapping.findForward("noReport"));

		boolean hasPurchased = false;
		Hashtable reportCosts = null;
		Period thisPeriod = null;
		Period nextPeriod = null;

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
		try
		{
			reportCosts = dbFacade.getGameParametersDBHandler().getConsultingCostsParameters(game);

			hasPurchased = dbFacade.getReportLedgerDBHandler().checkIfPurchased(
							   company.getCompanyID(), period, reportID);

			// The company has not yet purchased this report, and has decided to do so
			if (!hasPurchased && null != purchaseFlag && "yes".equals(purchaseFlag))
			{
				// Make a record of this payment request
				dbFacade.getReportLedgerDBHandler().recordPayment(
							   company.getCompanyID(), period, reportID, game.getCurrentPeriodNumber());

				hasPurchased = true;
			}

			thisPeriod = dbFacade.getPeriodDBHandler().getPeriod(game, game.getCurrentPeriodNumber());

			// Retrieve next period if the game has not yet been finalized
			if (game.getIsActive())
				nextPeriod = dbFacade.getPeriodDBHandler().getPeriod(game, game.getCurrentPeriodNumber() + 1);

			dbFacade.close();
		}
		catch (BIGDatabaseFacadeException e)
		{
			log.fatal(getClass().getName(), e);
 			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch (ParametersNotFoundException e)
		{
			log.fatal(getClass().getName(), e);
 			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}
		catch (ObjectNotFoundException e)
		{
			log.fatal(getClass().getName(), e);
 			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}


		ConsultingReportsForm consultingReportsForm = new ConsultingReportsForm();
		consultingReportsForm.setPeriod(period);
		consultingReportsForm.setReportID(reportID);
		consultingReportsForm.setReportCost(getReportCost(reportID, reportCosts));

		if (!hasPurchased)
		{

			if (game.getIsActive() || (reportID == COMPLETE_LIST_OF_BIDS) || (reportID == CAM_REPORT))
			{
				destination = mapping.findForward("purchaseVerification");
			}
			else
			{
				destination = mapping.findForward("noReport");
			}

		}
		else
		{
			if (reportID == COMPLETE_LIST_OF_BIDS) {

				String forward = "/completeListofBidsReport.big";

				if (null != periodString && !periodString.equals(""))
					forward += "?period=" + period;

				destination = new ActionForward(forward);

			}
			else if (reportID == CAM_REPORT)
			{

				String forward = "/camReport.big";

				if (null != periodString && !periodString.equals(""))
					forward += "?period=" + period;

				destination = new ActionForward(forward);


			}
			else if (reportID == CONSTRUCTION_DEMAND)
			{

				if (game.getIsActive() && nextPeriod != null)
				{
					consultingReportsForm.setNumberOfJobsAvailable(nextPeriod.getNumberOfNewBuildingJobs());
					destination = mapping.findForward("constructionDemand");
				}
				else
				{
					// No construction demand report for final period.
					destination = mapping.findForward("noReport");
				}

			}
			else if (reportID == LABOR_AVAILABILITY)
			{

				if (game.getIsActive() && nextPeriod != null)
				{
					consultingReportsForm.setLaborAvailability(decodeLaborAvailability(thisPeriod.getLaborAvailability()));
					destination = mapping.findForward("laborAvailability");
				}
				else
				{
					// No labor availability report for final period.
					destination = mapping.findForward("noReport");
				}
			}
			else if (reportID == MATERIAL_COST_INDEX)
			{

				if (game.getIsActive() && nextPeriod != null)
				{
					consultingReportsForm.setMaterialCostIndex(thisPeriod.getMaterialsCostIndex());
					destination = mapping.findForward("materialCostIndex");
				}
				else
				{
					// No material cost index report for final period.
					destination = mapping.findForward("noReport");
				}
			}
			else if (reportID == WEATHER_FORECAST)
			{

				if (game.getIsActive() && nextPeriod != null)
				{
					consultingReportsForm.setRainfall(thisPeriod.getRainfall());
					consultingReportsForm.setTemperature(thisPeriod.getTemperature());
					destination = mapping.findForward("weatherForecast");
				}
				else
				{
					// No weather forecast report for final period.
					destination = mapping.findForward("noReport");
				}

			}
			else
			{
				if(log.isWarnEnabled())
				{
					log.warn(getClass().getName() + ": Invalid report ID");
				}
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}

		}

		session.setAttribute("consultingReportsForm", consultingReportsForm);

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}

			saveErrors(request, errors);
			saveToken(request);
			return (mapping.findForward("error"));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to " + destination);
			}
			return (destination);
		}
	}


   /**
	* Determine the cost of a report, identified by <code>reportID</code>
	* (see constants defined at the beginning of this class)
	*
	* @param reportId the ID of the report
	* @param costs the <code>Hashtable</code> containing the
	*        report costs from the database.
	*/
	private int getReportCost(int reportID, Hashtable costs)
	{

		if (reportID == COMPLETE_LIST_OF_BIDS) {

			return ((Integer) costs.get("Complete_List_Of_Bids")).intValue();

		} else if (reportID == CONSTRUCTION_DEMAND) {

			return ((Integer) costs.get("Market_Demand")).intValue();

		} else if (reportID == LABOR_AVAILABILITY) {

			return ((Integer) costs.get("Labor_Availability")).intValue();

		} else if (reportID == MATERIAL_COST_INDEX) {

			return ((Integer) costs.get("Material_Cost_Index")).intValue();

		} else if (reportID == WEATHER_FORECAST) {

			return ((Integer) costs.get("Weather_Forecast")).intValue();

		} else if (reportID == CAM_REPORT) {

			return ((Integer) costs.get("CAM_Report")).intValue();
		}
		  else {
			return -1;
		}

	}


	/**
	 * Given a numeric labor availability value, determine the corresponding
	 * rating as a String.  Rating may be: "verygood", "good", "fair", "bad",
	 * or "verybad"
	 *
	 * @param laborAvail numeric labor availability value
	 * @returns labor availability rating in string form
	 */
	private String decodeLaborAvailability(int laborAvail)
	{
		if (laborAvail > 105)
			return "verygood";
		else if (laborAvail > 101)
			return "good";
		else if (laborAvail > 99)
			return "fair";
		else if (laborAvail > 95)
			return "bad";
		else if (laborAvail >= 92)
			return "verybad";
		else
			return "invalid";

	}
}
