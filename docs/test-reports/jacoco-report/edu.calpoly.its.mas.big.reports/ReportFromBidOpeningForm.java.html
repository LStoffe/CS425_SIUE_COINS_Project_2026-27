package edu.calpoly.its.mas.big.reports;
import java.util.Vector;
import java.util.TreeMap;
import java.util.Iterator;

import edu.calpoly.its.mas.big.game.*;
import edu.calpoly.its.mas.big.users.company.Bid;
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Job;

//BIG v3 imports
import edu.calpoly.lib.multimedia.big.game.Contractor;
import edu.calpoly.lib.multimedia.big.game.ComputerControlledContractor;

/**
 * Holds the following information for a Report from Bid Opening:
 * a list of Jobs, the winning bids on those jobs and a list of
 * a game's companies (including Company X).  The information stored
 * represents a single period in a single BIG game.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.4 $
 */
public class ReportFromBidOpeningForm extends ReportForm
{
	private TreeMap companies;
	private TreeMap jobs;
	private Vector jobsVector;
	private TreeMap bids; 	// indexed by jobID

	private String companyName;

	/*
	 * Creates empty containers for bids, jobs and companies.
	 * Adds company X object to list of companies since CompanyX
	 * is not stored as a company in the database
	 */
	public ReportFromBidOpeningForm()
	{
		bids = new TreeMap();
		companies = new TreeMap();
		jobs = new TreeMap();
	}

	/**
	 * Transforms a <code>Vector</code> of jobs from the database
	 * into a <code>TreeMap</code> keyed by job ID.  This conversion
	 * allows for simpler retrieval of <code>Job</code> objects.
	 *
	 * @param jobs Vector of jobs from one period within one game.
	 */
	public void setJobs(Vector jobs)
	{
		// Transform the jobs Vector into a TreeMap keyed by jobID
		// for easier access
		Iterator jobsIter = jobs.iterator();
		while (jobsIter.hasNext()) {
			Job j = (Job) jobsIter.next();
			this.jobs.put(new Integer(j.getJobID()), j);
		}
		jobsVector = jobs;
	}

	/**
	 * Transforms a <code>Vector</code> of companies from the database
	 * into a <code>TreeMap</code> keyed by company ID.  This conversion
	 * allows for simpler retrieval of specific <code>Contractor</code> objects.
	 *
	 * @param companies Vector of companies involved in one BIG game
	 */
	public void setCompanies(Vector companies)
	{
		// Store the companies in a TreeMap for easier retrieval
		Iterator cIter = companies.iterator();
		while (cIter.hasNext()) {
			Company c = (Company) cIter.next();
			this.companies.put(new Integer(c.getContractorID()), c);
		}
	}

	/**
	 * Adds a <code>Vector</code> of <code>ComputerControlledContractor</code> from the database
	 * to a <code>TreeMap</code> keyed by company ID.  This conversion
	 * allows for simpler retrieval of specific <code>Contractor</code> objects.
	 *
	 * @param companies Vector of companies involved in one BIG game
	 */
	public void setCCCs(Vector companies)
	{
		// Store the companies in a TreeMap for easier retrieval
		Iterator cIter = companies.iterator();
		while (cIter.hasNext()) {
			ComputerControlledContractor ccc = (ComputerControlledContractor) cIter.next();
			this.companies.put(new Integer(ccc.getContractorID()), ccc);
		}
	}

	/**
	 * Adds a winning bid to this <code>Actionform</code>'s
	 * list or winning bids that will eventually be displayed
	 * in the report
	 *
	 * @param bid the bid to be added
	 */
	public void addBid(Bid bid)
	{
		bids.put(new Integer(bid.getJobNumber()), bid);
	}



	public int getJobType(int jobID)
	{
		Job j = (Job) jobs.get(new Integer(jobID));
		if (null == j)
			return 1;
		else
			return j.getJobType();
	}


	public int getBidAmount(int jobID)
	{
		Bid bid = (Bid) bids.get(new Integer(jobID));
		if (null == bid)
			return 0;
		else
			return (int) bid.getBidAmount();
	}


	public String getCompanyTitle(int jobID)
	{
		Bid bid = (Bid) bids.get(new Integer(jobID));
		if (null == bid) {
			return "";
		} else {
			Contractor c = (Contractor) companies.get(new Integer(bid.getCompanyID()));
			return c.getTitle();
		}
	}


	public Vector getJobs()
	{
		return jobsVector;
	}

	public void setCompanyName(String name)
	{
		companyName = name;
	}

	public String getCompanyName()
	{
		return companyName;
	}

}
