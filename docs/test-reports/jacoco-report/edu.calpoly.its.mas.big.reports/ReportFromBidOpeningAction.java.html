package edu.calpoly.its.mas.big.reports;

/**
 * Builds an <code>ActionForm</code> containing the information needed
 * for a Report from Bid Opening: all winning bid amounts and the
 * companies who placed the winning bids for every job in a certain
 * period.
 *
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.9 $
 */
public class ReportFromBidOpeningAction extends Action
{

	public ActionForward execute(ActionMapping mapping,
								 ActionForm form,
								 HttpServletRequest request,
								 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		// Process request if it was not cancelled
        if(!isCancelled(request))
        {
	        ActionMessages errors = new ActionMessages();

            ServletContext context = servlet.getServletContext();

            HttpSession session = request.getSession();
            if(session == null)
            {
		       return (mapping.findForward("loginFailure"));
		    }

	        // Check for existance of Game object in user's session
	        Game game = (Game) session.getAttribute("game");
            if(game == null)
            {
		       return (mapping.findForward("loginFailure"));
		    }

            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

            // If for any reason the company should not be allowed to login, route the
            // company to an error message page.
	        if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	        {
		        return (mapping.findForward("loginFailure"));
		    }

			Company company = (Company) session.getAttribute("company");

			String periodString = request.getParameter("period");
			int reportPeriod;
		    if (periodString == null || periodString.equals(""))
		    {
				reportPeriod = game.getCurrentPeriodNumber() - 1;
			}
			else
			{
				reportPeriod = Integer.parseInt(periodString);
			}

			ReportFromBidOpeningForm reportFromBidOpeningForm = new ReportFromBidOpeningForm();
			reportFromBidOpeningForm.setPeriod(reportPeriod);

            // Make sure period is within an acceptable range
            if (reportPeriod < 1 || reportPeriod > game.getCurrentPeriodNumber() - 1)
            {
                return (mapping.findForward("noReport"));
            }

			// Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

			try
			{
				Vector jobs = dbFacade.getJobDBHandler().getJobs(game, reportPeriod);
				Vector companies = dbFacade.getCompanyDBHandler().getCompanies(game);
				Vector cccs = dbFacade.getCCCDBHandler().getAllCCCs(game);

				/* For each job, find the winning bid and add it to the
                   report ActionForm */
				Iterator jobIter = jobs.iterator();
				while (jobIter.hasNext())
				{
					Job theJob = (Job) jobIter.next();
					Vector bids = dbFacade.getBidDBHandler().getBids(game, theJob);

					Iterator bidIter = bids.iterator();
					while (bidIter.hasNext()) {
						Bid theBid = (Bid) bidIter.next();

						// We found the winning bid, add it to the report
						if (theBid.getOutcomeCode() == 1)
						{
							reportFromBidOpeningForm.addBid(theBid);
							break;    // Only one winning bid per job, go to the next Job
						}
					}

				}

				reportFromBidOpeningForm.setJobs(jobs);
				reportFromBidOpeningForm.setCompanies(companies);
				reportFromBidOpeningForm.setCCCs(cccs);

				dbFacade.close();

			}
			catch(ObjectNotFoundException onfe)
			{
				log.fatal("", onfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}


			reportFromBidOpeningForm.setCompanyName(company.getTitle());
			reportFromBidOpeningForm.setReportDate(game.getDateString(reportPeriod));
			reportFromBidOpeningForm.setPeriod(reportPeriod);

			session.setAttribute("reportFromBidOpening", reportFromBidOpeningForm);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				saveErrors(request, errors);
				saveToken(request);
				return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

		}
		else
		{  // Request was cancelled
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (new ActionForward(mapping.getInput()));
		}

	}
}
