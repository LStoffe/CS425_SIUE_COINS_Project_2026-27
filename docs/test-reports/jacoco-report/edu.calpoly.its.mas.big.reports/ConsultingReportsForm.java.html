package edu.calpoly.its.mas.big.reports;

/**
 * Since all consulting reports consist of just one or two values
 * beyond the standard report information, this
 * <code>ActionForm</code> is multi-purpose.  At any given time, it is
 * used to represent the information for exactly one consulting
 * report. Information that does not pertain to the current consulting
 * report is left as <code>null</code> or 0.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.3 $ */

public class ConsultingReportsForm extends ReportForm {
	private int reportID;                   // 1 - 4
	private int reportCost;

	private String laborAvailability = "";	// Very Good, Good, Balanced, etc.
	private int materialCostIndex;	        // 95%-105%
	private float rainfall;                 // inches
	private int temperature;                // deg. F
	private int numberOfJobsAvailable;

	private int period;

	private String companyName;

	// ---Simple Getters/Setters---

	public void setPeriod(int period) { this.period = period; }
	public int getPeriod()            { return period; }

	public void setReportID(int reportID) { this.reportID = reportID; }
	public int getReportID()              { return reportID; }

	public void setReportCost(int reportCost) { this.reportCost = reportCost; }
	public int getReportCost()                { return reportCost; }

	public void setLaborAvailability(String laborAvail) { laborAvailability = laborAvail;	}
	public String getLaborAvailability() { return laborAvailability==null?"":laborAvailability; }

	public void setMaterialCostIndex(int mci) { materialCostIndex = mci; }
	public int getMaterialCostIndex()         { return materialCostIndex; }

	public void setRainfall(float rain) { rainfall = rain; }
	public float getRainfall()          { return rainfall; }

	public void setTemperature(int temp) { temperature = temp; }
	public int getTemperature()          { return temperature; }

	public void setNumberOfJobsAvailable(int numOfJobs) { numberOfJobsAvailable = numOfJobs; }
	public int getNumberOfJobsAvailable() { return numberOfJobsAvailable; }

	public void setCompanyName(String name) { companyName = name; }
	public String getCompanyName()          { return companyName;      }

}
