package edu.calpoly.its.mas.big.reports;

import org.apache.struts.action.ActionForm;

/**
 * Serves as super class to all other report forms in BIG.  Contains information common to all report forms.
 * If a request to view a report is made, ReportForm can be used to receive the incoming data.
 * <br>
 * Report information to be returned to the user is generated and put into a ReportForm of the type matching the
 * type of report requested.  The populated ReportForm is put in the <code>Session</code> where its contents will later
 * be extracted and displayed to the user.
 *
 * @version $Revision: 1.6 $ $Date: 2002/02/25 21:16:03 $
 */
public class ReportForm extends ActionForm
{

  private int period;
  private int gameID;
  private String reportDate;

  //************************  Simple Set and Get Methods  ***********************************

  public int getPeriod()
  {
    return period;
  }

  public void setPeriod(int period)
  {
    this.period = period;
  }

  public int getGameID()
  {
    return gameID;
  }

  public void setGameID(int gameID)
  {
    this.gameID = gameID;
  }

  public String getReportDate()
  {
    return reportDate;
  }

  public void setReportDate(String reportDate)
  {
    this.reportDate = reportDate;
  }


}
