package edu.calpoly.its.mas.big.reports;

//JDK imports
import java.util.Hashtable;
import java.util.Vector;

//Servlet imports
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.users.User;
import edu.calpoly.its.mas.big.users.admin.Admin;
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 *  Handles requests to view a Complete List of Bids Report.
 *
 *  @version $Revision: 1.16 $
 */
public class CompleteListofBidsReportAction extends Action
{


	public ActionForward execute(ActionMapping mapping,
								 ActionForm form,
								 HttpServletRequest request,
								 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		// Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			MessageResources resources = getResources(request);

            HttpSession session = request.getSession();
            if(session == null)
            {
		       return (mapping.findForward("loginFailure"));
		    }

	        Game game = (Game) session.getAttribute("game");
            if(game == null)
            {
		       return (mapping.findForward("loginFailure"));
		    }

			Company company = null;

			User user = (User) session.getAttribute("user");
			if (user instanceof Admin)
				company = (Company) session.getAttribute("company");
			else if (user instanceof Company)
				company = (Company) user;

			String periodString = request.getParameter("period");
			int reportPeriod;
		    if (periodString == null || periodString.equals(""))
		    {
				reportPeriod = game.getCurrentPeriodNumber() - 1;
			}
			else
			{
				reportPeriod = Integer.parseInt(periodString);
			}

			// This accounts for the period numbering on 'View Period Reports'
			// admin page
			if ((user instanceof Admin) && (reportPeriod == game.getCurrentPeriodNumber()))
			{
				reportPeriod--;
			}

			CompleteListofBidsReportForm completeListofBidsForm = new CompleteListofBidsReportForm();

			// Make sure period is within an acceptable range
			if (reportPeriod < 1 || reportPeriod > (game.getCurrentPeriodNumber() - 1))
			{
				return (mapping.findForward("noReport"));
			}

			// Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

			try {
				// FIXME: Should it be assumed that there are in fact jobs, companies and bids
				// for this game?
				Vector jobs = dbFacade.getJobDBHandler().getJobs(game, reportPeriod);
				Vector companies = dbFacade.getCompanyDBHandler().getCompanies(game);
				Vector cccs = dbFacade.getCCCDBHandler().getAllCCCs(game);

				Hashtable miscParams = dbFacade.getGameParametersDBHandler().getMiscParameters(game);

				// Make sure that the company isn't trying any funny
				// business by going directly to this report rather than
				// through the ConsultingReportsAction gate
				if (user instanceof Company && null != company)
				{
					boolean hasPurchased = dbFacade.getReportLedgerDBHandler().checkIfPurchased(
												company.getCompanyID(), reportPeriod, 0);

					if (!hasPurchased)
					{
						//System.out.println("Trying to pull a fast one? " + company.getCompanyID() + ", " + reportPeriod);
						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
					}
				}

				completeListofBidsForm.setMaxJobsPerCompany(((Integer) miscParams.get("maxJobsPerTeam")).intValue());
				completeListofBidsForm.setJobs(jobs);
				completeListofBidsForm.addCompanies(companies);
				completeListofBidsForm.addCompanies(cccs);

				for (int i = 0; i < jobs.size(); i++)
				{
					Job theJob = (Job) jobs.get(i);

					Vector bids = dbFacade.getBidDBHandler().getBids(game, (Job) jobs.get(i));
					completeListofBidsForm.setBids(bids);
				}

				dbFacade.close();

			}
			catch(ObjectNotFoundException onfe)
			{
				log.fatal("", onfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (ParametersNotFoundException pnfe)
			{
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			if (null != company)
			{
				completeListofBidsForm.setCompanyName(company.getTitle());
			}

			completeListofBidsForm.setReportDate(game.getDateString(reportPeriod));
			completeListofBidsForm.setPeriod(reportPeriod);

			session.setAttribute("listofBids", completeListofBidsForm);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				saveErrors(request, errors);
				saveToken(request);
				return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			// Request was cancelled
			return (new ActionForward(mapping.getInput()));
		}

	}
}
