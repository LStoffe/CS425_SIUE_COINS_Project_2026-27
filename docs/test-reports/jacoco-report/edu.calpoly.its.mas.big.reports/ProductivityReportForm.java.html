package edu.calpoly.its.mas.big.reports;

import java.util.Vector;

import edu.calpoly.its.mas.big.game.Method;

/**
 * Holds info for a Productivity Report.
 *
 * @author Kevin Craig
 * @version $Revision: 1.2 $ $Date: 2002/02/26 00:56:37 $
 */
public class ProductivityReportForm extends ReportForm
{

  Method[][] methods;

  public ProductivityReportForm()
  {
  }

  //********************************  Simple Set and Get Methods  ******************************


  public Method[][] getMethods()
  {
       return methods;
  }

  public void setMethods(Method[][] methods)
  {
       this.methods = methods;
  }

  /**
   * Returns a method for a given methodID for and a given activityID
   * <br><br>
   * Pre: 1 <= methodID <= 5 <br>
   * Pre: 1 <= activityID <= 9
   */
  public Method getMethod(int methodID, int activityID)
  {
	  return methods[(activityID - 1)][(methodID - 1)];
  }


}
