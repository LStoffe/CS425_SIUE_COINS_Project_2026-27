package edu.calpoly.its.mas.big.reports;

//JDK imports
import java.util.Hashtable;

//Serlvet imports
import javax.servlet.ServletContext;
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;

//Servlet imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;

import org.apache.struts.action.ActionForm;

import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;

/**
 *  Handles requests to view a Progress Report.
 *
 *  @author Kevin Craig
 *  @version $Revision: 1.24 $ $Date: 2002/04/16 00:35:36 $
 */
public class ProgressReportAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
		           				 ActionForm form,
		           			     HttpServletRequest request,
		           			     HttpServletResponse response)
		{
			Log log = LogFactory.getLog(this.getClass());
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " beginning execute method");
			}

	       	ActionMessages errors = new ActionMessages();

	       	// Check for existance of Game object in user's session
           	ServletContext context = servlet.getServletContext();

           	HttpSession session = request.getSession();
           	if(session == null)
           	{
			   return (mapping.findForward("loginFailure"));
			}

			Game game = (Game) session.getAttribute("game");
			if(game == null)
			{
		       return (mapping.findForward("loginFailure"));
			}

			Company company = null;

			String periodNumber = (String) request.getParameter("period");
			User user = (User) session.getAttribute("user");

			if(user instanceof Company)
			{
			   LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
			   // If for any reason the company should not be allowed to login, route the
			   // company to an error message page.
			   if(!loginBroker.verifyGameLoginAllowed(game, context, request))
			   {
			       return (mapping.findForward("loginFailure"));
		   	   }
			   company = (Company) user;
			}
			else if(user instanceof Admin)
			{
			   company = (Company) session.getAttribute("company");
			}

		    Period period = null;
		    Vector jobs = null;
		    Vector jobTitles = new Vector();
		    Method[][] methods = new Method[9][5];
		    Method[][] methodsHC = new Method[9][5];

		    int reportPeriod;

		    if (periodNumber == null)
		    {
		 	   reportPeriod = game.getCurrentPeriodNumber() - 1;
		    }
		    else
		    {
			   reportPeriod = Integer.parseInt(periodNumber);
		    }

		    ProgressReportForm progressReportForm = new ProgressReportForm();

		    //Load report with company title, report date, and report period
		    progressReportForm.setCompanyName(company.getTitle());
		    progressReportForm.setReportDate(game.getDateString(reportPeriod));
		    progressReportForm.setPeriod(reportPeriod);

		    if(reportPeriod > 0)
		    {
		 	    MessageResources resources = getResources(request);

 			    BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

 			    try
				{
					for(int j = 0; j < 9; j++)
				    {
				  	    methods[j] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, (j+1));
				    }
					for(int j = 0; j < 9; j++)
				    {
				  	    methodsHC[j] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, (j+10));
				    }

				    period = dbFacade.getPeriodDBHandler().getPeriod(game, reportPeriod);
				    jobs = dbFacade.getJobDBHandler().getWorkedJobs(game, company, reportPeriod) ;

		  		    if(jobs != null)
					{
					    Iterator i = jobs.iterator();
					    while(i.hasNext())
					    {
					        Job job = (Job) i.next();
						    Activity[] activities = dbFacade.getActivityDBHandler().getActivities(game, job, reportPeriod);
						    job.setActivities(activities);
						    jobTitles.add(resources.getMessage("jobType.jobType" + job.getJobType()));

						}
					}

					dbFacade.close();

				}
				catch (BIGDatabaseFacadeException bdbe)
				{
					errors.add("database", new ActionMessage("error.database"));
					log.fatal("", bdbe);
				}
				catch(ObjectNotFoundException onfe)
				{
					log.fatal("", onfe);
				}

				//Determine the activity name in the resource bundle associated with the activity type number
				String[] activityTypes = new String[18];
				for(int i = 0; i < 18; i++)
				{
				   activityTypes[i] = resources.getMessage("activity.activity" + (i+1));
				}

				//Load report with period values
				loadTemperature(period, resources, progressReportForm);
				loadRainfall(period, resources, progressReportForm);
				loadMaterialsCostIndex(period, progressReportForm);
				loadLaborAvailability(period, resources, progressReportForm);

				progressReportForm.setJobs(jobs);
				progressReportForm.setJobTitles(jobTitles);
				progressReportForm.setActivityTypes(activityTypes);
				progressReportForm.setMethods(methods);
				progressReportForm.setMethodsHC(methodsHC);
			}
		    else
		    {

		    }

		    session.setAttribute("progressReportForm", progressReportForm);

		    if (!errors.isEmpty())
			{
			    // Report any errors we have discovered back to the original form
		    	saveErrors(request, errors);
				saveToken(request);
				return (new ActionForward(mapping.getInput()));
			}
			else
			{
			    return (mapping.findForward("success"));
			}

		} // end execute()


		//Pre: period contains the temperature value for the report period.  resources contains the key "progressReport.temperatureUnits"
		//which gives the key to the units string for temperature
		//Post: The temperature value is now a String with units appended in progressReportForm
		private void loadTemperature(Period period, MessageResources resources, ProgressReportForm progressReportForm)
		{
		    String temperatureUnits = resources.getMessage("progressReport.temperatureUnits");
		    progressReportForm.setTemperature(period.getTemperature() + temperatureUnits);
		}


		//Pre: period contains the rainfall value for the report period.  resources contains the key "progressReport.precipitationUnits"
		//which gives the key to the units string for rainfall
        //Post: The rainfall value is now a String with units appended in progressReportForm
		private void loadRainfall(Period period, MessageResources resources, ProgressReportForm progressReportForm)
		{
			NumberFormat formatter = NumberFormat.getInstance();
	        formatter.setMaximumFractionDigits(2);
			String rainfallUnits = resources.getMessage("progressReport.precipitationUnits");
			progressReportForm.setRainfall(formatter.format(period.getRainfall()) + " " + rainfallUnits);
		}


		//Pre: period contains the labor availability value for the report period.  resources contains keys to the String
		//representations of the labor availability values
		//Post: The labor availability is now a String in progressReportForm
		private void loadLaborAvailability(Period period, MessageResources resources, ProgressReportForm progressReportForm)
		{
			int laborAvailability = period.getLaborAvailability();
			String laborAvailabilityString = "Error";

			//Hardcoded cutoff values from SRS
			if(laborAvailability <= 95)
			{
			   laborAvailabilityString = resources.getMessage("laborAvailability.verybad");
			}
			else if(laborAvailability <= 99)
			{
			   laborAvailabilityString = resources.getMessage("laborAvailability.bad");
			}
			else if(laborAvailability <= 101)
			{
			   laborAvailabilityString = resources.getMessage("laborAvailability.fair");
			}
			else if(laborAvailability <= 105)
			{
			   laborAvailabilityString = resources.getMessage("laborAvailability.good");
			}
			else
			{
			   laborAvailabilityString = resources.getMessage("laborAvailability.verygood");
			}
			progressReportForm.setLaborAvailability(laborAvailabilityString);
		}


		//Pre: period contains the cost index value value for the report period.
		//Post: The materials cost index is now a number String with 2 decimal places, in proper percent format for the current
		//local, in progressReportForm
		private void loadMaterialsCostIndex(Period period, ProgressReportForm progressReportForm)
		{
		   //Setup NumberFormat to fix decimals to 2 places in percents
		   NumberFormat percentFormatter = NumberFormat.getPercentInstance();
		   percentFormatter.setMaximumFractionDigits(0);
		   progressReportForm.setMaterialsCostIndex(percentFormatter.format(((double) period.getMaterialsCostIndex()) / 100));
		}
}
