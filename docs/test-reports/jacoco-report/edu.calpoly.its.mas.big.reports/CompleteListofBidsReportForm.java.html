package edu.calpoly.its.mas.big.reports;

import java.util.TreeMap;
import java.util.TreeSet;
import java.util.Vector;
import java.util.Iterator;
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.users.company.Bid;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * Holds info for a Complete List of Bids Report
 *
 * @version $Revision: 1.9 $
 */

public class CompleteListofBidsReportForm extends ReportForm {

  //private Vector bids;
  private Vector jobs;
  private Vector companies = new Vector();
  private int period;
  private String periodString;
  private int maxJobsPerCompany;
  private int companyID;
  private String companyName;

  // Keyed by 'jobID.companyID' for now (there's probably
  // a much better way...)
  private TreeMap bids = new TreeMap();

  //******************************  Simple Get and Set Methods  **************************************

  public void setBids(Vector inBids)
  {
	  if (inBids != null && inBids.size() > 0) {
		  // Iterate through the bids for this job, populating the bids TreeMap
		  Iterator bidIterator = inBids.iterator();
		  while(bidIterator.hasNext()) {
			  Bid theBid = (Bid) bidIterator.next();

			  String key = theBid.getJobNumber() + "." + theBid.getCompanyID();
			  bids.put(key, theBid);
		  }
	  }

  }

  public String getBidAmount(int jobID, int companyID)
  {
	  String bidString = "";
	  String key = jobID + "." + companyID;

	  Bid theBid = (Bid) bids.get(key);

	  if (theBid == null) {
		  bidString = "";
	  } else {
		  bidString = NumberConverter.stringValue(theBid.getBidAmount());
	  }

	  return bidString;

  }

  public int getOutcomeCode(int jobID, int companyID)
  {
	  int outcomeCode = 0;
	  String key = jobID + "." + companyID;

	  Bid theBid = (Bid) bids.get(key);

	  if (theBid == null) {
		  outcomeCode = 5;
	  } else {
		  outcomeCode = theBid.getOutcomeCode();
	  }

	  return outcomeCode;
  }

  public String getOutcomeArgument(int jobID, int companyID)
  {
	  String outcomeArg = "";

	  int outcome = getOutcomeCode(jobID, companyID);

	  if (outcome == 3)
		  outcomeArg = getMaxJobsPerCompany();

	  return outcomeArg;
  }


  public Vector getJobs()
  {
    return jobs;
  }

  public void setJobs(Vector jobs)
  {
    this.jobs = jobs;
  }

  public Vector getCompanies()
  {
    return companies;
  }

  /**
   *  Add <code>Contractor</code> instances to an internal list of contractors.
   *  Each time
   *  The list is retrievable through the <code>getCompanies</code> method.
   *
   */
  public void addCompanies(Vector companies)
  {
	  this.companies.addAll(companies);

	  TreeSet sorter = new TreeSet(this.companies);

	  this.companies = new Vector(sorter);
  }


  public void setPeriodString(String periodString)
  {
	  this.periodString = periodString;
  }

  public String getPeriodString()
  {
	  return periodString;
  }

  public void setPeriod(int period)
  {
	  this.period = period;
  }

  public int getPeriod()
  {
	  return period;
  }

  public void setMaxJobsPerCompany(int maxJobs)
  {
	  this.maxJobsPerCompany = maxJobs;
  }

  public String getMaxJobsPerCompany()
  {
	  return String.valueOf(maxJobsPerCompany);
  }

  public int getCompanyID()
  {
	  return companyID;
  }

  public void setCompanyID(int companyID)
  {
	  this.companyID = companyID;
  }

  public String getCompanyName()
  {
	  return companyName==null?"":companyName;
  }

  public void setCompanyName(String companyName)
  {
	  this.companyName = companyName;
  }

}
