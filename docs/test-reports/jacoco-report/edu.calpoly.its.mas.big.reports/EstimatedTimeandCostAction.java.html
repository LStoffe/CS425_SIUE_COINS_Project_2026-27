package edu.calpoly.its.mas.big.reports;

/**
 * Construct an <code>ActionForm</code> which represents the data in
 * an Estimated Time and Cost Report.  The query parameters expected
 * are: "jobID" which identifies the job for which the report
 * will be generated and an optional "period" parameter, which
 * allows the user to view the Estimated Time and Cost Report for
 * any past period.  If the period parameter is omitted, the game's
 * current period is used.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.15 $
 */
public class EstimatedTimeandCostAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		ActionMessages errors = new ActionMessages();

        HttpSession session = request.getSession();

        if(session == null)
        {
		   return (mapping.findForward("loginFailure"));
		}

	    Game game = (Game) session.getAttribute("game");
        if(game == null) {
		   return (mapping.findForward("loginFailure"));
		}
		Company company = null;

		User user = (User) session.getAttribute("user");
		if (user instanceof Admin)
		{
			company = (Company) session.getAttribute("company");
		}
		else if (user instanceof Company)
		{
			company = (Company) user;
		}

		// Grab jobID parameter
		String jobIDString = request.getParameter("jobID");
		int jobID = 0;
		if (jobIDString != null && !jobIDString.equals(""))
		{
			jobID = Integer.parseInt(jobIDString);
		}
		else
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}

		String periodString = request.getParameter("period");
		int reportPeriod;
		if (periodString == null || periodString.equals(""))
		{
			reportPeriod = game.getCurrentPeriodNumber();
		}
		else
		{
			reportPeriod = Integer.parseInt(periodString);
			session.setAttribute("reportPeriod", periodString);
		}


		String referrer = request.getParameter("referrer");

		// Pass 'referrer' param to JSP. This allows the JSP to
		// build the proper navigation bar (there are two ways for
		// a company to access the Estimated Time and Cost Report)
		if (null != referrer) {
			session.setAttribute("referrer", referrer);
		}

		EstimatedTimeandCostForm reportForm = new EstimatedTimeandCostForm();

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);

		try
		{
			// Ensure that the jobId passed in as a request parameter
			// corresponds to a current job. If not, throw an exception.
			Vector jobs = dbFacade.getJobDBHandler().getJobs(game, reportPeriod);

			// Admin->Period Reports does not put a Company in session
			if (null != company)
			{
				Vector allJobsForCompany = dbFacade.getJobDBHandler().getJobs(game, company);
				jobs.addAll(allJobsForCompany);
			}

			Iterator jobsIter = jobs.iterator();
			boolean jobFound = false;
			while(jobsIter.hasNext()) {
				Job job = (Job) jobsIter.next();
				if (job.getJobID() == jobID) {
					reportForm.setJob(job);
					jobFound = true;
				}
			}
            // must close connection in case exception is thrown
            dbFacade.close();

			if (!jobFound)
				throw new ObjectNotFoundException();

			Activity [] activities = null;

			// true when requesting a progress report
			if (null != referrer && referrer.equals("progressReport"))
			{
				activities = dbFacade.getActivityDBHandler().getActivities(game,
																		   reportForm.getJob(),
																		   reportPeriod);

				// Tell the form to use lusRemaining rather than lusTotal in all calculations
				reportForm.setStaticReportFlag(false);
			}
			// true when called using "More Info" during bid on jobs
			else
			{
				activities = dbFacade.getActivityDBHandler().getActivities(game,
																		   reportForm.getJob(),
																		   reportForm.getJob().getPeriodCreated() - 1);
			}

			reportForm.setActivities(activities);

			int[][] retentionPolicies = dbFacade.getRetentionDBHandler().getRetentionPolicies();
			reportForm.setRetentionPolicies(retentionPolicies);

            // changed to 18 activities -- include the ones for Heavy jobs
            // getAllMethods() -- retrieves a list of all possible methods to use on a certain activity type
            // returns: Method[] containing 5 methods (every activity will have one of five possible methods).
			for (int activityNum = 1; activityNum <= 18; activityNum++) {
				Method[] methods = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, activityNum);

				reportForm.setMethods(activityNum, methods);
			}

			// Retrieve number of hours per day (a misc game parameter)
			Hashtable miscParams = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
			reportForm.setHoursPerDay(((Integer) miscParams.get("standardHoursPerDay")).intValue());

			//for Heavy jobs only
			Job job = reportForm.getJob();
			if(job.getTypeModifier() == Job.HEAVY)
			{
				//get quantities
				Activity[] jobsActivities = dbFacade.getActivityDBHandler().getActivities(game, job, game.getCurrentPeriodNumber() - 1 );
                int[] quantities = new int[9];
                for (int i = 0; i < 9; i++)
				{
				   quantities[i] = jobsActivities[i].getLusTotal();
		        }
                reportForm.setQuantities(quantities);

				//get unit prices
				double[][] unitPrices = new double[45][4];
			    Hashtable parameters;
			    int j = 0;

			    // Note: x = activityType, i = methodID
			    // Once for each activity (second argument to getMethodParameters() is --> int activityType)
			    for (int x=0; x < 9; x++)
			    {
					for (int i=0; i < 5; i++)
					{
			    		// retrieves method parameters for one activityType and one methodID.
			    		parameters = dbFacade.getGameParametersDBHandler().getMethodParameters(game, x+10, i+1);

			    		unitPrices[j][0] = ( (Integer) parameters.get("LaborCost")).intValue();
			    		unitPrices[j][1] = ( (Integer) parameters.get("MaterialsCost")).intValue();
			    		unitPrices[j][2] = ( (Integer) parameters.get("EquipmentCost")).intValue();
			    		unitPrices[j][3] = ( (Integer) parameters.get("SubcontractorCost")).intValue();

			    		j++;
					}
				}
				reportForm.setUnitPrices(unitPrices);

			} // end of: if(job.getTypeModifier() == Job.HEAVY)

			dbFacade.close();
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}
		catch(ParametersNotFoundException pnfe)
		{
			log.fatal("", pnfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		finally
		{
			try
			{
         		dbFacade.close();
			}
			catch (BIGDatabaseFacadeException e)
			{
				log.fatal(getClass().getName(), e);
 				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
		}

		reportForm.setReportDate(game.getCurrentDateString());

		session.setAttribute("estimatedTimeandCostReportForm", reportForm);

		if (!errors.isEmpty())
		{
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);

			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}

    }
}
