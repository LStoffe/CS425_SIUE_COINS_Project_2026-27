package edu.calpoly.its.mas.big.reports;

import edu.calpoly.its.mas.big.game.Job;

/**
 * Holds financial info for one period of a job
 *
 * @author Kevin Craig
 * @version $Revision: 1.3 $ $Date: 2002/02/12 00:13:58 $
 */
public class JobFinancialInfo
{
	private int jobid;

    private int jobTypeID;
	private String jobType;

	private short typeModifier;

	private long paymentsReceived;
	private long retainageReceived;
	private long directLaborCost;
	private long directMaterialsCost;
	private long directEquipmentCost;
	private long subcontractorCost;
	private long fieldOverhead;
	private long liquidatedDamages;
	private int daysLate;
	private long liquidatedDamagesPerDay;

	public JobFinancialInfo(Job job)
	{
		this.jobid = job.getJobID();
		jobType = null;
		this.jobTypeID = jobTypeID;
		this.typeModifier = job.getTypeModifier();
		this.paymentsReceived = 0;
		this.retainageReceived = 0;
		this.directLaborCost = 0;
		this.directMaterialsCost = 0;
		this.directEquipmentCost = 0;
		this.fieldOverhead = 0;
		this.liquidatedDamages = 0;
		this.subcontractorCost = 0;
		this.daysLate = 0;
		this.liquidatedDamagesPerDay = 0;

	}

	public JobFinancialInfo(int jobid,
	                        int jobTypeID,
	                        short typeModifier,
						    long paymentsReceived,
						    long retainageReceived,
						    long directLaborCost,
						    long directMaterialsCost,
						    long directEquipmentCost,
						    long subcontractorCost,
						    long fieldOverhead,
						    int  daysLate,
						    long liquidatedDamagesPerDay
						    )
	{
		this.jobid = jobid;
		jobType = null;
		this.jobTypeID = jobTypeID;
		this.typeModifier = typeModifier;
		this.paymentsReceived = paymentsReceived;
		this.retainageReceived = retainageReceived;
		this.directLaborCost = directLaborCost;
		this.directMaterialsCost = directMaterialsCost;
		this.directEquipmentCost = directEquipmentCost;
		this.subcontractorCost = subcontractorCost;
		this.fieldOverhead = fieldOverhead;
		this.liquidatedDamages = liquidatedDamagesPerDay * daysLate;
		this.daysLate = daysLate;
		this.liquidatedDamagesPerDay = liquidatedDamagesPerDay;
	}


//************ PUBLIC METHODS ***************

	public int getJobID()
	{
		return jobid;
	}

	public void setJobID(int jobid)
	{
		this.jobid = jobid;
	}

	public String getJobType()
	{
		return jobType;
	}

	public void setJobType(String jobType)
	{
		this.jobType = jobType;
	}

	public int getJobTypeID()
	{
		return jobTypeID;
	}

	public void setJobTypeID(int jobTypeID)
	{
		this.jobTypeID = jobTypeID;
	}

    public short getTypeModifier()
    {
		return typeModifier;
	}

    public void setTypeModifier(short typeModifier)
    {
		this.typeModifier = typeModifier;
	}

	public long getPaymentsReceived()
	{
		return paymentsReceived;
	}

	public void setPaymentsReceived(long paymentsReceived)
	{
		this.paymentsReceived = paymentsReceived;
	}

	public void increasePaymentsReceived(long amount)
	{
		this.paymentsReceived += amount;
	}

	public long getRetainageReceived()
	{
		return retainageReceived;
	}

	public void setRetainageReceived(long retainageReceived)
	{
		this.retainageReceived = retainageReceived;
	}

	public void increaseRetainageReceived(long amount)
	{
		this.retainageReceived += amount;
	}

	public long getDirectLaborCost()
	{
		return directLaborCost;
	}

	public void setDirectLaborCost(long directLaborCost)
	{
		this.directLaborCost = directLaborCost;
	}

	public void increaseDirectLaborCost(long amount)
	{
		this.directLaborCost += amount;
	}

	public long getDirectMaterialsCost()
	{
		return directMaterialsCost;
	}

	public void setDirectMaterialsCost(long directMaterialsCost)
	{
		this.directMaterialsCost = directMaterialsCost;
	}

	public void increaseDirectMaterialsCost(long amount)
	{
		this.directMaterialsCost += amount;
	}

	public long getDirectEquipmentCost()
	{
		return directEquipmentCost;
	}

	public void setDirectEquipmentCost(long directEquipmentCost)
	{
		this.directEquipmentCost = directEquipmentCost;
	}

	public void increaseDirectEquipmentCost(long amount)
	{
		this.directEquipmentCost += amount;
	}

	public long getSubcontractorCost()
	{
		return subcontractorCost;
	}

	public void setSubcontractorCost(long subcontractorCost)
	{
		this.subcontractorCost = subcontractorCost;
	}

	public void increaseSubcontractorCost(long amount)
	{
		this.subcontractorCost += amount;
	}

	public long getFieldOverhead()
	{
		return fieldOverhead;
	}

	public void setFieldOverhead(long fieldOverhead)
	{
		this.fieldOverhead = fieldOverhead;
	}

	public void increaseFieldOverhead(long amount)
	{
		this.fieldOverhead += amount;
	}

	public long getLiquidatedDamages()
	{
		return liquidatedDamages;
	}

	public void setLiquidatedDamages(long liquidatedDamages)
	{
		this.liquidatedDamages = liquidatedDamages;
	}

	public void increaseLiquidatedDamages(long amount)
	{
		this.liquidatedDamages += amount;
	}

	public int getDaysLate()
	{
		return daysLate;
	}

	public void setDaysLate(int daysLate)
	{
		this.daysLate = daysLate;
	}

	public void incrementDaysLate()
	{
		this.daysLate++;
	}

	public long getLiquidatedDamagesPerDay()
	{
		return liquidatedDamagesPerDay;
	}

	public void setLiquidatedDamagesPerDay(long liquidatedDamagesPerDay)
	{
		this.liquidatedDamagesPerDay = liquidatedDamagesPerDay;
	}

    public long getJobNetIncome()
    {
		return paymentsReceived + retainageReceived - directLaborCost - directMaterialsCost - directEquipmentCost - subcontractorCost - fieldOverhead - liquidatedDamages;
	}

}
