package edu.calpoly.its.mas.big.reports;

import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Method;
import edu.calpoly.its.mas.big.game.Activity;

/**
 * Holds info for an Estimated Time and Cost Report
 *
 * @version $Revision: 1.10 $
 */

public class EstimatedTimeandCostForm extends ReportForm {
	// The Job associated with this report
	private Job job;

    // 18 possible activity types -- each has five possible methods
	private Method[][] methods = new Method[18][5];  // First index is the activity id
	private Activity[] activities;

	private int[][] retentionPolicies = null;

	private int hoursPerDay;

	// true indicates that lusTotal should be used in calculations,
	// false indicates that lusRemaining should be used
	private boolean staticReportFlag = true;

    // for Heavy Jobs
    private double[][] unitPrices = new double[45][4];
    private int[] quantities = new int[9];

	/**
	 * Calculate the labor cost for a method of an activity using the formula:
	 * Total Labor Units * Labor Cost.
	 *
	 * @param activity the activity (1-9) in which we are interested
	 * @param method the method (1-5)
	 * @returns Labor cost for the activity/method combination
	 */
	public int getLaborCost(int activity, int method)
	{
		// Return 0 if bogus activity or method is passed in, or
		// if labor cost for this method/activity is 0
		if ((activity < 1 || activity > 18) ||
			(method < 1 || method > 5) ||
			(methods[activity - 1][method - 1].getBaseLaborCost() == 0))
		{
			return 0;
		}
		else
		{
		    // adjust for Heavy jobs. -- activities[] only has nine elements
		    int tempActivity = activity;
		    if (job.getTypeModifier() == Job.HEAVY) tempActivity = tempActivity - 9;

			int laborUnits = activities[tempActivity - 1].getLusTotal();
			if (!staticReportFlag)
			{
				laborUnits = activities[tempActivity - 1].getLusTotal() - activities[tempActivity - 1].getLusCompleted();
			}
			return (laborUnits * methods[activity - 1][method - 1].getBaseLaborCost());
		}
	}

	/**

	/**
	 * Calculate the estimated number of days an activity will take
	 * using a certain method using the following formula:
	 * (Total Labor Units / (Productivity * Method Scaling Factor)) / Hours per Day
	 *
	 * @param activity the activity (1-9) in which we are interested
	 * @param method the method (1-5)
	 * @returns The number of days it will take to complete an
	 *          activity using a given method
	 */
	public int getDays(int activity, int method)
	{
		// Return 0 if bogus activity or method is passed in, or
		// if productivity for this method/activity is 0
		if ((activity < 1 || activity > 18) ||
			(method < 1 || method > 5) ||
			(methods[activity - 1][method - 1].getProductivity() == 0))
		{
			return 0;
		}
		else
		{
		    // adjust for Heavy jobs. -- activities[] only has nine elements
		    int tempActivity = activity;
		    if (job.getTypeModifier() == Job.HEAVY) tempActivity = tempActivity - 9;

			int laborUnits = activities[tempActivity - 1].getLusTotal();
			if (!staticReportFlag)
			{
				laborUnits = activities[tempActivity - 1].getLusTotal() - activities[tempActivity - 1].getLusCompleted();
			}

			// Adjust productivity by Job's scaling factor
			float actualProductivity = methods[activity - 1][method - 1].getProductivity() * job.getMethodScalingFactor();
			return (int) ((laborUnits / actualProductivity)) / hoursPerDay;
		}
	}

	public String getMethodName(int method)
	{
		if (method < 1 || method > 5)
			return "Try Again";
		else
			// TODO: Make this return actual method name?
			return "Method #" + method;
	}

    public int getPercentofJob(int retentionID)
	{
		if (retentionPolicies == null ||
			retentionID < 1 || retentionID > 5) {
			return 0;
		} else {
			return retentionPolicies[retentionID - 1][0];
		}
	}

    public int getRetainedAmount(int retentionID)
	{
		if (retentionPolicies == null ||
			retentionID < 1 || retentionID > 5) {
			return 0;
		} else {
			return retentionPolicies[retentionID - 1][1];
		}
	}

	public void setMethods(int activity, Method[] methods)
	{
		this.methods[activity - 1] = methods;
	}

    public void setRetentionPolicies(int[][] retentions)
	{
		retentionPolicies = retentions;
	}

	public void setStaticReportFlag(boolean flag)

	}

	}

	public double[][] getUnitPrices()
	{

	}
	public void setUnitPrices(double[][] unitPrices)
	{
		this.unitPrices = unitPrices;
	}

	public int[] getQuantities()
	{
		return quantities;
	}
	public void setQuantities(int[] quantities)
	{
		this.quantities = quantities;
	}

}
