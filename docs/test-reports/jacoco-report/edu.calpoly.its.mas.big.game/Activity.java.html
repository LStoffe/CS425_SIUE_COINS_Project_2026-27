/**
 * Contains information about the current state of an activity of a job.
 * The <code>Activity</code> class represents the state of an activity
 * during a particular period rather than the activity itself.
 * <code>Activity</code> objects use their own period count that is not tied into the
 * main period count in <code>Game</code> so it is up to BIG to keep the two in sychronization (if sychronization is desired).
 * There are period sensitive values in each <code>Activity</code> (such as current LU's completed)
 * that are cleared everytime the <code>Activity</code>'s period is incremented.  By incrementing each <code>Activity</code>'s
 * period each time the game period is incremented each <code>Activity</code> will contain
 * work/cost values for each particular period for only work done during that period.
 * Each <code>Activity</code> also keeps a running total of all work/costs ever done on the activity.
 * The update method will do one days work on an activity with the work counting as beening done in the period the <code>Activity</code>
 * thinks is the current period.
 * <br><br>
 * The constructor for <code>Activity</code> does not create <code>Activity</code> objects that
 * contain all the values they need to perform updates because some values may not be available at
 * the time the <code>Activity</code> is created.  Before the update method can be called the <code>Activity</code>
 * object must be provided with network dependency values, and a method.
 * <br><br>
 * <b>How <code>Activity</code> fits into the update process</b><br>
 * Whe <code>Job</code>s are created a set of empty activities is saved to that database for that <code>Job</code>.
 * If the <code>Job</code> is never assigned then no more activites are ever added to the database for that <code>Job</code>.
 * If a <code>Job</code> is being worked on then each period the <code>Job</code> will be loaded with its activities as they were
 * at the end of the last period (or with its set of empty activities if this is the first time work is being done
 * on this <code>Job</code>).  The <code>Job</code> will then have to tell each <code>Activity</code> to increment its period
 * and clear out its period sensitive values from last period.
 * As work is done each <code>Activity</code> object will develop a new values for the current period that must be stored to the database
 * as a new <code>Activity</code>.  This is because each <code>Activity</code> is the <i>state</i> of an activity for a period
 * so all the saved activies in the database are actually saved states of how <code>Activity</code>'s were at different periods.
 * <br>When the game is being finalized one <code>Activity</code> is used without updating the period for that <code>Activity</code>
 * so that work done during multiple periods is all counted as being done at the same time for <code>Activity</code>'s.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.29 $ $Date: 2002/04/24 17:45:37 $
 */

   * @param activityType int [1...9] representing the type this <code>Activity</code> is.
   * @param lusTotal The total number of LUs that must be completed before this <code>Activity</code> is considered complete
   * @param lusCompleted The total number of LUs that have been done on this <code>Activity</code> since it was created
   * @param currentLaborCost The dollar amount of cost of labor accrued from work done on this <code>Activity</code> <i>in the current period only</i>
   * @param currentMaterialCost The dollar amount of cost of materials accrued from work done on this <code>Activity</code> <i>in the current period only</i>
   * @param isActive true if this <code>Activity</code> is active
   * @param lusCompletedThisPeriod The number of lus done on this <code>Activity</code> <i>in the current period only </i>
   * @param accumulatedCosts All costs accumulated (both labor and materials) since this <code>Activity</code> was created
   * @param currentPeriod The period number of this <code>Activity</code>.

   * @param overtimeDays The number of days that should be worked as overtime days on this <code>Activity</code>.
   * @param stdHours Number of hours in a standard workday
   * @param overtimeHours Number of hours that will be worked, in addition to the hours in a standard day, on an overtime day.

				  boolean isActive, int lusCompletedThisPeriod,

	  this.activityType = activityType;
	  this.lusTotal = lusTotal;
	  this.lusCompleted = lusCompleted;
	  this.currentLaborCost = currentLaborCost;
	  this.currentMaterialCost = currentMaterialCost;

	  this.currentSubcontractorCost = currentSubcontractorCost;
	  this.isActive = isActive;
	  this.lusCompletedThisPeriod = lusCompletedThisPeriod;
	  this.accumulatedCosts = accumulatedCosts;
	  this.currentPeriod = currentPeriod;
	  this.currentMethodUsed = currentMethodUsed;
	  this.overtimeDays = overtimeDays;
	  this.stdHours = stdHours;
	  this.overtimeHours = overtimeHours;
	  this.lusCompletedBeforeUpdates = lusCompleted;
  }

  /**
   * Advances the period number of this activity so that it represents the initial state of the activity for the next period.
   * Values such as lusCompletedThisPeriod are reset to 0, while persistent values such as lusCompleted are left the same.
   */
  public void incrementPeriodNumber()
  {
	  currentLaborCost = 0;
	  currentMaterialCost = 0;
	  currentEquipmentCost = 0;
	  currentSubcontractorCost = 0;
	  lusCompletedThisPeriod = 0;
	  lusCompletedBeforeUpdates = lusCompleted;
	  currentPeriod++;
  }



  /**
   * Sets the current method
   */
  public void setMethod(Method method)
  {
	  this.currentMethod = method;
	  currentMethodUsed = method.getMethodNumber();
  }


  /**
   * Does one days work on the activity.  If there are overtime days left on the activity the day will be
   * an overtime day.  If the activity cannot proceed due to a network dependency no work will be done.
   * <br><br>
   * <b>Precondtion: </b>The method to use and network dependency values have been set
   * via the appropiate set methods.
   * <br><br>
   * @param rainday Set to true if the current day is a rainday.  If true work will only be done on the job
   * if the current method used is not affected by rain;
   */
  public void update(boolean rainday, int jobType, int jobID, ExpandedCompany company, ExpandedGame game, PrintWriter out)
  {
	  int lusCompletedSoFar = lusCompleted;

	  int productivityReduced = 0;

	  if(isActive)  // set in NetworkDependencies.java (see Job.java line # 768) -- determines whether or not work can be performed on an
	                // activity based on its Network Dependences. If not, the activity is skipped. Completed activities are also skipped.
	  {
          AdjustProductivity adjustMe = new AdjustProductivity();

	      int productivity = currentMethod.getProductivity();
	      int laborCost = currentMethod.getLaborCost();
	      int materialsCost = currentMethod.getMaterialsCost();
	      int equipmentCost = currentMethod.getEquipmentCost();
	      int subcontractorCost = currentMethod.getSubcontractorCost();
	      int weatherSensitivity = currentMethod.getWeatherSensitivity();

	      //Scale the method values based on the scaling factor
	      productivity = (int) Math.round(productivity * methodScalingFactor);

	      //For Heavy jobs, adjust productivity and unit cost depending on what equipment a company has in its fleet
	      if(jobType > 5)
	      {
	         try
	         {
	            productivityReduced = adjustMe.adjustProductivity(productivity, jobID, activityType, currentMethodUsed, currentPeriod, game, company, currentMethod, out);
	            // log = LogFactory.getLog(this.getClass());
				// log.debug("entered adjustProductivity");
	         }
	         catch(Exception e)
	         {
                e.printStackTrace();
                throw new RuntimeException ("Error executing adjustProductivity. Cause: " + e);
             }

             // if productivity was reduced because some equipment is missing
             if(productivityReduced < productivity) {

                 productivity = productivityReduced;

                 // when productivity is reduced we increase the unit cost of equipment
                 equipmentCost = equipmentCost * 2;

			 }

          }
	      //Do work if it's not raining or the current method's is not affected by rain (weather sensitivity 0 or 1)
	      if(!rainday || (weatherSensitivity == 0) || (weatherSensitivity == 1) )
	      {
			  //Do work for each standard hour in the day
			  int i = 0;
			  while((i < stdHours) && (lusCompleted < lusTotal))
			  {
				  lusCompleted += productivity;
				  currentLaborCost += laborCost * productivity;
				  currentMaterialCost += materialsCost * productivity;
				  currentEquipmentCost += equipmentCost * productivity;
				  currentSubcontractorCost += subcontractorCost * productivity;
				  lusCompletedThisPeriod += productivity;
				  accumulatedCosts += laborCost * productivity;
				  accumulatedCosts += materialsCost * productivity;
				  accumulatedCosts += equipmentCost * productivity;
				  accumulatedCosts += subcontractorCost * productivity;
				  i++;
			  } //end while

			  //If there are still overtime days left, then work them
			  if(overtimeDays > 0)
			  {
				  i = 0;
				  overtimeDays--;
				  while((i < overtimeHours) && (lusCompleted < lusTotal) )
				  {
					  lusCompleted += productivity;
				      currentLaborCost += Math.round(laborCost * productivity * overtimeModifier);
				      currentMaterialCost += materialsCost * productivity;
				      currentEquipmentCost += equipmentCost * productivity;
				      currentSubcontractorCost += Math.round(subcontractorCost * productivity * overtimeModifier);
				      lusCompletedThisPeriod += productivity;
				      accumulatedCosts += Math.round(laborCost * productivity * overtimeModifier);
				      accumulatedCosts += materialsCost * productivity;
				      accumulatedCosts += equipmentCost * productivity;
				      accumulatedCosts += Math.round(subcontractorCost * productivity * overtimeModifier);
				  	  i++;
				  } // end while
			  } //end if

              //Check to make sure extra lus weren't completed.  This can happen because a full hour of lus is done on the final hour,
              //even if there weren't that many lus left to be completed

			  //Make sure the lus completed doesn't read higher than the amount possible
		      if(lusCompleted > lusTotal) lusCompleted = lusTotal;

			  //Make sure more lus weren't completed this period than were left to do
			  if(lusCompletedThisPeriod > lusTotal - lusCompletedBeforeUpdates) lusCompletedThisPeriod = lusTotal - lusCompletedBeforeUpdates;
		  } // end if
	  } //end if

  }
