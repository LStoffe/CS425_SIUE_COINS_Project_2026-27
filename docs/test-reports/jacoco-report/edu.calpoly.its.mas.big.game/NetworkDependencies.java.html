package edu.calpoly.its.mas.big.game;

import java.util.Vector;
import edu.calpoly.its.mas.big.game.Activity;

/**
 * Determines whether or not an activity is active (whether or not work can be performed on an activity)
 * based on the activity's network dependencies. The network dependencies for a particular job type
 * are set by the game's admin before the game begins.  Network dependencies for each activity consist of
 * a "start-to-start" (STS) activity and percent, and a "finish-to-finish" (FTF) activity and percent.
 *
 * An activity may or may not have a STS dependency.  If the activity does not have an STS dependency,
 * then work may begin on the activity immediately.  If the activity does have an STS dependency, then work
 * may not begin on that activity until the activity on which it depends is X% complete. For example, an
 * activity's STS activity=1, and STS percent=70.  This means that work may not begin on the activity until
 * activity #1 has reached 70% completion.
 *
 * Each activity may or may not have a "finish-to-finish" (FTF) dependency.  If the activity does not have a
 * FTF dependency, then work will continue on the activity non-stop until the activity is 100% complete.  If
 * the activity does have an FTF dependency, then once the activity has reached Y% completion, work on the activity
 * must be pause until the activity upon which it depends is 100% complete.  For example: an activity's
 * FTF activity=3 and its FTF percent=40.  This means that once this activity has reached 40% completion,
 * this activity must wait until activity #3 is 100% complete before any more work may be done on this activity.
 *
 * Note that NetworkDependencies.update(activities) method should be called at the beginning of the day before any
 * work is done on any of the activities. After the update method has updated the status of each of the activities,
 * work for each activity may then be calculated for the current day.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.6 $ $Date: 2002/04/12 19:06:03 $
 */
 public class NetworkDependencies
 {

  /**
   * Determines whether or not each activity should be set active (meaning that work may
   * be done on this activity) or inactive (work may not be done on this activity) based on
   * the amount of work that has already been performed on the activities, and the activity's
   * start-to-start and finish-to-finish dependencies.
   * @param activities A javax.util.Vector containing the current job's activities.
   */
  public void update(Vector activities)
  {
     // For each activity
     for(int i=0; i<9; i++)
     {

	    // Get the current activity
	    Activity activity = (Activity) activities.get(i);

        // If an activity does not exist for this job, the lusTotal will be zero.  For example, "Industrial Plant" jobs have
        // no "Basement" activity. Thus, the lusTotal for the "Basement" activity will be zero.
        // If the activity does not exist, make sure it is inactive, and skip further processing of it.
        if(activity.getLusTotal() == 0)
        {
			activity.setIsActive(false);
		}
        else if(activity.getLusTotal() != 0)
        {
           // Calculate the Percent Completion of this activity
	       float currentActivityPercentComplete = (activity.getLusCompleted() * 100 / activity.getLusTotal()) ;

           // If Activity is completed, mark it as inactive and skip further processing of it
           if(currentActivityPercentComplete >= 100)
           {
			   activity.setIsActive(false);
		   }
           // If Activity is not complete, determine whether or not it should be active based on its STS and FTF dependencies
	       else if(currentActivityPercentComplete < 100)
	       {
              // Perform STS logic only if activity has not yet started
		      if(currentActivityPercentComplete == 0)
	          {
				  // If activity has no STS dependency, set it active
				  if(activity.getStartToStartActivity() == 0)
				  {
					  activity.setIsActive(true);
				  }
				  else
				  {
					 // Set activity inactive just in case it was inadvertently created as active.
					 // (All activities except for the first activity should be created as inactive.)
					 // Then determine whether or not the activity should be made active by checking its STS dependency.
					 activity.setIsActive(false);
                     startToStartDependency(activities, i);
				  }
		      }

              // Perform FTF logic only if activity has an FTF dependency
              if(activity.getFinishToFinishActivity() != 0)
              {
		         finishToFinishDependency(activities, i);
		      }
	       }
	    }
     }
  }


   /**
    * Determines whether or not the current activity should be started. Note that the current activity's
    * "active" status is based on the status of its dependent activity at the beginning of the current day.
    * If at the beginning of the current day the dependent activity has progressed to or beyond the current
    * activity's startToStartPercent, the current activity will be set active. However, if the dependent activity
    * reaches the startToStartPercent because of the work performed on it this day, the current activity must wait
    * until the following day before it will be marked as "active."
    * @param activities A javax.util.Vector containing the current job's activities.
    * @param i An index into the "activities" Vector which specifieds the current activity. Note that this will be one less than the actual activity number.
    */
    private void startToStartDependency(Vector activities, int i)
    {
	    // Get the current activity
	    Activity activity = (Activity) activities.get(i);

	    // Get the STS dependent Activity number and percent
		int stsDependentActivityNumber = activity.getStartToStartActivity();
        int stsDependentActivityPercent = activity.getStartToStartPercent();

		// Get the STS dependent Activity itself - subtract 1 for the vector numbering system
		Activity dependentActivity = (Activity) activities.get(stsDependentActivityNumber - 1);

        // If the administrator mistakenly sets a STS dependency on a non-existant activity, allow this to happen.
        // This activity will always be set active.
        if(dependentActivity.getLusTotal() == 0)
        {
			activity.setIsActive(true);
		}
        else
        {

           // Determine the Percent Completion of the dependent Activity
           // MGet the amount done before this day's work was calculated in. The current activity must be set active or inactive
           // based on the amount of work done at the start of the day.
		   float stsDependentActivityActualPercentComplete = (dependentActivity.getLusCompleted() * 100)/ dependentActivity.getLusTotal() ;

           // Determine whether or not this activity can be started yet
           if(stsDependentActivityActualPercentComplete >= stsDependentActivityPercent)
           {
              activity.setIsActive(true);
		   }
	    }
    }

    /**
    * Determines whether or not the current activity should be stopped or restarted. Note that the current activity's
    * "active" status is based on the status of its dependent activity at the beginning of the current day. If the
    * current activity has progressed to finishToFinishPercent completion, the current activity will be set to inactive
    * until the dependent activity is 100% complete. When the dependent activity has reached 100% completion, the
    * current activity will again be set to active so that it work may resume on it on the next work day.
    * @param activities A javax.util.Vector containing the current job's activities.
    * @param i An index into the "activities" Vector which specifieds the current activity. Note that this will be one less than the actual activity number.
    */
    private void finishToFinishDependency(Vector activities, int i)
    {
	   // Get the current activity
	    Activity activity = (Activity) activities.get(i);

       // Get the FTF dependent Activity number and percent
       int ftfDependentActivityNumber = activity.getFinishToFinishActivity();
       int ftfPercent = activity.getFinishToFinishPercent();

       // Get the FTF dependent Activity itself- subtract 1 for the vector numbering system
       Activity dependentActivity = (Activity) activities.get(ftfDependentActivityNumber - 1);

       // If the administrator mistakenly sets a FTF dependency on a non-existant activity, allow this to happen.
       // This activity will always be set active.
       if(dependentActivity.getLusTotal() == 0)
       {
		   activity.setIsActive(true);
	   }
       else
       {
          // Calculate the percent completion of the dependent activity
          // The percent completion of the dependent activity is based on the amount of work done at the start of the day.
          // Once work is 100% complete on the dependent activity, the current activity will be marked as active so that work
          // may resume on the following work day.
          float ftfDependentActivityActualPercentComplete = (dependentActivity.getLusCompleted()  * 100) / dependentActivity.getLusTotal();

          // Temporarily stop the current activity if it has reached its finish-to-finish maximum and must wait for the dependent activity to complete
          float currentActivityPercentComplete = (activity.getLusCompleted() * 100 / activity.getLusTotal()) ;
          if((currentActivityPercentComplete >= (ftfPercent)) & (ftfDependentActivityActualPercentComplete < 100) & (activity.getIsActive()))
          {
             activity.setIsActive(false);
          }

         // Restart the current activity if the dependent activity is 100% complete and the current activity is not already active
          if( (ftfDependentActivityActualPercentComplete >= 100) && (!activity.getIsActive()))
          {
             activity.setIsActive(true);
	      }
       }

   }

}
