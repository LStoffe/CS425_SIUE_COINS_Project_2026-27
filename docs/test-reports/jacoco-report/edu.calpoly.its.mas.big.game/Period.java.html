package edu.calpoly.its.mas.big.game;

import java.util.Random;

/**
 * Period holds information about one period of one game of BIG.
 *
 *
 * @version $Revision: 1.10 $ $Date: 2001/11/26 21:47:37 $
 */
public class Period
{

	private Random r;

	private int periodNumber;
	private int workdaysFirstMonth;
    private int workdaysSecondMonth;
	private int raindaysFirstMonth;
    private int raindaysSecondMonth;
	private int laborAvailability;
	private int materialsCostIndex;
	private float rainfall;
	private int temperature;
    private int numberOfNewBuildingJobs;
    private int numberOfNewHeavyJobs;

    /**
     * Constructor for retrieving a Period from the database.
     * @param periodNumber The number of this period
     * @param laborAvailability The labor availability index value for this period
     * @param materialsCostIndex The materials cost index value for this period
     * @param rainfall The inches of rainfall this period
     * @param temperature The temperature this period
     * @param numberOfNewBuildingJobs The number of new Building jobs that will be generated this period
     * @param numberOfNewHeavyJobs The number of new Heavy jobs that will be generated this period
     * @param workdaysFirstMonth The number of workdays in the first month of this period
     * @param workdaysSecondMonth The number of workdays in the second month of this period
     * @param raindaysFirstMonth The number of raindays in the first month of this period
     * @param raindaysSecondMonth The number of raindays in the second month of this period
     */
	public Period(int periodNumber,
				  int laborAvailability,
				  int materialsCostIndex,
				  float rainfall,
				  int temperature,
				  int numberOfNewBuildingJobs,
				  int numberOfNewHeavyJobs,
				  int workdaysFirstMonth,
				  int workdaysSecondMonth,
				  int raindaysFirstMonth,
				  int raindaysSecondMonth)
	{
		this.periodNumber = periodNumber;
		this.laborAvailability = laborAvailability;
		this.materialsCostIndex = materialsCostIndex;
		this.rainfall = rainfall;
		this.temperature = temperature;
		this.numberOfNewBuildingJobs = numberOfNewBuildingJobs;
		this.numberOfNewHeavyJobs = numberOfNewHeavyJobs;
        this.workdaysFirstMonth = workdaysFirstMonth;
        this.workdaysSecondMonth = workdaysSecondMonth;
        this.raindaysFirstMonth = raindaysFirstMonth;
        this.raindaysSecondMonth = raindaysSecondMonth;
	}

    /**
     * Constructor for creating a new period from statistical information.
     * @param numberJobsMean The mean of the possible numbers of jobs to be created this period
     * @param numberJobsStddev The standard deviation of the possible numbers of jobs to be created this period
     * @param numberJobsMin The minimum possible value for number of jobs that will be created this period
     * @param numberJobsMax The maximum possible value for number of jobs that will be created this period
     * @param temperatureMean The mean of the possible temperatures this period
     * @param temperatureStddev The standard deviation of the possible temperature values this period
     * @param temperatureMin The minimum possible value for temperature this period
     * @param temperatureMax The maximum possible value for temperature this period
     * @param rainfallMean The mean of the possible rainfall amounts this period
     * @param rainfallStddev The standard deviation of the possible rainfall amounts this period
     * @param rainfallMin The minimum possible value for rainfall this period
     * @param rainfallMax The maximum possible value for rainfall this period
     * @param laborAvailabilityMean The mean of the possible labor availabilities this period
     * @param laborAvailabilityStddev The standard deviation of the possible labor availabilities this period
     * @param laborAvailabilityMin The minimum possible value for labor availability this period
     * @param laborAvailabilityMax The maximum possible value for labor availability this period
     * @param materialsCostMean The mean of the possible values for materials cost index
     * @param materialsCostStddev The standard deviation of the possible values for materials cost index
     * @param materialsCostMin The minimum possible value for the materials cost index
     * @param materialsCostMax The maximum possible value for the materials cost index
     * @param baseWorkdaysPerMonth The number of workdays each month before any adjustments
     * @param periodNumber The number of this period
     */
    public Period(//Number of new jobs per period statistics
		          int numberJobsMean,
    			  int numberJobsStddev,
    			  int numberJobsMin,
    			  int numberJobsMax,
    			  int numberHCJobsMin,
    			  int numberHCJobsMax,
    			  //Temperature statistics
    			  int temperatureMean,
    			  int temperatureStddev,
    			  int temperatureMin,
    			  int temperatureMax,
    			  //Rainfall statistics
    			  int rainfallMean,
    			  int rainfallStddev,
    			  int rainfallMin,
    			  int rainfallMax,
    			  //Labor availability statistics
    			  int laborAvailabilityMean,
    			  int laborAvailabilityStddev,
    			  int laborAvailabilityMin,
    			  int laborAvailabilityMax,
    			  //Materials Cost Index statistics
    			  int materialsCostMean,
    			  int materialsCostStddev,
    			  int materialsCostMin,
    			  int materialsCostMax,
    			  //Default number of days each month
    			  int baseWorkdaysPerMonth,
    			  //Period number
    			  int periodNumber)
    {
		this.periodNumber = periodNumber;
		workdaysFirstMonth = baseWorkdaysPerMonth;
		workdaysSecondMonth = baseWorkdaysPerMonth;

		r = new Random();
		numberOfNewBuildingJobs = Math.round(calculateValueFromStatistics(numberJobsMean, numberJobsStddev, numberJobsMin, numberJobsMax));
		numberOfNewHeavyJobs = Math.round(calculateValueFromStatistics(numberJobsMean, numberJobsStddev, numberHCJobsMin, numberHCJobsMax));
		temperature = Math.round(calculateValueFromStatistics(temperatureMean, temperatureStddev, temperatureMin, temperatureMax));
		rainfall = calculateValueFromStatistics(rainfallMean, rainfallStddev, rainfallMin, rainfallMax);
		laborAvailability = Math.round(calculateValueFromStatistics(laborAvailabilityMean, laborAvailabilityStddev, laborAvailabilityMin, laborAvailabilityMax));
        materialsCostIndex = Math.round(calculateValueFromStatistics(materialsCostMean, materialsCostStddev, materialsCostMin, materialsCostMax));

       	calculateRaindays();
        calculateLaborAdjustments();
	}


	//Calculates a value from the mean, standard deviation, minimum and maximum statistics for the value and
	//a random number from the normal distribution
	private float calculateValueFromStatistics(int mean, int stddev, int min, int max)
	{
		float value;
		float normal = (float) r.nextGaussian();
		value = (normal * stddev) + mean;
		if (value > max)
		    value = max;
		if (value < min)
		    value = min;

		return value;
	}


    //Calculates the number of raindays each month.
    //If rainfall < 2 there are 0 raindays both months
	//If  2 <= rainfall <= 3 there is 1 rainday the first month and 2 the second month
    //If rainfall > 3 there are 3 raindays each month
    private void calculateRaindays()
    {
        if (rainfall < 2)
        {
            raindaysFirstMonth = 0;
            raindaysSecondMonth = 0;
        }
        else if (rainfall <= 3)
        {
            raindaysFirstMonth = 1;
            raindaysSecondMonth = 2;
        }
        else
        {
            raindaysFirstMonth = 3;
			raindaysSecondMonth = 3;
        }

	}

    private void calculateLaborAdjustments()
    {
		if(laborAvailability <= 95)
		{
			workdaysFirstMonth = workdaysFirstMonth - 3;
			workdaysSecondMonth = workdaysSecondMonth - 3;
		}
		else if (laborAvailability <= 99)
		{
			workdaysFirstMonth = workdaysFirstMonth - 1;
			workdaysSecondMonth = workdaysSecondMonth - 2;
		}
	}

    /**
     * Returns the percentage to adjust productivities by due to the temperature this period.
     * Temperature affects productivies in the follow ways: <br>
     * <ul>
     * <li>Temperature <= 14       : -30% productivity</li>
     * <li>15 <= Temperature <= 23 : -20% productivity</li>
     * <li>24 <= Temperature <= 32 : -15% productivity</li>
     * <li>33 <= Temperature <= 41 : -10% productivity</li>
     * <li>42 <= Temperature <= 59 :  +0% productivity</li>
     * <li>60 <= Temperature <= 72 :  +5% productivity</li>
     * <li>73 <= Temperature <= 80 :  -5% productivity</li>
     * <li>81 <= Temperature <= 90 : -10% productivity</li>
     * <li>91 <= Temperature       : -20% productivity</li>
     * </ul>
     *
	 * @return int representing the percent
     */
    public int getProductivityAdjustmentDueToTemperature()
    {
		int productivityAdjustmentDueToTemperature;

		if (temperature <= 14)
		{
			productivityAdjustmentDueToTemperature = -30;
		}
		else if (temperature <= 23)
		{
			productivityAdjustmentDueToTemperature = -20;
		}
		else if(temperature <= 32)
		{
			productivityAdjustmentDueToTemperature = -15;
		}
		else if(temperature <= 41)
		{
			productivityAdjustmentDueToTemperature = -10;
		}
		else if(temperature <= 59)
		{
			productivityAdjustmentDueToTemperature = 0;
		}
		else if(temperature <= 72)
		{
			productivityAdjustmentDueToTemperature = 5;
		}
		else if(temperature <= 80)
		{
			productivityAdjustmentDueToTemperature = -5;
		}
		else if(temperature <= 90)
		{
			productivityAdjustmentDueToTemperature = -10;
		}
		else
		{
			productivityAdjustmentDueToTemperature = -20;
		}
        return productivityAdjustmentDueToTemperature;
    }

    /**
     * Returns the percentage to adjust labor cost by due to the effects of the labor availability index for this period.
     * The percent that labor cost is adjusted by is 100 - Labor availability Index
     *
	 * @return int representing the percent
     */
    public int getLaborCostAdjustment()
    {
		return 100 - laborAvailability;
	}


	/**
	 * Returns the percentage to adjust materials cost by due to the effects of the materials cost index for this period.
	 * The percent that materials cost is adjusted by is the Materials Cost Index minus 100;
	 * @return int representing the percent
	 */
	public int getMaterialsCostAdjustment()
	{
		return materialsCostIndex - 100;
	}
