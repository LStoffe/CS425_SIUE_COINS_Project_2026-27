package edu.calpoly.its.mas.big.game;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.game.NumberConverter;

public class AutoAwardJobsForm extends ActionForm
{
  //These values will be the initial values appearing in the form inside autoAddJobsInput.jsp
  //Upon validation failure, the unacceptable values will appear in the form.
  //Using value="-1" inside the form tag in autoAddJobsInput.jsp will cause that number to ALWAYS appear in the text box upon view regeneration
  private String percentMarkup = "-1";
  private String jobType = "0";
  private String jobSize = "0";
  private String numbOfJobs = "1";

  public void resetForm()
  {
	 this.percentMarkup = "-1";
     this.jobType = "0";
     this.jobSize = "0";
     this.numbOfJobs = "1";
  }

  public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
  {
    ActionErrors errors2 = new ActionErrors();

    //Check to see if percentMarkup is an acceptable value
    if ( NumberConverter.intValue(percentMarkup) < -1 || NumberConverter.intValue(percentMarkup) > 100 )
    {
        errors2.add("percentMarkup", new ActionMessage("error.addJobs.percentMarkup"));
    }

    //Check to see if jobType is an acceptable value
    if ( NumberConverter.intValue(jobType) < 0 || NumberConverter.intValue(jobType) > 5 )
    {
        errors2.add("jobType", new ActionMessage("error.addJobs.jobType"));
    }

    //Check to see if jobSize is an acceptable value
    if ( NumberConverter.intValue(jobSize) != 0 )
    {
       if ( NumberConverter.intValue(jobSize) < 10000 || NumberConverter.intValue(jobSize) > 99000 )
       {
          errors2.add("jobSize", new ActionMessage("error.addJobs.jobSize"));
       }
    }

    //Check to see if numbOfJobs is an acceptable value
    if ( NumberConverter.intValue(numbOfJobs) < 1 || NumberConverter.intValue(numbOfJobs) > 10 )
    {
        errors2.add("numbOfJobs", new ActionMessage("error.addJobs.numbOfJobs"));
    }

    return errors2;
  }

  public String getPercentMarkup()
  {
    return (this.percentMarkup);
  }

  public void setPercentMarkup(String percentMarkup)
  {
    this.percentMarkup = percentMarkup;
  }

  public String getJobType()
  {
    return (this.jobType);
  }

  public void setJobType(String jobType)
  {
    this.jobType = jobType;
  }

  public String getJobSize()
  {
    return (this.jobSize);
  }

  public void setJobSize(String jobSize)
  {
    this.jobSize = jobSize;
  }

  public String getNumbOfJobs()
  {
    return (this.numbOfJobs);
  }

  public void setNumbOfJobs(String numbOfJobs)
  {
    this.numbOfJobs = numbOfJobs;
  }

}
