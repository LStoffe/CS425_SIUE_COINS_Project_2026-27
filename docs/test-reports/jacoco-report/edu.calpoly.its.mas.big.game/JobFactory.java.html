/**
 * Creates new jobs for companies to bid on.
 *
 * New jobs are randomly given one of the 5 jobs types.  The job type determines which set of
 * job size statistics game parameters are used for the job size.  The job size is calculated by taking a random normal
 * value and modifying it with the job size statistics.  For each of the 9 activities of the job a random method is chosen.  This
 * set of 9 methods is the 'ideal' set of methods to complete this job.  Using these 9 methods and the network dependencies for this
 * job type <code>JobFactory</code> determines how long this job should take to complete.  This information comes in the form of the required
 * completion date of the job.  <code>JobFactory</code> also determines the direct cost to finish the job on that completion date using the 'ideal'
 * methods.  Liquidated damaged are assigned to all jobs which will be a random percentage of the direct
 * cost of the job, within the range specified by the liquidated damages game parameters, rounded to 1 significant digit.  Maximum liquidated damages are $100,000 a day.
 * All jobs must have liquidated damages otherwise companies would have no reason to complete the job on time.
 * <br><br>
 * The jobs created by the factory will contain 9 Activity objects for the current period.  These objects must be stored to the database seperately.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.32 $ $Date: 2002/05/02 23:59:24 $
 */

	/**
	 * Creates a new <code>Job</code> objects.

	public Job createJob(int jobID)
	{
        currentDate = periodDate;
        estimatedWorkdays = 0;
        directCost = 0;
        requiredCompletionDate = null;
        jobSize = 0;
        jobType = -1;
        activityLUS = null;
        currentMonth = null;
		activities = null;

        //If Admin specified values when "Automatically Add One Job To Every Company," use those instead of the normal methods.
		if (adminJobType == 0)
		{
		    jobType = getJobType();
	    }
	    else
	    {
		   if (jobFamily == 1) jobType = adminJobType - 1;
		   if (jobFamily == 2) jobType = adminJobType + 4;
	    }

		if (adminJobSize == 0)
		{
           jobSize = getJobSize(jobType);
	    }
	    else
	    {
		   jobSize = adminJobSize;
		}

		//Calculates how many LUS are in each activity based on percent takeoff parameters
		calculateActivitesLUS();

        //The methods used to make an estimate of the job length and cost are chosen as follows...
        //A random number between 0 and 3 is chosen.  This is the baseline method number for this job.  Note that
        //the baseline method can never be 4, the highest possible method number.  If the administrator
        //makes method 5 (the 4th method using zero based numbering) the quickest and most expensive method, then jobs will
        //never be created that require the fastest methods in order to be completed on time.
        //1 is randomly added or substracted from the baseline method number to create some random variation
        //in job cost and required completetion date.
        int[] activityMethods = new int[9];

        int methodNum = r.nextInt(3) + 1;  //baseline method number
        float methodAverage = 0;
        for(int i = 0; i < 9; i++)
        {
			activityMethods[i] = methodNum + r.nextInt(3) - 1; //Add -1, 0, or 1 to the baseline method number
															   //for each activity to get the method number for that activity

		    //Neither of these should actually ever happen but its simple to check for them:
		    //The method number can never be less than 0
		    if(activityMethods[i] < 0) activityMethods[i] = 0;
		    //The method number can never be greater than 4
			if(activityMethods[i] > 4) activityMethods[i] = 4;
			methodAverage += activityMethods[i];
			//System.out.println(activityMethods[i] + 1);

			if (game != null) game.recordJobMethod(jobID, i+1, activityMethods[i]+1);
		}

        methodAverage = methodAverage / 9;  //methodAverage isn't currently used for anything but could be useful
        								    //in future modifications to BIG, especially in the area of liquidated damges
        setupCalendar();

        activities = createActivities(activityMethods);

//OPTIONAL METHOD SCALING FACTOR
  //The method scaling factor is a number for each job that is multipled by the productivity
  //of any method that is used on any activity of the job.  The number is set by equation in the line of code
  //immediately following this comment.  This equation can be set to anything involving
  //job sizes or to 1 to disable method scaling
  //factor.  The purpose of the method scaling factor is to increase work done on jobs with large lu sizes
  //so that they finish in a reasonable amount of time.  Setting method scaling factor will solve
  //problems where big jobs take too long to complete while small jobs finish too quickly.
  //The suggested equation (commented out below) works well.
  //If this is set companies must be informed on the productivity report
        float methodScalingFactor = 1f; //(float) ( Math.sqrt(jobSize) / Math.sqrt(getLowestJobSize()) );
	    //System.out.println("Method Scaling Factor: " + methodScalingFactor);

        for(int i = 0; i <9; i++)
        {
			activities[i].setMethodScalingFactor(methodScalingFactor);
		}
//END METHOD SCALING FACTOR

        //runs through the 'ideal' activities -- sets the completion date, estimated number of workdays, and direct cost.
        walkThroughActivities();

        //Now that the direct cost and req completion date has been calculated reset all the activities
        //so they start off with no work done
        for(int i = 0; i < 9; i++)
            activityMethods[i] = 0;

        //create the activities for this job
        activities = null;
        activities = createActivities(activityMethods);

        int damagesPerDay = calculateLiquidatedDamages(methodAverage);

        //System.out.println("Direct Cost: " + directCost);
        //System.out.println("Estimated Workdays: " + estimatedWorkdays);
 	    SimpleDateFormat formatter = new SimpleDateFormat("MMMMM dd, yyyy");
 	    String dateString = formatter.format(requiredCompletionDate);
 	    //System.out.println("Required Completion Date: " + dateString);
 	    //System.out.println("Liquidated Damages per Day: " + damagesPerDay);

        //Randomly choose a retention policy
        int retentionPolicy = r.nextInt(5) + 1;

		short typeModifier = -1;
		if (jobFamily == 1) typeModifier = Job.PUBLIC;
		if (jobFamily == 2) typeModifier = Job.HEAVY;

    	Job job = new Job(jobID,
						  damagesPerDay,
						  null,
						  requiredCompletionDate,
						  jobSize,
						  0, //Leave percent retention and percent of job empty.  A retentionID
						  0, //will be assigned to the job and they will be populated when saved to the database
						  directCost,
						  0, //No amount retained yet
						  jobSize,  //amount remaining on job is equal to job size
						  0, //Not completed yet
						  currentPeriod,
						  true,  //Active
						  false, //not Auto
						  jobType + 1,
						  estimatedWorkdays,
						  methodScalingFactor,
						  0,  //No change orders yet
						  0,  // "
						  0,  //overhead doesn't factor into job generation
						  0,  // "
						  0,
						  0,
						  new Short(typeModifier)
						  );

		job.setRetentionID(retentionPolicy);
		//System.out.println("JobID: " + job.getJobID());
		//System.out.println("Job Size: " + job.getJobSize());
		//System.out.println("Job type: " + job.getJobType());
		job.setActivities(activities);
		return job;
	}

    //Returns the size of this job
	private int getJobSize(int jobType)
	{
		int size;

		//jobSizeParams will contain values for a heavy job if this is a heavy job.  This Hashtable is created
		//in ExpandedGame.java and sent here as part of JobFactory's constructor.  See Line # 530 in that file.
		if (jobFamily == 2) jobType = jobType - 5;  // Hashtable contains values 0 through 4

		int mean =   ((Integer) jobSizeParams[jobType].get("Median")).intValue();
		int stddev = ((Integer) jobSizeParams[jobType].get("StdDev")).intValue();
		int min =    ((Integer) jobSizeParams[jobType].get("Min")).intValue();
		int max =    ((Integer) jobSizeParams[jobType].get("Max")).intValue();

		float normal = (float) r.nextGaussian();
		size = Math.round((normal * stddev) + mean);
		if (size > max)
		    size = max;
		if (size < min)
		    size = min;

		return size;
	}

    //Returns the lowest job size median
	private int getLowestJobSize()
    {
        int lowestJobSize = ((Integer) jobSizeParams[0].get("Median")).intValue();
        for(int i = 1; i < 5; i++)
        {
            int temp = ((Integer) jobSizeParams[i].get("Median")).intValue();
            if (temp < lowestJobSize)
               lowestJobSize = temp;
        }
        return lowestJobSize;
    }


    //  For Building jobs, returns an int between 0 and 4 that will be the code for a type of job.
    //  The probability of each type being returned is as follows:
    //     0 - 50%
    //     1 - 15%
    //     2 - 15%
    //     3 - 10%
    //     4 - 10%
    //  If this is a Heavy job, it will get values between 5 and 9.
    //  Similar logic for probability of a certain type being returned.

    private int getJobType()
    {
		int spin = r.nextInt(100);

        if (jobFamily == 1)
        {
  		   if (spin < 50)  //spin < 50
		   {
		   	   return 0;
		   }
		   else if(spin < 65)  //spin < 65 && spin >= 50
		   {
			   return 1;
		   }
		   else if(spin < 80)  //spin < 80 && spin >= 65
		   {
			   return 2;
		   }
		   else if(spin < 90)  //spin < 90 && spin >= 80
		   {
			   return 3;
		   }
		   else
		   {
			   return 4;
		   }
        }

        else  // (jobFamily == 2)
        {
  		   if (spin < 30)      //spin < 30  --> 30%
		   {
		   	   return 5;
		   }
		   else if(spin < 50)  //spin < 50 && spin >= 30 --> 20%
		   {
			   return 6;
		   }
		   else if(spin < 65)  //spin < 65 && spin >= 50 --> 15%
		   {
			   return 7;
		   }
		   else if(spin < 85)  //spin < 85 && spin >= 65 --> 20%
		   {
			   return 8;
		   }
		   else                // spin >= 85 --> 15%
		   {
			   return 9;
		   }
        }
	}

    //Calculates how many lus are in each activity based on percent takeoff parameters
    //Postcondition: The activityLUS array now contains the number of lus for each activity where the array index+1 equals the activity number
    //Postcondition: The sum of all values in the activityLUS array is equal to jobSize
    private void calculateActivitesLUS()
    {
        //Get the percent takeoff for each activity
	    activityLUS = new int[9];
	    int sumSizes = 0;
		int thisJobType = jobType;

        if (jobFamily == 2) thisJobType = thisJobType - 5;  // the array contains only values 0 through 4 in the first element

        for(int i = 0; i < 9; i++)
        {
	        activityLUS[i] = (int) Math.round(jobSize * percentTakeoffs[thisJobType][i] / 100f);
	        sumSizes += activityLUS[i];
		}

		//Replace the job size with the sum of the sizes of the activities to remove the rounding error
		jobSize = sumSizes;
	}

    //Creates each activity of this job
    private Activity[] createActivities(int[] activityMethods)
    {
        Activity[] activities = new Activity[9];
        for(int i = 0; i<9; i++)
        {
			boolean active = false;
			if (i == 0) active = true;

            //activity type needs to be 10-18 for Heavy Jobs
			int tempIndex = i;
			if (jobFamily == 2) tempIndex = tempIndex + 9;

			activities[i] = new Activity(tempIndex + 1, //Activity type, 1 greater than the index
			                             activityLUS[i], //Total number of lu's in this activity
			                             0,   //no lu's completed yet
			                             0,   //no labor cost incurred yet
			                             0,   //no materials cost incurred yet
			                             0,   //no equipment cost incurred yet
			                             0,   //no sub cost incurred yet
			                             active,  //set activity active if it is the first activity
			                             0,   //no lu's completed yet
			                             0,   //no costs yet
			                             currentPeriod - 1, //store these as last period so work can begin on them this period
			                             0,   //Current method used will need to be set before work is done
			                             0,   //don't work any overtime days
			                             stdHours, //standard hours in a day
			                             overtimeHours   //overtime hours in a day
			                             );

            //Changes the effect of uncertainty each time a method is used
            methods[i][activityMethods[i]].reRollCosts();

			//assigns methods to this activity
			activities[i].setMethod(methods[i][activityMethods[i]]);

			int stsActivity, stsPercent, ftfActivity, ftfPercent;

		    int thisJobType = jobType;
            if (jobFamily == 2) thisJobType = thisJobType - 5;  // the array contains only values 0 through 4 in the first element

			stsActivity = ((Integer) networkDependencies[thisJobType][i].get("STSActivity")).intValue();
			stsPercent = ((Integer) networkDependencies[thisJobType][i].get("STSPercent")).intValue();
			ftfActivity = ((Integer) networkDependencies[thisJobType][i].get("FTFActivity")).intValue();
			ftfPercent = ((Integer) networkDependencies[thisJobType][i].get("FTFPercent")).intValue();

			activities[i].setStartToStartActivity(stsActivity);
			activities[i].setStartToStartPercent(stsPercent);
			activities[i].setFinishToFinishActivity(ftfActivity);
			activities[i].setFinishToFinishPercent(ftfPercent);

		}
		return activities;
	}

	//Runs through the activities doing work on them as if the job were really being worked on
	//Sets the completion date, estimated number of workdays, and direct cost.
	//It is assumed that no days are raindays and that temperature, materials cost index, and labor availability
	//have no effect.
	private void walkThroughActivities()
	{
		boolean complete = false;
		Vector activitiesVector = new Vector();
		NetworkDependencies networkDependencies = new NetworkDependencies();

		//necessary for compatability with Job.java
		ExpandedCompany dummyCompany = new ExpandedCompany();
		PrintWriter out = null;

		for(int i = 0; i < 9; i++)
		     activitiesVector.add(activities[i]);

	    while(!complete)
	    {
			if(!currentMonth.daysLeft())
			{
				setupCalendar();
			}

			while(currentMonth.daysLeft() && !complete)
			{
				if(currentMonth.getWorkday())
				{
					estimatedWorkdays++;
					networkDependencies.update(activitiesVector);
					for(int i = 0; i < 9; i++)
					{
					    activities[i].update(false, 0, 0, dummyCompany, game, out);
					/*  System.out.println(i+1);
					    System.out.println(activities[i].getLusTotal());
					    System.out.println(activities[i].getLusCompleted());  */
					}

				}
				if(allActivitiesComplete())
				{
				    requiredCompletionDate = currentMonth.getDate();
				    complete = true;
				}
				currentMonth.advanceDay();
			}
		}

		//Sum up the activity direct costs into the job direct cost
		for(int i = 0; i < 9; i++)
		{
			directCost += activities[i].getCurrentLaborCost();
			//System.out.println("lc: " + activities[i].getCurrentLaborCost());
			directCost += activities[i].getCurrentMaterialCost();
			//System.out.println("mc: " + activities[i].getCurrentMaterialCost());
			directCost += activities[i].getCurrentSubcontractorCost();
			//System.out.println("sc: " + activities[i].getCurrentSubcontractorCost());
			directCost += activities[i].getCurrentEquipmentCost();
			//System.out.println("sc: " + activities[i].getCurrentEquipmentCost());
		}

	}

	//Return true if all the activities are complete
	private boolean allActivitiesComplete()
	{
		boolean allComplete = true;
		for (int i = 0; i < 9; i++)
		{
			if(activities[i].getLusCompleted() < activities[i].getLusTotal())
			    allComplete = false;
		}
		return allComplete;
	}

    //Set up the calendar
    private void setupCalendar()
    {
	    Calendar calendar = new GregorianCalendar();
	    calendar.setTime(currentDate);
	    calendar.add(Calendar.MONTH, 1);
	    calendar.set(Calendar.DAY_OF_MONTH, 1);
	    currentDate = calendar.getTime();
	    currentMonth = new ScheduledMonth(currentDate, workdaysPerMonth, 0);
    }

    //Calculates what the liquidated damages should be on this job.
    //If the fasterJobsIncrease game parameter is set to true,
    //the amount of liquidated damages will be multiplied
    //by the method average.
    //If the largerJobsIncrease game parameter is set to true,
    //the amount of liquidated damages will be multiplied
    //by job size of the job divided by the mean job size for that job type.
    private int calculateLiquidatedDamages(float methodAverage)
    {
		//Get the liquidated damages parameters
		float median = ((Float) liquidatedDamagesParameters.get("Median")).floatValue();
		float stddev = ((Float) liquidatedDamagesParameters.get("StdDev")).floatValue();
		float min = ((Float) liquidatedDamagesParameters.get("Min")).floatValue();
		float max = ((Float) liquidatedDamagesParameters.get("Max")).floatValue();
		int percentJobs = ((Integer) liquidatedDamagesParameters.get("PercentJobs")).intValue();
		boolean fasterJobsIncrease = ((Boolean) liquidatedDamagesParameters.get("FasterJobsIncrease")).booleanValue();
		boolean largerJobsIncrease = ((Boolean) liquidatedDamagesParameters.get("LargerJobsIncrease")).booleanValue();

		int damages = 0;

        int chance = r.nextInt(100);  //number less than 100
        if( chance < percentJobs ) //if chance is less than percentJobs there will be liquidated damages for this job
		{

			float normal = (float) r.nextGaussian();
			float damagePercent = ((normal * stddev) + median);

			damages = Math.round((damagePercent / 100 ) * directCost);

			if(fasterJobsIncrease)
			{
				damages = Math.round(((float) damages) * (methodAverage / 3));
			}

			if(largerJobsIncrease)
			{
		        int thisJobType = jobType;

                if (jobFamily == 2) thisJobType = thisJobType - 5;  // the array contains only values 0 through 4

				float sizeAdjustment = ((float) jobSize) /  ((float) ((Integer) jobSizeParams[thisJobType].get("Median")).intValue());
				damages = Math.round(((float) damages) * sizeAdjustment);
			}


        	if (damages < 100)
        	    damages = 0;
        	else if (damages < 1000)
				damages = ((int) Math.round(damages / 100f) ) * 100;
			else if (damages < 10000)
			    damages = ((int) Math.round(damages / 1000f) ) * 1000;
			else if (damages < 100000)
			    damages = ((int) Math.round(damages / 10000f) ) * 10000;
			else
			    damages = 100000;

		}

		return damages;
	}

	//Sorts the method parameters by productivity.  This can be called to make the productivities ascend with the method numbers
	private void sortMethods()
	{
		for(int i = 0; i < 9; i++)
		{
			for(int j = 0; j <= 4; j++)
			{
				for(int k = 0; k < 4; k++)
				{
					if(methods[i][k].getProductivity() > methods[i][k+1].getProductivity())
					{
						Method temp = methods[i][k];
						methods[i][k] = methods[i][k+1];
						methods[i][k+1] = temp;
					}
				}
			}
		}
	}
