package edu.calpoly.its.mas.big.game;

import java.util.Random;

/**
 * Contains information about a method of doing an activity
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.18 $ $Date: 2002/03/05 19:19:27 $
 */
public class Method
{

  private String methodName;
  private int methodNumber; //1,2,3,4 or 5
  private float productivity;
  private float laborCost;
  private float materialsCost;
  private float equipmentCost;
  private float subcontractorCost;
  private int uncertaintyRange;
  private int weatherSensitivity;
  private boolean subcontracted = false;

  private float adjustmentAmount;

  private int gameID;

  private float uncertaintyPercent;
  private float laborCostAdjustmentPercent;
  private float materialsCostAdjustmentPercent;

  Random random;

  public Method(int methodNumber, String methodName, int productivity, int laborCost, int materialsCost, int equipmentCost, int subcontractorCost, int uncertaintyRange, int weatherSensitivity, boolean subcontracted)
  {
	  this.methodNumber = methodNumber;
	  this.methodName = methodName;
	  this.productivity = productivity;
	  this.laborCost = laborCost;
	  this.materialsCost = materialsCost;
	  this.equipmentCost = equipmentCost;
	  this.subcontractorCost = subcontractorCost;
	  this.uncertaintyRange = uncertaintyRange;
	  this.weatherSensitivity = weatherSensitivity;
 	  this.subcontracted = subcontracted;

	  random = new Random();

	  reRollCosts();
  }

  /**
   * Adjusts labor, materials and equipment costs by some percentage from the uncertainty range.
   * Each time this method is called the base labor, materials and equipment costs will be adjusted by a different amount.
   */
  public void reRollCosts()
  {

	  uncertaintyPercent = (uncertaintyRange * (2 * random.nextFloat() - 0.5f)) / 100f;
  }


  /**
   * Adjusts productivity by the given percentage if this method is sensitive to temperature.
   * If this method is not sensitive to temperature a call to this method will do nothing.
   */
  public void setProductivityAdjustment(int adjustmentPercent)
  {
	  //If this method is affected by temperature then add this amount, otherwise ignore it
	  if((weatherSensitivity == 1) || (weatherSensitivity == 3)) //Weather sensitivites 1 & 3 imply sensitivity to temperature
	  {
	      float adjustmentAmount =  productivity * (adjustmentPercent / 100f);
	      productivity += adjustmentAmount;
  	  }
  }

  /**
   * Increases (or decreases) this method's labor cost by adjustmentPercent.
   */
  public void adjustLaborCost(int adjustmentPercent)
  {
	  laborCostAdjustmentPercent = adjustmentPercent;
  }

  /**
   * Increases (or decreases) this method's materials cost by adjustmentPercent.
   */
  public void adjustMaterialsCost(int adjustmentPercent)
  {
	  materialsCostAdjustmentPercent = adjustmentPercent;
  }

  /**
   * Get the productivity of this method
   */
  public int getProductivity()
  {
      return (int) Math.round(productivity);
  }


  /**
   * Get the labor cost per hour of this method taking the uncertainty range and adjustments into account.
   */
  public int getLaborCost()
  {
	  //Adjust for period effects
	  float adjustmentAmount = laborCost * (laborCostAdjustmentPercent / 100f);

	  //Adjust for uncertainty
	  adjustmentAmount = laborCost * uncertaintyPercent + adjustmentAmount;
      return (int) Math.round(laborCost + adjustmentAmount);
  }

  /**
   * Get the materials cost per hour of this method taking the uncertainty range and adjustments into account.
   */
  public int getMaterialsCost()
  {
	  //Adjust for period effects
	  float adjustmentAmount = materialsCost * (materialsCostAdjustmentPercent / 100f);

	  //Adjust for uncertainty
	  adjustmentAmount = materialsCost * uncertaintyPercent + adjustmentAmount;
      return (int) Math.round(materialsCost + adjustmentAmount);
  }

  /**
   * Get the equipment cost per hour of this method taking the uncertainty range and adjustments into account.
   * Use the same "materialsCostAdjustmentPercent" as for MaterialsCost.
   */
  public int getEquipmentCost()
  {
	  //Adjust for period effects
	  float adjustmentAmount = equipmentCost * (materialsCostAdjustmentPercent / 100f);

	  //Adjust for uncertainty
	  adjustmentAmount = equipmentCost * uncertaintyPercent + adjustmentAmount;
      return (int) Math.round(equipmentCost + adjustmentAmount);
  }

  /**
   * Get the subcontractor cost per hour of this method.  there is no uncertainty using subcontracted
   * methods.
   */
  public int getSubcontractorCost()
  {
      return (int) Math.round(subcontractorCost);
  }


  /**
   * Returns true if this methods is subcontracted, false o/w
   */
  public boolean subcontracted()
  {
	  return subcontracted;
  }

  //**********************************  Simple Set and Get Methods  ***********************************

  public int getMethodNumber()
  {
	  return methodNumber;
  }

  public void setMethodNumber(int methodNumber)
  {
	  this.methodNumber = methodNumber;
  }

  public String getMethodName()
  {
      return methodName;
  }

  public void setMethodName(String methodName)
  {
      this.methodName = methodName;
  }

  public int getWeatherSensitivity()
  {
      return weatherSensitivity;
  }

  public int getGameID()
  {
      return gameID;
  }

  public void setGameID(int gameID)
  {
      this.gameID = gameID;
  }

  public int getUncertaintyRange()
  {
	  return uncertaintyRange;
  }

  public void setUncertaintyRange(int uncertaintyRange)
  {
	  this.uncertaintyRange = uncertaintyRange;
  }


  /**
   * Get the labor cost per hour of this method
   */
  public int getBaseLaborCost()
  {
      return (int) Math.round(laborCost);
  }

  /**
   * Get the materials cost per hour of this method
   */
  public int getBaseMaterialsCost()
  {
      return (int) Math.round(materialsCost);
  }

  /**
   * Get the equipment cost per hour of this method
   */
  public int getBaseEquipmentCost()
  {
      return (int) Math.round(equipmentCost);
  }


  //Unit Test
  public static void main(String args[])
  {
	  int prod = 11;
	  int lab = 20;
	  int mat = 15;
	  int sub = 0;
	  int uncertainty = 10;
	  int weather = (new java.util.Random()).nextInt(4);
	  boolean subd = false;
	  System.out.println("Weather sensitivity: " + weather);
	  Method method = new Method(1, "Method", prod, lab, mat, 1, sub, uncertainty, weather, subd);
	  System.out.println("Original productivity: " + prod);
	  System.out.println("Productivity plus uncertainty: " + method.getProductivity());
	  method.setProductivityAdjustment(15);
	  System.out.println("Productivity plus temperature adjustment: " + method.getProductivity());

	  System.out.println("Original labor cost: " + lab);
	  System.out.println("Labor cost plus uncertainty: " + method.getLaborCost());
	  method.adjustLaborCost(-10);
	  System.out.println("Labor cost after 10% decrease: " + method.getLaborCost());

	  System.out.println("Original materials cost: " + mat);
	  System.out.println("Materials cost plus uncertainty: " + method.getMaterialsCost());
	  method.adjustMaterialsCost(3);
	  System.out.println("Materials cost after 3% increase: " + method.getMaterialsCost());

  }

}
