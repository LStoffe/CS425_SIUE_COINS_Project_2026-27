/**
 * Contains information on a job.
 * The job class holds information about the current state of a job.  There is only one copy of a <code>Job</code> class stored in the
 * database, unlike Activities which have seperate copies for each period.  During period updates the state of the job is changed
 * and the updates are saved to the database.
 * <br><br>
 * The three most important functions of the <code>Job</code> class are to:
 * <ul>
 *   <li>Determine the bid winner of the job</li>
 *   <li>Do one month's work on the job</li>
 *   <li>Make payments for work done on the job</li>
 * </ul>
 * <br>
 * <code>Job</code> cannot perform the functions above without first receiving data from the <code>ExpandedCompany</code> and
 * <code>ExpandedGame</code> classes, which control the period updates.  When a <code>Job</code> is created using its constructor (such as when
 * it is received from a DBHandler class) it will be missing its company, its list of bids, and its activities.  Before bid winners can be determined the <code>Job</code>
 * must receive a <code>Vector</code> of bids.  Before it can be updated or income paid activities and a company must be set.
 * The <code>ExpandedGame</code> and <code>ExpandedCompany</code> classes provide
 * a <code>Job</code> with these components via set methods.
 *
 *
 * @version $Revision: 1.50 $ $Date: 2002/05/13 22:54:31 $
 */

  /**
   * Charges a bidding expense to each company who bid on this job.
   * A bidding expense is (Bid expense factor game parameter / 1000) * bid amount.
   * This expense is deducted from the cash-on-hand of each company who placed a bid on this job.
   */
  public void chargeBiddingExpenses(int bidExpensePercent)
  {
	  Iterator bidIterator = bids.iterator();
	  while(bidIterator.hasNext()) {
		  Bid theBid = (Bid) bidIterator.next();
		  if (theBid.getCompanyID() > -1) {   // negative companyIDs are CCCs
			  ExpandedCompany theCompany = (ExpandedCompany) game.getExpandedCompany(theBid.getCompanyID());
			  FinancialReportForm financialReportForm = theCompany.getFinancialReport();
			  financialReportForm.increaseBiddingExpenses((theBid.getBidAmount() * bidExpensePercent) / 1000);
			  theCompany.decreaseCashOnHand((theBid.getBidAmount() * bidExpensePercent) / 1000, "BID_EXPENSE");

			  //When the company is charged the bidding expense they also get name recognition
			  //points for placing the bid
			  game.modifyCAMUpdate(theCompany, CAMUpdater.NAME_RECOGNITION, CAMUpdater.NAME_RECOGNITION_BID_JOB);
		  }
	  }
  }

  /**
   * Determines which company is awarded this job.
   * Examines all bids made on this job and chooses one based on the following factors:
   * <ul>
   *   <li>The company has the required percentage of the bid amount as cash-on-hand</li>
   *   <li>The company's bid is greater than or equal to a lower limit determined for this job(80-90% of direct cost)</li>
   *   <li>The company's bid is lower than all other bids</li>
   * </ul>
   *
   * The bid outcome codes are:
   * <ul>
   *   <li>1 - The project was awarded to your firm</li>
   *   <li>2 - Your bid was deemed too low and was therefore disregarded</li>
   *   <li>3 - Your firm was found to have too many jobs in progress</li>
   *   <li>4 - Your firm was considered to be financially weak</li>
   *   <li>5 - Your firm did not bid on this project</li>
   *   <li>6 - Another firm made a lower bid</li>
   * </ul>
   * Company X's bid is treated as any other bid during this process.  If for some reason no company is
   * determined to be the winner, Company X will be the winner by default.
   * <br>
   * <br>
   * The job will be assigned to the winner.
   *
   * @precondition This Job's bid list (<code>bids</code>) has been populated by its parent ExpandedGame
   * @precondition The bid list contains a bid on behalf of CompanyX.  This bid is valid in that it will
   *               not be disqualified based on outcome codes 2, 3, or 4.
   * @postcondition Each <code>Bid</code> in the bid list has been marked with an outcome code.
   *                Exactly one bid has been assigned outcome code 1 (bid winner)  This winning bid
   *                is the one with the lowest bid amount among the bids that satisfy the conditions
   *                of acceptable bids.
   */
  public void determineBidWinner()
  {
	  // Note: Bids do not contain [Expanded]Company objects, just company ids
	  // for use in game.getExpandedCompany(companyID)

	  // Serves as a starting point when finding the lowest bid
	  double[] unitPrices = new double[9];
	  double theBidPrice = 0;
	  Bid lowBid = new Bid(-1, -1, -1, Long.MAX_VALUE, theBidPrice, unitPrices, -1);

      // Value to use as cutoff point for bids.  All bids below this percentage of the direct cost
      // are rejected with code 2.
	  // Random number between 70 and 80, inclusive (see BIG FAQ - 8/7/2001)
	  int minDirectCostPercent = 70 + (new Random()).nextInt(11);

	  // DEBUG - To Be Removed
	  //System.out.println("Before:");
	  Iterator bidIterat = bids.iterator();
	  while(bidIterat.hasNext()) {
		  Bid theBid = (Bid) bidIterat.next();
		  //System.out.println(" " + theBid);
	  }

	  /*
	   * Disqualify bids based on outcome codes 6, 2, 3, and 4
	   * (which don't require comparison with other bids)
	   */
	  Iterator bidIterator = bids.iterator();
	  while(bidIterator.hasNext())
	  {
		  Bid theBid = (Bid) bidIterator.next();

		  if (theBid.getCompanyID() > -1)
		  {
			  // Do not check CompanyX against codes 6, 2, 3 or 4
			  ExpandedCompany theCompany = (ExpandedCompany) game.getExpandedCompany(theBid.getCompanyID());
			  long bidAmount = theBid.getBidAmount();

			  if (bidAmount < (getDirectCost() * minDirectCostPercent) / 100)
			  {
				  theBid.setOutcomeCode(2);     // Bid is too low
          		  game.sendInGameMessageToCompany(new Message("PJC",
													  		   game.getExpandedCompany(theBid.getCompanyID()).getCompany(),
													  		   "We regret to inform you that we have chosen a different contractor for work on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ".  We appreciate your bid and look forward to possible future business opportunities with your company.",
													  		   "Your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );
			  }
			  else if (!theCompany.hasFewerThanMaxJobs())
			  {
				  theBid.setOutcomeCode(3);     // Company already has too many jobs
          		  game.sendInGameMessageToCompany(new Message("PJC",
													  		   game.getExpandedCompany(theBid.getCompanyID()).getCompany(),
													  		   "We regret to inform you that we have chosen a different contractor for work on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ".  We appreciate your bid and look forward to possible future business opportunities with your company.",
													  		   "Your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );
			  }
			  else if (!theCompany.jobWithinPerJobLimit(bidAmount) && typeModifier==PUBLIC)
			  {
				  theBid.setOutcomeCode(4);     // Per job limit exceeded
          		  game.sendInGameMessageToCompany(new Message("TBC",
													  		   game.getExpandedCompany(theBid.getCompanyID()).getCompany(),
													  		   "We are unable to provide you with a bond for your bid on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ", because the amount of your bid exceeds your per job limit for bonds.  Public jobs require a bond, so your bid has been disregared by the customer.  Please be aware of your bond limits when you bid to prevent this from happening in the future.",
													  		   "Bond for your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );
			  }
			  else if (!theCompany.jobWithinWIPLimit(bidAmount) && typeModifier==PUBLIC)
			  {
				  theBid.setOutcomeCode(6);     // WIP limit exceeded
          		  game.sendInGameMessageToCompany(new Message("TBC",
													  		   game.getExpandedCompany(theBid.getCompanyID()).getCompany(),
													  		   "We are unable to provide you with a bond for your bid on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ", because the amount of your bid plus the value of your work currently in progress would exceed your WIP limit.  Public jobs require a bond, so your bid has been disregared by the customer.  Please be aware of your bond limits when you bid to prevent this from happening in the future.",
													  		   "Bond for your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );
			  }

		  }

	  }

	  /*
	   * Find the lowest bid
	   */
	  bidIterator = bids.iterator();
	  while(bidIterator.hasNext()) {
		  Bid theBid = (Bid) bidIterator.next();

		  try
		  {
		  if ((theBid.getOutcomeCode() < 1) || (theBid.getOutcomeCode() > 7)) {
			  // Bid has not yet been assigned an outcome code, which means
			  // it should be checked against the lowest bid
			  if (theBid.getBidAmount() < lowBid.getBidAmount()) {
				  // We've found a lower bid
				  lowBid.setOutcomeCode(7);
				  Bid temp = lowBid;
				  lowBid = theBid;
          		  game.sendInGameMessageToCompany(new Message("PJC",
													  		   game.getExpandedCompany(temp.getCompanyID()).getCompany(),
													  		   "We regret to inform you that we have chosen a different contractor for work on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ".  We appreciate your bid and look forward to possible future business opportunities with your company.",
													  		   "Your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );

			  } else {
				  theBid.setOutcomeCode(7);
          		  game.sendInGameMessageToCompany(new Message("PJC",
													  		   game.getExpandedCompany(theBid.getCompanyID()).getCompany(),
													  		   "We regret to inform you that we have chosen a different contractor for work on the " + TEMPORARY_getJobType() + ", Job# " + jobID + ".  We appreciate your bid and look forward to possible future business opportunities with your company.",
													  		   "Your bid on job #" + jobID,
													  		   game.getGame().getCurrentPeriodNumber() + 1,
													  		   Message.TEXT_ONLY,
													  		   -1
													 		 )
										 		 );
			  }
		  }
	  	  }
	  	  catch(NullPointerException npe)
	  	  {
			  //A bit sloppy...  we want to ignore cases
			  //were the bid is by a CCC
		  }

	  }

	  // lowBid now contains the winning bid (may be CompanyX's bid)
	  lowBid.setOutcomeCode(1);

	  // DEBUG - To Be Removed
	  bidIterator = bids.iterator();
	  while(bidIterator.hasNext()) {
		  Bid theBid = (Bid) bidIterator.next();
		  //System.out.println(" " + theBid);
	  }

	  game.storeBids(this, bids);

	  // Associate the winning company with this job, leaving <code>company</code>
	  // null if a CCC placed the winning bid;
	  if (lowBid.getCompanyID() > -1) {
		  company = game.getExpandedCompany(lowBid.getCompanyID());
          game.getExpandedCompany(lowBid.getCompanyID()).awardJob(this, lowBid);
          game.sendInGameMessageToCompany(new Message("PJC",
													  game.getExpandedCompany(lowBid.getCompanyID()).getCompany(),
													  "We are pleased to inform you that we have chosen your company to be the contractor for work on our new " + TEMPORARY_getJobType() + ", Job# " + jobID,
													  "Job Award Notification",
													  game.getGame().getCurrentPeriodNumber() + 1,
													  Message.TEXT_ONLY,
													  -1
													 )
										 );

          //Give this company name recognition credit for getting this job
          int nameRecognitionBonus = Math.round(((float) jobSize) / ((float) CAMUpdater.NAME_RECOGNITION_GET_JOB_MODIFIER));
          game.modifyCAMUpdate(company, CAMUpdater.NAME_RECOGNITION, nameRecognitionBonus);

          //Give this company job type credit for getting this job
		  giveJobTypeCredit(CAMUpdater.JOB_TYPE_AWARD_BONUS);


		  //Need to store this bid amount in the job -- K.C. 1/9/2002
		  bidAmount = lowBid.getBidAmount();
		  createChangeOrder();
	  }

  }

  /**
   * For each bid made by a player company, determines how responsible that
   * bid is, using the criteria described in the BIGv3.2 SRS:
   * <ul>
   *   <li>
   *     If a company sumbits a bid that is within 1 standard deviation
   *     of the average of all bids for the job, then the value for the
   *     responsibility metric will be increased by the
   *     value of <code>CAMUpdater.BID_RESPONSIBILITY_1SD_INC</code>
   *   </li>
   *   <li>
   *     If a company sumbits a bid that is within 2 standard deviations
   *     of the average of all bids for the job, then the value for the
   *     responsibility metric will be increased by the
   *     value of <code>CAMUpdater.BID_RESPONSIBILITY_2SD_INC</code>
   *   </li>
   *   <li>
   *     If a company sumbits a bid that is below BIG's calculated
   *     direct cost of the job, then the value for the responsibility metric
   *     will be decreased by the
   *     value of <code>CAMUpdater.BID_RESPONSIBILITY_LOW_DC</code>
   *   </li>
   *   <li>
   *     If a company sumbits a bid that is over 3 standard deviations
   *     of the average of all bids for the job, then the value for the
   *     responsibility metric will be decreased by the
   *     value of <code>CAMUpdater.BID_RESPONSIBILITY_3SD_DEC</code>
   *   </li>
   * </ul>
   */
  public void determineResponsibility()
  {
	  // Note: Bids do not contain [Expanded]Company objects, just company ids
	  // for use in game.getExpandedCompany(companyID)

	  long[] bidPrices = new long[bids.size()];

	  for(int i = 0; i < bids.size(); i++)
	  {
		  Bid theBid = (Bid) bids.elementAt(i);
		  bidPrices[i] = theBid.getBidAmount();
	  }

	  long bidAverage = Utility.average(bidPrices);
	  long bidStdDev = (long) Utility.stdDev(bidPrices);

	  for(int i = 0; i < bids.size(); i++)
	  {
		  Bid theBid = (Bid) bids.elementAt(i);

		  if(theBid.getCompanyID() > -1)
		  {
		      ExpandedCompany ec = game.getExpandedCompany(theBid.getCompanyID());

		      //Bid is less than direct cost
		      if(theBid.getBidAmount() < directCost)
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_LOW_DC);
		      }
		      //Bid between 2 and 1 standard devs below average
		      else if ((theBid.getBidAmount() > (bidAverage - (2 * bidStdDev))) && (theBid.getBidAmount() < (bidAverage - (1 * bidStdDev))))
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_2SD_INC);
		      }
		      //Bid between 1 standard dev below average and average
		      else if ( (theBid.getBidAmount() <= bidAverage) && (theBid.getBidAmount() > (bidAverage - (1 * bidStdDev))))
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_1SD_INC);
		      }
		      //Bid between average and 1 standard dev above average
		      else if ( (theBid.getBidAmount() > bidAverage) && (theBid.getBidAmount() < (bidAverage + (1 * bidStdDev))))
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_1SD_INC);
		      }
		      //Bid between 1 and 2 standard devs above average
		      else if ((theBid.getBidAmount() < (bidAverage + (2 * bidStdDev))) && (theBid.getBidAmount() > (bidAverage + (1 * bidStdDev))))
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_2SD_INC);
		      }
		      //Bid above 3 standard devs above average
		      else if (theBid.getBidAmount() > (bidAverage + (3 * bidStdDev)))
		      {
			      game.modifyCAMUpdate(ec, CAMUpdater.BID_RESPONSIBILITY, CAMUpdater.BID_RESPONSIBILITY_3SD_DEC);
		      }
		  }
	  }

  }

  //Increase the job size of this job by some random amount upto 5% of the current job size.  The new LU's will be randomly
  //allocated amongst each activity that has LU's (ie where the percent takeoff for that activity != 0)
  //A call to ExpandedGame will be made to get the cost of doing this.  This cost will be stored in the
  //changeOrderPrice variable.
  protected void createChangeOrder()
  {
	  Random random = new Random();
	  int changeOrderPercent = random.nextInt(6); //Generates value from 0 to 5

	  changeOrderLus = Math.round((((float) changeOrderPercent) / 100f) * jobSize);

	  int tempChangeOrderLus = changeOrderLus;
	  int[] allocatedLus = new int[9];

      activities = game.getActivities(this);

	  //Randomly allocate lus to activities
	  while (tempChangeOrderLus > 0)
	  {
		  int activity = random.nextInt(9);  //Values from 0 to 8
		  if (activities[activity].getLusTotal() > 0) //Only allocate to activities that have some lus already
		  {
			  int lusToAllocate = random.nextInt(tempChangeOrderLus + 1);
			  allocatedLus[activity] += lusToAllocate;
			  activities[activity].setLusTotal(activities[activity].getLusTotal() + lusToAllocate);
			  tempChangeOrderLus -= lusToAllocate;
		  }
	  }

	  changeOrderPrice = game.getChangeOrderPrice(allocatedLus);

	  //System.out.println(changeOrderPrice);
      game.updateActivities(activities, this);
      activities = null;
  }

  /**
   * Does work on each activity of this job for each workday of the month, or until the job completes.
   * <br><br>
   * <b>Precondition: </b>This job must have been assigned an <code>ExpandedCompany</code> using the setCompany() method,
   * and an array of <code>Activity</code> using the setActivities() methods.
   * <br>
   * @
   *
   */
  public void update(ScheduledMonth month, FinancialReportForm financialReportForm, PrintWriter out)
  {
		//System.out.println("Did one month's work");
		boolean complete = false;
		Vector activitiesVector = new Vector();
		fixedFieldCharge = 0;

		//Place the Activities into a Vector for use in the NetworkDependencies object
		NetworkDependencies networkDependencies = new NetworkDependencies();


		for(int i = 0; i < 9; i++)
		     activitiesVector.add(activities[i]);

		while(month.daysLeft() && !complete)  // work through a single day as long as the month is not yet over and the job is not finished
		{
			if(month.getWorkday())
			{
				//Charge company for overhead for this workday
				fixedFieldCharge += fixedFieldOverhead;

                networkDependencies.update(activitiesVector);  // will set the "isActive" field for all activities

				for(int i = 0; i < 9; i++)
				{
				    activities[i].update(month.getRainday(), this.jobType, this.jobID, company, game, out);
				    /* System.out.println(i+1);
				    System.out.println(activities[i].getLusTotal());
				    System.out.println(activities[i].getLusCompleted()); */
				}

			}
	        updateLusRemaining();
			if(allActivitiesComplete())
			{
			    completionDate = month.getDate();
			    complete = true;
                periodCompleted = game.getGame().getCurrentPeriodNumber();

	  			java.text.SimpleDateFormat formatter = new java.text.SimpleDateFormat("MM/dd/yyyy");
	  			//System.out.println(formatter.format(completionDate));
	  			isActive = false;

	  			//Give this company name recognition credit if the job is finished on time
	  			//Give this company job type credit if the job is finished on time
	  			//Give this company pace metric credit if the job is finished on time
	  			if(month.getDate().getTime() < completionDeadline.getTime())
	  			{
					int nameRecognitionBonus = Math.round(((float) jobSize) / ((float) CAMUpdater.NAME_RECOGNITION_FINISH_JOB_MODIFIER));
					game.modifyCAMUpdate(company, CAMUpdater.NAME_RECOGNITION, nameRecognitionBonus);
				//	giveJobTypeCredit(CAMUpdater.JOB_TYPE_FINISHING_BONUS);
					game.modifyCAMUpdate(company, CAMUpdater.PACE, CAMUpdater.PACE_FINISHED_ONTIME);
				}

			}
			if(month.getDate().getTime() > completionDeadline.getTime())
			    incLiquidatedDamages(financialReportForm);
			month.advanceDay();
		}  // end of move through one day -- on to the next day!





		//Give this company name recognition credit for work done on this job
		int nameRecognitionBonus = Math.round(((float) jobSize) / ((float) CAMUpdater.NAME_RECOGNITION_WORK_JOB_MODIFIER));
		game.modifyCAMUpdate(company, CAMUpdater.NAME_RECOGNITION, nameRecognitionBonus);
  }

  public void recordDirectCosts(FinancialReportForm financialReportForm, BalanceSheetForm balanceSheet)
  {
	    laborExpense = 0;
	    materialsExpense = 0;
	    equipmentExpense = 0;
	    subcontractorExpense = 0;
	    for(int i = 0; i < 9; i++)
		{
			laborExpense += activities[i].getCurrentLaborCost();
			materialsExpense += activities[i].getCurrentMaterialCost();
			equipmentExpense += activities[i].getCurrentEquipmentCost();
			subcontractorExpense += activities[i].getCurrentSubcontractorCost();
		}

		financialReportForm.increaseDirectLaborCost(this, laborExpense);
		financialReportForm.increaseDirectMaterialsCost(this, materialsExpense);
		financialReportForm.increaseDirectEquipmentCost(this, equipmentExpense);
		financialReportForm.increaseSubcontractorCost(this, subcontractorExpense);

        //Determine the total amount retained from subcontractors
		int activityAmountRetained = 0;
		for(int i = 0; i < 9; i++)
		{
			activityAmountRetained += amountToRetain(activities[i], percentRetained, percentOfJob);
		}

		//Increase the running total of amounts retained from subs by the current amount
		amountRetainedFromSubcontractors += activityAmountRetained;

		balanceSheet.setCurrentAccountsPayable(  balanceSheet.getCurrentAccountsPayable()
		                                       + materialsExpense
		                                       + (subcontractorExpense - activityAmountRetained)
		                                       );

		balanceSheet.setRetentionPayable(balanceSheet.getRetentionPayable() + activityAmountRetained);

		//If the job completed this period, all amounts retained from subcontractors become payable
		if(!isActive)
		{
			balanceSheet.setRetentionPayable(balanceSheet.getRetentionPayable() - amountRetainedFromSubcontractors);
			balanceSheet.setCurrentAccountsPayable(balanceSheet.getCurrentAccountsPayable() + amountRetainedFromSubcontractors);
		}



  }

  //Calculates the amount to retain from subcontractor payments on an activity.
  //If a subcontracted method was not used on the activity this method will return 0.
  private int amountToRetain(Activity activity, int percentRetained, int percentOfJob)
  {
      int percentActivityComplete = Math.round((((float) activity.getLusCompleted()) / ((float) activity.getLusTotal())) * 100f);
      int retained = 0;

      //If the % work complete is greater than the percent of the activity to collect
      //retention for, return 0
      if(percentActivityComplete < percentOfJob)
      {
		  retained = Math.round(( ((float) percentRetained) / 100f) * activity.getCurrentSubcontractorCost());
	  }
	  else
	  {
		  return 0;
	  }

	  return retained;
  }

  private void updateLusRemaining()
  {
	  int totalLusComplete = 0;
	  for(int i = 0; i < 9; i++)
	  {
		  totalLusComplete += activities[i].getLusCompleted();
	  }
	  lusRemaining = (jobSize + changeOrderLus) - totalLusComplete;
  }

  //Return true if all the activities are complete
  private boolean allActivitiesComplete()
  {
  	  boolean allComplete = true;
  	  for (int i = 0; i < 9; i++)
  	  {
		  if(activities[i].getLusCompleted() < activities[i].getLusTotal())
			    allComplete = false;
  	  }
  	  return allComplete;
  }
