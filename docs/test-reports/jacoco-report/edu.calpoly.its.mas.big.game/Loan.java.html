package edu.calpoly.its.mas.big.game;

/**
 * Contains information about a loan requested by a company.
 * <code>Loan</code> keeps track of current balances and how many payments
 * a company has left on a loan they've requested.  When a company requests a loan,
 * a new <code>Loan</code> object is created to calculate what their monthly payment will
 * be based on the amount requested and other game parameters.  When it comes time to payback
 * loans, BIG will identify current loans and create <code>Loan</code> objects to deal with them.
 *
 * @author Kevin Craig
 * @version $Revision: 1.12 $ $Date: 2002/04/26 18:03:07 $
 */
public class Loan
{

  private int loanID;

  private int numberOfPayments;
  private long amount;
  private long balance;
  private int APR;
  private long paymentAmount;
  private int paymentsRemaining;
  private long interestPaid = 0;
  private boolean approved = false;
  private boolean repayImmediately = false;

  /**
   * Contructor for creating a brand new loan
   * @param periodNum The period this Loan object is for
   * @param amount Dollar amount of the loan
   * @param APR Annual percentage rate of interest applied to this loan
   */
  public Loan(long amount, int APR, int numberOfPayments)
  {
     this.amount = amount;
     this.APR = APR;
	 this.numberOfPayments = numberOfPayments;
     balance = amount;
     paymentsRemaining = numberOfPayments;
     determinePaymentAmount();
  }

  /**
   * Constructor for creating a loan object from a loan previously saved
   * in the database
   * @param loanID loanID used to identify this loan for this period, to the database
   * @param amount Initial dollar amount of the loan
   * @param balance Remaining balance on the loan
   * @param APR Annual percentage rate of interest applied to this loan
   * @param paymentAmount Dollar amount of each payment
   * @param paymentsRemaining Number of payments that must be made before this loan will be paidoff
   * @param interestPaid Amount of of last payment that went towards interest
   * @param approved True if the loan has been approved
   */
  public Loan(int loanID,
  			  long amount,
  			  long balance,
  			  int APR,
  			  long paymentAmount,
  			  int paymentsRemaining,
  			  long interestPaid,
  			  boolean approved,
  			  boolean repayImmediately)
  {
	 this.loanID = loanID;
     this.amount = amount;
     this.balance = balance;
     this.APR = APR;
     this.paymentAmount = paymentAmount;
     this.paymentsRemaining = paymentsRemaining;
     this.interestPaid = interestPaid;
     this.approved = approved;
     this.repayImmediately = repayImmediately;
  }

  /**
   * Uses a loan amortization equation to determine what the amount of the payments should be
   */
  private void determinePaymentAmount()
  {
     float interestPerPeriod = getInterestRatePerPeriod();
     paymentAmount = (int) Math.round(interestPerPeriod * (amount - (amount/(1 - Math.pow((double) (1d + interestPerPeriod) , (double) numberOfPayments)))));
  }

  /**
   * Returns the interest rate per period
   */
  private float getInterestRatePerPeriod()
  {
     //Calculate the per period interest rate.
     //APR is an integer so divide by 100 to get the decimal.  There are 6
     //periods per year so divide by 6.  The 600 comes from 100*6
     return (((float) APR) / 600f);
  }

  /**
   * Makes a payment to this loan.  If the repayImmediately flag is set, this
   * payment will be equal to the balance of the loan, no part of the payment will
   * go to interest, and the number of payments remaining will be set to zero.
   * <br><br>
   * If the repayImmediately flag is not set, a payment will be made that is
   * equal to the <code>paymentAmount</code> provided or calculated
   * when this object was constructed.  Part of this payment
   * equal to APR / 6 percent of the outstanding balance will go to
   * interest and the remainder of the payment will reduce the loan balance.
   * The APR is divided by 6 because there are six periods in a year.
   * The number of payments remaining will be reduced by 1.
   * <br><br>
   * <b>Note:</b>It is the responsibility of the method that
   * calls makePayment to deduct the dollar amount of that payment
   * from the company's cash-on-hand.  This dollar amount can be obtained
   * from the getPaymentAmount method.
   */
  public void makePayment()
  {
	 if(!repayImmediately)
	 {
		//Make a normal periodic payment

     	float periodPercentageRate = getInterestRatePerPeriod();

     	//Determine how much interest to charge this payment
     	interestPaid = (int) Math.round(periodPercentageRate * (float) balance);

     	//Deduct from the balance the amount of the payment that didn't go to paying interest
     	balance = balance - (paymentAmount - interestPaid);

     	paymentsRemaining--;
     }
     else
     {
		 //Pay off the loan in full
		 this.paymentAmount = balance;
		 balance = 0;
		 interestPaid = 0;
		 paymentsRemaining = 0;
	 }
  }

  /**
   * Returns true if there are no more payments to be made on this loan.
   */
  public boolean isPaidOff()
  {
      if (paymentsRemaining <= 0)
         return true;
      return false;
  }

  //**********************  Simple Get & Set Methods  *************************

  /**
   * Returns the initial amount of the loan
   */
  public long getAmount()
  {
     return amount;
  }

  /**
   * Sets the initial amount of the loan
   */
  public void setAmount(long amount)
  {
     this.amount = amount;
  }

  /**
   * Returns the remaining balance of the loan
   */
  public long getBalance()
  {
     return balance;
  }

  /**
   * Sets the remaining balance of the loan
   */
  public void setBalance(long balance)
  {
     this.balance = balance;
  }

  /**
   * Returns the annual percent rate of interest applied to this loan
   */
  public int getAPR()
  {
     return APR;
  }

  /**
   * Sets the annual percent rate of interested to be applied to this loan
   */
  public void setAPR(int APR)
  {
     this.APR = APR;
  }

  /**
   * Returns the dollar amount of each payment
   */
  public long getPaymentAmount()
  {
     return paymentAmount;
  }

  /**
   * Sets the dollar amount of each payment
   */
  public void setPaymentAmount(long paymentAmount)
  {
     this.paymentAmount = paymentAmount;
  }

  /**
   * Returns the number of payments remaining on this loan
   */
  public int getPaymentsRemaining()
  {
     return paymentsRemaining;
  }

  /**
   * Sets the number of payments remaining on this loan
   */
  public void setPaymentsRemaining(int paymentsRemaining)
  {
     this.paymentsRemaining = paymentsRemaining;
  }

  /**
   * Gets the dollar amount of the current payment that is interest.
   * <br><br><b>Note:</b>
   * If the makePayment method has not been called since the Loan object
   * was instantiated, this method will return 0.
   */
  public long getInterestPaid()
  {
     return interestPaid;
  }

  /**
   * Returns true if this loan is approved.
   */
  public boolean isApproved()
  {
     return approved;
  }

  /**
   * Set whether or not this loan is marked as approved.
   * @param approved True for approved, false for not approved
   */
  public void setApproved(boolean approved)
  {
     this.approved = approved;
  }


  /**
   * Returns the loanID assigned to this loan for this period by the database
   */
  public int getLoanID()
  {
	  return loanID;
  }

  /**
   * Sets the loan id used to identify this loan to the database
   */
  public void setLoanID(int loanID)
  {
	  this.loanID = loanID;
  }



  /**
   * Returns the  total value (total interest + unpaid principle) of this loan.
   * Note: The total value will always be the total value at that point in time, so if payments
   * have been made the value decreases.
   */
  public long getTotalLoanValue()
  {
	  return paymentAmount * paymentsRemaining;
  }

  public boolean repayImmediately()
  {
	  return repayImmediately;
  }

  public void setRepayImmediately(boolean repayImmediately)
  {
	  this.repayImmediately = repayImmediately;
  }
