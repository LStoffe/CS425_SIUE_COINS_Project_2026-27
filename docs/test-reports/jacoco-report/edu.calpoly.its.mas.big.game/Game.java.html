/**
 * Game is a bean that contains state information about a game of big.
 * The Game class is used to hold and distribute information about the current game.
 * A Game will almost always be stored in the session where it can be accessed to provide needed information.
 * The <code>Game</code> class contains some utility methods for working with dates and period numbers in
 * BIG.  Since the <code>Game</code> class keeps track of the start year and current period
 * it is the best place to put these methods.  Note that other utility methods are available in
 * <code>ExpandedGame</code> however these can only be used when <code>ExpandedGame</code> objects are
 * created during period updated.
 * <br><br>
 * Notes for v3:
 * <br>
 * Because this class contains the start year of a game, it has become the central class for working with dates
 * in BIG.  Whenever a period number needs to be converted into a date or vice versa, the necessary method
 * should reside in this class.  Any new methods to be added that manipulate dates or perform conversions
 * between dates and periods should be added to this class.
 *

 *
 */

   /**
    * Returns the date (months and year) for a particular period.
    * The date returned will use this game's starting year to determine the date.
    *
    */
   public String getDateString(int periodNumber)
   {
	   //Multiply 2 times the number of periods -1 to get the month id of the first month.  Month's are zero based in the Calendar class
	   int months = 2 * (periodNumber - 1);

	   int firstMonthOfThisPeriod = months;

	   Calendar calendar = new GregorianCalendar();
	   calendar.setTime(startYear);

	   //Add the number of months passed to the start date to get the date of the first month in this period
	   calendar.add(Calendar.MONTH, firstMonthOfThisPeriod);
	   Date firstMonth = calendar.getTime();

	   //Add one to the month of the first month date to get the date of the second month of this period
	   calendar.add(Calendar.MONTH, 1);
	   Date secondMonth = calendar.getTime();

	   SimpleDateFormat formatter = new SimpleDateFormat("MMMMM");

	   String firstMonthString = formatter.format(firstMonth);
	   String secondMonthString = formatter.format(secondMonth);

	   formatter = new SimpleDateFormat("yyyy");

	   String yearString = formatter.format(secondMonth);

	   String date = firstMonthString + " & " + secondMonthString + " " + yearString;

	   return date;
   }

   /**

   /**
    * Returns first day of each of the months in a period.
    * @param periodNumber Period number to return start dates for
    * @return <code>Date</code> array of size 2, containing 2 <code>Date</code> objects representing the date
    * of the first day of each of the two months in the period.
    */
   public Date[] getFirstOfMonth(int periodNumber)
   {
		//Multiply 2 times the number of periods -1 to get the month id of the first month.  Month's are zero based in the Calendar class
	   int months = 2 * (periodNumber - 1);

	   int firstMonthOfThisPeriod = months;

	   Calendar calendar = new GregorianCalendar();
	   calendar.setTime(startYear);

	   //Add the number of months passed to the start date to get the date of the first month in this period
	   calendar.add(Calendar.MONTH, firstMonthOfThisPeriod);
	   calendar.set(Calendar.DAY_OF_MONTH, 1); //set to first day of month
       Date firstMonth = calendar.getTime();


	   //Add one to the month of the first month date to get the date of the second month of this period
	   calendar.add(Calendar.MONTH, 1);
           calendar.set(Calendar.DAY_OF_MONTH, 1); //set to first day of month
	   Date secondMonth = calendar.getTime();

	   Date[] dates = {firstMonth, secondMonth};

	   return dates;
   }


   /**
    * Returns the period number in which a date occurs.
    * @param date A <code>Date</code> object representing any date in the Gregorian calendar
    * @return int The period number the date will fall in BIG.  If the date is before Jan 1st of
    * this game's start year, -1 is returned.
    */
   public int convertDateToPeriod(Date date)
   {
	   if( date.before(startYear) )
   	   {
		   return -1;
	   }
	   else
	   {
	       Calendar calendar = new GregorianCalendar();
	       calendar.setTime(date);

	       //Get the year and month of the date provided
	       int currentYear = calendar.get(Calendar.YEAR);
	       int currentMonth = calendar.get(Calendar.MONTH) + 1; //Make months start at 1

	       //Get the start year
	       calendar.setTime(startYear);
	       int startingYear = calendar.get(Calendar.YEAR);

	       //1 - 6 depending on the month
	       int requestedPeriodNumber = Math.round(((float) currentMonth / 2f));

	       //Number of years greater than starting year
	       int yearOffset = (currentYear - startingYear);

           //6 periods each year, so multiply the yearOffset by 6 to get the period offset
	       return requestedPeriodNumber + 6 * yearOffset;

	   }


   }

   /**
    * Given a required completion date this method estimates the percentage
    * of the last period that will be worked assuming a job finishes on its
    * required completion date.  This method is hardcoded to assume
    * two months per period.
    * <br><br>
    * Examples: If a job's required completion date is the first day of the second
    * month, this method would return around .5.
    * <br> If a job's required completion date is about halfway through the first month
    * this method would return around.25.
    * @return A float between (0 and 1]
    *
    */
   public float getEstimatedPercentageOfLastPeriodWorked(Date completionDate)
   {
	   float maxDaysPerMonth = 30f;  //Assume 30 days per month since it's simpler
	   							     //and this is an estimate anyway

	   Calendar calendar = new GregorianCalendar();
	   calendar.setTime(completionDate);

	   int month = calendar.get(Calendar.MONTH);
	   int day = calendar.get(Calendar.DAY_OF_MONTH);

	   //Calculate how far the completion day is into the completion month
	   float percentIntoMonth = ((float) day) / maxDaysPerMonth;


	   //Two months per period so divide by 2 to get the percent in the period
	   //not yet counting the offset
	   float percentIntoPeriod = percentIntoMonth / 2f;

	   //If this is the second month of a period (even numbered) add .5 because the completion
	   //day is at least halfway through the period
	   if ((month % 2) == 1)
	   {
		   percentIntoPeriod += .5;
	   }

	   //Make sure we didn't get over 1 somehow
	   if (percentIntoPeriod > 1)
	      percentIntoPeriod = 1f;

	   return percentIntoPeriod;
   }
