  //Increase the job size of this job by some random amount upto 5% of the current job size.  The new LU's will be randomly
  //allocated amongst each activity that has LU's (ie where the percent takeoff for that activity != 0)
  //A call to ExpandedGame will be made to get the cost of doing this.
  private void createChangeOrder(ExpandedGame xgame, Job job)
  {
	  Random random = new Random();

	  int changeOrderPercent = random.nextInt(6); //Generates value from 0 to 5
	  int changeOrderLus = Math.round((((float) changeOrderPercent) / 100f) * job.getJobSize() );
	  int tempChangeOrderLus = changeOrderLus;
	  int[] allocatedLus = new int[9];

      Activity[] activities = null;

      //Get activities from the database
      activities = getActivities(job);

	  //Randomly allocate lus to activities
	  while (tempChangeOrderLus > 0)
	  {
		  int activity = random.nextInt(9);  //Values from 0 to 8
		  if (activities[activity].getLusTotal() > 0) //Only allocate to activities that have some lus already
		  {
			  int lusToAllocate = random.nextInt(tempChangeOrderLus + 1);
			  allocatedLus[activity] += lusToAllocate;
			  activities[activity].setLusTotal(activities[activity].getLusTotal() + lusToAllocate);
			  tempChangeOrderLus -= lusToAllocate;
		  }
	  }

	  long changeOrderPrice = xgame.getChangeOrderPrice(allocatedLus);

	  //System.out.println("Change Order Price = " + changeOrderPrice);

      //store the CO data with the job
	  job.changeOrderPrice = changeOrderPrice;
	  job.changeOrderLus = changeOrderLus;

      //update job in the database
	  try
	  {
	  	  dbFacade.getJobDBHandler().updateJob(game, job);
	  }
	  catch (BIGDatabaseFacadeException bigdbe)
	  {
	  	  log.fatal("", bigdbe);
	  }
	  catch(ObjectNotFoundException onfe)
	  {
		  log.fatal("", onfe);
	  }

	  //need to advance the period number of the activities by one since they are behind where they should be
      for(int i = 0; i < 9; i++)
      {
	  //System.out.println("before - activity " + i + "= " + activities[i].getCurrentPeriod() );
      activities[i].incrementPeriodNumber();
      //System.out.println( "after - activity " + i + "= " + activities[i].getCurrentPeriod() );
      }

      //store these activities in the database (note: this is not an update because the activities now have a new period
      //number, so they will be a new set of activities in the database)
	  try
	  {
		  for(int i = 0; i < 9; i++)
		  {
			  dbFacade.getActivityDBHandler().addActivity(game, job, activities[i]);
		  }
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("", bigdbe);
	  }

  }
