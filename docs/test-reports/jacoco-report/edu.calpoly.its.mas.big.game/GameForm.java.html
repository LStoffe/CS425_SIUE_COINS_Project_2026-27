package edu.calpoly.its.mas.big.game;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.users.StringConstraints;

/**
 * ActionForm containing data needed to create a new game.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.5 $ $Date: 2001/08/14 19:58:45 $
 */
public class GameForm extends ActionForm
{

	private String gameName;
	private String startYear;
	private boolean building = false;
	private boolean heavy = false;
	int gameType;

	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
    	ActionErrors errors = new ActionErrors();

        //Check that game name is not null and is of acceptable length
    	if ((gameName == null) || (gameName.length() < 1))
    	{
    	    errors.add("gameName", new ActionMessage("error.game.name.required"));
		}
		else if ((gameName.length() < StringConstraints.GAMENAME_LOWER_LIMIT) || (gameName.length() > StringConstraints.GAMENAME_UPPER_LIMIT))
		{
			errors.add("gameName_length", new ActionMessage("error.game.name.length"));
		}

    	//Check that game start year is not null and that it is in the form YYYY
    	if ((startYear == null) || (startYear.length() < 1))
    	{
    	    errors.add("startYear", new ActionMessage("error.game.startyear.required"));
		}
		else
		{
            boolean validYear = true;

            try
            {
		    	int year = Integer.parseInt(startYear, 10); //10 for decimal base
                if ((year < 1000) || (year > 9999)) //make sure number is 4 digits long
                  validYear = false;
		    }
		    catch(NumberFormatException nfe)
		    {
		    	validYear = false;
		    }

		    if(!validYear)
		    	errors.add("validYear", new ActionMessage("error.game.startyear.format"));
		}

		//Make sure one of the two checkboxes has been checked
		if( !building && !heavy )
    	{
    	    errors.add("noGameType", new ActionMessage("error.game.gametype.required"));
		}

		//Make sure user did not check both boxes
		if( building && heavy )
    	{
    	    errors.add("twoGameTypes", new ActionMessage("error.game.gametype.onlyone"));
		}


    	return errors;
  	}

    //**************************************  Simple Set & Get Methods  ************************************

  	public String getGameName()
  	{
  	    return gameName;
  	}

  	public void setGameName(String gameName)
  	{
  	    this.gameName = gameName;
    }

  	public String getStartYear()
  	{
  	    return startYear;
  	}

  	public void setStartYear(String startYear)
  	{
  	    this.startYear = startYear;
  	}

  	public int getGameType()
  	{
  	    return gameType;
  	}

  	public void setGameType(int gameType)
  	{
  	    this.gameType = gameType;
  	}

public boolean getBuilding()
    {
		return building;
	}

public void setBuilding(boolean building)
	{
		this.building = building;
	}

public boolean getHeavy()
    {
		return heavy;
	}

public void setHeavy(boolean heavy)
	{
		this.heavy = heavy;
	}

}
