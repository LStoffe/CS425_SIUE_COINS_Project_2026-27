/**
 * ExpandedGame is the main controller of the components that perform period update process of BIG.
 * ExpandedGame differs from the Game class in that ExpandedGame will contain references to other complex components of
 * BIG, such as Jobs and Companies.  ExpandedGame controls the period update process by calling methods in other classes in the correct
 * sequence to carry out the process.  ExpandedGame also controls the starting of the game and the finishing of the game, which are both
 * special cases of the period update process.
 * <br><br>
 * The sequence of events in a period update occurs as follows:
 * <ul>
 *   <li>Create <code>Period</code> objects for future periods</li>
 *   <li>Process all company loans</li>
 *   <li>Generate new jobs for the next period</li>
 *   <li>Do work on jobs</li>
 *   <li>Increment the period number</li>
 * </ul>
 * <br><br>
 *
 *
 * @version $Revision: 1.57 $ $Date: 2002/05/17 00:03:09 $
 *
 */

  public ExpandedGame(Game g, BIGDatabaseFacadeInterface dbFacade)
  {
	  log = LogFactory.getLog(this.getClass());

	  game = g;
	  gameType = g.getGametype();

	  this.dbFacade = dbFacade;


	  //Retrieve all the companies in this game and create CompanyX instance
	  try
	  {
	      companies = dbFacade.getCompanyDBHandler().getCompanies(game);
	      customers = dbFacade.getCustomerDBHandler().getCustomers();
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("", bigdbe);
	  }

      //Created an ExpandedCompany for each company
      expandedCompanies = new TreeMap();
	  if(companies != null)
	  {
	  	  Iterator i = companies.iterator();
	  	  while(i.hasNext())
	  	  {
			  Company company = (Company) i.next();
			  ExpandedCompany ec = new ExpandedCompany(company, this, dbFacade);

			  expandedCompanies.put(new Integer(company.getCompanyID()), ec);
		  }
	  }
	  loadMethods();
	  loadNetworkDependencies();
	  loadMiscParameters();
	  loadLiquidatedDamagesParameters();
	  loadNegotiationParameters();
	  loadPersonnelParameters();

	  customer = new Customer(maxOverbillingPercent, overbillingPercentAffectingEthics);
  }


  /**
   * Starts the process of updating the period.
   */
  public void updatePeriod()
  {
	 //Perform Auto Billing on all jobs if that game component is turned on.
	 boolean autoBillingOn = false;

	 try
	 {
	   	autoBillingOn = dbFacade.getComponentsDBHandler().isAutoBillingComponentOn(game);
	 }
	 catch(BIGDatabaseFacadeException bigdbe)
	 {
		log.fatal("", bigdbe);
	 }

	 if(autoBillingOn)
	 {
	    doAutoBillingOnJobs();
	 }

   	 //Retrieve the Period object for this period from the database
   	 try
   	 {
   	 	currentPeriod = dbFacade.getPeriodDBHandler().getPeriod(game, game.getCurrentPeriodNumber());
   	 	nextPeriod = dbFacade.getPeriodDBHandler().getPeriod(game, game.getCurrentPeriodNumber() + 1);
	 }
	 catch(BIGDatabaseFacadeException bigdbe)
	 {
		 log.fatal("", bigdbe);
	 }
	 catch(ObjectNotFoundException onfe)
	 {
		 log.fatal("", onfe);
	 }

   	 //Create a new Period for period n+2.  n+2 needs to be created now so that in the next period, n+1,
   	 //companies will be able to view the consulting report values for the period after that
   	 //There will already be a Period in the database for n+1 but it is not used during this period's update
   	 createNewPeriod(game.getCurrentPeriodNumber() + 2);


	 makeCompaniesProcessLoans();
	 makeCompaniesAssignResponsibilityPenaltyForUnderbilling();
 	 makeCompaniesIncomingPayments();

	 // Determine which consulting reports each company purchased (if any)
	 // and subtract their costs from the company's cash on hand
	 processConsultingCosts();


	 //Companies must pay their current bills before bids are processed
	 makeCompaniesPayBills();

	 //Update the CCCs
	 updateCCCs();

	 // For each current job, generate CCC bids,
	 // determine the winning bid, and save the outcome codes of all bids
	 // to the DB.
	 processAllBids();

	 makeCompaniesApplyPersonnelToMosp();

	 if(gameType == 1 || gameType == 3)
	 {
        generateJobs(Job.PUBLIC);
 	 }

	 if(gameType == 2 || gameType == 3)
	 {
        generateJobs(Job.HEAVY);
 	 }

     makeCompaniesWorkOnePeriod();

	 makeCompaniesUpdatePersonnel();
	 makeCompaniesProcessCAMs();

	 //Increment the period number
   	 game.setCurrentPeriodNumber(game.getCurrentPeriodNumber() + 1);

	 //The period number has been incremented, check if this new period
	 //requires bond limit reevaluation
	 if((game.getCurrentPeriodNumber() % periodsBetweenBondLimitReeval) == 0)
	 {
		 makeCompaniesReevaluateBondLimits();
	 }

	 if(gameType == 1 || gameType == 3)
	 {
	    generateJobs(Job.PRIVATE);
	 }

	 //Save game changes
   	 try
   	 {
		 // Reset all overtime requests so they don't carry over to the next period
		 dbFacade.getOvertimeDBHandler().zeroOvertimeRequests(game.getGameID());

		 dbFacade.getGameDBHandler().updateGame(game);
	 }
	 catch (BIGDatabaseFacadeException bigdbe)
	 {
		 log.fatal("", bigdbe);
	 }
	 catch (ObjectNotFoundException onfe)
	 {
		 log.fatal("", onfe);
	 }

     //Finally e-mail everybody and let them know about the update
	 emailUpdateNotifications();
  }

  //Runs through the list of ExpandedCompany calling the doWorkOnJobs method for each company.
  private void makeCompaniesWorkOnePeriod()
  {
	 Date[] monthStartDates = game.getCurrentFirstOfMonths();

	 month1 = new ScheduledMonth(monthStartDates[0], currentPeriod.getWorkdaysFirstMonth(), currentPeriod.getRaindaysFirstMonth());
	 month2 = new ScheduledMonth(monthStartDates[1], currentPeriod.getWorkdaysSecondMonth(), currentPeriod.getRaindaysSecondMonth());

     //Modify method values based on the current period values for temperature, labor availability and materials cost index
     //This is not done until now, because these things don't factor into generating jobs
     applyPeriodValuesToMethods();

     //Have each company do work on each job it owns
     //Reset the iterator of ExpandedCompany
     Iterator i = expandedCompanies.values().iterator();
     i = expandedCompanies.values().iterator();
     while(i.hasNext())
     {
		 ExpandedCompany ec = (ExpandedCompany) i.next();
		 ec.chargeOfficeOverhead(fixedOfficeOverhead);
		 ec.doWorkOnJobs(month1, month2);
		 ec.saveChanges();
	 }

  }

  //Creates some new jobs and stores them to the database.  The number of new jobs
  //to be created is currently stored in the Period object for this period.
  //Type is either Job.PUBLIC, Job.PRIVATE or Job.HEAVY
  private void generateJobs(int type)
  {
	 //First get all the necessary parameters out of the database
	 Hashtable[] jobSizes = new Hashtable[5];
     int[][] percentTakeoffs = new int[5][9];
     int newJobs;

	 try
	 {
		 //Get the job size parameters -- Note: method getJobSizeParameters(Game game, int jobType)
		 //obtains values from table: game_params_job_size.  One of its fields is: job_type_id
         //there are ten possible values for job type (including the five for Heavy jobs)
         //jobType will be i+1 for Building jobs, i+6 for Heavy jobs
         if(type == Job.PUBLIC || type == Job.PRIVATE )
         {
		    for(int i = 0; i < 5; i++)
		    {
		 	   jobSizes[i] = dbFacade.getGameParametersDBHandler().getJobSizeParameters(game, i+1);
		    }
		 }
         if(type == Job.HEAVY )
         {
		    for(int i = 0; i < 5; i++)
		    {
		 	   jobSizes[i] = dbFacade.getGameParametersDBHandler().getJobSizeParameters(game, i+6);
		    }
		 }

		 //Get the percent takeoff parameters and convert them into ints
		 //same as above -- getPercentTakeoffParameters(Game game, int jobType)
         //jobType will be i+1 for Building jobs, i+6 for Heavy jobs
         if(type == Job.PUBLIC || type == Job.PRIVATE )
         {
		    for(int i = 0; i < 5; i++)
		    {
		        Hashtable jobTypePT = dbFacade.getGameParametersDBHandler().getPercentTakeoffParameters(game, i+1);

		        for(int j = 0; j < 9; j++)
		        {
				    percentTakeoffs[i][j] = ((Integer) jobTypePT.get("Activity" + (j+1))).intValue();
			    }
		    }
		 }
         if(type == Job.HEAVY )
         {
		    for(int i = 0; i < 5; i++)
		    {
		        Hashtable jobTypePT = dbFacade.getGameParametersDBHandler().getPercentTakeoffParameters(game, i+6);

		        for(int j = 0; j < 9; j++)
		        {
				    percentTakeoffs[i][j] = ((Integer) jobTypePT.get("Activity" + (j+1))).intValue();
			    }
		    }
		 }
	 }
	 catch (BIGDatabaseFacadeException bigdbe)
	 {
		 log.fatal("", bigdbe);
	 }
	 catch(ParametersNotFoundException pnfe)
	 {
		 log.fatal("", pnfe);
	 }

     Date[] monthStartDates = game.getFirstOfMonth(game.getCurrentPeriodNumber() + 1);

	 JobFactory jf = null;

	 if(type == Job.PRIVATE)
	 {
		Vector v = new Vector();
		Vector v1 = new Vector();
		v1.addAll(expandedCompanies.values());

		// limited to companies that have a "B" license -- otherwise, companies with only an "A" license
		// will receive invitations to bid on private jobs.

		// original code --> v.addAll(expandedCompanies.values());

	    //iterator.hasNext() returns true if the iteration has more elements
	    for (Iterator iterator = v1.iterator(); iterator.hasNext(); )
	    {
	        //iterator.next() returns the next element in the iteration
	        ExpandedCompany exco = (ExpandedCompany) iterator.next();
			int license = exco.getCompany().getLicense();
			if (license == 1 || (license == 0 && gameType == 1) || license == 3)
			{
				v.add(exco);
				log.debug("Company named " + exco.getCompany().getUsername() + " added to the vector." );
			}
	    }

	 	jf = new NegotiatedJobFactory(this,
			                          jobSizes,
			  		   			 	  percentTakeoffs,
							 		  methods,
							 		  networkDependencies,
							 		  liquidatedDamagesParameters,
							 		  game.getCurrentPeriodNumber(),
							 		  monthStartDates[0],
							 		  workdaysPerMonth,
							 		  stdHours,
							 		  overtimeHours,
							 		  customers,
							 		  negotiationParameters,
							 		  v);
	 }

	 if(type == Job.PUBLIC)
	 {
	 	jf = new JobFactory(this,
							jobSizes,
			  				 percentTakeoffs,
							 methods,
							 networkDependencies,
							 liquidatedDamagesParameters,
							 game.getCurrentPeriodNumber() + 1,
							 monthStartDates[0],
							 workdaysPerMonth,
							 stdHours,
							 overtimeHours,
							 0,
							 0,
							 1);
	 }

	 if(type == Job.HEAVY)
	 {
	 	jf = new JobFactory(this,
							jobSizes,
			  				 percentTakeoffs,
							 methodsHC,
							 networkDependenciesHC,
							 liquidatedDamagesParameters,
							 game.getCurrentPeriodNumber() + 1,
							 monthStartDates[0],
							 workdaysPerMonth,
							 stdHours,
							 overtimeHours,
							 0,
							 0,
							 2);
	 }

     if(type == Job.HEAVY)
     {
		 newJobs = nextPeriod.getNumberOfNewHeavyJobs();
	 }
	 else
	 {
		 newJobs = nextPeriod.getNumberOfNewBuildingJobs();
	 }

     //Have the job factory create each job.  Save each one to the database
     while(newJobs > 0)
     {
		 game.incrementCurrentJobNumber();
		 int nextJobID = game.getCurrentJobNumber();

		 Job job = jf.createJob(nextJobID);

		 storeNewJob(job);
         newJobs--;

	 }

  }

  //Creates a new Period object.  Pulls values from database as needed.
  private Period createNewPeriod(int periodNumber)
  {
	  //Mod the period number so that if it is greater than 12 it will wrap around to
	  //use game parameters from previous period
	  int paramsPeriodNumber = periodNumber % 12;
	  if (paramsPeriodNumber == 0) paramsPeriodNumber = 12;

      Hashtable materialsCostIndexParams = null;
      Hashtable laborAvailabilityParams = null;
      Hashtable numberOfNewJobsParams = null;
      Hashtable temperatureParams = null;
      Hashtable rainfallParams = null;
      Hashtable miscParams = null;

	  try
	  {
	  	  materialsCostIndexParams = dbFacade.getGameParametersDBHandler().getMaterialCostIndexParameters(game, paramsPeriodNumber);
	  	  laborAvailabilityParams = dbFacade.getGameParametersDBHandler().getLaborAvailabilityParameters(game, paramsPeriodNumber);
	  	  numberOfNewJobsParams = dbFacade.getGameParametersDBHandler().getNumberOfJobsParameters(game, paramsPeriodNumber);
	  	  temperatureParams = dbFacade.getGameParametersDBHandler().getTemperatureParameters(game, paramsPeriodNumber);
	  	  rainfallParams = dbFacade.getGameParametersDBHandler().getRainfallParameters(game, paramsPeriodNumber);
	  	  miscParams = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  bigdbe.printStackTrace();
	  }
	  catch(ParametersNotFoundException pnfe)
	  {
		  pnfe.printStackTrace();
	  }

	  Period period = new Period( /*Number of new jobs parameters */
	  						      ((Integer) numberOfNewJobsParams.get("Median")).intValue(),
	  						      ((Integer) numberOfNewJobsParams.get("StdDev")).intValue(),
	  						      ((Integer) numberOfNewJobsParams.get("Min")).intValue(),
	  							  ((Integer) numberOfNewJobsParams.get("Max")).intValue(),
	  							  ((Integer) numberOfNewJobsParams.get("MinHC")).intValue(),
								  ((Integer) numberOfNewJobsParams.get("MaxHC")).intValue(),
	  							  /* Temperature parameters */
	  							  ((Integer) temperatureParams.get("Median")).intValue(),
	  							  ((Integer) temperatureParams.get("StdDev")).intValue(),
	  							  ((Integer) temperatureParams.get("Min")).intValue(),
	  							  ((Integer) temperatureParams.get("Max")).intValue(),
	  							  /* Rainfall parameters */
	  							  ((Integer) rainfallParams.get("Median")).intValue(),
	  							  ((Integer) rainfallParams.get("StdDev")).intValue(),
	  							  ((Integer) rainfallParams.get("Min")).intValue(),
	  							  ((Integer) rainfallParams.get("Max")).intValue(),
	  							  /* Labor availability parameters */
	  							  ((Integer) laborAvailabilityParams.get("Median")).intValue(),
	  							  ((Integer) laborAvailabilityParams.get("StdDev")).intValue(),
	  							  ((Integer) laborAvailabilityParams.get("Min")).intValue(),
	  							  ((Integer) laborAvailabilityParams.get("Max")).intValue(),
	  							  /* Materials Cost Index parameters */
	  							  ((Integer) materialsCostIndexParams.get("Median")).intValue(),
	  							  ((Integer) materialsCostIndexParams.get("StdDev")).intValue(),
	  							  ((Integer) materialsCostIndexParams.get("Min")).intValue(),
	  							  ((Integer) materialsCostIndexParams.get("Max")).intValue(),
	  							  /* Number of hours per month from misc parameters */
	  							  ((Integer) miscParams.get("workdaysPerMonth")).intValue(),
	  							  /* Period number */
	  							  periodNumber);


	  try
	  {
		 dbFacade.getPeriodDBHandler().addPeriod(game, period);
	  }
	  catch (BIGDatabaseFacadeException bigdbe)
	  {
		 log.fatal("", bigdbe);
	  }
      return period;
  }

	/**
	 * Updates the database records associated with a <code>Vector</code> of
	 * <code>Bid</code> objects.  This method is used by <code>Job.determineBidWinners()</code>
	 * after bid outcome codes have been assigned.
	 */
	public void storeBids(Job job, Vector bids)
	{
		Iterator bidIterator = bids.iterator();
		while(bidIterator.hasNext()) {
			Bid bid = (Bid) bidIterator.next();
			Company dummyCompany = new Company();
			dummyCompany.setCompanyID(bid.getCompanyID());

			try
			{
				if (bid.getCompanyID() < 0)
				{ // CCC
					dbFacade.getBidDBHandler().addBid(game, job, dummyCompany, bid);
				}
				else
				{
					dbFacade.getBidDBHandler().updateBid(game, job, dummyCompany, bid);
				}
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
			    log.fatal("", bigdbe);
			}
			catch(ObjectNotFoundException onfe)
			{
			    log.fatal("", onfe);
		    }
		}
	}

  /**
   * Returns the total amount the bid price will be increased by due to a change order.
   * Assumes the fastest method is used with overtime for each activity.  A markup equal
   * to company X's bid markup is added to the cost as profit for the company.
   * @param lusPerActivity Int array of length 9, each index represents an activity.  The value for
   * each index is the amount of lus that will be added to that activity
   */
  protected long getChangeOrderPrice(int[] lusPerActivity)
  {
	  int fastestMethod = 4;

	  long changeOrderCost = 0;

	  for(int i = 0; i < 9; i++)
	  {
		  //Add labor cost assuming overtime
		  changeOrderCost += lusPerActivity[i] * methods[i][fastestMethod].getLaborCost() * overtimeModifier;
		  changeOrderCost += lusPerActivity[i] * methods[i][fastestMethod].getMaterialsCost();
		  changeOrderCost += lusPerActivity[i] * methods[i][fastestMethod].getSubcontractorCost() * overtimeModifier;
  	  }

  	  changeOrderCost = changeOrderCost + Math.round((((float) ((lowerPercentJobMarkup + upperPercentJobMarkup) / 2)) / 100f) * ((float) changeOrderCost));

	  return changeOrderCost;
  }
