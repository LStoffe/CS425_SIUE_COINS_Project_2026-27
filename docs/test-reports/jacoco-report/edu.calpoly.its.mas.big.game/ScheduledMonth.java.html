package edu.calpoly.its.mas.big.game;

import java.util.Calendar;
import java.util.GregorianCalendar;
import java.util.Date;
import java.util.Random;

import java.text.SimpleDateFormat;
import java.text.ParsePosition;

import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Represents a month that has had its workdays and raindays scheduled on certain days.
 * <br><br>
 * The amount of workdays in a BIG month is is set as a game parameter.
 * After a possible reduction for poor labor availability, there will be X workdays in a given month.  The total number of days
 * in the month will be greater than (or equal to) X by some amount. <code>ScheduledMonth</code> decides which of the
 * days in the month are part of the group of X workdays.  If X is larger than the number of days in the month then all days that month will be
 * workdays.
 * <br><br>
 * Each month also has a certain number of workdays that are also raindays.  Activities whose
 * current method is affected by rainfall will not be worked on raindays.  Depending on how many inches of rain occur in a given month
 * (determined by game parameters) a number of days, R, will be raindays as well as workdays.
 * <br><br>
 * A <code>ScheduledMonth</code> object will contain a calendar for a given month that marks which days are
 * workdays and which of those are raindays.  The algorithm used to determine which days are workdays will
 * first try to schedule days based on an internal order.  By default this order goes: Wed, Thurs, Tue, Fri, Mon, Sat, Sun.
 * In this case all Wednesday would be marked workdays, then all Thursdays, and so on until all workdays are scheduled.
 * Raindays are randomly scheduled amongst those days which are workdays.  Only workdays can be raindays
 * because BIG only cares about rain when it occurs on a workday.
 * <br><br>
 * <code>ScheduledMonth</code> is built around <code>java.util.GregorianCalendar</code> and is able to handle all
 * months handled by this class.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.7 $ $Date: 2002/05/02 23:59:36 $
 *
 */
public class ScheduledMonth
{
  private static final Log log = LogFactory.getLog(ScheduledMonth.class);

  private Calendar calendar;

  private int daysThisMonth;
  private int currentDay;

  boolean[] workdays;
  boolean[] raindays;

  private int workdaysToSchedule;
  private int raindaysToSchedule;

  int[] order;

  /**
   * Constructor for ScheduledMonth.
   * The current day will be set to the first day of the month.
   * @param date Date set to the first day of the month to be scheduled.  ie if the month to be
   *             scheduled is January 2001, date should be Jan. 1, 2001
   * @param workdays The number of workdays this month
   * @param raindays The number of raindays this month
   * <br><br><b>Precondition:</b>rainday must be less than or equal to workdays<br>
   */
  public ScheduledMonth(Date date, int workdays, int raindays)
  {
	  if(workdays < raindays) {
	      log.debug("Precondition not met in ScheduledMonth.  raindays must be <= workdays, clamping raindays to workdays");
	      raindays = workdays;
	  }

	  workdaysToSchedule = workdays;
	  raindaysToSchedule = raindays;

	  calendar = Calendar.getInstance();
	  calendar.setLenient(false);
	  calendar.setTime(date);
	  daysThisMonth = calendar.getActualMaximum(Calendar.DAY_OF_MONTH);
	  currentDay = 0; //0 is 1st index in array which corresponds to 1st of month

	  this.workdays = new boolean[daysThisMonth];
	  this.raindays = new boolean[daysThisMonth];

	  order = new int[7];
	  setOrderDefaults();

	  scheduleWorkdays();
	  scheduleRaindays();
  }

  /**
   * Sets the internal order which days of the week are scheduled first.  The ints in the array must correspond
   * to the static ints for days of the week in the java.util.Calendar class.
   * <br><br>
   * <b>Precondition:</b> The int array order must only contain ints that are used for days of the week in java.util.Calendar<br>
   * <b>Precondition:</b> Each int in the array can appear no more than once
   * <b>Precondition:</b> int[] order must be of length 7
   */
  public void setScheduleOrder(int[] order)
  {
	  this.order = order;
  }

  /**
   * Returns an int array listing the order in which days will be scheduled.  The ints in the array correspond to the static ints for
   * days of the week in the java.util.Calendar class
   */
  public int[] getScheduleOrder()
  {
	  return order;
  }

  /**
   * Returns true if the current day is a workday
   *
   */
  public boolean getWorkday()
  {
	  return workdays[currentDay];
  }

  /**
   * Returns true if the current day is a rainday.  Only workdays can be raindays.
   *
   */
  public boolean getRainday()
  {
	  return raindays[currentDay];
  }

  /**
   * Returns the current day's date as a <code>java.util.Date</code> object.
   *
   */
  public Date getDate()
  {
	  return calendar.getTime();
  }

  /**
   * Advances the current day to the next day in the month.
   * If there are no more days left in the month, does nothing
   */
  public void advanceDay()
  {

	  currentDay++;

      if (currentDay < daysThisMonth)
      {
	      calendar.set(Calendar.DAY_OF_MONTH, (currentDay + 1));

	  }


  }

  /**
   * Returns true if there are days left in the current month.  False if the current days
   * is the last day of the month.
   */
  public boolean daysLeft()
  {
      if (currentDay < daysThisMonth)
         return true;
      else
         return false;
  }

  /**
   * Sets the current day to the first day in the month.
   *
   */
  public void firstDay()
  {
	  currentDay = 0; //0 is 1st index in array which corresponds to 1st of month
	  calendar.set(Calendar.DAY_OF_MONTH, 1);
  }


  //Schedules which days are to be workdays
  private void scheduleWorkdays()
  {
	  int i;
	  int j = 0;

      //While their are still days of the week left and there are still workdays to schedule
	  while((j < 7) && (workdaysToSchedule > 0))
	  {
		  //Set the day type (Monday, Tuesday, etc) to be scheduled
	  	  int day_being_scheduled = order[j];
	      i = 1;

	      //While there are still days left in the month
          while((i <= daysThisMonth) && (workdaysToSchedule > 0))
          {
              calendar.set(Calendar.DAY_OF_MONTH, i);
	          int day = calendar.get(Calendar.DAY_OF_WEEK);

	          if(day == day_being_scheduled)
	          {
		          workdays[i - 1] = true;
		          workdaysToSchedule--;
	          }
	          i++;
  	      }
  	      j++;
	  }
	  calendar.set(Calendar.DAY_OF_MONTH, 1); //reset the calendar back to the first day of the month

  }

  //Schedule raindays random amongst the workdays
  private void scheduleRaindays()
  {
	  Random random = new Random();

	  int i = 0;

	  while (raindaysToSchedule > 0)
	  {
	      i = random.nextInt(daysThisMonth);
	      if ((workdays[i] == true) && !(raindays[i] == true)) //the day is a workday and not already a rainday
	      {
			  raindays[i] = true;
			  raindaysToSchedule--;
		  }
	  }
  }


  //Populates order with a default order of days
  private void setOrderDefaults()
  {
	  order[0] = Calendar.WEDNESDAY;
	  order[1] = Calendar.THURSDAY;
	  order[2] = Calendar.TUESDAY;
	  order[3] = Calendar.FRIDAY;
	  order[4] = Calendar.MONDAY;
	  order[5] = Calendar.SATURDAY;
	  order[6] = Calendar.SUNDAY;
  }


  /**
   * Unit test for scheduled month
   */
  public static void main(String args[])
  {
	  SimpleDateFormat formatter = new SimpleDateFormat("MM/dd/yyyy");

	  Date date = formatter.parse("1/1/2004", new ParsePosition(0));
	  ScheduledMonth scheduledMonth = new ScheduledMonth(date, 20, 3);

	  int i = 0;
	  while(scheduledMonth.daysLeft())
	  {
		  String today = formatter.format(scheduledMonth.getDate());
		  i++;
		  if(scheduledMonth.getWorkday())
		      System.out.print(today + ": Workday: ");
		  else
		      System.out.print(today + ": Not a workday: ");
		  if(scheduledMonth.getRainday())
		      System.out.println("Rainday");
		  else
		  	  System.out.println("Not a rainday");
		  scheduledMonth.advanceDay();
	  }
	  scheduledMonth.firstDay();
	  while(scheduledMonth.daysLeft())
	  {
		  String today = formatter.format(scheduledMonth.getDate());
		  i++;
		  if(scheduledMonth.getWorkday())
		      System.out.print(today + ": Workday: ");
		  else
		      System.out.print(today + ": Not a workday: ");
		  if(scheduledMonth.getRainday())
		      System.out.println("Rainday");
		  else
		  	  System.out.println("Not a rainday");
		  scheduledMonth.advanceDay();
	  }
  }

}
