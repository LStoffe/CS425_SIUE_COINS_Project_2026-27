package edu.calpoly.its.mas.big.game;

import java.lang.String;
import java.text.NumberFormat;
import java.text.ParseException;
import java.util.Locale;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Converts money amounts, in the form of ints/longs, into strings in the format
 * used to display money amounts in BIG (and vice-versa).
 *
 * @author Kevin Craig
 * @version $Revision: 1.5 $ $Date: 2002/02/07 00:43:09 $
 */

public class NumberConverter
{
	private static NumberFormat nf;
	private static Log log;

	static
	{
		log = LogFactory.getLog("edu.calpoly.its.mas.big.game.NumberConverter");
	}

    /**
     * Converts a float amount to a string with two decimal places
     * added by me in November 2014
     */
	public static String stringFloatTo2Decimals(float number)
	{
        return String.format("%.2f", number);
    }

    /**
     * Converts an int money amount to a string in the format "$1,234"
     */
	public static String stringValue(int money)
    {
       	prepareNumberFormat();
		return nf.format(money);
    }

	/**
     * Converts a long money amount to a string in the format "$1,234"
     */
	public static String stringValue(long money)
    {
       	prepareNumberFormat();
		return nf.format(money);
    }

    /**
     * Converts a long number amount into a string with commas
     */
    public static String stringValueCommasOnly(long number)
    {
		boolean negative = false;

		if(number < 0)
		   negative = true;

		number = Math.abs(number);

	    nf = NumberFormat.getInstance();
		nf.setParseIntegerOnly(true);
        nf.setMaximumFractionDigits(0);

        String retValue = nf.format(number);

        if(negative)
        {
        	retValue = "(" + retValue + ")";
		}

        return retValue;
	}

    /**
     * Converts a String, in the money format used by BIG, into an int.
     * The following String formats are considered valid: "1234"
     * "1,234" "$1234" "1,234" <br>
     * <br>Strings with decimal digits will be accepted but truncated to integers.
     * <br>-1 is returned for invalid strings.
     */
    public static int intValue(String money)
    {
		if(log.isDebugEnabled())
		{
			log.debug("converting " + money + " to int");
		}

		boolean negative = isNegative(money);
		if(negative)
		    money = removeNegatingCharacters(money);


        //Add a '$' to the start of the string is there isn't one
        int k = money.indexOf('$');
        if(k != 0)
        {
           	money = "$" + money;
        }
		int value = -1;
		prepareNumberFormat();
		try
		{
			Long l = (Long) nf.parse(money);
                        value = l.intValue();
		}
		catch(ParseException pe)
		{
			value = -1;
		}

		if(negative)
		{
			value = -value;
		}

		if(log.isDebugEnabled())
		{
			log.debug("int value " + value);
		}
		return value;
    }

    /**
     * Converts a String, in the money format used by BIG, into a long.
     * The following String formats are considered valid: "1234"
     * "1,234" "$1234" "$1,234" <br>
     * <br>Strings with decimal digits will be accepted but truncated to integers.
     * <br>-1 is returned for invalid strings.
     */
    public static long longValue(String money)
    {
		if(log.isDebugEnabled())
		{
			log.debug("converting " + money + " to long");
		}

		boolean negative = isNegative(money);
		if(negative)
		    money = removeNegatingCharacters(money);

        //Add a '$' to the start of the string is there isn't one
        int k = money.indexOf('$');
        if(k != 0)
        {
           	money = "$" + money;
        }
		long value = -1;
		prepareNumberFormat();
		try
		{
			Long l = (Long) nf.parse(money);
                        value = l.longValue();
		}
		catch(ParseException pe)
		{
			if (log.isDebugEnabled())
				log.debug("Unparseable number: " + money, pe);
			value = -1;
		}

		if(negative)
		{
			value = -value;
		}

		if(log.isDebugEnabled())
		{
			log.debug("long value " + value);
		}
		return value;
    }

    /**
     * Returns true if the given String is in a money format known to
     * BIG
     */
    public static boolean isValidMoneyString(String money)
    {
		if(log.isDebugEnabled())
		{
			log.debug("checking if " + money + " is valid money string");
		}

		boolean valid = true;

		money = removeNegatingCharacters(money);

        //Add a '$' to the start of the string is there isn't one
        int k = money.indexOf('$');
        if(k != 0)
        {
           	money = "$" + money;
        }

		prepareNumberFormat();
		try
		{
			Number n = nf.parse(money);
		}
		catch(ParseException pe)
		{
			valid = false;
		}
		return valid;
    }

	private static void prepareNumberFormat()
    {
        Locale.setDefault(Locale.US);
       	nf = NumberFormat.getCurrencyInstance();
		nf.setParseIntegerOnly(true);
        nf.setMaximumFractionDigits(0);
	}

	//Determines whether or not the String parameter represents a
	//negative amount of money (if the String represents a money amount at all).
	//There are two possible indicators that an amount of money is negative.  The
	//first is the first character in the String being a hypen ('-').
	//The second indicator is the first character being a left parentheses ('(').
	//If either of this characters is the first character the money amount is considered
	//negative
	//Return true if the string is negative, false if positive
	private static boolean isNegative(String money)
	{
		if(money.indexOf('-') == 0)
		{
			if(log.isDebugEnabled())
			{
				log.debug(money + " is negative");
			}
			return true;
	  	}
		if(money.indexOf('(') == 0)
		{
			if(log.isDebugEnabled())
			{
				log.debug(money + " is negative");
			}
		    return true;
		}

		return false;
	}


	//Returns a String where characters that indicate the amount of money is negative have been removed.
	//If the first character is a '-' is it removed.

	//If the first character is a '(' is it removed.
	//If the first character is a '(' and the last character is a ')' they are both removed
	private static String removeNegatingCharacters(String money)
	{
		if(money.indexOf('-') == 0)
		  return money.substring(1, money.length());
		if(money.indexOf('(') == 0)
		{
			if(money.indexOf(')') == money.length()-1)
			{
			  return money.substring(1, money.length()-1);
			}
			else
			  return money.substring(1, money.length());
		}

		if(log.isDebugEnabled())
		{
			log.debug("After removing negating characters, string is " + money);
		}

		return money;
	}
}
