package edu.calpoly.its.mas.big.db;

//iBATIS imports
import com.ibatis.sqlmap.client.SqlMapClient;
import com.ibatis.sqlmap.client.SqlMapClientBuilder;
import com.ibatis.common.resources.Resources;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * A connection to the database using iBatis' SqlMapClient should only be established <b>once</b> in an application.
 * Each time SqlMapClientBuilder is used to build an SqlMapClient, a new database connection pool is created.
 * This class works as a singleton pattern (used to restrict instantiation of a class to one object).
 * <br><br>
 * See: http://en.wikipedia.org/wiki/Singleton_pattern
 */

public class BIGGetSqlMapClient
{
  private static BIGGetSqlMapClient singleton = null;
  private static SqlMapClient sqlMap;
  private static Log log;

  protected BIGGetSqlMapClient()
  {
    // Exists only to defeat instantiation.
  }

  public static BIGGetSqlMapClient getInstance()
  {
     if(singleton == null)
     {
        log = LogFactory.getLog("edu.calpoly.its.mas.big.db.BIGGetSqlMapClient");
	 	try
	 	{
			sqlMap = SqlMapClientBuilder.buildSqlMapClient(Resources.getResourceAsReader("sqlMaps.xml"));
	 	}
	 	catch(Exception e)
	 	{
			log.fatal("Class = edu.calpoly.its.mas.big.db.BIGGetSqlMapClient: Error creating iBatis SqlMapClient", e);
	 	}
        singleton = new BIGGetSqlMapClient();

		if(log.isDebugEnabled())
		{
			log.debug("Class = edu.calpoly.its.mas.big.db.BIGGetSqlMapClient --> iBATIS SqlMapClient created successfully");
		}
     }
     return singleton;
  }

  public static SqlMapClient sqlMap()
  {
	 return sqlMap;
  }
}
