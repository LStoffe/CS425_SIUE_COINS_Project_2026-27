package edu.calpoly.its.mas.big.db;

import edu.calpoly.its.mas.big.game.Method;

import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.Job;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>CurrentMethodDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.13 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultCurrentMethodDBHandler extends DefaultDBHandler implements CurrentMethodDBHandler
{

    public DefaultCurrentMethodDBHandler(Connection connection)
    {
		super(connection);
	}

    public int[] getCurrentMethods(Game game, Job job) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {

        int[] methods = new int[9]; //9 activities

		//SQL statement to retrieve the game matching the given gameID
		String query = "SELECT method_id FROM current_methods cm WHERE cm.gameid = ? AND cm.jobid =? AND cm.activity_type_id = ?";

        for (int i = 0; i < 9; i ++)
        {
  			try
  			{
				PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, job.getJobID());
				pstmt.setInt(3, i+1);

				ResultSet rset = pstmt.executeQuery();

				//Check if no rows are returned
				//(rset.last() returns false if there are no rows in the result set)
    		    if(!rset.last())
    		    {
    		 	   throw new ObjectNotFoundException();
    		    }
    		    else
    		    {
    		       rset.first();

    		       methods[i] = rset.getInt("method_id");

    		    }
    		    rset.close();
    		    pstmt.close();
			}
			catch(SQLException sqle)
			{
				//if an exception occurs close the connection
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}
		}
		return methods;
	}


    public Method[] getAllMethods(Game game, int activityType) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
        Method[] methods = new Method[5]; //5 methods for each activity

		//SQL statement to retrieve the game matching the given gameID
		String query = "SELECT methodid, method_name, productivity, laborcost, " +
		                       " materialcost, equipmentcost, subcontractorcost, uncertainty_range, weather_sensitivity, subcontracted " +
		                       " FROM method WHERE gameid = ? AND activity_type_id = ? ORDER BY methodid ";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(query);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, activityType);

			ResultSet rset = pstmt.executeQuery();

      	                int i = 0;
    	                while(rset.next())
    	                {
                 	     methods[i] = new Method(rset.getInt("methodid"),
                                                 rset.getString("method_name"),
    	        	                             rset.getInt("productivity"),
    	        	                             rset.getInt("laborcost"),
    	        	                             rset.getInt("materialcost"),
    	        	                             rset.getInt("equipmentcost"),
    	        	                             rset.getInt("subcontractorcost"),
    	        	                             rset.getInt("uncertainty_range"),
    	        	                             rset.getInt("weather_sensitivity"),
    	        	                             rset.getBoolean("subcontracted"));
    	                     i++;
			}
                        if (i < 4)
                            throw new ObjectNotFoundException();
        		rset.close();
    		        pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return methods;
	}


    public void updateCurrentMethods(Game game, Job job, int[] methods) throws BIGDatabaseFacadeException
    {
		String update = "UPDATE current_methods SET method_id = ? WHERE gameid = ? AND jobid = ? AND activity_type_id = ? ";
		for (int i = 0; i < 9; i++)
		{
			try
 			{
				PreparedStatement pstmt = connection.prepareStatement(update);
				pstmt.setInt(1, methods[i]);
				pstmt.setInt(2, game.getGameID());
				pstmt.setInt(3, job.getJobID());
				pstmt.setInt(4, i+1);

                int rows = pstmt.executeUpdate();
                connection.commit();

    		    pstmt.close();
			}
			catch(SQLException sqle)
			{
				//if an exception occurs close the connection
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}
		}
	}


    public void addCurrentMethods(Game game, Job job, int[] methods) throws BIGDatabaseFacadeException
	{
		String update = "INSERT INTO current_methods (gameid, jobid, activity_type_id, method_id) VALUES (?, ?, ?, ?)";
		for (int i = 0; i < 9; i++)
		{
			try
 			{
				PreparedStatement pstmt = connection.prepareStatement(update);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, job.getJobID());
				pstmt.setInt(3, i+1);
				pstmt.setInt(4, methods[i]);

                int rows = pstmt.executeUpdate();
                connection.commit();

    		    pstmt.close();
			}
			catch(SQLException sqle)
			{
				//if an exception occurs close the connection
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}
		}
	}


    public void removeCurrentMethods(int gameID) throws BIGDatabaseFacadeException
    {
		String remove = "DELETE FROM current_methods WHERE gameid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameID);

			int rows = pstmt.executeUpdate();
			connection.commit();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}
}
