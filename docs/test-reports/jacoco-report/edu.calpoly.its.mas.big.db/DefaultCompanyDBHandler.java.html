    public Vector getCompanies(Game game) throws BIGDatabaseFacadeException
    {
		Vector companies = new Vector();

		try
		{
			//Select all companies in this game
			String query = "SELECT c.companyID, c.title, c.username, c.password, c.cash_on_hand, b.equity, c.wip_limit, c.wip_limit_percent, c.per_job_limit, c.per_job_wip_percent, c.wip, c.license, c.missionstatement, c.corevalues, c.companyurl " +
			       		   "FROM company c, game g, balance_sheet b " +
			       		   "WHERE c.gameID= ? AND c.gameid = g.gameid " +
			               "AND g.gameid = b.gameid AND c.companyid = b.companyid " +
			               "AND ((b.periodid = (g.current_period - 1) AND g.current_period > 0) OR (g.current_period = b.periodid AND g.current_period = 0)) " +
			       		   "ORDER BY c.title";

			PreparedStatement pstmt = connection.prepareStatement(query);
			pstmt.setInt(1, game.getGameID());

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				//Create a new company and store it in the Vector
				Company tempCompany = new Company(rset.getInt("companyID"),
												  rset.getString("title").trim(),
												  rset.getString("username").trim(),
												  rset.getString("password").trim(),
												  rset.getLong("cash_on_hand"),
												  rset.getLong("equity"),
												  rset.getInt("wip"),
												  rset.getLong("wip_limit"),
												  rset.getInt("wip_limit_percent"),
												  rset.getLong("per_job_limit"),
												  rset.getInt("per_job_wip_percent"),
												  rset.getInt("license"),
												  rset.getString("missionstatement"),
												  rset.getString("corevalues"),
												  rset.getString("companyurl")
												  );
				companies.add(tempCompany);
			}
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//Close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return companies;
    }

    public void updateCompany(Game game, Company theCompany) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		//Create the SQL string used for the update
		String update = "UPDATE company SET username = ?, password = ?, title = ?, cash_on_hand = ?, wip_limit_percent = ?, per_job_limit = ?, per_job_wip_percent = ?, wip_limit = ?, wip = ?, license = ?, missionstatement = ?, corevalues = ?, companyurl = ? " +
				" WHERE gameid = ? AND companyid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setString(1, theCompany.getUsername());
			pstmt.setString(2, theCompany.getPassword());
			pstmt.setString(3, theCompany.getTitle());
			pstmt.setLong(4, theCompany.getCashOnHand());
			pstmt.setInt(5, theCompany.getWIPLimitPercent());
			pstmt.setLong(6, theCompany.getPerJobLimit());
			pstmt.setInt(7, theCompany.getPerJobLimitAsWIPPercent());
			pstmt.setLong(8, theCompany.getWIPLimit());
			pstmt.setInt(9, theCompany.getWIP());
			pstmt.setInt(10, theCompany.getLicense());
			pstmt.setString(11, theCompany.getMissionStatement());
			pstmt.setString(12, theCompany.getCoreValues());
			pstmt.setString(13, theCompany.getCompanyUrl());
			pstmt.setInt(14, game.getGameID());
			pstmt.setInt(15, theCompany.getCompanyID());

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

			if (rowsChanged < 1)
			   throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)
		{
			//Close connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    public void addCompany(Game theGame, Company theCompany) throws BIGDatabaseFacadeException
    {
		int companyID = -1;

		try
		{
			//Get the next companyID in the sequence
			String sequenceName = "big_companyid_sequence";
            String query = "SELECT nextval ('" + sequenceName + "')";

            PreparedStatement stmt = connection.prepareStatement(query);

			ResultSet rset = stmt.executeQuery();

			if(rset.next()) //A companyID value was returned from the database
			{
				companyID = rset.getInt(1);

			}
			rset.close();
			stmt.close();

			if(companyID >= 0)
			{
				//Assign the new companyID to the Company to be stored
				theCompany.setCompanyID(companyID);

				//Create the SQL statement used to insert the new admin into the database.
				String insert = "INSERT INTO company (companyID, gameID, username, password, title, cash_on_hand, wip_limit, wip_limit_percent, wip, per_job_limit, per_job_wip_percent, license, missionstatement, corevalues, companyurl ) ";
				insert = insert + "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

				//Perform the insert
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, companyID);
				pstmt.setInt(2, theGame.getGameID());
				pstmt.setString(3, theCompany.getUsername());
				pstmt.setString(4, theCompany.getPassword());
				pstmt.setString(5, theCompany.getTitle());
				pstmt.setLong(6, theCompany.getCashOnHand());
				pstmt.setLong(7, theCompany.getWIPLimit());
				pstmt.setInt(8, theCompany.getWIPLimitPercent());
				pstmt.setInt(9, theCompany.getWIP());
				pstmt.setLong(10, theCompany.getPerJobLimit());
				pstmt.setInt(11, theCompany.getPerJobLimitAsWIPPercent());
				pstmt.setInt(12, theCompany.getLicense());
				pstmt.setString(13, theCompany.getMissionStatement());
				pstmt.setString(14, theCompany.getCoreValues());
				pstmt.setString(15, theCompany.getCompanyUrl());

				int rowsAdded = pstmt.executeUpdate();
				connection.commit();

				pstmt.close();

			}
			else  //A companyID was not returned from the database
			{
				throw new BIGDatabaseFacadeException("Couldn't retrieve next value from " + sequenceName );
			}
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

    public void removeCompany(int gameID, int companyID) throws BIGDatabaseFacadeException
    {
        //Remove all members names in this company
        (new DefaultMemberNamesDBHandler(connection)).removeMemberNames(gameID, companyID);

        //Remove all loans owned by this company
        (new DefaultLoanDBHandler(connection)).removeLoans(companyID);

		//Bids owned by this company
	    (new DefaultBidDBHandler(connection, sqlMap)).removeBids(companyID);

		//Remove contracts reports
		(new edu.calpoly.lib.multimedia.big.db.DefaultContractsReportsDBHandler(connection)).removeContractsInfo(gameID, companyID);

	    //Remove BalanceSheets for this company
	    (new edu.calpoly.lib.multimedia.big.db.DefaultBalanceSheetDBHandler(connection)).removeBalanceSheets(gameID, companyID);

		//Remove cash flow reports
		(new edu.calpoly.lib.multimedia.big.db.DefaultCashFlowDBHandler(connection)).removeCashFlowReports(gameID, companyID);

		//Remove billings
		(new edu.calpoly.lib.multimedia.big.db.DefaultBillingDBHandler(connection)).removeBillingForms(gameID, companyID);

		//Remove ratios
		(new edu.calpoly.lib.multimedia.big.db.DefaultRatiosDBHandler(connection)).removeRatios(gameID, companyID);

		//Remove appraisal metrics updates
		(new edu.calpoly.lib.multimedia.big.db.DefaultCAMDBHandler(connection)).deleteCAMUpdateRecord(companyID);

		//Remove appraisal metrics
		(new edu.calpoly.lib.multimedia.big.db.DefaultCAMDBHandler(connection)).removeCAMs(companyID);

		//Remove inbox messages
		(new edu.calpoly.lib.multimedia.big.db.DefaultInboxDBHandler(connection)).deleteMessages(companyID);

        //SQL statement to remove a company from the database
		String update = "DELETE FROM company WHERE gameID = ? AND companyID = ?";

		try
		{

   		    PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, gameID);
            pstmt.setInt(2, companyID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

    public void removeCompanies(int gameID) throws BIGDatabaseFacadeException
    {
		//Note: rows in other tables related to the company must also be deleted, so the multiple calls are made to the removeCompany() method
		//which handles this

		//Construct a Game with the given gameID
		Game game = new Game(gameID, 0, null, null, 0, false, 0, 0);

		//Get all companies in this game
		Vector companies = getCompanies(game);

		Iterator companyIterator = companies.iterator();

		//Iterate over the list of companies, removing each one
		while(companyIterator.hasNext())
		{
			Company nextCompany = (Company) companyIterator.next();
			removeCompany(gameID, nextCompany.getCompanyID());
		}

	}
