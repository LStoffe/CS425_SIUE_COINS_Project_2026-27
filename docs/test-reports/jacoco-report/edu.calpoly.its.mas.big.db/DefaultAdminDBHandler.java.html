package edu.calpoly.its.mas.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.game.Game;

//JDK imports
import java.util.Vector;

//JDBC imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

//iBATIS imports
import com.ibatis.sqlmap.client.SqlMapClient;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Default implementation of <code>AdminDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.2 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultAdminDBHandler extends DefaultDBHandler implements AdminDBHandler
{
	private Log log;
	private SqlMapClient sqlMap;

    public DefaultAdminDBHandler(Connection connection, SqlMapClient sqlMap)
    {
		super(connection);
		this.sqlMap = sqlMap;
		log = LogFactory.getLog(this.getClass());
	}

  	public Admin getAdmin(String adminName) throws BIGDatabaseFacadeException
  	{
		Admin theAdmin = null;

		try
		{
			String query = "SELECT adminID, password, email, is_head_admin FROM admin WHERE username = ?";

		    PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
		    pstmt.setString(1, adminName);

		    ResultSet rset = pstmt.executeQuery();

		  	// Check if no rows are returned
		   	// (rset.last() returns false if there are no rows in the result set)
		   	if(!rset.last())
		   	{
		   	}
  	     	// A record exists for this admin username
		   	else
		   	{
		   		  rset.first();
		   		  theAdmin = new Admin(rset.getInt("adminID"), adminName, rset.getString("password").trim(), rset.getString("email"), rset.getBoolean("is_head_admin"));
		   	}
       		//done with the result set
	   		rset.close();
	   		pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
  		}
    	return theAdmin;
    }

    public Vector getAdmins() throws BIGDatabaseFacadeException
    {
	  	Vector admins = new Vector();

	  	try
	    {
		  	  //The head admin should not be included in the search results
		  	  String query = "SELECT adminID, username, email FROM admin WHERE is_head_admin = FALSE ORDER BY username";

		  	  PreparedStatement pstmt = connection.prepareStatement(query);

			  ResultSet rset = pstmt.executeQuery();

			  while(rset.next())
			  {
				  //Create a new admin and store it in the Vector
				  Admin tempAdmin =  new Admin(rset.getInt("adminID"), rset.getString("username"), null, rset.getString("email"), false);
				  admins.add(tempAdmin);
			  }
			  rset.close();
			  pstmt.close();
	  	}
	  	catch(SQLException sqle)
	  	{
			  //close the connection if an exception occurs
			  close();
			  throw new BIGDatabaseFacadeException(sqle);
	  	}
		return admins;
  	}

    public void updateAdmin(Admin theAdmin) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		//SQL statement to update password field of admin
		String update = "UPDATE admin SET password = ?, email = ? WHERE adminid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setString(1, theAdmin.getPassword());
			pstmt.setString(2, theAdmin.getEmail());
			pstmt.setInt(3, theAdmin.getAdminID());

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

			if (rowsChanged < 1)
				throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)
		{
			//Close connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

    public void addAdmin(Admin theAdmin) throws BIGDatabaseFacadeException
    {
		int adminID;

		try
		{
			//Get the next adminID in the sequence
			String sequenceName = "big_adminid_sequence";
            String query = "SELECT nextval ('" + sequenceName + "')";

			PreparedStatement stmt = connection.prepareStatement(query);
			ResultSet rset = stmt.executeQuery();

			if(rset.next()) //An adminID value was returned from the database
			{
				adminID = rset.getInt(1);

				rset.close();

				//Assign the new adminID to the Admin to be stored
				theAdmin.setAdminID(adminID);

				//Create the SQL statement used to insert the new admin into the database.
				//The value for is_head_admin is hardcoded to false because only normal admins can be added using this method
				String insert = "INSERT INTO admin (adminID, username, password, email, is_head_admin) ";
				insert = insert + "VALUES (?, ?, ?, ?, FALSE)";

				//Perform the insert
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, adminID);
				pstmt.setString(2, theAdmin.getUsername());
				pstmt.setString(3, theAdmin.getPassword());
				pstmt.setString(4, "");

				int rowsAdded = pstmt.executeUpdate();
				connection.commit();

				pstmt.close();

			}
			else  //An adminID was not returned from the database
			{
				rset.close();
				close();
				throw new BIGDatabaseFacadeException("Couldn't retrieve next value from " + sequenceName );
			}

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    public void removeAdmin(int adminID) throws BIGDatabaseFacadeException
    {
		//Remove all games belonging to this admin
		(new DefaultGameDBHandler(connection, sqlMap)).removeGames(adminID);

		//Remove this admin's set of default parameters
		(new DefaultGameParametersDBHandler(connection, sqlMap)).removeAdminDefaultParameters(adminID);

		//String to remove the admin matching the ID number passed in
		String update = "DELETE FROM admin WHERE adminID = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, adminID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

}
