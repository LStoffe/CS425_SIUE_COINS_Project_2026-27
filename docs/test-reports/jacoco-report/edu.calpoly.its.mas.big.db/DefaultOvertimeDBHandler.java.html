package edu.calpoly.its.mas.big.db;

import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Game;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>OvertimeDBHandler</code>.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.3 $ $Date: 2002/03/14 19:08:41 $
 */
public class DefaultOvertimeDBHandler extends DefaultDBHandler implements OvertimeDBHandler
{

    public DefaultOvertimeDBHandler(Connection connection)
    {
		super(connection);
	}

    public int getOvertime(Game game, Job job, int activityType)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
        int overtimeDays = 0;

        // activity types must be 10-18 for Heavy jobs
        int tempType = activityType;
        if (job.getTypeModifier() == Job.HEAVY) tempType = tempType + 9;

        String query = "SELECT overtime_days " +
                       "FROM overtime " +
			"WHERE gameid = ? AND jobid = ? AND activity_type_id = ?";

        try {
            PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
            pstmt.setInt(1, game.getGameID());
            pstmt.setInt(2, job.getJobID());
            pstmt.setInt(3, tempType);

            ResultSet rset = pstmt.executeQuery();

            if(!rset.last()) {
                //Close the database connection if the parameters aren't found.
                close();
                throw new ObjectNotFoundException();
            } else {
                rset.first();
                overtimeDays = rset.getInt("overtime_days");
            }

            rset.close();
            pstmt.close();

        } catch (SQLException sqle) {
            // close the connection
            close();
            throw new BIGDatabaseFacadeException(sqle);
        }

        return overtimeDays;
    }

    public int[] getOvertimes(Game game, Job job) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
        int[] overtimeDays = new int[9];

        String query = "SELECT overtime_days, activity_type_id " +
                       "FROM overtime " +
			"WHERE gameid = ? AND jobid = ?";

        try {
            PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
            pstmt.setInt(1, game.getGameID());
            pstmt.setInt(2, job.getJobID());

            ResultSet rset = pstmt.executeQuery();

            // Is it safe to expect nine rows?
            while(rset.next()) {
				if(job.getTypeModifier() == Job.PUBLIC || job.getTypeModifier() == Job.PRIVATE)
				{
                   overtimeDays[rset.getInt("activity_type_id") - 1] = rset.getInt("overtime_days");
			    }
			    if(job.getTypeModifier() == Job.HEAVY)
				{
				   overtimeDays[rset.getInt("activity_type_id") - 10] = rset.getInt("overtime_days");
				}
            }

            rset.close();
            pstmt.close();

        } catch (SQLException sqle) {
            // close the connection
            close();
            throw new BIGDatabaseFacadeException(sqle);
        }

        return overtimeDays;
    }


    public void addOvertime(Game game, Job job) throws BIGDatabaseFacadeException
    {

        //SQL statement to update overtime days
        String update = "INSERT INTO overtime (gameid, jobid, activity_type_id, overtime_days) VALUES (?, ?, ?, ?)";

        int tempIndex = -1;
        try
			{
				for(int i = 0; i < 9; i++)
					{
						// activity types must be 10-18 for Heavy jobs
						tempIndex = i;
						if (job.getTypeModifier() == Job.HEAVY) tempIndex = tempIndex + 9;

						PreparedStatement pstmt = connection.prepareStatement(update);
						pstmt.setInt(1, game.getGameID());
						pstmt.setInt(2, job.getJobID());
						pstmt.setInt(3, tempIndex + 1);
						pstmt.setInt(4, 0);

						int rowsChanged = pstmt.executeUpdate();
						pstmt.close();
					}
			}
        catch(SQLException sqle)
			{
				//Close connection if an exception occurs
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}

    }


    public void updateOvertime(Game game, Job job, int activityType, int overtimeDays)
        throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
        // activity types must be 10-18 for Heavy jobs
        int tempType = activityType;
        if (job.getTypeModifier() == Job.HEAVY) tempType = tempType + 9;

        //SQL statement to update overtime days
        String update = "UPDATE overtime " +
                        "SET overtime_days = ? " +
			"WHERE gameid = ? AND jobid = ? AND activity_type_id = ?";

        try
			{
				PreparedStatement pstmt = connection.prepareStatement(update);
				pstmt.setInt(1, overtimeDays);
				pstmt.setInt(2, game.getGameID());
				pstmt.setInt(3, job.getJobID());
				pstmt.setInt(4, tempType);

				int rowsChanged = pstmt.executeUpdate();

				pstmt.close();

				if (rowsChanged < 1)
					throw new ObjectNotFoundException();
			}
        catch(SQLException sqle)
			{
				//Close connection if an exception occurs
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}

    }


    public void updateOvertimes(Game game, Job job, int[] overtimeDays)
        throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
        for (int activityNumber = 1; activityNumber <= 9; activityNumber++) {
            updateOvertime(game, job, activityNumber, overtimeDays[activityNumber - 1]);
        }

    }


    public void removeOvertimeRequests(int gameID) throws BIGDatabaseFacadeException
	{
		String delete = "DELETE FROM overtime WHERE gameid = ?";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(delete);
			pstmt.setInt(1, gameID);

			pstmt.executeUpdate();
			connection.commit();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}


    public void zeroOvertimeRequests(int gameID)  throws BIGDatabaseFacadeException
	{
		String update = "UPDATE overtime SET overtime_days = 0 WHERE gameid = ?";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, gameID);

			pstmt.executeUpdate();
			connection.commit();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}


}
