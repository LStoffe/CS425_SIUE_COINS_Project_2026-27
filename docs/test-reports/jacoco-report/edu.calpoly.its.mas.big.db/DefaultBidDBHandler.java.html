	public Bid getBid(Game game, Job job, Company company)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		Bid bid = null;

		String query = "SELECT b.bid_amount, b.bid_amount_up, b.unitprice1, b.unitprice2, b.unitprice3, b.unitprice4, b.unitprice5, b.unitprice6, b.unitprice7, b.unitprice8, b.unitprice9, b.outcome_code " +
			           "FROM bid b " +
                       "WHERE b.gameid = ? AND b.companyid = ? AND b.jobid = ?";
		try
		{
			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, job.getJobID());

			ResultSet rset = pstmt.executeQuery();

            //Check if no rows are returned
            //(rset.last() returns false if there are no rows in the result set)
            if(!rset.last())
            {
				throw new ObjectNotFoundException();
			}
			else
			{           // We found a bid
				rset.first();

				double[] unitPrices = new double[9];
				unitPrices[0] = rset.getDouble("unitprice1");
				unitPrices[1] = rset.getDouble("unitprice2");
				unitPrices[2] = rset.getDouble("unitprice3");
				unitPrices[3] = rset.getDouble("unitprice4");
				unitPrices[4] = rset.getDouble("unitprice5");
				unitPrices[5] = rset.getDouble("unitprice6");
				unitPrices[6] = rset.getDouble("unitprice7");
				unitPrices[7] = rset.getDouble("unitprice8");
				unitPrices[8] = rset.getDouble("unitprice9");

				bid = new Bid(game.getGameID(), company.getCompanyID(),
							  job.getJobID(),
							  rset.getLong("bid_amount"),
							  rset.getDouble("bid_amount_up"),
							  unitPrices,
							  rset.getInt("outcome_code"));
			}

			// Retrieve bid methods
			bid.setMethods( getBidMethods(bid) );

			rset.close();
			pstmt.close();
		} catch(SQLException sqle) {
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return bid;
	}

	public Vector getBids(Game game, Job job)
		throws BIGDatabaseFacadeException, ObjectNotFoundException
	{
		Vector bids = new Vector();

		String bidQuery = "SELECT bid_amount, bid_amount_up, unitprice1, unitprice2, unitprice3, unitprice4, unitprice5, unitprice6, unitprice7, unitprice8, unitprice9, outcome_code, companyid " +
			              "FROM bid " +
                          "WHERE gameid = ? AND jobid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(bidQuery);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());

			ResultSet rset = pstmt.executeQuery();

			int rowCount = 0;

			double[] unitPrices = new double[9];

			while (rset.next())
			{
				unitPrices[0] = rset.getDouble("unitprice1");
				unitPrices[1] = rset.getDouble("unitprice2");
				unitPrices[2] = rset.getDouble("unitprice3");
				unitPrices[3] = rset.getDouble("unitprice4");
				unitPrices[4] = rset.getDouble("unitprice5");
				unitPrices[5] = rset.getDouble("unitprice6");
				unitPrices[6] = rset.getDouble("unitprice7");
				unitPrices[7] = rset.getDouble("unitprice8");
				unitPrices[8] = rset.getDouble("unitprice9");






				Bid bid = new Bid(game.getGameID(), rset.getInt("companyid"),
								  job.getJobID(),
								  rset.getLong("bid_amount"),
								  rset.getDouble("bid_amount_up"),
								  unitPrices,
								  rset.getInt("outcome_code"));

				bid.setMethods( getBidMethods(bid) );

				bids.add(bid);
				rowCount++;
			}

			/* Do all jobs necessarily have bids?  If so, this should
			   be uncommented...
			if (rowCount < 1) {
				throw new ObjectNotFoundException();
			}
			*/

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return bids;
	}


	private int[] getBidMethods(Bid bid)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		int[] methods = new int[9];

		String query = "SELECT bid_method FROM bid_method " +
                       "WHERE gameid = ? AND companyid = ? AND " +
			           "jobid = ? ORDER BY bid_activity";

		if (bid.getCompanyID() == -1) {
			return methods;
		}

		try
  		{
			PreparedStatement pstmt = connection.prepareStatement(query);
			pstmt.setInt(1, bid.getGameID());
			pstmt.setInt(2, bid.getCompanyID());
			pstmt.setInt(3, bid.getJobNumber());

			ResultSet rset = pstmt.executeQuery();

			int rowCount = 0;
			while (rset.next())
			{
				methods[rowCount++] = rset.getInt("bid_method");
			}

			// Each of the nine activities should have a method
			if (rowCount < 9)
			{
				throw new ObjectNotFoundException();
			}

			rset.close();
			pstmt.close();
		} catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return methods;
	}

	public void updateBid(Game game, Job job, Company company, Bid bid)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		int[] methods = bid.getMethods();
		double[] unitPrices = new double[9];

		String update = "UPDATE bid " +
			            "SET bid_amount = ?, bid_amount_up = ?, unitprice1 = ?, unitprice2 = ?, unitprice3 = ?, unitprice4 = ?, unitprice5 = ?, unitprice6 = ?, unitprice7 = ?, unitprice8 = ?, unitprice9 = ?, outcome_code = ? " +
			            "WHERE jobid = ? AND companyid = ? AND gameid = ?";

		String update2 = "UPDATE bid_method " +
			             "SET bid_method = ? " +
                         "WHERE gameid = ? AND jobid = ? " +
			             "AND companyid = ? AND bid_activity = ? ";

		try
		{
			unitPrices = bid.getUnitPrices();

			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setLong(1, bid.getBidAmount());
			pstmt.setDouble(2, bid.getBidAmountUP());
			pstmt.setDouble(3, unitPrices[0]);
			pstmt.setDouble(4, unitPrices[1]);
			pstmt.setDouble(5, unitPrices[2]);
			pstmt.setDouble(6, unitPrices[3]);
			pstmt.setDouble(7, unitPrices[4]);
			pstmt.setDouble(8, unitPrices[5]);
			pstmt.setDouble(9, unitPrices[6]);
			pstmt.setDouble(10, unitPrices[7]);
			pstmt.setDouble(11, unitPrices[8]);
			pstmt.setInt(12, bid.getOutcomeCode());
			pstmt.setInt(13, job.getJobID());
			pstmt.setInt(14, company.getCompanyID());
			pstmt.setInt(15, game.getGameID());

			int rows = pstmt.executeUpdate();
			if (rows < 1) {
				throw new ObjectNotFoundException();
			}

			connection.commit();

			pstmt.close();
		} catch(SQLException sqle) {
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		// Bid methods are not stored for CompanyX
		if (company.getCompanyID() != -1) {
			for (int i = 0; i < 9; i++)
				{
					try
						{
							PreparedStatement pstmt = connection.prepareStatement(update2);
							pstmt.setInt(1, methods[i]);
							pstmt.setInt(2, game.getGameID());
							pstmt.setInt(3, job.getJobID());
							pstmt.setInt(4, company.getCompanyID());
							pstmt.setInt(5, i+1);

							int rows = pstmt.executeUpdate();

							connection.commit();

							pstmt.close();
						}
					catch(SQLException sqle)
						{
							//if an exception occurs close the connection
							close();
							throw new BIGDatabaseFacadeException(sqle);
						}
				}
		}


	}

	public void addBid(Game game, Job job, Company company, Bid theBid) throws BIGDatabaseFacadeException
	{
		int[] methods = theBid.getMethods();
		double[] unitPrices = new double[9];

		String insert = "INSERT INTO bid (gameid, jobid, companyid, bid_amount, bid_amount_up, unitprice1, unitprice2, unitprice3, unitprice4, unitprice5, unitprice6, unitprice7, unitprice8, unitprice9, outcome_code) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

		String insert2 = "INSERT INTO bid_method (gameid, jobid, companyid, bid_activity, bid_method) VALUES (?, ?, ?, ?, ?)";

		try {
			unitPrices = theBid.getUnitPrices();

			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());
			pstmt.setInt(3, company.getCompanyID());
			pstmt.setLong(4, theBid.getBidAmount());
			pstmt.setDouble(5, theBid.getBidAmountUP());
			pstmt.setDouble(6, unitPrices[0]);
			pstmt.setDouble(7, unitPrices[1]);
			pstmt.setDouble(8, unitPrices[2]);
			pstmt.setDouble(9, unitPrices[3]);
			pstmt.setDouble(10, unitPrices[4]);
			pstmt.setDouble(11, unitPrices[5]);
			pstmt.setDouble(12, unitPrices[6]);
			pstmt.setDouble(13, unitPrices[7]);
			pstmt.setDouble(14, unitPrices[8]);
			pstmt.setInt(15, theBid.getOutcomeCode());

			int rows = pstmt.executeUpdate();
			connection.commit();

			pstmt.close();
		} catch(SQLException sqle) {
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		// Don't store CompanyX bid methods
		if (company.getCompanyID() != -1) {
			for (int i = 0; i < 9; i++)	{
				try	{
					PreparedStatement pstmt = connection.prepareStatement(insert2);
					pstmt.setInt(1, game.getGameID());
					pstmt.setInt(2, job.getJobID());
					pstmt.setInt(3, company.getCompanyID());
					pstmt.setInt(4, i+1);
					pstmt.setInt(5, methods[i]);

					int rows = pstmt.executeUpdate();
					connection.commit();

					pstmt.close();
				} catch(SQLException sqle) {
					// if an exception occurs close the connection
					close();
					throw new BIGDatabaseFacadeException(sqle);
				}
			}
		}

	}

	public void removeBids(int companyID) throws BIGDatabaseFacadeException

			connection.commit();
		} catch(SQLException sqle) {
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}
