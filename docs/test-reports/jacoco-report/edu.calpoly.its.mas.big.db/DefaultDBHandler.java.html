package edu.calpoly.its.mas.big.db;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;


/**
 * Default implementation of <code>DBHandler</code>.  Can be extended for other DBHandlers.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.2 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultDBHandler implements DBHandler
{
    protected Connection connection;

    public DefaultDBHandler(Connection connection)
    {
		this.connection = connection;
	}

	public void close() throws BIGDatabaseFacadeException
	{
		try
		{
			connection.close();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
	}
}
