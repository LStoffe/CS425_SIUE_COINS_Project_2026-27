package edu.calpoly.its.mas.big.db;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>ReportLedgerDBHandler</code>.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.5 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultReportLedgerDBHandler extends DefaultDBHandler implements ReportLedgerDBHandler
{

    public DefaultReportLedgerDBHandler(Connection connection)
    {
		super(connection);
	}


    public void recordPayment(int companyID, int periodID, int reportID, int purchasePeriod) throws BIGDatabaseFacadeException
	{
		String insert = "INSERT INTO report_ledger (companyid, periodid, reportid, purchase_period) " +
                        "VALUES (?, ?, ?, ?)";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, companyID);
			pstmt.setInt(2, periodID);
			pstmt.setInt(3, reportID);
			pstmt.setInt(4, purchasePeriod);

			pstmt.executeUpdate();
			connection.commit();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}


	public boolean checkIfPurchased(int companyID, int periodID, int reportID)  throws BIGDatabaseFacadeException
	{
		boolean purchased = false;
		String query = "SELECT * FROM report_ledger " +
                       "WHERE companyid = ? AND periodid = ? AND reportid = ?";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, companyID);
			pstmt.setInt(2, periodID);
			pstmt.setInt(3, reportID);

			ResultSet rset = pstmt.executeQuery();

			// If a row was returned, the report has been purchased,
			// otherwise leave purchased as false
			if(rset.next())
				purchased = true;

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return purchased;
	}


	public int[] getPurchasesForPeriod(int companyID, int purchasePeriod) throws BIGDatabaseFacadeException
	{
		int[] purchaseMap = new int[6];

		String query = "SELECT reportid FROM report_ledger " +
                       "WHERE companyid = ? AND purchase_period = ?";

		try
 		{
			PreparedStatement pstmt = connection.prepareStatement(query);
			pstmt.setInt(1, companyID);
			pstmt.setInt(2, purchasePeriod);

			ResultSet rset = pstmt.executeQuery();

			// If a row was returned, the report has been purchased,
			// otherwise leave purchased as false
			while (rset.next())
				purchaseMap[rset.getInt("reportid")] += 1;

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return purchaseMap;

	}
}
