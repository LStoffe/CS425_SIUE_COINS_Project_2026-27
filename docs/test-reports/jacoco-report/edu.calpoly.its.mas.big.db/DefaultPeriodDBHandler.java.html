package edu.calpoly.its.mas.big.db;

import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.Period;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>PeriodDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.6 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultPeriodDBHandler extends DefaultDBHandler implements PeriodDBHandler
{

    public DefaultPeriodDBHandler(Connection connection)
    {
		super(connection);
	}


	public Period getPeriod(Game game, int periodNumber) throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		Period period = null;

		try
		{
		   String query = "SELECT temperature, rainfall, labor_availability, material_cost_index, number_of_new_jobs, number_of_new_heavy_jobs, workdays_first_month, " +
		   				  " workdays_second_month, raindays_first_month, raindays_second_month " +
		   				  " FROM period WHERE gameid = ? AND periodid = ?";

		   PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
		   pstmt.setInt(1, game.getGameID());
		   pstmt.setInt(2, periodNumber);

		   ResultSet rset = pstmt.executeQuery();

		   //Check if no rows are returned
		   //(rset.last() returns false if there are no rows in the result set)
    	   if(!rset.last())
           {
    		   throw new ObjectNotFoundException();
    	   }
           else
           {
		       period = new Period(periodNumber,
				 				   rset.getInt("labor_availability"),
				 				   rset.getInt("material_cost_index"),
				  			       rset.getFloat("rainfall"),
				  				   rset.getInt("temperature"),
				  				   rset.getInt("number_of_new_jobs"),
				  				   rset.getInt("number_of_new_heavy_jobs"),
				  				   rset.getInt("workdays_first_month"),
				  				   rset.getInt("workdays_second_month"),
				  				   rset.getInt("raindays_first_month"),
				     			   rset.getInt("raindays_second_month"));
		   }

		   rset.close();
		   pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
    	}

    	return period;
	}

	public void updatePeriod(Game game, Period thePeriod) throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		try
		{
		   String update = "UPDATE period SET temperature = ?, rainfall = ?, labor_availability = ?, material_cost_index = ?, " +
		                   " number_of_new_jobs = ?, number_of_new_heavy_jobs = ?, workdays_first_month = ?, workdays_second_month = ?, raindays_first_month = ?, " +
		                   " raindays_second_month = ? WHERE gameid = ? AND periodid = ?";

		   PreparedStatement pstmt = connection.prepareStatement(update);
		   pstmt.setInt(1, thePeriod.getTemperature());
		   pstmt.setFloat(2, thePeriod.getRainfall());
		   pstmt.setInt(3, thePeriod.getLaborAvailability());
		   pstmt.setInt(4, thePeriod.getMaterialsCostIndex());
		   pstmt.setInt(5, thePeriod.getNumberOfNewBuildingJobs());
		   pstmt.setInt(6, thePeriod.getNumberOfNewHeavyJobs());
		   pstmt.setInt(7, thePeriod.getWorkdaysFirstMonth());
		   pstmt.setInt(8, thePeriod.getWorkdaysSecondMonth());
		   pstmt.setInt(9, thePeriod.getRaindaysFirstMonth());
		   pstmt.setInt(10, thePeriod.getRaindaysSecondMonth());
		   pstmt.setInt(11, game.getGameID());
		   pstmt.setInt(12, thePeriod.getPeriodNumber());

		   int rowsChanged = pstmt.executeUpdate();

           if (rowsChanged < 1)
           {
			   throw new ObjectNotFoundException();
		   }
		   else if (rowsChanged > 1)
		   {
			   throw new BIGDatabaseFacadeException("More than one Period in database matching game: " + game.getGameID() + " period: " + thePeriod.getPeriodNumber());
		   }

		   pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
    	}

	}

	public void addPeriod(Game game, Period thePeriod) throws BIGDatabaseFacadeException
	{
		try
		{
		   String add = "INSERT INTO period (gameid, periodid, temperature, rainfall, labor_availability, material_cost_index, " +
		                " number_of_new_jobs, number_of_new_heavy_jobs, workdays_first_month, workdays_second_month, raindays_first_month, raindays_second_month) " +
		                " VALUES (?,?,?,?,?,?,?,?,?,?,?,?)";

		   PreparedStatement pstmt = connection.prepareStatement(add);
		   pstmt.setInt(1, game.getGameID());
		   pstmt.setInt(2, thePeriod.getPeriodNumber());
		   pstmt.setInt(3, thePeriod.getTemperature());
		   pstmt.setFloat(4, thePeriod.getRainfall());
		   pstmt.setInt(5, thePeriod.getLaborAvailability());
		   pstmt.setInt(6, thePeriod.getMaterialsCostIndex());
		   pstmt.setInt(7, thePeriod.getNumberOfNewBuildingJobs());
		   pstmt.setInt(8, thePeriod.getNumberOfNewHeavyJobs());
		   pstmt.setInt(9, thePeriod.getWorkdaysFirstMonth());
		   pstmt.setInt(10, thePeriod.getWorkdaysSecondMonth());
		   pstmt.setInt(11, thePeriod.getRaindaysFirstMonth());
		   pstmt.setInt(12, thePeriod.getRaindaysSecondMonth());

		   int rowsChanged = pstmt.executeUpdate();

		   pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
    	}

	}

	public void removePeriods(int gameID) throws BIGDatabaseFacadeException
	{
		try
		{
		   String remove = "DELETE FROM period WHERE gameid = ?";

    	   PreparedStatement pstmt = connection.prepareStatement(remove);
  		   pstmt.setInt(1, gameID);

		   int rowsChanged = pstmt.executeUpdate();

		   pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
    	}
	}

}
