package edu.calpoly.its.mas.big.db;

import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Game;

import edu.calpoly.its.mas.big.reports.FinancialReportForm;
import edu.calpoly.its.mas.big.reports.JobFinancialInfo;

import java.util.Vector;
import java.util.Iterator;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>FinancialReportDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.8 $ $Date: 2002/04/11 23:36:16 $
 */
public class DefaultFinancialReportDBHandler extends DefaultDBHandler implements FinancialReportDBHandler
{

    public DefaultFinancialReportDBHandler(Connection connection)
    {
		super(connection);
	}

    public FinancialReportForm getFinancialReport(Game game, int periodNum, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		FinancialReportForm financialReportForm = null;

		try
		{
			String query = "SELECT interest_paid, bidding_expenses, consulting_expenses, office_overhead, settlement, " +
						   " previous_cash_on_hand, loan_principle_payments " +
						   " FROM financial_report WHERE gameid = ? AND periodid = ? AND companyid = ? " ;

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
				close();
				throw new ObjectNotFoundException();
			}
			else
			{
				rset.first();

				financialReportForm = new FinancialReportForm(company.getTitle(),
										   getJobFinancialInfo(game, company, periodNum),
										   rset.getLong("interest_paid"),
										   rset.getLong("bidding_expenses"),
										   rset.getLong("consulting_expenses"),
										   rset.getLong("previous_cash_on_hand"),
										   rset.getLong("office_overhead"),
										   rset.getLong("loan_principle_payments")
										);
				financialReportForm.setSettlementExpenses(rset.getLong("settlement"));
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return financialReportForm;
    }


    private Vector getJobFinancialInfo(Game game, Company company, int periodID) throws BIGDatabaseFacadeException
    {
	    Vector jobInfoList = new Vector();

	  	try
	  	{
			  String query = "SELECT jfi.jobid, payments_received, retainage_received, direct_labor_costs, " +
			  				 " direct_materials_costs, direct_equipment_costs, field_overhead, subcontractor_costs, days_late, j.liquidation_damages, j.job_type_id, j.type_modifier" +
			  				 " FROM job_financial_info jfi, job j WHERE jfi.gameid = ? AND jfi.companyid = ? AND jfi.periodid = ? " +
			  				 " AND jfi.gameid = j.gameid AND jfi.companyid = j.companyid AND jfi.jobid = j.jobid";

			  PreparedStatement pstmt = connection.prepareStatement(query);
			  pstmt.setInt(1, game.getGameID());
			  pstmt.setInt(2, company.getCompanyID());
			  pstmt.setInt(3, periodID);

			  ResultSet rset = pstmt.executeQuery();

			  while(rset.next())
			  {
				  JobFinancialInfo jobInfo =  new JobFinancialInfo(rset.getInt("jobid"),
				  												   rset.getInt("job_type_id"),
				  												   rset.getShort("type_modifier"),
				  												   rset.getLong("payments_received"),
				  												   rset.getLong("retainage_received"),
				  												   rset.getLong("direct_labor_costs"),
				  												   rset.getLong("direct_materials_costs"),
				  												   rset.getLong("direct_equipment_costs"),
				  												   rset.getLong("subcontractor_costs"),
				  												   rset.getLong("field_overhead"),
				  												   rset.getInt("days_late"),
				  												   rset.getLong("liquidation_damages"));
				  jobInfoList.add(jobInfo);
			  }
			  rset.close();
			  pstmt.close();
	  	}
	  	catch(SQLException sqle)
	  	{
			  //If an exception occurs close the connection
			  close();
			  throw new BIGDatabaseFacadeException(sqle);
	  	}

	  	return jobInfoList;
	}


    /**
     * Adds a new FinancialReport to the database for a given company
     */
    public void addFinancialReport(Game game, Company company, FinancialReportForm theReport) throws BIGDatabaseFacadeException
    {
		Vector jobInfoList = theReport.getJobInfoList();
		Iterator i = jobInfoList.iterator();
		while(i.hasNext())
		{
			JobFinancialInfo jobInfo = (JobFinancialInfo) i.next();
			addJobFinancialInfo(game, company, theReport.getPeriod(), jobInfo);

		}

		String insert = "INSERT INTO financial_report (gameid, companyid, periodid, interest_paid, bidding_expenses, " +
						" consulting_expenses, office_overhead, settlement, previous_cash_on_hand, loan_principle_payments " +
						" ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, theReport.getPeriod());
			pstmt.setLong(4, theReport.getInterestPaidOnLoans());
			pstmt.setLong(5, theReport.getBiddingExpenses());
			pstmt.setLong(6, theReport.getConsultingExpenses());
			pstmt.setLong(7, theReport.getOfficeOverhead());
			pstmt.setLong(8, theReport.getSettlementExpenses());
			pstmt.setLong(9, theReport.getPreviousCashOnHand());
			pstmt.setLong(10, theReport.getLoanPrinciplePayments());

			int rowsChanged = pstmt.executeUpdate();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

    }


    //Adds a new JobFinancialInfo object to the database
    private void addJobFinancialInfo(Game game, Company company, int periodID, JobFinancialInfo jobFinancialInfo)
    	throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO job_financial_info (gameid, companyid, periodid, jobid, payments_received, " +
						"retainage_received, direct_labor_costs, direct_materials_costs, direct_equipment_costs, subcontractor_costs, field_overhead, " +
						"days_late) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";
		try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, periodID);
			pstmt.setInt(4, jobFinancialInfo.getJobID());
			pstmt.setLong(5, jobFinancialInfo.getPaymentsReceived());
			pstmt.setLong(6, jobFinancialInfo.getRetainageReceived());
			pstmt.setLong(7, jobFinancialInfo.getDirectLaborCost());
			pstmt.setLong(8, jobFinancialInfo.getDirectMaterialsCost());
			pstmt.setLong(9, jobFinancialInfo.getDirectEquipmentCost());
			pstmt.setLong(10, jobFinancialInfo.getSubcontractorCost());
			pstmt.setLong(11, jobFinancialInfo.getFieldOverhead());
			pstmt.setInt(12, jobFinancialInfo.getDaysLate());

			int rowChanged = pstmt.executeUpdate();

			pstmt.close();

		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    /**
     * Deletes, from the database, all financial reports in the <code>Game</code> with id matching <code>gameid</code>
     */
    public void removeFinancialReports(int gameID) throws BIGDatabaseFacadeException
    {
		removeJobFinancialInfo(gameID);

		String remove = "DELETE FROM financial_report WHERE gameid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}


    }


    //Removes all rows in the JOB_FINANCIAL_INFO table belonging to a given game
    private void removeJobFinancialInfo(int gameid) throws BIGDatabaseFacadeException
    {
		String remove = "DELETE FROM job_financial_info WHERE gameid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameid);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


}
