 	public Game getGame(int gameID) throws ObjectNotFoundException, BIGDatabaseFacadeException
  	{
		if(log.isDebugEnabled())
		{
			log.debug("Starting method getGame(int gameID)");
		}

		Game game;

		//SQL statement to retrieve the game matching the given gameID
		String query = "SELECT adminid, gamename, start_year, current_period, is_active,";
		query = query +"  current_job_number, gametype FROM game WHERE gameid= ?";

  		try
  		{
			PreparedStatement pstatement = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstatement.setInt(1, gameID);

			ResultSet rset = pstatement.executeQuery();

			//Check if no rows are returned
			//(rset.last() returns false if there are no rows in the result set)
       		if(!rset.last())
       	 	{
     	    	throw new ObjectNotFoundException();
        	}
        	else
        	{
        	   rset.first();

        	   game = new Game(gameID, rset.getInt("adminid"), rset.getString("gamename").trim(), rset.getDate("start_year"),
			          rset.getInt("current_period"), rset.getBoolean("is_active"),
			          rset.getInt("current_job_number"), rset.getInt("gametype") );
        	}
        	rset.close();
        	pstatement.close();
		}
		catch(SQLException sqle)
		{
			//if an exception occurs close the connection
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	 	if(log.isDebugEnabled())
		{
			log.debug("Returning from method getGame(int gameID)");
		}

		return game;
  	}

  	public Vector getGames(Admin admin) throws BIGDatabaseFacadeException
  	{
		if(log.isDebugEnabled())
		{
			log.debug("Starting method getGames(Admin admin)");
		}

	    Vector games = new Vector();

	  	try
	  	{
			  //Select all games owned by this administrator
			  String query = "SELECT gameID, gamename, start_year, current_period, is_active,";
			  query = query +" current_job_number, gametype FROM game WHERE adminID= ? ORDER BY gamename";

			  PreparedStatement pstmt = connection.prepareStatement(query);
			  pstmt.setInt(1, admin.getAdminID());

			  ResultSet rset = pstmt.executeQuery();

			  while(rset.next())
			  {
				  //Create a new admin and store it in the Vector
				  Game tempGame =  new Game(rset.getInt("gameID"), admin.getAdminID(), rset.getString("gamename").trim(), rset.getDate("start_year"),
			                            rset.getInt("current_period"), rset.getBoolean("is_active"),
			                            rset.getInt("current_job_number"), rset.getInt("gametype") );
				  games.add(tempGame);
			  }
			  rset.close();
			  pstmt.close();
	  	}
	  	catch(SQLException sqle)
	  	{
			  //If an exception occurs close the connection
			  close();
			  throw new BIGDatabaseFacadeException(sqle);
	  	}

	 	if(log.isDebugEnabled())
		{
			log.debug("Returning from method getGames(Admin admin)");
		}

	  	return games;
  	}

    public void updateGame(Game theGame) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method updateGame(Game theGame)");
		}

		//Create the SQL string used for the update
		String update = "UPDATE game SET current_period = ?, current_job_number = ?, is_active = ? " +
				" WHERE gameid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, theGame.getCurrentPeriodNumber());
			pstmt.setInt(2, theGame.getCurrentJobNumber());
			pstmt.setBoolean(3, theGame.getIsActive());
                        pstmt.setInt(4, theGame.getGameID());

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

			if (rowsChanged < 1)
			   throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)
		{
			//Close connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		if(log.isDebugEnabled())
		{
			log.debug("Returning from method updateGame(Game theGame)");
		}
	}

    public void addGame(Game theGame) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method addGame(Game theGame)");
		}

		int gameID;

		try
		{
			//Get the next gameID in the sequence
			String sequenceName = "big_gameid_sequence";
			String query = "SELECT nextval ('" + sequenceName + "')";

			PreparedStatement stmt = connection.prepareStatement(query);
			ResultSet rset = stmt.executeQuery();

			if(rset.next()) //A gameID value was returned from the database
			{
				gameID = rset.getInt(1);

				//Assign the new gameID to the Game to be stored
				theGame.setGameID(gameID);

				//Convert the game's start year to a SQL date
				java.sql.Date sqlDate = new java.sql.Date(theGame.getStartYear().getTime());

				//Create the SQL statement used to insert the new game into the database
				String insert = "INSERT INTO game (gameID, adminID, gamename, start_year, current_period, is_active, current_job_number, gametype) ";
				insert = insert + "VALUES (?, ?, ?, ?, ?, ?, ?, ?)";

				//Perform the insert
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, gameID);
				pstmt.setInt(2, theGame.getAdminID());
				pstmt.setString(3, theGame.getName());
				pstmt.setDate(4, sqlDate);
				pstmt.setInt(5, theGame.getCurrentPeriodNumber());
				pstmt.setBoolean(6, theGame.getIsActive());
				pstmt.setInt(7, theGame.getCurrentJobNumber());
				pstmt.setInt(8, theGame.getGametype());

				int rowsAdded = pstmt.executeUpdate();
				connection.commit();

				rset.close();
				stmt.close();
				pstmt.close();
			}
			else  //A gameID was not returned from the database
			{
				rset.close();
				stmt.close();
				close();
				throw new BIGDatabaseFacadeException("Couldn't retrieve next value from " + sequenceName );
			}

		}
		catch(SQLException sqle)
		{
			//Close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		if(log.isDebugEnabled())
		{
			log.debug("Returning from method addGame(Game theGame)");
		}
	}

    public void removeGame(int adminID, int gameID) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method removeGame(int adminID, int gameID)");
		}

		//Remove all jobs, companies, etc from this game
		(new DefaultCompanyDBHandler(connection, sqlMap)).removeCompanies(gameID);
		(new DefaultPeriodDBHandler(connection)).removePeriods(gameID);
		(new DefaultJobDBHandler(connection, sqlMap)).removeJobs(gameID);
		(new DefaultFinancialReportDBHandler(connection)).removeFinancialReports(gameID);
		(new DefaultCurrentMethodDBHandler(connection)).removeCurrentMethods(gameID);
		(new DefaultOvertimeDBHandler(connection)).removeOvertimeRequests(gameID);
		(new DefaultBidDBHandler(connection, sqlMap)).removeBids(gameID, true);
		(new DefaultCCCDBHandler(connection)).removeCCCs(gameID);
		(new DefaultComponentsDBHandler(connection)).removeEstimatingComponentEntries(gameID);
		(new DefaultEstimatingComponentDBHandler(connection)).finalizeEstimatingComponentDatabase(gameID);

        //Remove all parameters for this game
        (new DefaultGameParametersDBHandler(connection, sqlMap)).removeGameParameters(adminID, gameID);

		//String to remove the game matching the ID number passed in
		String update = "DELETE FROM game WHERE gameID = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, gameID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		if(log.isDebugEnabled())
		{
			log.debug("Returning from method removeGame(int adminID, int gameID)");
		}
	}


    public void removeGames(int adminID) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method removeGames(int adminID)");
		}

		//Note: The SQL statement "DELETE FROM game WHERE adminid = ?" does not work in this case because deleting a game needs to
		//trigger the deletion of all the values in other tables associated with that game.  Multiple calls to the removeGame method must be made

		//Construct a fake admin with the given adminID
		Admin admin = new Admin(adminID, null, null, null, false);

		//Get all games owned by this admin
		Vector games = getGames(admin);

		Iterator gameIterator = games.iterator();

		//Iterate over the list of games, removing each one
		while(gameIterator.hasNext())
		{
			Game nextGame = (Game) gameIterator.next();
			removeGame(adminID, nextGame.getGameID());
		}

		if(log.isDebugEnabled())
		{
			log.debug("Returning from method removeGames(int adminID)");
		}

	}
