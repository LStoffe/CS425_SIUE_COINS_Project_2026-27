	public Job getJob(Game game, int jobID) throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		Job job = null;

		try {
			String query = "SELECT j.jobid, percent_retention, percent_of_job, liquidation_damages, " +
				"completion_date, job_lu_size, lus_remaining, " +
				"direct_cost, completion_deadline, period_created, " +
				"period_completed, job_type_id, estimated_workdays, " +
				"retained_amount, is_active, is_auto, method_scaling_factor, change_order_lus, change_order_pay, " +
				"field_overhead_fixed, field_overhead_variable, amount_retained_from_subs, billed_to_date, type_modifier, " +
				"j.retentionid, j.customerid " +
				"FROM job j, retention r, game_params_overhead gpo " +
				"WHERE j.gameid = ? AND jobid = ? AND j.retentionid = r.retentionid AND j.gameid = gpo.gameid " +
				"ORDER BY jobid ";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, jobID);

			ResultSet rset = pstmt.executeQuery();

			while(rset.next()) {
				//Create a new job and store it in the Vector
				job = createJob(rset, game);
			}
			rset.close();
			pstmt.close();
		} catch(SQLException sqle) {
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		if (job == null)
			throw new ObjectNotFoundException();

		return job;
	}

	public Vector getActiveJobs(Game game, Company company) throws BIGDatabaseFacadeException
	{
		Vector jobs = new Vector();

		try {
			// All active jobs
			String query = "SELECT j.jobid, percent_retention, percent_of_job, liquidation_damages, " +
				"completion_date, job_lu_size, lus_remaining, " +
				"direct_cost, completion_deadline, period_created, " +

				"j.retentionid, j.customerid " +
				"FROM job j, bid b, retention r, game_params_overhead gpo " +
				"WHERE j.companyid = ? AND is_active = TRUE AND j.companyid = b.companyid AND b.jobid = j.jobid AND j.retentionid = r.retentionid " +

		} catch(SQLException sqle) {

	public Vector getWorkedJobs(Game game, Company company, int periodNum) throws BIGDatabaseFacadeException
	{
		Vector jobs = new Vector();

		try
		{
			//Each activity of a job is saved for a given period if work is done on any activity of that job in that period.
			//Check if there is a saved state of activity1 or activity10 for periodNum, if so work was done so the job is returned
			String query = "SELECT j.jobid, percent_retention, percent_of_job, liquidation_damages, " +
				"completion_date, job_lu_size, lus_remaining, " +
				"direct_cost, completion_deadline, period_created, " +
				"period_completed, job_type_id, estimated_workdays, " +
				"retained_amount, j.is_active, j.is_auto, method_scaling_factor, change_order_lus, change_order_pay, " +

				"bid_amount, j.retentionid, j.customerid " +
				"FROM job j, retention r, game_params_overhead gpo, activity a, bid b " +
				"WHERE j.companyid = ? AND j.retentionid = r.retentionid AND j.gameid = gpo.gameid " +
				"AND a.jobid = j.jobid AND (a.activity_type_id = 1 OR a.activity_type_id = 10) AND a.periodid = ? AND a.gameid = j.gameid AND j.period_created-1 != a.periodid " +

			pstmt.setInt(1, company.getCompanyID());
			pstmt.setInt(2, periodNum);

				//Create a new job and store it in the Vector

	/**
	 * Given a <code>ResultSet</code> containing all the columns from the jobs table,
	 * builds a <code>Job</code> object.
	 */
	private Job createJob(ResultSet rset, Game game) throws SQLException, BIGDatabaseFacadeException
	{
		Job job =  new Job(rset.getInt("jobid"), rset.getInt("liquidation_damages"),
						   rset.getDate("completion_date"), rset.getDate("completion_deadline"),
						   rset.getInt("job_lu_size"), rset.getInt("percent_retention"),
						   rset.getInt("percent_of_job"),
						   rset.getLong("direct_cost"), rset.getInt("retained_amount"),
						   rset.getInt("lus_remaining"), rset.getInt("period_completed"),
						   rset.getInt("period_created"), rset.getBoolean("is_active"),
                           rset.getBoolean("is_auto"),
						   rset.getInt("job_type_id"), rset.getInt("estimated_workdays"),
						   rset.getFloat("method_scaling_factor"),
						   rset.getLong("change_order_pay"),
						   rset.getInt("change_order_lus"),
						   rset.getInt("field_overhead_fixed"),
						   rset.getInt("field_overhead_variable"),
						   rset.getInt("amount_retained_from_subs"),
						   rset.getInt("billed_to_date"),
						   rset.getShort("type_modifier")
						   );

		// FIXME: Temporary hack to obviate constructor modification
		job.setRetentionID(rset.getInt("retentionid"));

		if(job.getTypeModifier() == Job.PRIVATE)
		{
			job.setCompaniesAllowedToBid(getAllowedCompanies(game, job.getJobID()));
			job.setCustomer((new edu.calpoly.lib.multimedia.big.db.DefaultCustomerDBHandler(connection)).getCustomer(rset.getInt("customerid")));
			job = getNegotiatedJob(game, job);
		}

		return job;
	}

	private Vector getAllowedCompanies(Game game, int jobid) throws BIGDatabaseFacadeException
	{
		Vector companyIDs = new Vector();
		String select = "SELECT companyid FROM job_allowed_companies WHERE jobid = ? AND gameid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select);
			pstmt.setInt(1, jobid);
			pstmt.setInt(2, game.getGameID());

			ResultSet rset = pstmt.executeQuery();
			while(rset.next())
			{
				companyIDs.add(new Integer(rset.getInt("companyid")));
			}

			rset.close();
			pstmt.close();

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return companyIDs;
	}

	private Job getNegotiatedJob(Game game, Job job) throws BIGDatabaseFacadeException
	{
		Job j = job;
		try
		{
			Hashtable negotiationValues = (new edu.calpoly.lib.multimedia.big.db.DefaultNegotiationDBHandler(connection)).getNegotiationValues(game, job.getJobID());
			if(negotiationValues != null)
			{
				Integer agreedProfit = (Integer) negotiationValues.get("agreedProfit");
				Long agreedCost = (Long) negotiationValues.get("agreedCost");
				Integer state = (Integer) negotiationValues.get("state");
				j = new NegotiatedJob(job, agreedProfit.intValue(), agreedCost.longValue(), state.intValue());
			}
		}
		catch(ObjectNotFoundException onfe)
		{
			//do nothing, job is not negotiated
		}
		return j;
	}


    public void updateJob(Game game, Job job) throws BIGDatabaseFacadeException, ObjectNotFoundException
    {
		//SQL statement to a job
		String update = "UPDATE job SET companyid = ?, job_type_id = ?, liquidation_damages = ?, " +
		                "completion_date = ?, job_lu_size = ?, lus_remaining = ?, is_active = ?, is_auto = ?, " +

		                "retained_amount = ?, estimated_workdays = ?, method_scaling_factor = ?, change_order_pay = ?, " +
		                "change_order_lus = ?, amount_retained_from_subs = ?, billed_to_date = ? WHERE gameid = ? AND jobid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);

			int companyID = 0;
			if(job.getCompany() != null)
			{
				companyID = job.getCompany().getCompany().getCompanyID();
			}

	   		java.sql.Date completionDate;
            if(job.getCompletionDate() != null)
                   completionDate  = new java.sql.Date(job.getCompletionDate().getTime());
            else
                   completionDate = new java.sql.Date(0);
            java.sql.Date completionDeadline = new java.sql.Date(job.getCompletionDeadline().getTime());

			pstmt.setInt(1, companyID);
			pstmt.setInt(2, job.getJobType());
			pstmt.setInt(3, job.getLiquidatedDamages());
			pstmt.setDate(4, completionDate);
			pstmt.setInt(5, job.getJobSize());
			pstmt.setInt(6, job.getLusRemaining());
			pstmt.setBoolean(7, job.getIsActive());
			pstmt.setBoolean(8, job.getIsAuto());
			pstmt.setLong(9, job.getDirectCost());
			pstmt.setDate(10, completionDeadline);
			pstmt.setInt(11, job.getPeriodCreated());
			pstmt.setInt(12, job.getPeriodCompleted());
			pstmt.setLong(13, job.getRetainedAmount());
			pstmt.setInt(14, job.getEstimatedWorkdays());
			pstmt.setFloat(15, job.getMethodScalingFactor());
			pstmt.setLong(16, job.getChangeOrderPrice());
			pstmt.setInt(17, job.getChangeOrderLus());
			pstmt.setInt(18, job.getAmountRetainedFromSubcontractors());
			pstmt.setInt(19, job.getBilledToDate());
			pstmt.setInt(20, game.getGameID());
			pstmt.setInt(21, job.getJobID());


			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

			if (rowsChanged < 1)
				throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)

			//Close connection if an exception occurs

    public void addJob(Game game, Job job) throws BIGDatabaseFacadeException
    {
		try
		{
			//Create the SQL statement used to insert the new job into the database.
			String insert = "INSERT INTO job (gameid, jobid, companyid, job_type_id, retentionid, liquidation_damages, " +
		                "completion_date, job_lu_size, lus_remaining, is_active, is_auto, " +
		                "direct_cost, completion_deadline, period_created, period_completed, " +
		                "retained_amount, estimated_workdays, method_scaling_factor, change_order_pay, change_order_lus, " +
		                " amount_retained_from_subs, billed_to_date, type_modifier, customerid) " +
		                " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ";

			int companyID = 0;
			if(job.getCompany() != null)
			{
				companyID = job.getCompany().getCompany().getCompanyID();
			}

			java.sql.Date completionDate;
            if (job.getCompletionDate() != null)
	            completionDate = new java.sql.Date(job.getCompletionDate().getTime());
	        else
	        {
				completionDate = new java.sql.Date(0);
			}
            java.sql.Date completionDeadline = new java.sql.Date(job.getCompletionDeadline().getTime());

			//Perform the insert
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());
			pstmt.setInt(3, companyID);
			pstmt.setInt(4, job.getJobType());
			pstmt.setInt(5, job.getRetentionID());
			pstmt.setInt(6, job.getLiquidatedDamages());
			pstmt.setDate(7, completionDate);
			pstmt.setInt(8, job.getJobSize());
			pstmt.setInt(9, job.getLusRemaining());
			pstmt.setBoolean(10, job.getIsActive());
			pstmt.setBoolean(11, job.getIsAuto());
			pstmt.setLong(12, job.getDirectCost());
			pstmt.setDate(13, completionDeadline);
			pstmt.setInt(14, job.getPeriodCreated());
			pstmt.setInt(15, job.getPeriodCompleted());
			pstmt.setLong(16, job.getRetainedAmount());
			pstmt.setInt(17, job.getEstimatedWorkdays());
			pstmt.setFloat(18, job.getMethodScalingFactor());
			pstmt.setLong(19, job.getChangeOrderPrice());
			pstmt.setInt(20, job.getChangeOrderLus());
			pstmt.setInt(21, job.getAmountRetainedFromSubcontractors());
			pstmt.setInt(22, job.getBilledToDate());
			pstmt.setShort(23, job.getTypeModifier());
			pstmt.setInt(24, job.getCustomerID());

			int rowsAdded = pstmt.executeUpdate();
			connection.commit();
			pstmt.close();

			//If this is a Negotiated Job, also store the negotiated
			//job parameters to the database in the NEGOTIATED_JOB table
			if (job instanceof NegotiatedJob)
			{

				NegotiatedJob nj = (NegotiatedJob) job;

				int negotiatedCompanyID = 0;
				if(nj.getCompaniesAllowedToBid().size() != 1)
				{
					System.out.println("DefaultJobDBHandler.addJob: incorrect number of allowed companies for negotiated job");
				}
				else
				{
					negotiatedCompanyID = ((Integer) nj.getCompaniesAllowedToBid().elementAt(0)).intValue();
				}



				//Create the SQL statement used to insert the new job into the database.
				String insertNegotiated = "INSERT INTO negotiated_job (gameid, jobid, companyid, customerid, agreed_cost, agreed_percent_profit, state, customer_target_profit, customer_haggling) " +
		    	            " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?) ";

				//Perform the insert
				pstmt = connection.prepareStatement(insertNegotiated);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, nj.getJobID());
				pstmt.setInt(3, negotiatedCompanyID);
				pstmt.setInt(4, nj.getCustomerID());
				pstmt.setLong(5, nj.getAgreedCost());
				pstmt.setInt(6, nj.getAgreedProfit());
				pstmt.setInt(7, nj.getState());
				pstmt.setInt(8, -1);
				pstmt.setInt(9, -1);

				pstmt.executeUpdate();
				connection.commit();
				pstmt.close();
			}

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		if((job.getTypeModifier() == Job.PRIVATE) && (job.getCompaniesAllowedToBid() != null))
		{
			addAllowedCompanies(game, job.getJobID(), job.getCompaniesAllowedToBid());
		}
	}

	private void addAllowedCompanies(Game game, int jobid, Vector companyIDs) throws BIGDatabaseFacadeException
	{
		String insert = "INSERT INTO job_allowed_companies VALUES (?, ?, ?)";

		try
		{
			PreparedStatement pstmt = null;

			for(int i = 0; i < companyIDs.size(); i++)
			{
				pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, jobid);
				pstmt.setInt(3, ((Integer) companyIDs.elementAt(i)).intValue());


				pstmt.executeUpdate();
				pstmt.close();
			}
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

    public void removeJob(int gameid, int jobid) throws BIGDatabaseFacadeException
    {
		//Remove all activities belonging to this activities
		(new DefaultActivityDBHandler(connection)).removeActivities(gameid, jobid);

		//String to remove the admin matching the ID number passed in
		String update = "DELETE FROM job WHERE gameID = ? AND jobid = ?";
		String delete2 = "DELETE FROM negotiated_job WHERE gameid = ? AND jobid = ?";

		removeAllowedCompanies(jobid);

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, gameid);
			pstmt.setInt(2, jobid);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

 			pstmt = connection.prepareStatement(delete2);
			pstmt.setInt(1, gameid);
			pstmt.setInt(2, jobid);

			rowsChanged = pstmt.executeUpdate();

			pstmt.close();

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

	public void removeJobs(int gameid) throws BIGDatabaseFacadeException
	{
           //NOTE: The SQL statement "DELETE FROM job WHERE gameid = ?" does not work in this case becase deleting a job needs to
           //trigger the deletion of all the values in other tables associated with that job.  Multiple calls to the removeJob method must be made

           //Construct a fake game with the given gameid
           Game game = new Game(gameid, 0, null, null, 0, false, 0, 0);

           //Get all jobs in this game
           Vector jobs = getJobs(game);

           Iterator jobIterator = jobs.iterator();

           //Iterate over the list of jobs, removing each one
           while(jobIterator.hasNext())
           {
           	Job nextJob = (Job) jobIterator.next();
                removeJob(gameid, nextJob.getJobID());
           }
	}

	}
