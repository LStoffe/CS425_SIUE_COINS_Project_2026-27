package edu.calpoly.its.mas.big.db;

public class DefaultMemberNamesDBHandler extends DefaultDBHandler implements MemberNamesDBHandler

    public DefaultMemberNamesDBHandler(Connection connection)
    {
		super(connection);
	}

    public Vector getMemberNames(Game game, Company company) throws BIGDatabaseFacadeException
    {
        Vector names = new Vector();
		edu.calpoly.its.mas.big.users.admin.MemberNameForm name;

	  	String query = "SELECT firstname, lastname, email FROM company_members WHERE gameid = ? AND companyid = ? ORDER BY lastname, firstname" ;

	  	try
	  	{
			  PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			  pstmt.setInt(1, game.getGameID());
			  pstmt.setInt(2, company.getCompanyID());

			  ResultSet rset = pstmt.executeQuery();

			  while(rset.next())
			  {
				  //Place the first and last name into a MemberNameForm object
				  name = new edu.calpoly.its.mas.big.users.admin.MemberNameForm();
				  name.setFirst(rset.getString("firstname").trim());
				  name.setLast(rset.getString("lastname").trim());
				  name.setEmail(rset.getString("email").trim());

				  //Add the MemberNameForm object to the Vector
				  names.add(name);
			  }
			  rset.close();
			  pstmt.close();
	  	}
	  	catch(SQLException sqle)
	  	{
			  //close the connection if an exception occurs
			  close();
			  throw new BIGDatabaseFacadeException(sqle);
	  	}

	  	return names;
  	}

    public void addMemberName(Game game, Company company, String first, String last, String email) throws BIGDatabaseFacadeException
    {
		//SQL statement to insert a member name into the database
		String insert = "INSERT INTO company_members (gameID, companyID, firstname, lastname, email)";
		insert = insert + "VALUES (?, ?, ?, ?, ?)";

       	try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setString(3, first);
			pstmt.setString(4, last);
			pstmt.setString(5, email);

			//perform the insert
			int rowsAdded = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}
