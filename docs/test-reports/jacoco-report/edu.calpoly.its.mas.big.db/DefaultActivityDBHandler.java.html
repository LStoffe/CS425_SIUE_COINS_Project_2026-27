    public Activity getActivity(Game game, Job job, int periodNum, int activityType) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		Activity theActivity = null;

		float overtimeExpenseFactor = getOvertimeExpenseFactor(game);

		try	{
			String query = "SELECT a.lus_total, a.lus_completed, a.current_labor_cost, " +
				           "a.current_material_cost, a.current_equipment_cost, a.current_subcontractor_cost, a.is_active, " +
				           "a.lus_completed_this_period, a.periodid, " +
				           "a.current_method_used, " +
				           "a.accumulated_costs, o.overtime_days, " +
				           "gpm.standard_hrs_per_day, gpm.overtime_hrs_per_day, j.method_scaling_factor " +
				           "FROM activity a, overtime o, game_params_misc gpm, job j " +
				           "WHERE a.activity_type_id = ? AND a.gameid = ? AND " +
				           "a.jobid = ? AND a.periodid = ? AND o.activity_type_id = a.activity_type_id " +
				           "AND o.gameid = a.gameid AND o.jobid = a.jobid AND a.gameid = gpm.gameid AND j.jobid = a.jobid " +
				           "AND a.gameid = j.gameid" ;

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, activityType);
			pstmt.setInt(2, game.getGameID());
			pstmt.setInt(3, job.getJobID());
			pstmt.setInt(4, periodNum);

			ResultSet rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last()) {
				// Just return null or... ?
				// throw new ObjectNotFoundException()
			} else {
				rset.first();
				theActivity = new Activity(activityType,
										   rset.getInt("lus_total"),
										   rset.getInt("lus_completed"),
										   rset.getLong("current_labor_cost"),
										   rset.getLong("current_material_cost"),
										   rset.getLong("current_equipment_cost"),
										   rset.getLong("current_subcontractor_cost"),
										   rset.getBoolean("is_active"),
										   rset.getInt("lus_completed_this_period"),
										   rset.getLong("accumulated_costs"),
										   periodNum,
										   rset.getInt("current_method_used"),
										   rset.getInt("overtime_days"),
										   rset.getInt("standard_hrs_per_day"),
										   rset.getInt("overtime_hrs_per_day"));

			    theActivity.setMethodScalingFactor(rset.getFloat("method_scaling_factor"));
				theActivity.setOvertimeModifier(overtimeExpenseFactor);
			}
			//done with the result set
			rset.close();
			pstmt.close();
		} catch(SQLException sqle) {
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return theActivity;
	}

    public Activity[] getActivities(Game game, Job job, int periodNum) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		Activity[] activities = new Activity[9];

		float overtimeExpenseFactor = getOvertimeExpenseFactor(game);

		String query = "SELECT a.activity_type_id, a.lus_total, a.lus_completed, a.current_labor_cost, " +
				           "a.current_material_cost, a.current_equipment_cost, a.current_subcontractor_cost, a.is_active, " +
				           "a.lus_completed_this_period, a.periodid, " +
				           "a.current_method_used, " +
				           "a.accumulated_costs, o.overtime_days, " +
				           "gpm.standard_hrs_per_day, gpm.overtime_hrs_per_day, j.method_scaling_factor " +
				           "FROM activity a, overtime o, game_params_misc gpm, job j " +
				           "WHERE a.gameid = ? AND " +
				           "a.jobid = ? AND a.periodid = ? AND o.activity_type_id = a.activity_type_id " +
				           "AND o.gameid = a.gameid AND o.jobid = a.jobid AND a.gameid = gpm.gameid " +
				           " AND j.jobid = a.jobid AND j.gameid = a.gameid ORDER BY activity_type_id";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(query);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());
			pstmt.setInt(3, periodNum);

			ResultSet rset = pstmt.executeQuery();

			int i = 0;
			while(rset.next())
			{
				activities[i] = new Activity(rset.getInt("activity_type_id"),
											 rset.getInt("lus_total"),
											 rset.getInt("lus_completed"),
											 rset.getLong("current_labor_cost"),
											 rset.getLong("current_material_cost"),
											 rset.getLong("current_equipment_cost"),
											 rset.getLong("current_subcontractor_cost"),
											 rset.getBoolean("is_active"),
											 rset.getInt("lus_completed_this_period"),
											 rset.getLong("accumulated_costs"),
											 periodNum,
											 rset.getInt("current_method_used"),
											 rset.getInt("overtime_days"),
											 rset.getInt("standard_hrs_per_day"),
											 rset.getInt("overtime_hrs_per_day"));

				activities[i].setMethodScalingFactor(rset.getFloat("method_scaling_factor"));
				activities[i].setOvertimeModifier(overtimeExpenseFactor);
				i++;
			}

			rset.close();

			if (i < 8)
			{
				throw new ObjectNotFoundException();
			}
		}
		catch(SQLException sqle)
		{
			//Close the connection if there's an error
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return activities;
	}


	private float getOvertimeExpenseFactor(Game game) throws BIGDatabaseFacadeException
	{
		float overtimeExpenseFactor = 2.0f;

		try	{

			// Retrieve the overtime expense factor
			String miscParamQuery = "SELECT overtime_expense_factor " +
				                    "FROM game_params_misc WHERE gameid = ? and adminid = ?";

			PreparedStatement pstmt = connection.prepareStatement(miscParamQuery, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, game.getAdminID());

			ResultSet rset = pstmt.executeQuery();

			if (rset.last())
			{
				rset.first();
				overtimeExpenseFactor = rset.getFloat("overtime_expense_factor");
			}

			rset.close();
			pstmt.close();

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return overtimeExpenseFactor;
	}

    public void updateActivity(Game game, Job job, Activity theActivity) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		//SQL statement to update password field of admin
		String update = "UPDATE activity SET lus_total = ?, lus_completed = ?, current_labor_cost = ?, " +
		                "current_material_cost = ?, current_equipment_cost = ?, current_subcontractor_cost = ?, is_active = ?, lus_completed_this_period = ?, " +
		                "accumulated_costs = ?, current_method_used = ? WHERE gameid = ? AND jobid = ? AND activity_type_id = ? " +
		                "AND periodid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, theActivity.getLusTotal());
			pstmt.setInt(2, theActivity.getLusCompleted());
			pstmt.setLong(3, theActivity.getCurrentLaborCost());
			pstmt.setLong(4, theActivity.getCurrentMaterialCost());
			pstmt.setLong(5, theActivity.getCurrentEquipmentCost());
			pstmt.setLong(6, theActivity.getCurrentSubcontractorCost());
			pstmt.setBoolean(7, theActivity.getIsActive());
			pstmt.setLong(8, theActivity.getLusCompletedThisPeriod());
			pstmt.setLong(9, theActivity.getAccumulatedCosts());
			pstmt.setInt(10, theActivity.getCurrentMethodUsed());
			pstmt.setInt(11, game.getGameID());
			pstmt.setInt(12, job.getJobID());
			pstmt.setInt(13, theActivity.getActivityType());
			pstmt.setInt(14, theActivity.getCurrentPeriod());

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();

			if (rowsChanged < 1)
				throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)
		{
			//Close connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}
