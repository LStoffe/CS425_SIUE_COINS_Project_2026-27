package edu.calpoly.its.mas.big.db;

import java.util.Vector;

import edu.calpoly.its.mas.big.game.Loan;
import edu.calpoly.its.mas.big.users.company.Company;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>LoanDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.4 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultLoanDBHandler extends DefaultDBHandler implements LoanDBHandler
{

    public DefaultLoanDBHandler(Connection connection)
    {
		super(connection);
	}

    public Vector getLoans(Company company, int periodNum) throws BIGDatabaseFacadeException
    {
		Vector loans = new Vector();
		try
		{
		   String query = "SELECT loanid, amount, balance, payment_amount, payments_remaining, interest_paid, is_approved, loan_interest_per_year, repay_immediately " +
		                  " FROM loan, game_params_misc WHERE companyid = ? AND periodid = ? AND loan.gameid=game_params_misc.gameid ORDER BY loanid";

		   PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
		   pstmt.setInt(1, company.getCompanyID());
		   pstmt.setInt(2, periodNum);

		   ResultSet rset = pstmt.executeQuery();

           while(rset.next())
           {
		      Loan l = new Loan(rset.getInt("loanid"), rset.getInt("amount"), rset.getInt("balance"), rset.getInt("loan_interest_per_year"),
		                        rset.getInt("payment_amount"), rset.getInt("payments_remaining"), rset.getInt("interest_paid"),
		                        rset.getBoolean("is_approved"), rset.getBoolean("repay_immediately"));
		      loans.add(l);
		   }

		   rset.close();
		   pstmt.close();
		}
		catch(SQLException sqle)
		{
		   //close the connection if an exception occurs
		   close();
		   throw new BIGDatabaseFacadeException(sqle);
    	}

    	return loans;
    }

    public void updateLoan(Loan theLoan) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		//SQL statement to update a loan
		String update = "UPDATE loan SET amount = ?, balance = ?, payments_remaining = ?, payment_amount = ?, interest_paid = ?, repay_immediately = ? WHERE loanid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setLong(1, theLoan.getAmount());
			pstmt.setLong(2, theLoan.getBalance());
			pstmt.setInt(3, theLoan.getPaymentsRemaining());
			pstmt.setLong(4, theLoan.getPaymentAmount());
			pstmt.setLong(5, theLoan.getInterestPaid());
			pstmt.setBoolean(6, theLoan.repayImmediately());
			pstmt.setInt(7, theLoan.getLoanID());

			int rowsChanged = pstmt.executeUpdate();
			connection.commit();

			pstmt.close();

			if (rowsChanged < 1)
				throw new ObjectNotFoundException();
		}
		catch(SQLException sqle)
		{
			//Close connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}


    public void addLoan(Company company, int periodNum, Loan theLoan) throws BIGDatabaseFacadeException
    {
		int loanid = 0;
		int gameid = 0;

		try
		{
			//Get the next loanID in the sequence
			String sequenceName = "big_loanid_sequence";
            String query = "SELECT nextval ('" + sequenceName + "')";

            PreparedStatement stmt = connection.prepareStatement(query);

			ResultSet rset = stmt.executeQuery();

			if(rset.next()) //A loanID value was returned from the database
			{
				loanid = rset.getInt(1);
		    }

			rset.close();
			stmt.close();

            //Get the gameid matching this company
            query = "SELECT gameid FROM company WHERE companyid = ?";

			stmt = connection.prepareStatement(query);
			stmt.setInt(1, company.getCompanyID());
            rset = stmt.executeQuery();

            if(rset.next()) //A gameid was found
            {
				gameid = rset.getInt(1);
			}

			rset.close();
			stmt.close();

            if((loanid > 0) && (gameid > 0))
            {

				//Assign the new loanID to the Loan to be stored
				theLoan.setLoanID(loanid);

				//Create the SQL statement used to insert the new loan into the database.
				String insert = "INSERT INTO loan (loanid, periodid, companyid, gameid, amount, balance, payment_amount, interest_paid, payments_remaining, is_approved, repay_immediately) ";
				insert = insert + "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

				//Perform the insert
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, loanid);
				pstmt.setInt(2, periodNum);
				pstmt.setInt(3, company.getCompanyID());
                pstmt.setInt(4, gameid);
                pstmt.setLong(5, theLoan.getAmount());
                pstmt.setLong(6, theLoan.getBalance());
                pstmt.setLong(7, theLoan.getPaymentAmount());
                pstmt.setLong(8, theLoan.getInterestPaid());
                pstmt.setInt(9, theLoan.getPaymentsRemaining());
                pstmt.setBoolean(10, theLoan.isApproved());
                pstmt.setBoolean(11, false);

				int rowsAdded = pstmt.executeUpdate();
				connection.commit();

				pstmt.close();

			}
			else  //A loanID was not returned from the database
			{
				rset.close();
				close();
				throw new BIGDatabaseFacadeException("Couldn't retrieve necessary data values" );
			}

		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    public void removeLoan(int companyID, int periodID) throws BIGDatabaseFacadeException
    {
		//String to remove the loans for this period that aren't approved
		String update = "DELETE FROM loan WHERE companyID = ? AND  is_approved = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, companyID);
			pstmt.setBoolean(2, false);

			int rowsChanged = pstmt.executeUpdate();

			if (rowsChanged > 1)   //Exactly 0 or 1 loans should have been removed.  Anymore is an error.
			{
				connection.rollback();
				throw new BIGDatabaseFacadeException("removeLoan: \n " + rowsChanged + " loans were attempted to be removed from the database.  Only 0 or 1 loans can be removed.");
			}

			connection.commit();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    public void removeLoans(int companyID) throws BIGDatabaseFacadeException
    {
		//String to remove the loans owned by the company passed in
		String update = "DELETE FROM loan WHERE companyID = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, companyID);

			int rowsChanged = pstmt.executeUpdate();
			connection.commit();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

}
