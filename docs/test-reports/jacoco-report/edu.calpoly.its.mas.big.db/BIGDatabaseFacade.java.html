/**
 * Implementation of <code>BIGDatabaseFacadeInterface</code>.  See <code>BIGDatabaseFacadeInterface</code>.
 * <br><br>
 * A connection to the database will be established automatically whenever a method is called that needs to access it.
 * By default this connection will be closed as soon as the method finishes.  Optionally the database connection can be held
 * until the user requests its release using the close() method.  This will save overhead if multiple database accesses are needed.
 * <br><br>
 * <b>Note:</b>All components of BIG should access the database via the BIGDatabaseFacade and never directly.
 * <br>
 * <br>
 * <b> NOTE-4/2/2004: it seems the connection is held until the close()
 * method is called, no matter what.  Always call the close() method whenever you
 * finish with an instance of BIGDatabaseFacade.</b>
 * @author Kevin Craig
 *
 * @version $Revision: 1.1 $ $Date: 2002/03/14 18:53:05 $
 */
public class BIGDatabaseFacade implements BIGDatabaseFacadeInterface
{
   private DataSource dataSource;
   private Connection connection;
   private Statement stmt;
   private static SqlMapClient sqlMap;
   private static Log log;
   boolean holdConnection = false;

   /**
    * Constructor for BIGDatabaseFacade.
    * Uses the iBatis sqlMaps.xml configuration file to setup <code>SqlMapClient</code>
    */
   public BIGDatabaseFacade()
   {
       initializeClass();
  	   this.dataSource = sqlMap.getDataSource();
   }

   /**
    * Constructor for BIGDatabaseFacade
    * @param data A javax.sql.DataSource associated with the database
    */
   public BIGDatabaseFacade(DataSource dataSource)
   {
       this.dataSource = dataSource;
       initializeClass();
       this.dataSource = sqlMap.getDataSource();
   }

  /**
   * Constructor for BIGDatabaseFacade that allows user to specify whether or not to hold the database connection
   * or release it after every method call.  This saves the overhead of having to establish a connection each time
   * when multiple methods of the <code>BIGDatabaseFacade</code> are going to be called in sequence.
   * @param data A javax.sql.DataSource associated with the database
   * @param holdConnection true if connection should be held until BIGDatabaseFacade.close() is called.  False
   *        if the connection should be released after each method call.
   */
   public BIGDatabaseFacade(DataSource dataSource, boolean holdConnection)
   {
        this.dataSource = dataSource;
	    this.holdConnection = holdConnection;
	    // never hold connections open.
	    this.holdConnection = false;
        initializeClass();
        this.dataSource = sqlMap.getDataSource();
   }


   // called by the constructors
   private void initializeClass()
   {
	    log = LogFactory.getLog(this.getClass());

		BIGGetSqlMapClient singleton = BIGGetSqlMapClient.getInstance();
		sqlMap = singleton.sqlMap();

	    if(log.isDebugEnabled())
	    {
			log.debug(this.getClass() + ": object 'dbFacade' created successfully");
	    }
   }


  /**
   * Initializes a connection to the database and prepares a <code>Statement</code>
   */
  private void prepare() throws BIGDatabaseFacadeException
  {
	 if (connection == null)
	 {
	 	try
	 	{
	 	   connection = dataSource.getConnection();
	 	   stmt = connection.createStatement();
     	}
     	catch (SQLException sqle)
     	{
			throw new BIGDatabaseFacadeException(sqle);
	 	}
	 }
  }


  /**
   * Releases connection to the database if <code>holdConnection</code> is set to false.
   */
  private void release() throws BIGDatabaseFacadeException
  {
	 try
     {
		if (stmt != null) stmt.close();
		stmt = null;

		if (connection != null) connection.close();
		connection = null;
	 }
     catch (SQLException sqle)
     {
		throw new BIGDatabaseFacadeException(sqle);
	 }
  }


  /**
   * Closes this object's connection to the database.  Should be called whenever the client code is done
   * with the database or will not be using it for a while.  A new connection is established if another database access method
   * is called.
   */
  public void close() throws BIGDatabaseFacadeException
  {
	  holdConnection = false;
	  release();
  }
