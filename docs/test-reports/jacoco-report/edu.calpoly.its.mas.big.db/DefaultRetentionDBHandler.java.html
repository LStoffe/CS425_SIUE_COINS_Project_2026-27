package edu.calpoly.its.mas.big.db;

import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>RetentionDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.2 $ $Date: 2002/03/14 18:53:05 $
 */
public class DefaultRetentionDBHandler extends DefaultDBHandler implements RetentionDBHandler
{

    public DefaultRetentionDBHandler(Connection connection)
    {
		super(connection);
	}


	public int[][] getRetentionPolicies() throws BIGDatabaseFacadeException
	{
		int[][] retentionPolicies = new int[5][2];

		String query = "SELECT retentionid, percent_retention, percent_of_job FROM retention ORDER BY retentionid";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			ResultSet rset = pstmt.executeQuery();

			int i = 0;
			while(rset.next())
			{
				retentionPolicies[i][0] = rset.getInt("percent_of_job");
				retentionPolicies[i][1] = rset.getInt("percent_retention");
				i++;
			}

			rset.close();

		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return retentionPolicies;
	}

}
