package edu.calpoly.its.mas.big.db;



import java.sql.SQLException;



/**

 * Exception thrown when the <code>BIGDatabaseFacadeInterface</code> cannot run the requested query or update.

 * Can signify a problem connecting to the database.

 *

 * @author Kevin Craig

 *

 * @version $Revision: 1.1 $ $Date: 2002/03/14 18:53:05 $

 */

public class BIGDatabaseFacadeException extends Exception

{



   String message;

   SQLException sqle;



   public BIGDatabaseFacadeException()

   {

   }



   public BIGDatabaseFacadeException(String message)

   {

	   this.message = message;

   }





   public BIGDatabaseFacadeException(SQLException sqle)

   {

	   this.sqle = sqle;

   }









   /**

    * @return SQLException A <code>SQLException</code> that may provide more information about the error.

    * Null if there was no <code>SQLException</code>.

    *

    */

   public SQLException getSQLException()

   {

	   return sqle;

   }



   /**

    * @return String This exception as a String.

    *

    */

   public String toString()

   {

	   String exceptionString = "edu.calpoly.its.mas.big.game.BIGDatabaseFacadeException";



	   if(message != null)

	   {

		   exceptionString = exceptionString + "\n" + message;

	   }



	   if (sqle != null)

	   {

		   exceptionString = exceptionString + "\n " + sqle.toString();

	   }



	   return exceptionString;

   }

}
