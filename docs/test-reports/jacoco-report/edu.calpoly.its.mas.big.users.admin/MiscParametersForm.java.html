	// This is now a misnomer, overtimeExpenseFactor is a Float rather than an Integer
	private Hashtable miscParameterIntegers = new Hashtable(17);
	private Hashtable miscParameters = new Hashtable(17);
	boolean makeDefaults = false;

    public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
    {
        ActionErrors errors = new ActionErrors();
        boolean nullParams = false;
        boolean integerParseError = false;
        boolean negativeFound = false;

        // If the 'Restore Defaults' button was pressed, validation should not take place
        // This makes the assumption that the defaults are valid, but prevents
		// problem that arises if the user enters invalid data and then tries to
    	// restore defaults.

        String testRestoreDefaults = (String) request.getParameter("restoreDefaults");
        if (testRestoreDefaults != null)
        {
            return(null);
        }

        // Eeek!
		for (Enumeration keys = miscParameters.keys(); keys.hasMoreElements() ;)
		{
			String currentKey = (String) keys.nextElement();

            // Check for null or empty Strings
            if((((String) miscParameters.get(currentKey)).trim() == null) ||
               (((String) miscParameters.get(currentKey)).trim().length() < 1))
            {
                nullParams = true;
            }
            else
            {
                try
                {
					// overtimeExpense factor is a float, the remaining parameters are integers
					if (currentKey.equals("overtimeExpenseFactor"))
					{
						// Attempt String->flota conversion and populate 'int' table
						miscParameterIntegers.put(currentKey,

						new Float((String) miscParameters.get(currentKey)));

						// Verify that this number isn't negative

						if (((Float) miscParameterIntegers.get(currentKey)).floatValue() < 0.0f)
						{
							negativeFound = true;

						}

					}
					else
					{
						// Attempt String->int conversion and populate int table
						miscParameterIntegers.put(currentKey,
												  new Integer((String) miscParameters.get(currentKey)));


						// Verify that this number isn't negative
						if (((Integer) miscParameterIntegers.get(currentKey)).intValue() < 0)
						{
							negativeFound = true;
						}
                    }

                }
                catch(NumberFormatException nfe)
                {
                    integerParseError = true;
                }
            }
		}

        if (nullParams == true)
        {
            errors.add(ActionMessages.GLOBAL_MESSAGE,
                       new ActionMessage("error.parameters.null"));
        }

        if (integerParseError == true)
        {
            errors.add(ActionMessages.GLOBAL_MESSAGE,
                       new ActionMessage("error.parameters.number.format"));
        }

        if (negativeFound == true)
        {
            errors.add(ActionMessages.GLOBAL_MESSAGE,
                       new ActionMessage("error.parameters.number.negative"));
        }

        return errors;
	}

	public Hashtable getMiscParameters()
	{
		return miscParameterIntegers;
	}

	public void setMiscParameters(Hashtable miscParameters)
	{
		this.miscParameterIntegers = miscParameters;

		// Populate the String Hashtable
		for (Enumeration keys = miscParameters.keys(); keys.hasMoreElements() ;)
		{
			String currentKey = (String) keys.nextElement();

			if (currentKey.equals("overtimeExpenseFactor"))
			{
				this.miscParameters.put(currentKey,
										((Float) this.miscParameterIntegers.get(currentKey)).toString());
			}
			else
			{
				this.miscParameters.put(currentKey,
										((Integer) this.miscParameterIntegers.get(currentKey)).toString());
			}
		}
	}
