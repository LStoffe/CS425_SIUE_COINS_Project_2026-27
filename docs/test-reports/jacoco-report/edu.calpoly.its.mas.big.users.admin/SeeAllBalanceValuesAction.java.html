package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Vector;
import java.util.Enumeration;
import java.util.List;
import java.util.ArrayList;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.lib.multimedia.big.reports.BalanceSheetForm;
import edu.calpoly.lib.multimedia.big.reports.ContractsInProgressReportForm;
import edu.calpoly.lib.multimedia.big.reports.CompletedContractsReportForm;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Handles a request to see the balance sheet values (A = L + E) for all companies in all games.
 * Only the head administrator is able to perform this action.
 *
 */
public class SeeAllBalanceValuesAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			//Check to make sure the user who made the request is the head admin
	        HttpSession session = request.getSession();
	        Admin admin = (Admin) session.getAttribute("user");

	        if(!admin.getIsHeadAdmin())  //User is not the head admin
	        {
	        	errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.user.permission"));
			}
			else  //User is the head admin
			{
				// Get a list of all companies in all games
				BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
				try
				{
                    //list of all games
                    Vector gameidVector = dbFacade.getGameDBHandler().getAllGames();
         			if((gameidVector != null) && (gameidVector.size() != 0))
         			{
						List listOfAllBalShts = new ArrayList();
             			for (Enumeration e = gameidVector.elements(); e.hasMoreElements();  )
             			{
		        			Game game = (Game) e.nextElement();
		        			//list of all companies in this game
		        			Vector companies = dbFacade.getCompanyDBHandler().getCompanies(game);
							for(int i = 0; i < companies.size(); i++)
							{
								// if a game has not yet been started, and has companies added to it, this routine will hang because
								// there is no balance sheet created yet for the company.  Note: period = 0 for games not yet started.
								if (game.getCurrentPeriodNumber() > 0)
         						{
									Company company = (Company) companies.get(i);
									BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, game.getCurrentPeriodNumber() - 1, company);
									ContractsInProgressReportForm contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, game.getCurrentPeriodNumber() - 1, company);
 		 	      					CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, game.getCurrentPeriodNumber() - 1, company);

				  					balanceSheet.setUnderbilledAmount(contractsInProgress.getTotalUnderbilling() );
				  					balanceSheet.setUnbilledAmount(completedContracts.getTotalUnbilled() );
				  					balanceSheet.setOverbilledAmount(contractsInProgress.getTotalOverbilling() );

									balanceSheet.setGameid(game.getGameID() );
									balanceSheet.setCompanyid(company.getCompanyID() );
									balanceSheet.setCurrentPeriodNum(game.getCurrentPeriodNumber() - 1 );
									balanceSheet.setUsername(company.getUsername() );
									balanceSheet.setPassword(company.getPassword() );

									listOfAllBalShts.add(balanceSheet);
								}
							}
             			}
             			session.setAttribute("listOfAllBalShts", listOfAllBalShts);
             			log.debug("Size of List is: " + listOfAllBalShts.size() + " items.");
	      			}
				}
				catch(BIGDatabaseFacadeException BIGDbe)
				{
					//Error with the database
					log.fatal("", BIGDbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch (ObjectNotFoundException onfe)
				{
					log.fatal("", onfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}
				finally
				{
					try
					{
				   		dbFacade.close();
					}
					catch (BIGDatabaseFacadeException e)
					{
						log.fatal(getClass().getName(), e);
				 		errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
					}
				}
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}
	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}
}
