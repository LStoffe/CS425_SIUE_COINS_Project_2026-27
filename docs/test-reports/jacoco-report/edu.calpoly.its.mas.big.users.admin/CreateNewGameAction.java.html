/**
 * Handles a request to create a new game.
 *
 * @author Kevin Craig
 *  Modified 3/21/02 Dara Manker: added code to add company login checking
 *
 * @version $Revision: 1.16 $ $Date: 2002/05/01 16:58:00 $
 */
public class CreateNewGameAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			String gameName = ( (GameForm) form).getGameName();
			String startYear = ( (GameForm) form).getStartYear();
			boolean building = ( (GameForm) form).getBuilding();
			boolean heavy = ( (GameForm) form).getHeavy();

			HttpSession session = request.getSession();
			Admin admin = (Admin) session.getAttribute("user");

            int adminID = admin.getAdminID();

            //Ensure user's adminID is >= 1.  Valid adminID's begin at 1.
            if(adminID > 0)
            {

            	//Create a date string for Jan. 1st of the start year
            	String startDateString = "1/1/" + startYear;

            	//Convert date string to Date object
            	Date startDate = (new SimpleDateFormat("MM/dd/yyyy")).parse(startDateString, new ParsePosition(0));

                //Determine value for gametype
                int gametype = 0;
                if (building && !heavy) { gametype = 1; }
                if (!building && heavy) { gametype = 2; }
                if (building && heavy)  { gametype = 3; }
                //System.out.println("gametype = " + gametype);

            	Game game = new Game(adminID, gameName, startDate, gametype);

            	try
            	{
                    BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
					dbFacade.getGameDBHandler().addGame(game);

                    //Create a set of starting parameters for the new game by making a copy of the administrators defaults
                    dbFacade.getGameParametersDBHandler().copyAdminDefaultParameters(admin, game.getGameID()); //game was assigned a gameID by the dbFacade.addGame method

					dbFacade.getComponentsDBHandler().newEstimatingComponentEntry(game);
					dbFacade.getEstimatingComponentDBHandler().initializeEstimatingComponentDatabase(game);
					dbFacade.getGradingTemplateDBHandler().createGradingTemplate(admin, game.getGameID());

                    dbFacade.close();

					//Place this game in the session for the administrator to edit
					session.setAttribute("game", game);

					// Place a GameLoginInfo object into the ServletContext.
					// The GameLoginInfo object will be used by BIG to determine when
					// companies should be allowed to log into this game.
					ServletContext context = servlet.getServletContext();

					if(context != null)
					{
                       LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
					   loginBroker.addGameLoginInfoToServletContext(game, context);
				    }
				}
				catch(BIGDatabaseFacadeException BIGdbe)
				{
					log.fatal("", BIGdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
			}
			else
			{
				if(log.isWarnEnabled())
				{
					log.warn(this.getClass() + " user permission error");
				}
				//User's adminID was < 1.  Valid adminID's begin at 1
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.user.permission"));
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));

		}

	}

}
