package edu.calpoly.its.mas.big.users.admin;

//Serlvet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.ServletContext;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.ExpandedGame;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Begins the process of updating the period of a game.
 *
 * <code>UpdatePeriodAction</code> simply instantiates an <code>ExpandedGame</code>
 * object, gives it a reference to a <code>BIGDatabaseFacade</code> and lets it
 * handle the update process.
 *
 * @author Kevin Craig
 *  Modified 3/21/02 Dara Manker: added code to facilitate company login verification
 *
 * @version $Revision: 1.5 $ $Date: 2002/04/12 17:44:20 $
 */

public class UpdatePeriodAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		HttpSession session = request.getSession();
		Game game = (Game) session.getAttribute("game");

		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method for game # " + game.getGameID() );
		}

		//Process request if it was not cancelled or initiated by an Admin's double-click on a menu option
        if(  (!isCancelled(request)) && (session.getAttribute("doNotProcess")==null)  )
        {
			// Set flag to prevent an Admin's double-click from rerunning this update (flag is removed in the jsp)
			session.setAttribute("doNotProcessFlag", "true");
		    //System.out.println("doNotProcessFlag is now true");

		    Admin admin = (Admin) session.getAttribute("user");
		    if(admin == null)
		    {
				return mapping.findForward("error");
			}

			ActionMessages errors = new ActionMessages();
            ServletContext context = servlet.getServletContext();

            // Temporarily disable company logins to this game
            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
            loginBroker.disableGameLogins(game, context);

            if(log.isInfoEnabled())
            {
            	log.info("UpdatePeriodAction: game logins disabled.");
			}

  			MessageResources resources = getResources(request);
  			//BIGDatabaseFacade dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
  			BIGDatabaseFacade dbFacade = new BIGDatabaseFacade();

            try
            {
				//Load default value for from e-mail account
				String fromAddress = resources.getMessage("emailFromDefaultAddress");

				//Load first part of e-mail message
			    String emailMessage = resources.getMessage("email.message");
				emailMessage += "\n";

			    //If the admin has provided their e-mail address, use that as the from
			    //rather than the default.  Include the message that
			    //informs students they can reply to the e-mail to
			    //e-mail the admin.  Otherwise include message that
			    //askes students not to reply to the e-mail
				if(admin.getEmail().trim().length() > 1)
				{
					fromAddress = admin.getEmail();
					emailMessage += resources.getMessage("email.replyToInstructor");
				}
				else
				{
					emailMessage += resources.getMessage("email.autogenerated");
			    }

				ExpandedGame xGame = new ExpandedGame(game, dbFacade);
				xGame.setEmailValues(resources.getMessage("emailServer"),
									 resources.getMessage("emailPort"),
									 fromAddress,
									 resources.getMessage("email.subject"),
									 emailMessage
									);
				xGame.updatePeriod();
				dbFacade.close();
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}


            // Now that period has been updated, record this information in the GameLoginInfo object/
            // This will set the number of logged in companies to zero, increment the period number,
            // and reenable logins.
            loginBroker.initializePeriod(game, context);

            if(log.isInfoEnabled())
            {
            	log.info("UpdatePeriodAction: game logins have been reenabled, period has been updated for game # " + game.getGameID() );
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }  // end if doNotProcess

   	    else  //Request was cancelled or a double-click initiated this servlet
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}



	}
}
