package edu.calpoly.its.mas.big.users.admin;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;

//BIG imports
import edu.calpoly.its.mas.big.users.company.Company;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Handles a request to add an administrator.
 * Only the head administrator is able to perform this action.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.17 $ $Date: 2002/03/14 18:53:06 $
 */
public class AddAdminAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			//Check to make sure the user who made the request is the head admin
	        HttpSession session = request.getSession();
	        Admin admin = (Admin) session.getAttribute("user");

	        if(!admin.getIsHeadAdmin())  //User is not the head admin
	        {
	        	errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.user.permission"));
			}
			else  //User is the head admin
			{
				//Get username and password for new admin from form
                                   //Username will be stored as lower case for case insensitivty.  Passwords are case sensitive
				String adminName = ((AddAdminForm) form).getAdminName().toLowerCase();
				String password = ((AddAdminForm) form).getPassword();

                //These will be used to test if an administrator or company with the adminName given already exists in the database
				Admin testAdmin = null;
				Company testCompany = null;

				try
				{
					//Create a new BIGDatabaseFacade that hold connections - it will be used more than once
				    BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

					//Check if there is already an admin with this username
					testAdmin = dbFacade.getAdminDBHandler().getAdmin(adminName);

					//Check if there is already a company with this username
					testCompany = dbFacade.getCompanyDBHandler().getCompany(adminName);


					if((testAdmin == null) && (testCompany == null))  //Username is not in use
					{
						Admin newAdmin = new Admin(0, adminName, password, null, false);
						dbFacade.getAdminDBHandler().addAdmin(newAdmin);

						//Create a set of default parameters for the new admin.  Note: newAdmin was given an admin id by the addAdmin method of dbFacade
						dbFacade.getGameParametersDBHandler().copyGlobalDefaultParameters(newAdmin.getAdminID());

						//Put the admin that was just added into the session so it can be displayed
						//in a confirmation message to the user
						session.setAttribute("adminJustAdded", newAdmin);
					}
					else
					{
						//The username was already in use
						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.username.duplicate"));
					}

					//Close connection
					dbFacade.close();

				}
				catch(BIGDatabaseFacadeException BIGDbe)
				{
					//Error with the database
					log.fatal("", BIGDbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}

			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));

		}

	}



}
