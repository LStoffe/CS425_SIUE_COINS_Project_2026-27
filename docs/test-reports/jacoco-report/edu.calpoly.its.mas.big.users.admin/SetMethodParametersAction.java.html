package edu.calpoly.its.mas.big.users.admin;


//JDK imports
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Determines whether the administrator wishes to set parameters or restore defaults for method parameters then performs the appropriate action.
 * If the administrator wishes to store his current parameters settings to the database s/he would have pushed the set parameters button
 * (which must have the name "submit").  In this case the values currently in the MethodParametersForm in the session will be
 * stored to the database.  These parameters will also be made the admin's defaults if the getMakeDefaults method of the form returns true.
 * <br><br>
 * If the administrator pressed some other submit button (any name other than "submit" or "cancel") the parameters in the MethodParametersForm
 * will be overwritten with a copy of the admin's defaults for the current parameters.  These parameters will also be re-copied
 * into the database overwriting the current game parameters.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.9 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class SetMethodParametersAction extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();

			//Retrieve the current game and admin from the session
			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

			//Get the game parameters from the action form
			MethodParametersForm methodParametersForm = (MethodParametersForm) form;

            //Determine which submit button was pressed by the Administrator. If the SetParameters button is null the other RestoreDefaults button had to be pushed
            String testSetParametersSubmitButton = (String) request.getParameter("submit");

			//The parameters are in arrays
			String[] names = methodParametersForm.getMethodNames();
			int[] productivities = methodParametersForm.getProductivities();
			int[] laborCosts = methodParametersForm.getLaborCosts();
			int[] materialsCosts = methodParametersForm.getMaterialsCosts();
			int[] equipmentCosts = methodParametersForm.getEquipmentCosts();
			int[] subcontractorCosts = methodParametersForm.getSubcontractorCosts();
			int[] uncertaintyRanges = methodParametersForm.getUncertaintyRanges();
			int[] weatherSensitivities = methodParametersForm.getWeatherSensitivities();
			boolean[] subcontracted = methodParametersForm.getSubcontracted();

			//Create a database facade that will hold the connection since it will be used repeatedly
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

			//Hashtable used to transfer the parameters to the database facade
			Hashtable parameters = null;

			if (testSetParametersSubmitButton != null)  //"Set Parameters" submit button was pushed.
			{
				//Store the current parameters values from the form into the database
				try
				{
					for(int i = 0; i < 5; i++)
					{
						parameters = new Hashtable(9); //9 parameters to store

						parameters.put("Name", names[i]);
						parameters.put("Productivity", new Integer(productivities[i]));
						parameters.put("LaborCost", new Integer(laborCosts[i]));
						parameters.put("MaterialsCost", new Integer(materialsCosts[i]));
						parameters.put("EquipmentCost", new Integer(equipmentCosts[i]));
						parameters.put("SubcontractorCost", new Integer(subcontractorCosts[i]));
						parameters.put("ROU", new Integer(uncertaintyRanges[i]));
						parameters.put("WeatherSensitivity", new Integer(weatherSensitivities[i]));
						parameters.put("Subcontracted", new Boolean(subcontracted[i]));

						dbFacade.getGameParametersDBHandler().updateMethodParameters(game, methodParametersForm.getActivityTypeID(), i + 1, parameters);

						if (methodParametersForm.getMakeDefaults())
						{
							//Store these as default parameters for this admin as well
							dbFacade.getGameParametersDBHandler().updateDefaultMethodParameters(admin, methodParametersForm.getActivityTypeID(), i + 1, parameters);
						}

					}
					dbFacade.close();
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
				   //Problem with the database
				   log.fatal("", bigdbe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
			       //All games should have parameters.  If there are no parameters found for a game it is an error.
				   log.fatal("", pnfe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.notfound"));
				}
			}
			else //"Restore defaults" submit button was pushed
			{
				try
				{
					for(int i = 0; i < 5; i++)
					{
					    //Retrieve this admins default parameters from the database and re-store them for the current game
						parameters = dbFacade.getGameParametersDBHandler().getDefaultMethodParameters(admin, methodParametersForm.getActivityTypeID(), i+1);
						dbFacade.getGameParametersDBHandler().updateMethodParameters(game, methodParametersForm.getActivityTypeID(), i + 1, parameters);

						//Record the values in the arrays so they can be put in the Action Form
						names[i] = (String) parameters.get("Name");
						productivities[i] = ( (Integer) parameters.get("Productivity")).intValue();
						laborCosts[i] = ( (Integer) parameters.get("LaborCost")).intValue();
						materialsCosts[i] = ( (Integer) parameters.get("MaterialsCost")).intValue();
						equipmentCosts[i] = ( (Integer) parameters.get("EquipmentCost")).intValue();
						subcontractorCosts[i] = ((Integer) parameters.get("SubcontractorCost")).intValue();
						uncertaintyRanges[i] = ( (Integer) parameters.get("ROU")).intValue();
						weatherSensitivities[i] = ( (Integer) parameters.get("WeatherSensitivity")).intValue();
						subcontracted[i] = ((Boolean) parameters.get("Subcontracted")).booleanValue();

					}
					dbFacade.close();

					//The parameter vaules also need to be put in the Action Form so the user can see them
					methodParametersForm.setMethodNames(names);
					methodParametersForm.setProductivities(productivities);
					methodParametersForm.setLaborCosts(laborCosts);
					methodParametersForm.setMaterialsCosts(materialsCosts);
					methodParametersForm.setEquipmentCosts(equipmentCosts);
					methodParametersForm.setSubcontractorCosts(subcontractorCosts);
					methodParametersForm.setUncertaintyRanges(uncertaintyRanges);
					methodParametersForm.setWeatherSensitivities(weatherSensitivities);
					methodParametersForm.setSubcontracted(subcontracted);

				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
				   //Problem with the database
				   log.fatal("", bigdbe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
			       //All games should have parameters.  If there are no parameters found for a game it is an error.
				   log.fatal("", pnfe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.notfound"));
				}


			}
			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else if (testSetParametersSubmitButton != null)
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"params_set\"");
				}
				return (mapping.findForward("params_set"));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"defaults_restored\"");
				}
				return (mapping.findForward("defaults_restored"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}
}
