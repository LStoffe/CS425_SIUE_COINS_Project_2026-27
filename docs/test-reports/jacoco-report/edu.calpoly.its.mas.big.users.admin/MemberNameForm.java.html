/**
 * Contains a member name to be added to the database.
 *
 * @author Kevin Craig

package edu.calpoly.its.mas.big.users.admin;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

public class MemberNameForm extends ActionForm
{

  private String first;
  private String last;
  private String email;

  public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
  {
    ActionErrors errors = new ActionErrors();

    //Make sure there is either a first or last name
    if( ((last == null) || (last.length() < 1)) &&  ((first == null) || (first.length() < 1)) )
    	errors.add("name_required", new ActionMessage("error.editCompany.name.required"));

    return errors;
  }

  //*********************************  Simple Set and Get Methods  **************************

  public String getFirst()
  {
	  return first;
  }

  public void setFirst(String first)
  {
	  this.first = first;
  }

  public String getLast()
  {
	  return last;
  }

  public void setLast(String last)
  {
	  this.last = last;
  }

  public String getEmail()
  {
	  return email;
  }

  public void setEmail(String email)
  {
	  this.email = email;
  }


}
