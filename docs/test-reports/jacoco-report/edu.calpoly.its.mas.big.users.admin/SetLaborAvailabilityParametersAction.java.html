package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles Administrator's request to set a game's "labor availability per period" parameters.
 * Options are:
 * 1. the "Set Parameters" button was pressed; therefore parameters are saved to the database.
 *    (If the Administrator also specified that these parameters should become his defaults, they will be recorded as such in the database.)
 * 2. the "Restore Default Parameters" button was pressed: therefore game's default parameters are retrieved from the database.
 *    Currently, these default parameters are not saved as the game's default labor availability parameters.  It is left to the admin to set them by
 *    clicking on the "Set Parameters" button after these params have been displayed to him.
 *
 * Inputs: laborAvailabilityParameterForm parameters are valid
 *         laborAvailParams.jsp form button "Set Parameters" must be named "submit", and "Set To Defaults" button must be named "restoreDefaults".
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.8 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class SetLaborAvailabilityParametersAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {

            BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			ActionMessages errors = new ActionMessages();
			HttpSession session = request.getSession();

            // Get the game information
			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

            // Get the labor availability parameters form bean
            LaborAvailabilityParametersForm laborAvailParamForm = (LaborAvailabilityParametersForm) form;

            // This hashtable will be used to save the labor availability parameters to or get the labor availability parameters from the database
			Hashtable laborAvailParams = null;

            // Determine which submit button was pressed by the Administrator. The button that was not pressed will have a null value.
            String testSetParametersSubmitButton = (String) request.getParameter("submit");
            String testRestoreDefaultParametersSubmitButton = (String) request.getParameter("restoreDefaults");

            try
            {
               // Admin pressed "Set Parameters" button: save Admin's parameters to database
               if(testSetParametersSubmitButton != null)
               {
                  // Determine if Admin also wants to make these parameters his defaults
                  boolean setAsDefaults = laborAvailParamForm.getMakeDefaults();

                  // Save these parameters to the database. "setAsDefaults" will be 'true' if the parameters should also be set as the admin's defaults,
			   	  laborAvailParams = laborAvailParamForm.setAllParameters();
			      dbFacade.getGameParametersDBHandler().updateAllLaborAvailabilityParameters(game, admin, laborAvailParams, setAsDefaults);
	           }

               // Admin pressed "Reset" button: ignore the values in the form, and retrieve the Admin's default parameters from the database
               else if(testRestoreDefaultParametersSubmitButton != null)
               {
		          Hashtable laborAvailParamsHashtable = dbFacade.getGameParametersDBHandler().getAllLaborAvailabilityParameters(game, admin, true);
			      laborAvailParamForm.setAllParameters(laborAvailParamsHashtable);

				  dbFacade.close();
			      return (mapping.findForward("defaultParamsObtained"));
		       }

               // Admin pressed neither submit button: this should never happen since the Admin must press a submit button to get to this Action
               else
               {
				   if(log.isWarnEnabled())
				   {
				       log.warn(this.getClass() + ": could not determine which submit button was pressed");
				   }
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		       }  // end "if" test of submit buttons

			   dbFacade.close();
		    }
		    catch(ParametersNotFoundException pnfe)
		    {
		       //All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		    }
		    catch(BIGDatabaseFacadeException bigdbe)
		    {
			   //Problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			if (!errors.isEmpty())
			{
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"paramsSetSuccessfully\"");
				}
				return (mapping.findForward("paramsSetSuccessfully"));
			}
    	}

	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
	    }
   }
}
