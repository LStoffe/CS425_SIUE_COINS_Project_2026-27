package edu.calpoly.its.mas.big.users.admin;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

/**
 * ActionForm used to hold the liquidated damages parameters.
 *
 *
 * @author Kevin Craig
 *
 */
public final class LiqdamParametersForm  extends ActionForm
{
	private boolean makeDefaults = false;

    //Int variables are stored twice, ints to work with and String for the html form
    //Conversions are done when a get method is called.  For example if the int is set
    //for a variable and a get String method is called, the int will be converted to a
    //String by the getter.
    //If a String is defined for a variable it will be converted to an int
    //when the get methods is called.  All get methods that can return ints
    //might throw <code>NumberFormatException</code>s if the user entered an invalid value.
    //This can be avoided by calling the validate method before any get methods that return ints.
    private float median;
    private float stddev;
    private float min;
    private float max;
    private int percentJobs;
    private boolean fasterJobsIncrease = false;
    private boolean largerJobsIncrease = false;


    private String medianString = null;
    private String stddevString = null;
    private String minString = null;
    private String maxString = null;
    private String percentJobsString = null;


    //**************** Public Methods ******************



	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * that all numbers are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;
        boolean minsValid = true;


		if((medianString.trim() == null) || (medianString.trim().length() < 1))
			nullParams = true;
		if((stddevString.trim() == null) || (stddevString.trim().length() < 1))
		    nullParams = true;
		if((minString.trim() == null) || (minString.trim().length() < 1))
			nullParams = true;
		if((maxString.trim() == null) || (maxString.trim().length() < 1))
			nullParams = true;
		if((percentJobsString.trim() == null) || (percentJobsString.trim().length() < 1))
			nullParams = true;


		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the number Strings contain only numeric characters.

			//Check that the strings are valid numbers by trying to convert them to ints and checking for a NumberFormatException
			try
			{
				median = Float.parseFloat(medianString);
				stddev = Float.parseFloat(stddevString);
				min = Float.parseFloat(minString);
				max = Float.parseFloat(maxString);
				percentJobs = Integer.parseInt(percentJobsString);
			}
			catch(NumberFormatException nfe)
			{
				validNumbers = false;
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			if(median < 0)
				positiveNumbers = false;
			if(stddev < 0)
				positiveNumbers = false;
			if(min < 0)
				positiveNumbers = false;
			if(max < 0)
				positiveNumbers = false;
			if(percentJobs < 0)
				positiveNumbers = false;

		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		//Check that all min values are greater than or equal to their corresponding max value
		if(validNumbers && !nullParams)  //validNumbers == true AND nullParams == false
		{
			if(min > max)
				minsValid = false;
		}

		if(!minsValid && validNumbers && !nullParams)  //minsValid == false AND validNumbers == true AND nullParams == false
		{
			errors.add("minmax", new ActionMessage("error.parameters.minmax"));
		}

		return errors;
	}



//**************************** Set & Get Methods ************************
