package edu.calpoly.its.mas.big.users.admin;

import edu.calpoly.its.mas.big.users.BIGAction;

/**
 * Puts a <code>Vector</code> of gameIDs, stored as <code>Integer</code>s in the session.
 * Games in the <code>Vector</code> will be all those owned by the administrator making the request.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.6 $ $Date: 2002/03/14 18:53:06 $
 */
public class ChooseExistingGameAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
		  				         ActionForm form,
 		  				         HttpServletRequest request,
					             HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		ActionMessages errors = new ActionMessages();
		HttpSession session = request.getSession();

		Admin admin = (Admin) session.getAttribute("user");

		if (admin == null)  // "user" object no longer exists because the session has timed out
		   return (mapping.findForward("sessionTimedOut"));

		//Ensure user's adminID >= 1.  Valid adminID's begin at 1
		if (admin.getAdminID() > 0)
		{
			//Create a database facade and get a list of games from it that belong to this admin

			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			try
			{
				Vector games = dbFacade.getGameDBHandler().getGames(admin);
				dbFacade.close();

				//Put the Vector of games in the session for a scriplet to extract
				session.setAttribute("games", games);
			}
			catch(BIGDatabaseFacadeException BIGDbe)
			{
				log.fatal("", BIGDbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
		}
		else
		{
			if(log.isWarnEnabled())
			{
				log.warn(this.getClass() + " user permission error");
			}
			//User's adminID was < 1.  Valid adminID's begin at 1
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.user.permission"));
		}

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
		}
		return (mapping.findForward("success"));
	}
}
