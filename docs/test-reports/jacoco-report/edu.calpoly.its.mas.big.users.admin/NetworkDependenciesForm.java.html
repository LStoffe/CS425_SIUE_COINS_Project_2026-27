/**
 * ActionForm used to hold the game parameters for the network dependencies of a job type.
 * <br>
 * The network dependencies for an activity can be expressed as 4 integers:
 * <ul>
 *   <li>"STSActivity" - The activity id (activity number) of the activity that the current activity depends on to start</li>
 *   <li>"STSPercent" - The percentage the start-to-start activity that must be complete before work can begin on the current activity</li>
 *   <li>"FTFActivity" - The activity id (activity number) of the activity that the current activity depends on to finish</li>
 *   <li>"FTFPercent" - The percentage the current activity cannot complete past until the finish-to-finish activity has completed</li>
 * </ul>
 * <br><br>
 * <code>NetworkDependenciesForm</code> will store dependencies in this format with the integers being accessible in two different ways:
 * Individual parameters (ie "activity1StartToStartPercent" or "activity4FinishToFinishActivity") are available as a Strings via get and set methods.
 * The parameters are also available as Integers in an array of hashtables, with one hashtable for each activity.  The keys in the hashtables are
 * "<code>STSActivity</code>," "<code>STSPercent</code>," "<code>FTFActivity</code>," and "<code>FTFPercent</code>" and match the values described in
 * the list above.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.6 $ $Date: 2001/10/15 23:24:46 $
 */
public class NetworkDependenciesForm extends ActionForm
{

    /**
     * Returns the array of Hashtables.  The activity type id for each array is its array index plus one.
     * (ie the Hashtable in index 0 is the network dependency parameters for activity type 1)
     */
	public Hashtable[] getNetworkDependencies()
	{
		for(int i = 0; i < 9; i++)
		{
			activities[i] = new Hashtable(4);
			activities[i].put("STSActivity", new Integer(stsActivities[i]));
			activities[i].put("STSPercent", new Integer(stsPercents[i]));
			activities[i].put("FTFActivity", new Integer(ftfActivities[i]));
			activities[i].put("FTFPercent", new Integer(ftfPercents[i]));
		}

		return activities;
	}

	public void setNetworkDependencies(Hashtable[] activities)
	{
		this.activities = activities;

		for(int i = 0; i < 9; i++)
		{
			stsActivities[i] = ((Integer) activities[i].get("STSActivity")).toString();
			stsPercents[i] = ((Integer) activities[i].get("STSPercent")).toString();
			ftfActivities[i] = ((Integer) activities[i].get("FTFActivity")).toString();
			ftfPercents[i] = ((Integer) activities[i].get("FTFPercent")).toString();
		}
	}

	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * that all percentages are between 0 and 100 (inclusive) and that all activity ids are between
     * 1 and 9 (inclusive)
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean percentsOutOfRange = false;
        boolean idsOutOfRange = false;

        int[] stsActivitiesTemp = new int[9];
        int[] stsPercentsTemp = new int[9];
        int[] ftfActivitiesTemp = new int[9];
        int[] ftfPercentsTemp = new int[9];

        //First make sure none of the Strings are null
        for(int i = 0; i < 9; i++)
        {
			if((stsActivities[i].trim() == null) || (stsActivities[i].trim().length() < 1))
				nullParams = true;
		    if((stsPercents[i].trim() == null) || (stsPercents[i].trim().length() < 1))
		    	nullParams = true;
			if((ftfActivities[i].trim() == null) || (ftfActivities[i].trim().length() < 1))
				nullParams = true;
			if((ftfPercents[i].trim() == null) || (ftfPercents[i].trim().length() < 1))
				nullParams = true;
		}

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the number Strings contain only numeric characters.
			for(int i = 0; i < 9; i++)
			{
				//Check that the strings that should represent numbers only contain numberic characters by trying to convert them to ints and checking for a NumberFormatException
				try
			    {
					stsActivitiesTemp[i] = Integer.parseInt(stsActivities[i]);
					stsPercentsTemp[i] = Integer.parseInt(stsPercents[i]);
					ftfActivitiesTemp[i] = Integer.parseInt(ftfActivities[i]);
					ftfPercentsTemp[i] = Integer.parseInt(ftfPercents[i]);
				}
				catch(NumberFormatException nfe)
				{
					validNumbers = false;
				}
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the percent and ids are within their respective correct ranges
			for(int i = 0; i < 9; i++)
			{
				if ((stsActivitiesTemp[i] > 9) || (stsActivitiesTemp[i] < 0))
					idsOutOfRange = true;
				if ((ftfActivitiesTemp[i] > 9) || (ftfActivitiesTemp[i] < 0))
				    idsOutOfRange = true;
				if ((stsPercentsTemp[i] > 100) || (stsPercentsTemp[i] < 0))
				    percentsOutOfRange = true;
				if ((ftfPercentsTemp[i] > 100) || (ftfPercentsTemp[i] < 0))
				    percentsOutOfRange = true;

                                //If an activity dependency is set to none then it's associated percentage
                                //should be zero.  No reason to make this a validation error, since this can be fixed here
                                if (stsActivitiesTemp[i] == 0)
                                	stsPercents[i] = "0";
                                if (ftfActivitiesTemp[i] == 0)
					ftfPercents[i] = "0";
			}
		}

		if(percentsOutOfRange)
			errors.add("percentsOutOfRange", new ActionMessage("error.parameters.percentage.outOfRange"));

		//ID values should never be out of range because they aren't directly input by the user.  If they are out of range something weird has happened
		if(idsOutOfRange)
			errors.add("idsOutOfRange", new ActionMessage("error.unknown"));

		return errors;
	}
