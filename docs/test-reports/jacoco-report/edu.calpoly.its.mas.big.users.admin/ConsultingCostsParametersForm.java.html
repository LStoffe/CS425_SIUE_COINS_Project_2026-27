package edu.calpoly.its.mas.big.users.admin;

import java.util.Hashtable;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

/**
 * ActionForm used to hold the game parameters for the fees to view consulting reports for a game.
 * <br>
 * Each individual report cost can be accessed as a string through get and set method or all 5
 * can be retrieved at once in a <code>Hashtable</code> of <code>Integer</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.3 $ $Date: 2001/10/19 19:34:54 $
 */
public class ConsultingCostsParametersForm extends ActionForm
{
	private boolean makeDefaults = false;

	private String laborAvailability;
	private String materialsCostIndex;
	private String weatherForecast;
	private String marketAnalysis;
	private String completeListOfBids;
	private String camReport;



//****************************** Public Methods *************************

    /**
     * Returns the array of Hashtables.  The activity type id for each array is its array index plus one.
     * (ie the Hashtable in index 0 is the network dependency parameters for activity type 1)
     */
	public Hashtable getConsultingCosts()
	{
		Hashtable parameters = new Hashtable(5);
		parameters.put("Labor_Availability", new Integer(laborAvailability));
		parameters.put("Material_Cost_Index", new Integer(materialsCostIndex));
		parameters.put("Weather_Forecast", new Integer(weatherForecast));
		parameters.put("Market_Demand", new Integer(marketAnalysis));
		parameters.put("Complete_List_Of_Bids", new Integer(completeListOfBids));
		parameters.put("CAM_Report", new Integer(camReport));

		return parameters;
	}

    public void setConsultingCosts(Hashtable parameters)
    {
		laborAvailability = ((Integer) parameters.get("Labor_Availability")).toString();
		materialsCostIndex = ((Integer) parameters.get("Material_Cost_Index")).toString();
		weatherForecast = ((Integer) parameters.get("Weather_Forecast")).toString();
		marketAnalysis = ((Integer) parameters.get("Market_Demand")).toString();
		completeListOfBids = ((Integer) parameters.get("Complete_List_Of_Bids")).toString();
		camReport = ((Integer) parameters.get("CAM_Report")).toString();
	}


	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * and that all numbers are non-negative
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

		int laborAvailabilityInt = 0;
		int materialsCostIndexInt = 0;
		int weatherForecastInt = 0;
		int marketAnalysisInt = 0;
		int completeListOfBidsInt = 0;
		int camReportInt = 0;

        //Ensure all values are non-null
        if ( (laborAvailability.trim()) == null || (laborAvailability.trim().length() < 1) )
        	errors.add("laborAvailability", new ActionMessage("error.parameters.null"));
        if ( (materialsCostIndex.trim()) == null || (materialsCostIndex.trim().length() < 1) )
        	errors.add("materialsCostIndex", new ActionMessage("error.parameters.null"));
        if ( (weatherForecast.trim()) == null || (weatherForecast.trim().length() < 1) )
        	errors.add("weatherForecast", new ActionMessage("error.parameters.null"));
        if ( (marketAnalysis.trim()) == null || (marketAnalysis.trim().length() < 1) )
        	errors.add("marketAnalysis", new ActionMessage("error.parameters.null"));
        if ( (completeListOfBids.trim()) == null || (completeListOfBids.trim().length() < 1) )
        	errors.add("completeListOfBids", new ActionMessage("error.parameters.null"));
        if ( (camReport.trim()) == null || (camReport.trim().length() < 1) )
        	errors.add("camReport", new ActionMessage("error.parameters.null"));

        //Ensure all number strings contain only numeric characters
        try
        {
			laborAvailabilityInt = Integer.parseInt(laborAvailability);
		}
		catch (NumberFormatException e)
		{
			errors.add("laborAvailability", new ActionMessage("error.parameters.number.format"));
		}

		try
		{
			materialsCostIndexInt = Integer.parseInt(materialsCostIndex);
		}
		catch (NumberFormatException e)
		{
			errors.add("materialsCostIndexInt", new ActionMessage("error.parameters.number.format"));
		}

		try
		{
			weatherForecastInt = Integer.parseInt(weatherForecast);
		}
		catch (NumberFormatException e)
		{
			errors.add("weatherForecast", new ActionMessage("error.parameters.number.format"));
		}

		try
		{
			marketAnalysisInt = Integer.parseInt(marketAnalysis);
		}
		catch (NumberFormatException e)
		{
			errors.add("marketAnalysis", new ActionMessage("error.parameters.number.format"));
		}

		try
		{
			completeListOfBidsInt = Integer.parseInt(completeListOfBids);
		}
		catch (NumberFormatException e)
		{
			errors.add("completeListOfBids", new ActionMessage("error.parameters.number.format"));
		}

		try
		{
			camReportInt = Integer.parseInt(camReport);
		}
		catch (NumberFormatException e)
		{
			errors.add("camReport", new ActionMessage("error.parameters.number.format"));
		}

		//Ensure no numbers are negative
		if (laborAvailabilityInt < 0)
			errors.add("laborAvailability", new ActionMessage("error.parameters.number.negative"));
		if (materialsCostIndexInt < 0)
			errors.add("materialsCostIndex", new ActionMessage("error.parameters.number.negative"));
		if (weatherForecastInt < 0)
			errors.add("weatherForecase", new ActionMessage("error.parameters.number.negative"));
		if (marketAnalysisInt < 0)
			errors.add("marketAnalysis", new ActionMessage("error.parameters.number.negative"));
		if (completeListOfBidsInt < 0)
			errors.add("compelteListOfBids", new ActionMessage("error.parameters.number.negative"));
		if (camReportInt < 0)
			errors.add("camReport", new ActionMessage("error.parameters.number.negative"));


		return errors;
	}

//****************************** Simple Get and Set Methods *************************

	public boolean getMakeDefaults()
	{
		return makeDefaults;
	}

	public void setMakeDefaults(boolean makeDefaults)
	{
		this.makeDefaults = makeDefaults;
	}

	public String getLaborAvailability()
	{
		return laborAvailability;
	}

	public void setLaborAvailability(String laborAvailability)
	{
		this.laborAvailability = laborAvailability;
	}

	public String getMaterialsCostIndex()
	{
		return materialsCostIndex;
	}

	public void setMaterialsCostIndex(String materialsCostIndex)
	{
		this.materialsCostIndex = materialsCostIndex;
	}

	public String getWeatherForecast()
	{
		return weatherForecast;
	}

	public void setWeatherForecast(String weatherForecast)
	{
		this.weatherForecast = weatherForecast;
	}

	public String getMarketAnalysis()
	{
		return marketAnalysis;
	}

	public void setMarketAnalysis()
	{
		this.marketAnalysis = marketAnalysis;
	}

	public String getCompleteListOfBids()
	{
		return completeListOfBids;
	}

	public void setCompleteListOfBids(String completeListOfBids)
	{
		this.completeListOfBids = completeListOfBids;
	}

	public String getCamReport()
	{
		return camReport;
	}

	public void setCamReport(String camReport)
	{
		this.camReport = camReport;
	}

}
