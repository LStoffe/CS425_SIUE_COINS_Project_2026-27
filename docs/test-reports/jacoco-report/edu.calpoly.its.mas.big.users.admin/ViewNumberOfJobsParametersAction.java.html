package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import javax.sql.DataSource;
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles use case of viewing the number of jobs per period game parameters.
 * Uses the session attribute "user", which is an <code>Admin</code>, and the session attribute "game"
 * which is a <code>edu.calpoly.its.mas.big.Game</code>, to determine the current administrator
 * and game.  Retrieves from the database the number of jobs game parameters that go
 * with this admin and game and places them in a <code>NumberOfJobsForm</code> in the session
 * under the name "numberOfJobsParametersForm".  The game parameters in this form can be displayed to the user
 * using jsp.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.8 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class ViewNumberOfJobsParametersAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

            HttpSession session = request.getSession();

			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			//Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			Hashtable parameters = null;

			int[] medians = new int[12];
			int[] stddevs = new int[12];
			int[] mins = new int[12];
			int[] maxes = new int[12];
			int[] minsHC = new int[12];
			int[] maxesHC = new int[12];

			try
			{
				//Get the parameters for each period
				for (int i=0; i < 12; i++)
				{
					parameters = dbFacade.getGameParametersDBHandler().getNumberOfJobsParameters(game, i+1);

					if(parameters != null)
					{
						medians[i] = ( (Integer) parameters.get("Median")).intValue();
						stddevs[i] = ( (Integer) parameters.get("StdDev")).intValue();
						mins[i] = ( (Integer) parameters.get("Min")).intValue();
						maxes[i] = ( (Integer) parameters.get("Max")).intValue();
						minsHC[i] = ( (Integer) parameters.get("MinHC")).intValue();
						maxesHC[i] = ( (Integer) parameters.get("MaxHC")).intValue();
					}
					else
					{
						//The parameters for this game weren't there.  There should never be a game without parameters
						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
					}
				}
				dbFacade.close();

			}
			catch(ParametersNotFoundException pnfe)
			{
				// All games should have parameters.  If there are no parameters
				// found for a game it is an error.
				log.fatal(pnfe);

				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				// Some problem with the database
				log.fatal(bigdbe);

				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}


			NumberOfJobsParametersForm numJobsForm = new NumberOfJobsParametersForm();

			numJobsForm.setMedians(medians);
			numJobsForm.setStddevs(stddevs);
			numJobsForm.setMins(mins);
			numJobsForm.setMaxes(maxes);
			numJobsForm.setMinsHC(minsHC);
			numJobsForm.setMaxesHC(maxesHC);

			session.setAttribute("numberOfJobsParametersForm", numJobsForm);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}
}
