package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles use case of viewing the job size game parameters.
 * Uses information contained in the session to determine the current administrator
 * and game.  Retrieves from the database the job size game parameters that go
 * with this admin and game and places them in a <code>JobSizeParametersForm</code> in the session
 * under the name "jobSizeParams".  The game parameters in this form can be displayed to the user
 * using jsp.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.5 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class ViewJobSizeParametersAction extends BIGAction
{
    int gametype;

	public ActionForward execute(ActionMapping mapping,
							     ActionForm form,
							     HttpServletRequest request,
						         HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			Hashtable jobSizeParamsHashtable = null;
            BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			ActionMessages errors = new ActionMessages();
			JobSizeParametersForm jobSizeParams = new JobSizeParametersForm();

			HttpSession session = request.getSession();

			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

			// Pull the game type number off the query string
			gametype = (new Integer((String) request.getParameter("gametype"))).intValue();

            try
            {
                //jobSizeParamsHashtable will have ten rows
			    jobSizeParamsHashtable = dbFacade.getGameParametersDBHandler().getAllJobSizeParameters(game, admin, false);

                if (gametype == 2)  // Heavy Construction - remove rows 1-5 -> copy 6-10 back in as 1-5
                {
					int j=51;
					for(int i=0;i<5;i++) {
                       jobSizeParamsHashtable.remove("Median" + i+1);
                       jobSizeParamsHashtable.put("Median" + i+1, (Integer) jobSizeParamsHashtable.get("Median" + j) );

                       jobSizeParamsHashtable.remove("StdDev" + i+1);
                       jobSizeParamsHashtable.put("StdDev" + i+1, (Integer) jobSizeParamsHashtable.get("StdDev" + j) );

                       jobSizeParamsHashtable.remove("Min" + i+1);
                       jobSizeParamsHashtable.put("Min" + i+1,    (Integer) jobSizeParamsHashtable.get("Min"    + j) );

                       jobSizeParamsHashtable.remove("Max" + i+1);
                       jobSizeParamsHashtable.put("Max" + i+1,    (Integer) jobSizeParamsHashtable.get("Max"    + j) );

                       jobSizeParamsHashtable.remove("JobType" + i+1);
                       jobSizeParamsHashtable.put("JobType" + i+1,    (Integer) jobSizeParamsHashtable.get("JobType"    + j) );

                       j=j+10;
					}
                    //remove rows 6-10)
					for(int i=5;i<10;i++) {
                       jobSizeParamsHashtable.remove("Median" + i+1);
                       jobSizeParamsHashtable.remove("StdDev" + i+1);
                       jobSizeParamsHashtable.remove("Min" + i+1);
                       jobSizeParamsHashtable.remove("Max" + i+1);
                       jobSizeParamsHashtable.remove("JobType" + i+1);
					}
				}
				else  // Building Construction - remove rows 6-10
				{
					for(int i=5;i<10;i++) {
                       jobSizeParamsHashtable.remove("Median" + i+1);
                       jobSizeParamsHashtable.remove("StdDev" + i+1);
                       jobSizeParamsHashtable.remove("Min" + i+1);
                       jobSizeParamsHashtable.remove("Max" + i+1);
                       jobSizeParamsHashtable.remove("JobType" + i+1);
					}
				}
				//now the hashtable has only five rows (numbered 1-5) which is all that is accepted by JobSizeParametersForm
			    jobSizeParams.setAllParameters(jobSizeParamsHashtable);
			    dbFacade.close();
		    }
		    catch(ParametersNotFoundException pnfe)
		    {
			    //All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.notfound"));
		    }
		    catch(BIGDatabaseFacadeException bigdbe)
		    {
				//Problem with the database
				log.fatal("", bigdbe);
			    errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			session.setAttribute("jobSizeParams", jobSizeParams);
			session.setAttribute("gametype", gametype);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
	    }

   }

}
