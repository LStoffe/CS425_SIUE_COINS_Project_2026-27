package edu.calpoly.its.mas.big.users.admin;

import javax.servlet.http.HttpServletRequest;

/**
 * ActionForm used to hold the statistics for the rainfall game parameters (min, max, standard deviation, and median values) per period.
 * <code>RainfallParametersForm</code> can hold information on 12 periods.
 *
 * <br>
 * <code>RainfallParametersForm</code> deals with the statistic values both as Strings and ints.  The numbers need to be Strings
 * to communicate with the HTML form.  Ints are needed for practical use of the values. To access the values as String simple get and set
 * methods are provided.  To access the values as ints, methods are provided that return
 * and take as a parameter arrays of ints, 12 long.  The index value of the int in the array corresponds with the period number that int is for.
 *
 *
 *
 * @author Kevin Craig, modified by Dara Manker
 *
 * @version $Revision: 1.5 $ $Date: 2001/09/07 02:54:06 $
 */
public class RainfallParametersForm  extends ActionForm
{
	//Developers Note:  All array lengths must be hard coded to 12 because there are 12 of each type of method.
	//To change this number this whole class must be modified
	//The initial conversion from strings to ints takes place during the validate method because the conversion also has
	//the effect of validating that the strings contain numeric values

	private boolean makeDefaults = false;

	private int[] medians = new int[12];
	private int[] stddevs = new int[12];
	private int[] mins = new int[12];
	private int[] maxes = new int[12];

    private String[] medianStrings = new String[12];
    private String[] stddevStrings = new String[12];
    private String[] minStrings = new String[12];
    private String[] maxStrings = new String[12];


    //**************** Public Methods ******************

    /**
     * Returns an array of 12 ints representing the median number of jobs value for the 12 periods.
     */

		stringsToInts(medianStrings, medians);

        return medians;
	}

    /**
     * Returns an array of 12 ints representing the standard deviation of the number of jobs value for the 12 periods.
     */

		stringsToInts(stddevStrings, stddevs);

		return stddevs;
	}

    /**
     * Populates a hashtable with all of the rainfall parameters for the 12 periods from the action form.
     * <b>Precondition:</b>All int values must be >= 0 <br>
     * @param parameters Hashtable containing all rainfall parameter values for the 12 periods. Keys to the hashtable are in the form "Median#", "Stddev#", "Min#", "Max#", where "#" is the period number.
     */
    public Hashtable setAllParameters()
    {
		Hashtable parameters = new Hashtable();

        for(int i=0;i<12;i++)
        {
		   parameters.put("Median" + i+1, new Integer(this.medians[i]));
		   parameters.put("StdDev" + i+1, new Integer(this.stddevs[i]));
		   parameters.put("Min" + i+1, new Integer(this.mins[i]));
		   parameters.put("Max" + i+1, new Integer(this.maxes[i]));
	    }

		return parameters;
	}


    /**
     * Populates the action form with all of the rainfall parameters for the 12 periods.
     * The values as String will automatically be set to match the ints
     * <br><br>
     * <b>Precondition:</b>The hashtable is in the form output by BIGDatabaseFacade.getRainfallParameters(Game game).
     * <b>Precondition:</b>All int values must be >= 0 <br>
     * @param parameters Hashtable containing all rainfall parameter values for the 12 periods
     */
    public void setAllParameters(Hashtable parameters)
    {
        for(int i=0;i<12;i++)
        {
		   this.medians[i] = ((Integer) parameters.get("Median" + i+1)).intValue();
		   this.stddevs[i] = ((Integer) parameters.get("StdDev" + i+1)).intValue();
		   this.mins[i] = ((Integer) parameters.get("Min" + i+1)).intValue();
		   this.maxes[i] = ((Integer) parameters.get("Max" + i+1)).intValue();
	    }

		intsToStrings(medians, medianStrings);
		intsToStrings(stddevs, stddevStrings);
		intsToStrings(mins, minStrings);
		intsToStrings(maxes, maxStrings);
	}


    /**
     * Reset fields to their initial values

    public void reset()

		makeDefaults = false;
	}


	/**
	 * Ensures all values are non-null, all number strings contain only numeric characters,
     * all number are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{

        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;
        boolean minsValid = true;

        //First make sure none of the Strings are null
        for(int i = 0; i < 12; i++)
        {
			if((medianStrings[i].trim() == null) || (medianStrings[i].trim().length() < 1))
				nullParams = true;
		    if((stddevStrings[i].trim() == null) || (stddevStrings[i].trim().length() < 1))
		    	nullParams = true;
			if((minStrings[i].trim() == null) || (minStrings[i].trim().length() < 1))
				nullParams = true;
			if((maxStrings[i].trim() == null) || (maxStrings[i].trim().length() < 1))
				nullParams = true;
		}

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the Strings contain only numeric characters.
			for(int i = 0; i < 12; i++)
			{
				//Check that the strings are valid numbers by trying to convert them to ints and checking for a NumberFormatException
				//Note that this also populated the int arrays
				try
				{
					medians[i] = Integer.parseInt(medianStrings[i]);
					stddevs[i] = Integer.parseInt(stddevStrings[i]);
					mins[i] = Integer.parseInt(minStrings[i]);
					maxes[i] = Integer.parseInt(maxStrings[i]);
				}
				catch(NumberFormatException nfe)
				{
					validNumbers = false;
				}
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			for(int i = 0; i < 12; i++)
			{
				if(medians[i] < 0)
					positiveNumbers = false;
				if(stddevs[i] < 0)
					positiveNumbers = false;
				if(mins[i] < 0)
					positiveNumbers = false;
				if(maxes[i] < 0)
					positiveNumbers = false;
			}
		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		if(validNumbers && !nullParams)  //validNumbers == true AND nullParams == false
		{
			//Check that all min values are greater than or equal to their corresponding max value
			for(int i = 0; i < 12; i++)
			{
				if(mins[i] > maxes[i])
				{
					minsValid = false;
				}

			}
		}

		if(!minsValid && validNumbers && !nullParams)  //minsValid == false AND validNumbers == true AND nullParams == false
		{
			errors.add("minmax", new ActionMessage("error.parameters.minmax"));
		}

		return errors;
	}

//**************************** Private Methods *********************************

    /* Converts a set of strings into ints, stored in an array
     * Param: stringValues - Array of strings to be converted into ints
     * Param: intValues - Array of ints where the int values of the Strings will be stored
     * Precondition: intValues and stringValues must have equal length
     */
	private void stringsToInts(String[] stringValues, int[] intValues)
	{

		for (int i = 0; i < stringValues.length; i++)
		{
			intValues[i] = Integer.parseInt(stringValues[i]);

		}

	}

    /* Converts a set of ints into into strings, stored in the appropriate variable
     * Param: intValues - Array of ints to be converted into Strings
     * Param: stringValues - Array of Strings where the String value of the ints will be stored
     * Precondition: intValues and stringValues must have equal length
     */
	private void intsToStrings(int[] intValues, String[] stringValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			stringValues[i] = Integer.toString(intValues[i]);

		}


	}

        //Retrieve the int arrays triggering the String to int conversion.  Compare the int arrays retrieved to those originally entered.
        //If they are the same then the unit test passed.

		int[] initMedians = new int[12];
		int[] initStddevs = new int[12];
		int[] initMins = new int[12];
		int[] initMaxes = new int[12];

		int[] finalMedians = new int[12];
		int[] finalStddevs = new int[12];
		int[] finalMins = new int[12];
		int[] finalMaxes = new int[12];

		for (int i=0; i < 12; i++)
		{
			initMedians[i] = i;
			initStddevs[i] = i - 10;
			initMins[i] = i * i;
			initMaxes[i] = i % 2;
		}


		rainForm.setMedians(initMedians);
		rainForm.setStddevs(initStddevs);
		rainForm.setMins(initMins);
		rainForm.setMaxes(initMaxes);

		finalMedians = rainForm.getMedians();
		finalStddevs = rainForm.getStddevs();
		finalMins = rainForm.getMins();
		finalMaxes = rainForm.getMaxes();


		boolean unitTestPassed = true;

		//Test that the initial and final median values are the same
 		System.out.println("The initial medians were: ");
 		for (int i = 0; i < 12; i++)
 		{
			System.out.print(initMedians[i] + " ");
		}
 		System.out.println("\nAfter being converted to strings and back the medians are: ");
 		for (int i = 0; i < 12; i++)
 		{
			System.out.print(finalMedians[i] + " ");
		}

		boolean mediansEqual = true;

	 	for (int i = 0; i < 12; i++)
	 	{
			if(!(initMedians[i] == finalMedians[i]))
