package edu.calpoly.its.mas.big.users.admin;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.users.StringConstraints;

/**
 * ActionForm containing information for a request to add an administrator.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.9 $ $Date: 2001/08/14 20:20:43 $
 */
public class AddAdminForm extends ActionForm
{
  private String adminName;
  private String password;
  private String passwordConfirm;

  public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
  {
    ActionErrors errors = new ActionErrors();

    //Check that username is not null and is acceptable length
    if ((adminName == null) || (adminName.length() < 1))
    {
    	errors.add("adminName", new ActionMessage("error.addAdmin.adminName.required"));
	}
	else if ((adminName.length() < StringConstraints.USERNAME_LOWER_LIMIT) || (adminName.length() > StringConstraints.USERNAME_UPPER_LIMIT))
	{
		errors.add("username_length", new ActionMessage("error.username.length"));
	}

    //Check that password is not null and is acceptable length
    if ((password == null) || (password.length() < 1))
    {
        errors.add("password", new ActionMessage("error.addAdmin.password.required"));
	}
	else if ((password.length() < StringConstraints.PASSWORD_LOWER_LIMIT) || (password.length() > StringConstraints.PASSWORD_UPPER_LIMIT))
	{
		errors.add("password_length", new ActionMessage("error.password.length"));
	}
	else if (!(password.equals(passwordConfirm)))
	{
        errors.add("passwordConfirm", new ActionMessage("error.password.mismatch"));
	}

    return errors;
  }


  //*********************************  Simple Set and Get Methods  **************************
  public String getAdminName()
  {
    return adminName;
  }

  public void setAdminName(String adminName)
  {
    this.adminName = adminName;
  }

  public String getPassword()
  {
    return password;
  }

  public void setPassword(String password)
  {
    this.password = password;
  }

  public String getPasswordConfirm()
  {
	return passwordConfirm;
  }

  public void setPasswordConfirm(String passwordConfirm)
  {
	this.passwordConfirm = passwordConfirm;
  }



}
