package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles use case of setting percent takeoff parameters. Uses the
 * session attribute "user", which is an <code>Admin</code>, and the
 * session attribute "game" which is a <code>edu.calpoly.its.mas.big.Game</code>,
 * to determine the current administrator and game.
 *
 * The job type for which percent takeoff parameters where requested
 * is determined by the "jobtype" property in the
 * <code>PercentTakeoffForm</code> bean on the session. The percent
 * takeoff parameters that go with this admin, game, and job type are
 * retrieved from a <code>PercentTakeoffParametersForm</code> in the
 * session using the key "percentTakeoffParams" and stored into the
 * database, provided the user didn't press Cancel, Restore Defaults,
 * or Reset.
 *
 * If "Make these parameters my defaults" was checked and the
 * user pressed the "Set Parameters" button, the submitted parameters
 * are also stored in the database as the current Admin's defaults.
 *
 * If the user pressed the "Restore Defaults" button, the Administrator's
 * default parameter values are retrieved from the database.  These parameters
 * are then stored in the database for this game and the user is returned to
 * the viewPercentTakeoffs page.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.10 $ $Date: 2002/03/14 18:53:06 $
 **/

public class SetPercentTakeoffAction extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		// Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();
			HttpSession session = request.getSession();

			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

			String testRestoreDefaults = (String) request.getParameter("restoreDefaults");

			PercentTakeoffForm percentTakeoffForm = (PercentTakeoffForm) form;

			Hashtable params = percentTakeoffForm.getTakeoffPercents();

			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

			try
			{
				// "Restore Defaults" button was pressed
				if (testRestoreDefaults != null)
				{
					Hashtable defaultParams = dbFacade.getGameParametersDBHandler().getDefaultPercentTakeoffParameters(
													   admin,
												       percentTakeoffForm.getJobTypeID());

					percentTakeoffForm.setTakeoffPercents(defaultParams);

					dbFacade.getGameParametersDBHandler().updatePercentTakeoffParameters(game,
														percentTakeoffForm.getJobTypeID(),
														defaultParams);
					dbFacade.close();

					return (new ActionForward(mapping.getInput()));

 				// "Set Parameters" button was pressed -- perhaps this should be
				// made explicit with a "submit" param check?

				}
				else
				{
					dbFacade.getGameParametersDBHandler().updatePercentTakeoffParameters(game,
														percentTakeoffForm.getJobTypeID(),
														params);

					if (percentTakeoffForm.getMakeDefaults())
					{
						//Store these as default parameters for this admin as well
						dbFacade.getGameParametersDBHandler().updateDefaultPercentTakeoffParameters(admin,
														  percentTakeoffForm.getJobTypeID(),
														  params);
					}
				}

				dbFacade.close();
			}
			catch(BIGDatabaseFacadeException bigdbe)
			{
				//Problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
			catch(ParametersNotFoundException pnfe)
			{
			    //All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.notfound"));
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"paramsSetSuccessfully\"");
				}
				return (mapping.findForward("paramsSetSuccessfully"));
			}

	    }
	    else
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			// Request was cancelled
			return (mapping.findForward("cancelled"));
		}
	}
}
