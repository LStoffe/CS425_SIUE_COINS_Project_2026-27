	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * and that all numbers are greater than or equal to zero
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;

        //First make sure none of the Strings are null
        for(int i = 0; i < 5; i++)
        {
			if((methodNames[i].trim() == null) || (methodNames[i].trim().length() < 1))
				nullParams = true;
		    if((productivityStrings[i].trim() == null) || (productivityStrings[i].trim().length() < 1))
		    	nullParams = true;
			if((laborCostStrings[i].trim() == null) || (laborCostStrings[i].trim().length() < 1))
				nullParams = true;
			if((materialsCostStrings[i].trim() == null) || (materialsCostStrings[i].trim().length() < 1))
				nullParams = true;
			if((equipmentCostStrings[i].trim() == null) || (equipmentCostStrings[i].trim().length() < 1))
				nullParams = true;
			if((subcontractorCostStrings[i].trim() == null) || (subcontractorCostStrings[i].trim().length() < 1))
				nullParams = true;
			if((uncertaintyRangeStrings[i].trim() == null) || (uncertaintyRangeStrings[i].trim().length() < 1))
			    nullParams = true;
			if((weatherSensitivityStrings[i].trim() == null) || (weatherSensitivityStrings[i].trim().length() < 1))
			    nullParams = true;
		}

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the number Strings contain only numeric characters.
			for(int i = 0; i < 5; i++)
			{
				//Check that the strings that should represent numbers only contain numberic characters by trying to convert them to ints and checking for a NumberFormatException
				//Note that this also populated the int arrays
				try
				{
					productivities[i] = Integer.parseInt(productivityStrings[i]);
					laborCosts[i] = Integer.parseInt(laborCostStrings[i]);
					materialsCosts[i] = Integer.parseInt(materialsCostStrings[i]);
					equipmentCosts[i] = Integer.parseInt(equipmentCostStrings[i]);
					subcontractorCosts[i] = Integer.parseInt(subcontractorCostStrings[i]);
					uncertaintyRanges[i] = Integer.parseInt(uncertaintyRangeStrings[i]);
				}
				catch(NumberFormatException nfe)
				{
					validNumbers = false;
				}
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			for(int i = 0; i < 5; i++)
			{
				if(productivities[i] < 0)
					positiveNumbers = false;
				if(laborCosts[i] < 0)
					positiveNumbers = false;
				if(materialsCosts[i] < 0)
					positiveNumbers = false;
				if(equipmentCosts[i] < 0)
					positiveNumbers = false;
				if(subcontractorCosts[i] < 0)
					positiveNumbers = false;
				if(uncertaintyRanges[i] < 0)
					positiveNumbers = false;
				if(weatherSensitivities[i] < 0)
					positiveNumbers = false;
			}
		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}


		return errors;
	}

    /* Converts a set of strings into ints, stored in an array
     * Param: stringValues - Array of strings to be converted into ints
     * Param: intValues - Array of ints where the values as ints will be stored
     * Precondition: intValues and stringValues must have equal length
     */
	private void stringsToInts(String[] stringValues, int[] intValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			intValues[i] = Integer.parseInt(stringValues[i]);

		}
	}

    /* Converts a set of ints into strings, stored in the appropriate variable
     * Param: intValues - Array of ints to be converted into Strings
     * Param: stringValues - Array of Strings where the values as Strings will be stored
     * Precondition: intValues and stringValues must have equal length
     */
	private void intsToStrings(int[] intValues, String[] stringValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			stringValues[i] = Integer.toString(intValues[i]);

		}
	}



	private void subOrSelfStringsToBooleans(String[] stringValues, boolean[] booleanValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			if(stringValues[i].equals("subcontracted"))
			{
				booleanValues[i] = true;
			}
			else
			{
				booleanValues[i] = false;
			}

		}
	}


	private void booleansToSubOrSelfStrings(boolean[] booleanValues, String[] stringValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			if(booleanValues[i])
			{
				stringValues[i] = "subcontracted";
			}
			else
			{
				stringValues[i] = "self-performed";
			}

		}
	}


	/* Converts a set of weather sensitivity values from strings to ints using the rules described
	 * under the setWeatherSensitivities method above.
	 * Param: stringValues - Array of weather sensitivity strings to be converted into ints
	 * Param: intValues - Array of ints where the values as ints will be stored
	 * Precondition: intValues and stringValues must have equal length
	 * Precondition: stringValues can only contain the strings "Rainfall", "Temperature", "Both", and "Neither"
	 */
	private void weatherSensitivityStringsToInts(String[] stringValues, int[] intValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			if (stringValues[i].equals("neither"))
				intValues[i] = 0;
			else if (stringValues[i].equals("temperature"))
				intValues[i] = 1;
			else if (stringValues[i].equals("rainfall"))
			    intValues[i] = 2;
			else if (stringValues[i].equals("both"))
			    intValues[i] = 3;
		}
	}


	/* Converts a set of weather sensitivity values from ints to strings using the rules described
	 * under the getWeatherSensitivities method above.
	 * Param: intValues - Array of weather sensitivity ints to be converted into strings
	 * Param: stringValues - Array of Strings where the values as Strings will be stored
	 * Precondition: intValues and stringValues must have equal length
	 * Precondition: for each value x, in intValues, 0 <= x <= 3
	 */
	private void weatherSensitivityIntsToStrings(int[] intValues, String[] stringValues)
	{
		for (int i = 0; i < stringValues.length; i++)
		{
			if (intValues[i] == 0)
				stringValues[i] = "neither";
		 	else if (intValues[i] == 1)
		 		stringValues[i] = "temperature";
		 	else if (intValues[i] == 2)
		 		stringValues[i] = "rainfall";
		 	else if (intValues[i] == 3)
		 		stringValues[i] = "both";

		}
	}
