package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;
import javax.sql.DataSource;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Determines whether the administrator wishes to set parameters or restore defaults for number of jobs parameters then performs the appropriate action.
 * If the administrator wishes to store his current parameters settings to the database s/he would have pushed the set parameters button
 * (which must have the name "submit").  In this case the values currently in the NumberOfJobsParametersForm in the session will be
 * stored to the database.  These parameters will also be made the admin's defaults if the getMakeDefaults method of the form returns true.
 * <br><br>
 * If the administrator pressed some other submit button (any name other than "submit" or "cancel") the parameters in the NumberOfJobsParametersForm
 * will be overwritten with a copy of the admin's defaults for the current parameters.  The defaults will also be re-copied into the database overwriting the current
 * game parameters.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.10 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class SetNumberOfJobsParametersAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();

			//Retrieve the current game and admin from the session
			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

			//Get the game parameters from the action form
			NumberOfJobsParametersForm numberOfJobsParametersForm = (NumberOfJobsParametersForm) form;

            //Determine which submit button was pressed by the Administrator. If the SetParameters button is null the other RestoreDefaults button had to be pushed
            String testSetParametersSubmitButton = (String) request.getParameter("submit");

			//Create a database facade that will hold the connection since it will be used repeatedly
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

			//Hashtable used to transfer the parameters to the database facade
			Hashtable parameters = null;

			//Int arrays to hold the parameters from the ActionForm
			int[] medians = numberOfJobsParametersForm.getMedians();
			int[] stddevs = numberOfJobsParametersForm.getStddevs();
			int[] mins = numberOfJobsParametersForm.getMins();
			int[] maxes = numberOfJobsParametersForm.getMaxes();
			int[] minsHC = numberOfJobsParametersForm.getMinsHC();
			int[] maxesHC = numberOfJobsParametersForm.getMaxesHC();

			if (testSetParametersSubmitButton != null)  //"Set Parameters" submit button was pushed.
			{
				//Store the current parameters values from the form into the database
				try
				{
					for(int i = 0; i < 12; i++)
					{
						parameters = new Hashtable(4); //4 parameters to store

						parameters.put("Median", new Integer(medians[i]));
						parameters.put("StdDev", new Integer(stddevs[i]));
						parameters.put("Min", new Integer(mins[i]));
						parameters.put("Max", new Integer(maxes[i]));
						parameters.put("MinHC", new Integer(minsHC[i]));
						parameters.put("MaxHC", new Integer(maxesHC[i]));

						dbFacade.getGameParametersDBHandler().updateNumberOfJobsParameters(game, i + 1, parameters);

						if (numberOfJobsParametersForm.getMakeDefaults())
						{
							//Store these as default parameters for this admin as well
							dbFacade.getGameParametersDBHandler().updateDefaultNumberOfJobsParameters(admin, i + 1, parameters);
						}

					}
					dbFacade.close();
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					//Problem with the database
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
					//The parameters for this game weren't there.  There should never be a game without parameters
					log.fatal("", pnfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}
			}
			else //"Restore defaults" submit button was pushed
			{
				try
				{
					for(int i = 0; i < 12; i++)
					{
					    //Retrieve this admins default parameters from the database and re-store them for the current game
						parameters = dbFacade.getGameParametersDBHandler().getDefaultNumberOfJobsParameters(admin, i+1);
						dbFacade.getGameParametersDBHandler().updateNumberOfJobsParameters(game, i + 1, parameters);

						//Record the values in the arrays so they can be put in the Action Form
						medians[i] = ( (Integer) parameters.get("Median")).intValue();
						stddevs[i] = ( (Integer) parameters.get("StdDev")).intValue();
						mins[i] = ( (Integer) parameters.get("Min")).intValue();
						maxes[i] = ( (Integer) parameters.get("Max")).intValue();
						minsHC[i] = ( (Integer) parameters.get("MinHC")).intValue();
						maxesHC[i] = ( (Integer) parameters.get("MaxHC")).intValue();
					}
					dbFacade.close();

					//The parameter vaules also need to be put in the Action Form so the user can see them
					numberOfJobsParametersForm.setMedians(medians);
					numberOfJobsParametersForm.setStddevs(stddevs);
					numberOfJobsParametersForm.setMins(mins);
					numberOfJobsParametersForm.setMaxes(maxes);
					numberOfJobsParametersForm.setMinsHC(minsHC);
					numberOfJobsParametersForm.setMaxesHC(maxesHC);
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					//Problem with the database
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
					//The parameters for this game weren't there.  There should never be a game without parameters
					log.fatal("", pnfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}

			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else if (testSetParametersSubmitButton != null)
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"params_set\"");
				}
				return (mapping.findForward("params_set"));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"defaults_restored\"");
				}
				return (mapping.findForward("defaults_restored"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));

		}

	}
}
