package edu.calpoly.its.mas.big.users.admin;

/**
 * Puts a <code>Vector</code> of <code>Company</code> in the session.  The companies in the <code>Vector</code> will be those
 * with gameID of the <code>Game</code> currently in the session.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.4 $ $Date: 2002/03/14 18:53:06 $
 */
public class ListCompaniesAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		ActionMessages errors = new ActionMessages();

		HttpSession session = request.getSession();
		Game game = (Game) session.getAttribute("game");
		Admin admin = (Admin) session.getAttribute("user");

		//Create a database facade and get a list of companies from it
		BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

        String template  = "";
        String template1 = "";
        String template2 = "";
        String template3 = "";

        HashMap m = new HashMap();

		try
		{
			Vector companies = dbFacade.getCompanyDBHandler().getCompanies(game);
			Vector camForms = new Vector();

			m = dbFacade.getGradingTemplateDBHandler().getGradingTemplate(game);
			template  = (String) m.get("template");
			template1 = (String) m.get("template1");
			template2 = (String) m.get("template2");
			template3 = (String) m.get("template3");

			NamedVariableDataFacade namedVariableDataFacade = new NamedVariableDataFacadeImpl(null, game, dbFacade);
			BIGFormulaInterpreter interpreter = new BIGFormulaInterpreter(namedVariableDataFacade);

			for(int i = 0; i < companies.size(); i++)
			{
				Company company = (Company) companies.get(i);

				try
				{
					camForms.add(new CAMReportForm(dbFacade.getCAMDBHandler().getCAMs(company, game.getCurrentPeriodNumber() -1)));
				} //This exception will occur during the first period, because CAMs haven't been created yet.
				catch(ObjectNotFoundException onfe)
				{
					camForms.add(new CAMReportForm());
				}

				company.setGrade( interpreter.interpretFormula(template, company.getCompanyID()));
				company.setGrade1(interpreter.interpretFormula(template1, company.getCompanyID()));
				company.setGrade2(interpreter.interpretFormula(template2, company.getCompanyID()));
				company.setGrade3(interpreter.interpretFormula(template3, company.getCompanyID()));
			}

			dbFacade.close();

			//Put the Vector of companies in the session for a scriplet to extract
			session.setAttribute("companies", companies);
			session.setAttribute("camForms", camForms);


			/* now do the ranking for each of the company's scores */

			// holds the unranked values
			int[][] rankTemplate  = new int[ companies.size() ][3];
			int[][] rankTemplate1 = new int[ companies.size() ][3];
			int[][] rankTemplate2 = new int[ companies.size() ][3];
			int[][] rankTemplate3 = new int[ companies.size() ][3];

			// contains the ranking values
			int[][] doneRankTemplate  = new int[ companies.size() ][3];
			int[][] doneRankTemplate1 = new int[ companies.size() ][3];
			int[][] doneRankTemplate2 = new int[ companies.size() ][3];
			int[][] doneRankTemplate3 = new int[ companies.size() ][3];

			// load values into unranked arrays
			for(int i = 0; i < companies.size(); i++)
			{
				Company company = (Company) companies.get(i);

				// note: replaceAll takes out the commas in strings when a number is very large
				// Double is used because sometimes the number has decimal value and needs to be an integer
				rankTemplate[i][0]  = Double.valueOf( company.getGrade().replaceAll(",", "") ).intValue();
				rankTemplate1[i][0] = Double.valueOf( company.getGrade1().replaceAll(",", "") ).intValue();
				rankTemplate2[i][0] = Double.valueOf( company.getGrade2().replaceAll(",", "") ).intValue();
				rankTemplate3[i][0] = Double.valueOf( company.getGrade3().replaceAll(",", "") ).intValue();

				// the index values
				rankTemplate[i][1]  = i;
				rankTemplate1[i][1] = i;
				rankTemplate2[i][1] = i;
				rankTemplate3[i][1] = i;
			}

			// perform the ranking analysis

			doneRankTemplate  = doRanking( rankTemplate, companies.size() );
			doneRankTemplate1 = doRanking( rankTemplate1, companies.size() );
			doneRankTemplate2 = doRanking( rankTemplate2, companies.size() );
			doneRankTemplate3 = doRanking( rankTemplate3, companies.size() );

        	// save rank positions to company beans
        	for (int i = 0; i < companies.size(); i++)
        	{
				Company company = (Company) companies.get(i);

				company.setRankGrade(  Integer.toString( doneRankTemplate[i][2] ) );
				company.setRankGrade1( Integer.toString( doneRankTemplate1[i][2] ) );
				company.setRankGrade2( Integer.toString( doneRankTemplate2[i][2] ) );
				company.setRankGrade3( Integer.toString( doneRankTemplate3[i][2] ) );
			}

			/* *** end of ranking *** */

		}
		catch(BIGDatabaseFacadeException BIGDbe)
		{
			log.fatal("", BIGDbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		finally
		{
			try
			{
        	 	dbFacade.close();
			}
			catch (BIGDatabaseFacadeException e)
			{
				log.fatal(getClass().getName(), e);
 				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
		}

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
		}
		return (mapping.findForward("success"));
	}


 /**
   * There are four grading templates.  Each has a two-dimensional array holding necessary values.
   * Vertical dimension of array equals number of companies.  Horizontal is three values.
   * Horizontally --> [0] holds the score, [1] holds the index value, [2] holds the rank
   * This method receives the unsorted values, does the sort, stores the ranking value for each in [2].
   * Returns the array (now with ranking information) to the calling method.
   */
  private int[][] doRanking( int[][] theArray, int numbComp )
  {
 	 int temp;

 	 // rank in descending order (highest to lowest)
 	 for (int i = 0; i < (numbComp - 1); i++)
     {
         for (int j = 0; j < numbComp - i - 1; j++)
         {
             if (theArray[j][0] < theArray[j+1][0])
             {
                 temp = theArray[j][0];
                 theArray[j][0] = theArray[j+1][0];
                 theArray[j+1][0] = temp;

                 temp = theArray[j][1];
			     theArray[j][1] = theArray[j+1][1];
                 theArray[j+1][1] = temp;
             }
         }
     }

     // store rank values in array
     for (int i = 0; i < numbComp; i++)
     {
		 theArray[i][2] = i + 1;
	 }

	 // restore array to original sequence - sort by index value in ascending order (smallest to largest)
 	 for (int i = 0; i < numbComp; i++)
     {
         for (int j = i + 1; j < numbComp; j++)
         {
             if (theArray[i][1] > theArray[j][1])
             {
                 temp = theArray[i][1];
                 theArray[i][1] = theArray[j][1];
                 theArray[j][1] = temp;

                 temp = theArray[i][0];
                 theArray[i][0] = theArray[j][0];
                 theArray[j][0] = temp;

                 temp = theArray[i][2];
                 theArray[i][2] = theArray[j][2];
                 theArray[j][2] = temp;
             }
         }
     }

     return theArray;
  }

}
