package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;

//BIG v2 imports
import edu.calpoly.its.mas.big.game.Game;

import edu.calpoly.its.mas.big.users.company.CompanyForm2;

//BIG v3 imports
import edu.calpoly.lib.multimedia.big.game.Message;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Handles a request to create a new company.
 *
 * @version $Revision: 1.18 $ $Date: 2002/05/07 21:30:05 $
 */
public class CreateNewCompanyAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}


		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();


			//Get datafields for new company from form
                //Username will be stored as lower case for case insensitivty.  Passwords are case sensitive
			String username = ((CompanyForm2) form).getUsername().toLowerCase();
			String password = ((CompanyForm2) form).getPassword();
            String title = ((CompanyForm2) form).getTitle();
            long cashOnHand = ((CompanyForm2) form).getCashOnHand();
            long WIPLimit = ((CompanyForm2) form).getWIPLimit();
            boolean aaLicense = ((CompanyForm2) form).getAaLicense();
            boolean bbLicense = ((CompanyForm2) form).getBbLicense();
            int perJobLimitAsWIPPercent = ((CompanyForm2) form).getPerJobLimitAsWIPPercent();
            long perJobLimit = Math.round((((float) perJobLimitAsWIPPercent) / 100f) * WIPLimit);  //perJobLimitAsWIPPercent = perJobLimit / WIPLimit
            int WIPLimitPercent = Math.round((((float) cashOnHand) / ((float) WIPLimit)) * 100f); //The percent of WIPLimit that cash-on-hand is equal to

			//Company cannot be created with negative cash-on-hand because some
			//values which must be positive are setup as percentages of cash-on-hand
			if(cashOnHand < 0)
			{
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.company.cashOnHand.negative"));

				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}


            //Enforce min. bond limits
            perJobLimit = BondLimitConstraints.applyPerJobLimitMinimum(perJobLimit);
            WIPLimit = BondLimitConstraints.applyWIPLimitMinimum(WIPLimit);

            //Determine a value for license
            int license = 0;
            if (bbLicense && !aaLicense) { license = 1; }
            if (!bbLicense && aaLicense) { license = 2; }
            if (bbLicense && aaLicense)  { license = 3; }
            //System.out.println("license = " + license);

            HttpSession session = request.getSession();

            Game game = (Game) session.getAttribute("game");

            //These will be used to test if an administrator or company with the username given already exists in the database
			Admin testAdmin = null;
			Company testCompany = null;

			if(game.getIsActive())
			{
				try
				{
					//Create a new BIGDatabaseFacade that hold connections - it will be used more than once
					BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

					//Check if there is already an admin with this username
					testAdmin = dbFacade.getAdminDBHandler().getAdmin(username);

					//Check if there is already a company with this username
					testCompany = dbFacade.getCompanyDBHandler().getCompany(username);


					if((testAdmin == null) && (testCompany == null))  //Username is not in use
					{
                        String mission = "Mission Statement";
                        String core = "Core Values";
                        String company_url = "http://";
						Company newCompany =  new Company(title, username, password, cashOnHand, WIPLimit, WIPLimitPercent, perJobLimit, perJobLimitAsWIPPercent, license, mission, core, company_url);
						dbFacade.getCompanyDBHandler().addCompany(game, newCompany);

						//Call method necessary to setup appraisal metrics
						dbFacade.getCAMDBHandler().createCAMUpdateRecord(newCompany);

						//Give company initial amounts for equipment
						      //Note the values are hard coded for now because at the current
						      //time there are plans for a future update to BIG
						      //that will completely rework the equipment,
						      //Right now we are just using these values to be placeholders
						      //on the balance sheet until a realistic equipment component can
						      //be added
						int constructionEquipment = Math.round((.004f) * cashOnHand);
						int autos = Math.round((.015f) * cashOnHand);
						int officeEquipment = Math.round((.006f) * cashOnHand);
						int land = Math.round((.097f) * cashOnHand);

                        //If the game has not yet started (current period = 0)
                        //then then only an intial balance sheet needs to be added.
                        if(game.getCurrentPeriodNumber() == 0)
                        {
						   //Create empty balance sheets for this company
						   dbFacade.getBalanceSheetDBHandler().newCompanyBalanceSheet(game,
						                                                           newCompany,
						                                                           0,
						                                                           cashOnHand,
						                                                           constructionEquipment,
						                                                           autos,
						                                                           officeEquipment,
					                                                               land);
						}
						else
						{
							//Game has already started: Empty financial reports and balance sheets need to be created
							//for past periods in the games.  Technically we only need
							//them for the previous 6 periods but we'll create them for every
							//previous period so that when the company views past reports, something is
							//there

                       		for(int i = 0;i < game.getCurrentPeriodNumber(); i++)
							{
							   //Create empty balance sheets for this company
							   dbFacade.getBalanceSheetDBHandler().newCompanyBalanceSheet(game,
							                                                           newCompany,
							                                                           i,
							                                                           cashOnHand,
							                                                           constructionEquipment,
							                                                           autos,
							                                                           officeEquipment,
						                                                               land);

						       FinancialReportForm emptyReport = new FinancialReportForm();
						       emptyReport.setPeriod(i);
						       dbFacade.getFinancialReportDBHandler().addFinancialReport(game, newCompany, emptyReport);

						    }
						}


						//Setup initial personnel for the company
						dbFacade.getCompanyPersonnelDBHandler().addEmployeeTypeRecord(game, newCompany, game.getCurrentPeriodNumber()+1, "CEO", 1, 0, 0);
						dbFacade.getCompanyPersonnelDBHandler().addEmployeeTypeRecord(game, newCompany, game.getCurrentPeriodNumber()+1, "Assistant Estimator", 1, 0, 0);
						dbFacade.getCompanyPersonnelDBHandler().addEmployeeTypeRecord(game, newCompany, game.getCurrentPeriodNumber()+1, "Assistant Project Manager", 1, 0, 0);
						dbFacade.getCompanyPersonnelDBHandler().addEmployeeTypeRecord(game, newCompany, game.getCurrentPeriodNumber()+1, "Secretary", 1, 0, 0);


						//Send a welcome message to this new company
						int messagePeriod = 1;
						String welcomeString = "<b>Welcome to BIG!</b><br><br> ";
						welcomeString += "You'll want to look over the instruction manual to get started.  Click on the \"?\" button in the menu bar to begin reading it.  Your first goal will be to ";
						welcomeString += "get some construction jobs to start earning money, so you'll want to pay close attention to the ";
						welcomeString += "sections on bidding.<br><br>Good Luck!";

						if(game.getCurrentPeriodNumber() > 1)
						{
						    messagePeriod = game.getCurrentPeriodNumber();
						}
						dbFacade.getInboxDBHandler().storeMessage(new Message("BIG",
																	  		  newCompany,
																	  		  welcomeString,
																	  		  "Welcome to BIG",
																	  		  messagePeriod,
																	  		  Message.TEXT_ONLY,
																	  		  -1
																	 		 )
														 		 );


						//Put the company that was just added into the session so it can be displayed
						//in a confirmation message to the user
						session.setAttribute("company", newCompany);
					}
					else
					{
						//The username was already in use
						errors.add("usernameInUse", new ActionMessage("error.username.duplicate"));
					}

					//Close connection
					dbFacade.close();

				}
				catch(BIGDatabaseFacadeException BIGDbe)
				{
					log.fatal("", BIGDbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
			}
			else
			{
				if(log.isErrorEnabled())
				{
					log.error(this.getClass() + " game is not active");
				}
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}


}
