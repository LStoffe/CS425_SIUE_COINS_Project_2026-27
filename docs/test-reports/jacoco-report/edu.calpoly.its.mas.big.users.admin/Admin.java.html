package edu.calpoly.its.mas.big.users.admin;

import java.util.Vector;
import edu.calpoly.its.mas.big.game.Game;

/**
 * Represents an administrator or head administrator of BIG games.
 *
 * @version $Revision: 1.5 $ $Date: 2001/07/06 22:15:04 $
 */

public class Admin extends edu.calpoly.its.mas.big.users.User
{
  private int adminID;
  private boolean isHeadAdmin;
  private String email;

  public Admin(int adminID, String username, String password, String email, boolean isHeadAdmin)
  {
	  super(username, password);
      this.adminID = adminID;
	  this.isHeadAdmin = isHeadAdmin;
	  this.email = email;
  }

  //*****************************  Simple Get and Set Methods  *********************************

  public int getAdminID()
  {
	  return adminID;
  }

  public void setAdminID(int adminID)
  {
	  this.adminID = adminID;
  }

  public boolean getIsHeadAdmin()
  {
     return isHeadAdmin;
  }

  public void setIsHeadAdmin(boolean answer)
  {
     this.isHeadAdmin = answer;
  }

  public String getEmail()
  {
	  return email;
  }

  public void setEmail(String email)
  {
	  this.email = email;
  }

}
