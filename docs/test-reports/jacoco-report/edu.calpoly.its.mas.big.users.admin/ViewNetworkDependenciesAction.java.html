package edu.calpoly.its.mas.big.users.admin;

//JDK imports
import java.util.Hashtable;
import java.util.ArrayList;
import java.util.List;
import javax.sql.DataSource;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Handles use case of viewing network dependency parameters.
 * Uses the session attribute "user", which is an <code>Admin</code>, and the session attribute "game"
 * which is a <code>edu.calpoly.its.mas.big.Game</code>, to determine the current administrator
 * and game.  The job type for which network dependency parameters were requested is determined by the "jobtype" attribute in the query string
 * The network dependency parameters that go with this admin, game, and job type are retrieved from the database and placed them in a
 * <code>NetworkDependenciesForm</code> in the session under the name "networkDependenciesForm".
 * The game parameters in this form can be displayed to the user using jsp.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.13 $ $Date: 2002/03/14 18:53:06 $
 *
 */
public class ViewNetworkDependenciesAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{

		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();
			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			//Pull the job type id off the query string -- NOTE: Url for calling this action includes "?jobtype=x"
			int jobType = (new Integer((String) request.getParameter("jobtype"))).intValue();

			//Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

			//Create array of hashtables to hold the parameters we receive from the database
			Hashtable[] parameters = new Hashtable[9];

			//this Action is called when admin clicks on one of the ten choices for job type.  Therefore, the job type
			//is known and parameters will hold info for the nine activities of that job type
			try
			{
				//Get the parameters for each activity
				for (int i=0; i < 9; i++)
				{
					parameters[i] = dbFacade.getGameParametersDBHandler().getNetworkDependencyParameters(game, jobType, i+1);
				}
				dbFacade.close();

			}
			catch(ParametersNotFoundException pnfe)
			{
				// All games should have parameters.  If there are no parameters
				// found for a game it is an error.
				log.fatal(pnfe);

				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Some problem with the database
				log.fatal(bigdbe);

				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			//Determine the activity name in the resource bundle associated with the activity type number
			MessageResources resources = getResources(request);
			String jobTypeName = resources.getMessage("jobType.jobType" + jobType);

			NetworkDependenciesForm networkDependenciesForm = new NetworkDependenciesForm();

			networkDependenciesForm.setJobTypeName(jobTypeName);
			networkDependenciesForm.setJobTypeID(jobType);

			//determine the activity type name and number in the resource bundle associated with the activity type number
			//Note: this refers to content of the "html:select" tags that have a drop-down list for choices
			ArrayList[] allOptions = new ArrayList[18];
            String activityTypeName = null;

			for (int i=0; i<18; i++)
			{
				allOptions[i] = new ArrayList();
				//Every list of options contains the "none" option.
				activityTypeName = resources.getMessage("networkDependencies.none");
				allOptions[i].add(new LabelValueBean(activityTypeName, "0"));

				//Each list of options will be missing the option equal to its own index number. ie ArrayList #1 won't have an option 1
				//This is so an activity can never depend on itself

                if (i < 9)
                {
                    //first nine times through - use names for activities 1-9
				    for (int j = 1; j < 10; j++)
				    {
				  	    if(j != (i+1))
					    {
						    activityTypeName = resources.getMessage("activity.activity" + j + "short");
						    allOptions[i].add(new LabelValueBean(activityTypeName, Integer.toString(j)));
					    }
				    }
                }
                else
                {
                    //next nine times through - use names for activities 10-18
				    for (int j = 1; j < 10; j++)
				    {
					    if(j != (i-8))
					    {
						    activityTypeName = resources.getMessage("activity.activity" + (j+9) + "short");
						    allOptions[i].add(new LabelValueBean(activityTypeName, Integer.toString(j)));
					    }
				    }
                }
			}

			networkDependenciesForm.setActivity1TypeArray(allOptions[0]);
			networkDependenciesForm.setActivity2TypeArray(allOptions[1]);
			networkDependenciesForm.setActivity3TypeArray(allOptions[2]);
			networkDependenciesForm.setActivity4TypeArray(allOptions[3]);
			networkDependenciesForm.setActivity5TypeArray(allOptions[4]);
			networkDependenciesForm.setActivity6TypeArray(allOptions[5]);
			networkDependenciesForm.setActivity7TypeArray(allOptions[6]);
			networkDependenciesForm.setActivity8TypeArray(allOptions[7]);
			networkDependenciesForm.setActivity9TypeArray(allOptions[8]);
			networkDependenciesForm.setActivity10TypeArray(allOptions[9]);
			networkDependenciesForm.setActivity11TypeArray(allOptions[10]);
			networkDependenciesForm.setActivity12TypeArray(allOptions[11]);
			networkDependenciesForm.setActivity13TypeArray(allOptions[12]);
			networkDependenciesForm.setActivity14TypeArray(allOptions[13]);
			networkDependenciesForm.setActivity15TypeArray(allOptions[14]);
			networkDependenciesForm.setActivity16TypeArray(allOptions[15]);
			networkDependenciesForm.setActivity17TypeArray(allOptions[16]);
			networkDependenciesForm.setActivity18TypeArray(allOptions[17]);

            networkDependenciesForm.setNetworkDependencies(parameters);
			session.setAttribute("networkDependenciesForm", networkDependenciesForm);

			//create a list of all the equipment in the database
			List listOfAllEquip = null;
			try
			{
				listOfAllEquip = dbFacade.getEquipmentParametersDBHandler().getListOfAllEquipment(game);

			}
			catch(ParametersNotFoundException pnfe)
			{
				//All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Some problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

            //add "none" at top of list
            EquipmentList addNone = new EquipmentList();
            addNone.setEquipmentTitle("none");
            addNone.setEquipmentId(0);
            listOfAllEquip.add(0, addNone);

			session.setAttribute("listOfAllEquip", listOfAllEquip);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}
	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}
}
