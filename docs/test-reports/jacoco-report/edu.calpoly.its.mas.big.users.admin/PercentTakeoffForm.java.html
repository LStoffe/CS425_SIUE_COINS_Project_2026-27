package edu.calpoly.its.mas.big.users.admin;

import java.util.Hashtable;

import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;


/**
 * Stores the percent take-off values of all 9 activities for one
 * job type.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.6 $ $Date: 2001/10/24 17:40:58 $
 */
public class PercentTakeoffForm extends ActionForm
{
	private String [] takeoffPercentStrings = new String[9];
	private int [] takeoffPercents = new int[9];

	private boolean makeDefaults = false;
	private String jobTypeName;
	private int jobTypeID;

	/**
	 * Validate the submitted values.  Percent Take-off values must be between
	 * 0 and 100.  Also, the 9 Percent Take-off values for each Job Type
	 * must sum to 100.  If these requirements are not met, the appropriate
	 * <code>ActionError</code> objects are created and returned to the user
	 * through the normal Struts mechanism.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
		boolean integerParseError = false;
		boolean negativeFound = false;

		// Percent Takeoffs should sum to 100
		int takeoffSum = 0;

		// If the Restore Defaults button was pressed, validation should not take place
		// This makes the assumption that the defaults are valid
		String testRestoreDefaults = (String) request.getParameter("restoreDefaults");
		if (testRestoreDefaults != null)
		{
			return(null);
		}

        // Sorry, this is ugly
        for(int i = 0; i < 9; i++)
        {
			// Check for null or empty Strings
			if((takeoffPercentStrings[i].trim() == null) ||
			   (takeoffPercentStrings[i].trim().length() < 1)) {
				nullParams = true;
			} else {
				try {
					// Attempt String->int conversion and populate int array
                    takeoffPercents[i] = Integer.parseInt(takeoffPercentStrings[i]);

					// Verify that this Percent isn't negative
					if (takeoffPercents[i] < 0) {
						negativeFound = true;
					} else {
						// Add it to the running sum if it passes all tests above
						takeoffSum += takeoffPercents[i];
					}
                } catch(NumberFormatException nfe) {
                    integerParseError = true;
                }
			}
		}

		if (takeoffSum != 100)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.sumTo100"));
		}

		if (nullParams == true)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.null"));
		}

		if (integerParseError == true)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.format"));
		}

		if (negativeFound == true)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.negative"));
		}

		return errors;
	}

    /**
     * Reset fields to their initial values
     */
    public void reset()
    {
		makeDefaults = false;
	}

	/**
	 * Populate this <code>ActionForm</code>'s fields using the provided
	 * int array.  The array index plus 1 is equal to the BIG activity to
	 * which the PercentTake-off corresponds.
	 */
	public void setTakeoffPercents(int[] takeoffs)
	{
		takeoffPercents = takeoffs;
		intsToStrings(takeoffPercents, takeoffPercentStrings);
	}

	/**
	 * Populate this <code>ActionForm</code>'s fields using the provided
     * <code>Hashtable</code>.  The <code>Hashtable</code>'s keys are
	 * ActivityX, where X is an Activity number from 1-9.
	 */
	public void setTakeoffPercents(Hashtable params)
	{

		for (int i = 0; i < 9; i++) {
			takeoffPercents[i] = ((Integer) params.get("Activity" + (i+1))).intValue();
		}
		intsToStrings(takeoffPercents, takeoffPercentStrings);

	}

	/**
	 * Return the Takeoff Percents for the Job Type represented by this
	 * <code>ActionForm</code>.  The <code>Hashtable</code>'s keys are
	 * ActivityX, where X is an Activity number from 1-9.
	 */
	public Hashtable getTakeoffPercents() {
		Hashtable tempTakeoffs = new Hashtable();

		for (int i = 0; i < 9; i++) {
			tempTakeoffs.put("Activity" + (i+1), new Integer(takeoffPercents[i]));
		}

		return tempTakeoffs;

	}

	// I'm sure more get/set methods will be needed...

	//**************************** Private Methods *********************************

	/* Converts a set of strings into ints, stored in an array
	 * Param: stringValues - Array of strings to be converted into ints
	 * Param: intValues - Array of ints where the int values of the Strings will be stored
	 * Precondition: intValues and stringValues must have equal length
	 */
	private void stringsToInts(String[] stringValues, int[] intValues)
	{
		for (int i = 0; i < stringValues.length; i++) {
			intValues[i] = Integer.parseInt(stringValues[i]);
		}

	}

	 /* Converts a set of ints into into strings, stored in the appropriate variable
	  * Param: intValues - Array of ints to be converted into Strings
	  * Param: stringValues - Array of Strings where the String value of the ints will be stored
	  * Precondition: intValues and stringValues must have equal length
	  */
     private void intsToStrings(int[] intValues, String[] stringValues)
	 {
		 for (int i = 0; i < stringValues.length; i++) {

			 stringValues[i] = Integer.toString(intValues[i]);

		 }

	 }
