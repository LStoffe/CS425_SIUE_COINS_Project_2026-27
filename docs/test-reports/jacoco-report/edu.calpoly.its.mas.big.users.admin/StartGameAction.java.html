package edu.calpoly.its.mas.big.users.admin;

/**
 * Begins the process of starting a game.
 * Starting a game is a similiar process to incrementing the period of a game
 * but has some differences.  Loans do not need to be processed, Period objects for both 1 and 2
 * need to be created, and jobs do not need to be updated.
 *
 *
 * @author Kevin Craig
 *  Modified 3/21/02 Dara Manker: added code to facilitate company login verification
 *
 * @version $Revision: 1.7 $ $Date: 2002/05/02 18:57:07 $
 */
public class StartGameAction extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{

		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    HttpSession session = request.getSession();

		//Process request if it was not cancelled or initiated by an Admin's double-click on a menu option
        if(  (session.getAttribute("doNotProcess")==null) && (!isCancelled(request))  )
        {
			// Set flag to prevent an Admin's double-click from rerunning this update (flag is removed in the jsp)
			session.setAttribute("doNotProcessFlag", "true");

			ActionMessages errors = new ActionMessages();

			Game game = (Game) session.getAttribute("game");

            BIGDatabaseFacadeInterface dbFacade = getDbFacade(request, true);

            try
            {
				ExpandedGame xGame = new ExpandedGame(game, dbFacade);
				xGame.startGame();
				dbFacade.close();
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				bigdbe.printStackTrace();
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				ServletContext context = servlet.getServletContext();
				LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
				// Update the GameLoginInfo period and allow companies to log in
                loginBroker.initializePeriod(game, context);

				if(log.isDebugEnabled())
				{
					// Test that GLI was stored with proper values:
					GameLoginInfo gamelogininfo = (GameLoginInfo) context.getAttribute(Integer.toString(game.getGameID()));
					log.debug("StartGameAction: Stored GLI in Servlet Context with values:");
					log.debug("GameLoginInfo name=" + gamelogininfo.getGameID());
					log.debug("GameLoginInfo currentPeriod=" + gamelogininfo.getCurrentPeriod());
					log.debug("GameLoginInfo loginAllowed=" + gamelogininfo.getLoginAllowed());
					log.debug("GameLoginInfo numberCurrentlyLoggedIn=" + gamelogininfo.getNumberCurrentlyLoggedIn());
				}

				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}
}
