/**
 * Chooses a company to edit.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.11 $ $Date: 2001/10/31 19:39:14 $
 */
public class ChooseCompanyAction extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
					   			 ActionForm form,
					   			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		ActionMessages errors = new ActionMessages();

        //Get the parameter off the end of the URL
		String index = (String) request.getParameter("companyIndex");

		int companyIndex = (new Integer(index)).intValue();

        HttpSession session = request.getSession();

        Admin admin = (Admin) session.getAttribute("user");
		Game game = (Game) session.getAttribute("game");
        Vector companies = (Vector) session.getAttribute("companies");

        Company company = null;

        if (companies != null)
        {
	        company = (Company) companies.elementAt(companyIndex);
			session.setAttribute("company", company);
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"noCompanyFound\"");
			}
			return (mapping.findForward("noCompanyFound"));
		}

		//Create a new BIGDatabaseFacadeInterface which will hold connections
		BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

        //Place a CompanyForm action form into the session.  Struts will use it to populate the values to edit
        CompanyForm companyForm = new CompanyForm();

        companyForm.setUsername(company.getUsername());
        companyForm.setPassword(company.getPassword());
        companyForm.setConfirmPassword(company.getPassword());
        companyForm.setTitle(company.getTitle());
        companyForm.setCashOnHand(company.getCashOnHand());
        companyForm.setWIPLimit(company.getWIPLimit());
        companyForm.setPerJobLimitAsWIPPercent(company.getPerJobLimitAsWIPPercent());

        session.setAttribute("companyForm", companyForm);


		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}

			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
			saveToken(request);
			return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}

	}
}
