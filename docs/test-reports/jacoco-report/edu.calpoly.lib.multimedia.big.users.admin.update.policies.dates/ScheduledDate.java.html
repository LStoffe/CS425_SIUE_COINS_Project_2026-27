package edu.calpoly.lib.multimedia.big.users.admin.update.policies.dates;

import java.util.Date;
import java.text.SimpleDateFormat;
import java.text.DateFormat;

/**
 * Class representing a date/time for which a period update is schedule.
 * This class meets the java bean specifications.
 */
public class ScheduledDate
{
	private Date date;
	private Date time;
	private boolean complete;
	private long dateHashCode;

	public ScheduledDate()
	{

	}

	public void setDate(Date date)
	{
		this.date = date;
	}

	public Date getDate()
	{
		return date;
	}

	public void setTime(Date time)
	{
		this.time = time;
	}

	public Date getTime()
	{
		return time;
	}

	/**
	 * Returns whether or not this update is complete (i.e. has already occurred)
	 */
	public boolean getComplete()
	{
		return complete;
	}

	/**
	 * Sets whether or not this update is complete (i.e. has already occurred)
	 */
	public void setComplete(boolean complete)
	{
		this.complete = complete;
	}

	public long getDateHashCode()
	{
		return dateHashCode;
	}

	public void setDateHashCode(long dateHashCode)
	{
		this.dateHashCode = dateHashCode;
	}

	/**
	 * Returns the date/time of this update in a human readable format.
	 * If date or time data is missing, the word "Error" is returned.
	 */
	public String toString()
	{

		if( (this.date == null) || (this.time == null) )
		{
			return "error";
		}
		else
		{
			DateFormat dateFormat = new SimpleDateFormat("MMM d, yyyy" );
			DateFormat timeFormat = new SimpleDateFormat("h:mm a" );

			String scheduledUpdateString = "";
			scheduledUpdateString += dateFormat.format(this.date) + "   ";
			scheduledUpdateString += timeFormat.format(this.time);

			return scheduledUpdateString;
		}
	}

	public String getFormattedOutput()
	{
		return this.toString();
	}
}
