package edu.calpoly.lib.multimedia.big.users.admin.update.policies.dates;

//Java imports
import javax.servlet.http.HttpServletRequest;
import java.util.Date;
import java.text.DateFormat;
import java.text.SimpleDateFormat;
import java.util.Calendar;

//Struts imports
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * ActionForm used to hold days update policy settings.
 *
 *
 * @author Kevin Craig
 *
 */
public final class DatesPolicySettingsForm  extends ActionForm
{
	private int day;
	private int month;
	private int year;

	private String updateTime;

	private Log log;

	public DatesPolicySettingsForm()
	{
		super();

		log = LogFactory.getLog(this.getClass());
	}

    //**************** Public Methods ******************


	/**
	 * All String parameters can be correctly converted to their expected type.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

        boolean error = false;

		DateFormat df = new SimpleDateFormat("h:mm a" );

		try
		{
			df.parse(updateTime);
		}
		catch(Exception e)
		{
			error = true;
		}

		if(error)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.invalid"));
		}

		return errors;
	}


	public void reset (ActionMapping mapping, HttpServletRequest request)
	{
	}


//**************************** Set & Get Methods ************************

	public int getDay()
	{
		return day;
	}

	public void setDay(int day)
	{
		this.day = day;
	}

	public int getMonth()
	{
		return month;
	}

	public void setMonth(int month)
	{
		this.month = month;
	}

	public int getYear()
	{
		return year;
	}

	public void setYear(int year)
	{
		this.year = year;
	}

	public String getUpdateTime()
	{
		return updateTime;
	}

	public void setUpdateTime(String updateTime)
	{
		this.updateTime = updateTime;
	}

	public Date getUpdateDateAsDate()
	{
		Calendar cal = Calendar.getInstance();
		cal.set(year, month, day);

		return cal.getTime();
	}

	public Date getUpdateTimeAsDate()
	{
		Date time = null;
		DateFormat df = new SimpleDateFormat("h:mm a" );

		try
		{
			time = df.parse(updateTime);
		}
		catch(Exception e)
		{
			log.error("", e);
		}

		return time;
	}

	public void setUpdateTimeAsDate(Date updateTime)
	{
		DateFormat df = new SimpleDateFormat("h:mm a" );

		this.updateTime = df.format(updateTime);
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": updateTime is " + updateTime);
		}

	}

}
