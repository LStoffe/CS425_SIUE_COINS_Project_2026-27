package edu.calpoly.lib.multimedia.big.users.admin.update.policies.dates;

import java.util.Date;
import java.util.Calendar;

/**
 * Creates Date hash codes as required to store schedule updates
 * under the dates policy.
 */
public class DateHasher
{
	/**
	 * Returns a Date hash code created from two <code>Date</code>
	 * objects.  The year, month, and day fields will be used from the
	 * first <code>Date</code> object and the hours, minutes
	 * fields will be used from the second <code>Date</code> object.
	 * <br><br>
	 * All hash codes are 12 digit integers in the format YYYYMMDDhhmm.
	 *
	 */
	public static long getDateHashCode(Date date, Date time)
	{
		Calendar dateCal = Calendar.getInstance();
		Calendar timeCal = Calendar.getInstance();
		dateCal.setTime(date);
		timeCal.setTime(time);

		org.apache.commons.logging.Log log = org.apache.commons.logging.LogFactory.getLog("edu.calpoly.lib.multimedia.big.users.admin.update.policies.dates.DateHasher");

		long dateHash = 0l;

		dateHash += dateCal.get(Calendar.YEAR) * 100000000l;
		log.debug(dateCal.get(Calendar.YEAR) + " " + dateHash);

		dateHash += dateCal.get(Calendar.MONTH) * 1000000l;
		log.debug(dateCal.get(Calendar.MONTH) + " " + dateHash);
		dateHash += dateCal.get(Calendar.DAY_OF_MONTH) * 10000l;
		log.debug(dateCal.get(Calendar.DAY_OF_MONTH) + " " + dateHash);
		dateHash += timeCal.get(Calendar.HOUR_OF_DAY) * 100l;
		log.debug(timeCal.get(Calendar.HOUR_OF_DAY) + " " + dateHash);
		dateHash += timeCal.get(Calendar.MINUTE);
		log.debug(timeCal.get(Calendar.MINUTE) + " " + dateHash);

		return dateHash;
	}
}
