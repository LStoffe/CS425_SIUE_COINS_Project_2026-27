package edu.calpoly.its.mas.big.tags;


import java.io.IOException;
import javax.servlet.http.HttpSession;
import javax.servlet.jsp.JspException;
import javax.servlet.jsp.JspWriter;
import javax.servlet.jsp.PageContext;
import javax.servlet.jsp.tagext.TagSupport;

import edu.calpoly.its.mas.big.users.User;

/**
 * Tag that checks if a user is logged in.
 * This tag can be placed on a page that is available only to users who are logged in.
 * If a user who is not logged in tries to access the page (through a bookmark or similiar) they
 * will be returned to the login screen.
 * <br><br>
 * A user is defined as logged-in if in their session they have a <code>User</code> object
 * whose getIsLoggedIn() method returns true.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.3 $ $Date: 2001/07/10 23:10:26 $
 */
public class CheckLoggedIn extends TagSupport
{

	public int doStartTag() throws JspException
	{
		return (SKIP_BODY);
    }

	//Check is there is a User in the session.  If there is an its getLoggedIn method() returns true evaluate the rest of the page
	//Otherwise return them to the login screen
    public int doEndTag() throws JspException
    {
		boolean loggedIn = false;
		HttpSession session = pageContext.getSession();
		if (session != null)
		{
			User user = (User) session.getAttribute("user");
			if (user != null && user.getIsLoggedIn())
				loggedIn = true;
		}

		if (loggedIn)
		{
		    return (EVAL_PAGE);
		}
		else
		{
		    try
		    {
				pageContext.forward("/login.jsp");
		    }
		    catch (Exception e)
		    {
				throw new JspException(e.toString());
		    }
		    return (SKIP_PAGE);
		}

    }
}
