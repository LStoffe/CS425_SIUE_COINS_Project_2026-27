package edu.calpoly.its.mas.big.tags;

import java.io.IOException;
import java.io.PrintStream;

import javax.servlet.jsp.JspException;
import javax.servlet.jsp.JspWriter;
import javax.servlet.jsp.PageContext;
import javax.servlet.jsp.tagext.TagSupport;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.ServletContext;
import javax.servlet.ServletException;

import edu.calpoly.its.mas.big.game.Game;

import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.users.User;

/**
 * <code>IncrementNumberOfCosLoggedInTag.java</code> increments the number of
 * companies logged in to a game by incrementing the value stored in the GameLoginInfo
 * object in the ServletContext for this game.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.2 $ $Date: 2002/04/12 19:17:22 $
 *
 * @preconditon An attribute named "newLogin" is placed in the company's request
 * after the company has been authenticated.
 *
 * @postcondition The value which tracks the number of companies currently logged in to
 * the game in the GameLoginInfo object in the BIG SessionContext is incremented by one.
 * The attribute named "newLogin" is removed from the company's request
 */

public class IncrementNumberOfCosLoggedInTag extends TagSupport
{


   public int doStartTag()
   {
      return SKIP_BODY;
   }


   public int doEndTag()
   {

	  // Determine if this page access is by a newly authenticated in user
	  ServletContext context = pageContext.getServletContext();
	  HttpServletRequest request = (HttpServletRequest) pageContext.getRequest();

      if(  (request.getAttribute("newLogin")!=null) )
      {
		 // Increment the number of Companies currently logged in
	     HttpSession session = pageContext.getSession();
	     Game game = (Game) session.getAttribute("game");

         try
         {
            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
            loginBroker.incrementNumberOfLoggedInCompanies(game, context);
            request.removeAttribute("newLogin");
         }
	     catch(Exception e){}

      }  //end if newLogin

      return EVAL_PAGE;
   }

}
