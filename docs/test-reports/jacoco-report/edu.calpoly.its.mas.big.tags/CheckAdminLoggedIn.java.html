package edu.calpoly.its.mas.big.tags;


import java.io.IOException;
import javax.servlet.http.HttpSession;
import javax.servlet.jsp.JspException;
import javax.servlet.jsp.JspWriter;
import javax.servlet.jsp.PageContext;
import javax.servlet.jsp.tagext.TagSupport;

import edu.calpoly.its.mas.big.users.User;
import edu.calpoly.its.mas.big.users.admin.Admin;

/**
 * Tag that checks if an administrator is logged in.
 * This tag can be placed on a page that is available only to administrators who are logged in.
 * If an administrator who is not logged in tries to access the page (through a bookmark or similiar) they
 * will be returned to the login screen.  If user who is not an administrator tries to view the page they will be
 * returned to the login screen.
 * <br><br>
 * An administrator is defined as logged-in if in their session they have a <code>Admin</code> object
 * whose getIsLoggedIn() method returns true.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.3 $ $Date: 2001/07/13 17:14:03 $
 */
public class CheckAdminLoggedIn extends TagSupport
{

	public int doStartTag() throws JspException
	{
		return (SKIP_BODY);
    }

	//Check is there is an Admin in the session.  If there is and its getLoggedIn method() returns true then evaluate the rest of the page
	//Otherwise return them to the login screen
    public int doEndTag() throws JspException
    {    
		boolean loggedIn = false;
		HttpSession session = pageContext.getSession();
		if (session != null)
		{
			User user = (User) session.getAttribute("user");

            if (user != null && user.getIsLoggedIn())
            {
				if (user instanceof Admin)
				{
					loggedIn = true;
				}

			}
		}

		if (loggedIn)
		{
		    return (EVAL_PAGE);
		}
		else
		{
		    try
		    {
				pageContext.forward("/login.jsp");
		    }
		    catch (Exception e)
		    {
				throw new JspException(e.toString());
		    }
		    return (SKIP_PAGE);
		}

    }
}
