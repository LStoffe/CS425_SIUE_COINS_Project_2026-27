package edu.calpoly.its.mas.big.tags;

//JDK imports
import java.io.IOException;

//Servlet/JSP imports
import javax.servlet.jsp.JspException;
import javax.servlet.jsp.JspWriter;

import javax.servlet.jsp.tagext.TagSupport;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.ServletContext;
import javax.servlet.ServletException;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.Globals;

//BIG imports
import edu.calpoly.its.mas.big.game.Game;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * <code>LoginVerificationTag.java</code> verifies that a company should be allowed
 * to access a BIG jsp page.  Access to a jsp page is based upon the following
 * criteria:
 * <ol>
 *   <li> Company has been authenticated
 *   <li> Game has started (is at period 1 or greater)
 *   <li> Admin is not currently updating the period
 *   <li> Admin has not updated the period between the time the company last accessed a JSP page and this access of a JSP page (while company was still logged in).
 * </ol>
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.5 $ $Date: 2002/04/18 15:44:13 $
 *
 */

public class LoginVerificationTag extends TagSupport
{


   public int doStartTag()
   {
      return SKIP_BODY;
   }


   public int doEndTag()
   {
	  Log log = LogFactory.getLog(this.getClass());

	  int returnValue = EVAL_PAGE;
	  String loginMessagePage = new String("/logoutAlert.jsp");

	  ServletContext context = pageContext.getServletContext();
	  HttpServletRequest request = (HttpServletRequest) pageContext.getRequest();

      try
      {

	     // Check that user's session is valid
	     HttpSession session = pageContext.getSession();
	     if(session == null)
	     {
		    ActionErrors errors = new ActionErrors();
		    errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.userNotLoggedIn"));
		    request.setAttribute(Globals.ERROR_KEY, errors);
		    pageContext.forward(loginMessagePage);
		    returnValue = SKIP_PAGE;
	     }
         else
         {
	        // Check for existance of Game object in user's session
	        Game game = (Game) session.getAttribute("game");
	        if(game == null)
	        {
		       ActionErrors errors = new ActionErrors();
		       errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.userNotLoggedIn"));
		       request.setAttribute(Globals.ERROR_KEY, errors);
		       pageContext.forward(loginMessagePage);
		       returnValue = SKIP_PAGE;
	        }
	        else
	        {
               LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

               // If for any reason the company should not be allowed to login, route the
               // company to an error message page.
	           if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	           {
			      returnValue = SKIP_PAGE;
			      pageContext.forward(loginMessagePage);
  	           }
  	           else
  	           {
			      returnValue = EVAL_PAGE;
		       }
	        }
         }
	  }
      catch(ServletException serv)
      {
         log.fatal("", serv);
      }
      catch(IOException ioe)
      {
		  log.fatal("", ioe);
	  }

      return returnValue;
   }

}
