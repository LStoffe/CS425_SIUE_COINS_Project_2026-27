	public List<EmployeeTypeRecord> getCompanyEmployment(Game game, Company company, int periodNumber) throws BIGDatabaseFacadeException
	{
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Starting method getCompanyEmployment");
		}

		List<EmployeeTypeRecord> list = new ArrayList<EmployeeTypeRecord>();

		try
		{
			sqlMap.startTransaction();

			Map m = new HashMap();
			m.put("gameid" , game.getGameID());
			m.put("companyid" , company.getCompanyID());
			m.put("periodid", periodNumber);
			list = (List<EmployeeTypeRecord>) sqlMap.queryForList("getCompanyEmployment", m);

			sqlMap.commitTransaction();

			log.debug(list.size() + "LIST SIZE");
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Returning from method getCompanyEmployment");
		}

		return list;
	}

	public Map<String, Integer> getUnfilledMosp(Game game, Company company, int periodNumber) throws BIGDatabaseFacadeException
	{
		Map<String, Integer> skillsMap = new HashMap<String, Integer>();

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Starting method getUnfilledMosp");
		}

		try
		{
			sqlMap.startTransaction();

			Map<String, Object> params = new HashMap<String, Object>();
			params.put("gameid" , game.getGameID());
			params.put("companyid" , company.getCompanyID());
			params.put("periodid", periodNumber);

			List maps = (List) sqlMap.queryForList("getUnfilledMospEntries", params);

			sqlMap.commitTransaction();

			for(Object o : maps)
			{
				Map m = (Map) o;
				skillsMap.put( (String) m.get("skill_code"), (Integer) m.get("points"));
			}
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Returning from method getUnfilledMosp");
		}

		return skillsMap;
	}
