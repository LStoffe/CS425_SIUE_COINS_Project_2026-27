    public ContractsInProgressReportForm getContractsInProgressReport(Game game, int periodNum, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		ContractsInProgressReportForm contractsInProgressReport = null;
		Vector contracts = new Vector();
		try
		{
			String query = "SELECT cr.jobid, j.job_type_id, j.type_modifier, previous_cost, current_cost, cr.billed_to_date, est_cost_to_complete, big_computes_est_cost, " +
					       " revenue_recorded, bid_amount, change_order_pay, bl.amount_billed, bl.rejected " +
						   " FROM contract_reports cr, job j, bid b, billings bl WHERE cr.gameid = ? AND cr.periodid = ? AND cr.companyid = ? " +
						   " AND cr.jobid = j.jobid AND b.jobid = cr.jobid AND cr.companyid = b.companyid AND cr.gameid = j.gameid AND cr.gameid = b.gameid AND " +
						   " ( ( j.is_active = TRUE) OR (cr.periodid < j.period_completed)) AND j.jobid = bl.jobid AND cr.periodid = bl.periodid AND cr.gameid = bl.gameid " +
						   " ORDER BY cr.jobid";

						   /*
						      The contracts table is joined to the job table to get the change order amounts and to determine if the job has finished.
						      It is joined to the bid table to get the original contract price.
						      The last part of the WHERE clause determines if a job has completed, and if so should not appear in the contracts
						      in progress report.  Until a job completes its is_active field will be true.
						      As long as the job is active is hasn't completed so it should be displayed.  The left-hand side of the OR statement handles this case.
						      Once a job is complete its period_completed field is set to the period it
						      completed.  If the period requested for the report is before the job completed then the job should appear on the report.
						      The right-hand side of the OR handles this case.
						   */

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				int currentBillingAmount = 0;
				//if(!rset.getBoolean("rejected"))
				//{
					currentBillingAmount = rset.getInt("amount_billed");
				//}
				Contract c = new Contract(rset.getInt("jobid"),
										   rset.getInt("job_type_id"),
										   rset.getInt("type_modifier"),
										   rset.getInt("bid_amount"),
										   rset.getInt("change_order_pay"),
										   rset.getInt("est_cost_to_complete"),
										   rset.getBoolean("big_computes_est_cost"),
										   rset.getInt("billed_to_date") + currentBillingAmount,
										   rset.getInt("current_cost"),
										   rset.getInt("previous_cost"),
										   rset.getInt("revenue_recorded")
				                          );

			    c.setPeriodID(periodNum);

			    //Check if this job is negotiated
				String negotiatedCheck = "SELECT jobid FROM negotiated_job WHERE jobid = ? AND gameid = ?";
				PreparedStatement pstmt2 = connection.prepareStatement(negotiatedCheck, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
				pstmt2.setInt(1, rset.getInt("jobid"));
				pstmt2.setInt(2, game.getGameID());

				ResultSet rset2 = pstmt2.executeQuery();

				// Check if no rows are returned
				// (rset.last() returns false if there are no rows in the result set)
				if(!rset2.last())
				{
				}
				else
				{
					//System.out.println("DefaultContractsReportDBHandler.getContractsInProgressReport: Job " + rset.getInt("jobid") + " is negotiated");
					c.setNegotiated(true);
					c.setNegotiatedJob((NegotiatedJob) (new edu.calpoly.its.mas.big.db.DefaultJobDBHandler(connection, null)).getJob(game, rset.getInt("jobid")));
				}
				rset2.close();
				pstmt2.close();

            	contracts.add(c);
			}

			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			sqle.printStackTrace();
			throw new BIGDatabaseFacadeException(sqle);
		}

		contractsInProgressReport = new ContractsInProgressReportForm(company.getTitle(), contracts);

		return contractsInProgressReport;
    }

    public Contract getContract(Game game, int periodNum, Job job, Company company) throws BIGDatabaseFacadeException
    {
		Contract contract = null;
		try
		{
			String query = "SELECT cr.jobid, j.job_type_id, j.type_modifier, previous_cost, current_cost, cr.billed_to_date, est_cost_to_complete, big_computes_est_cost, " +
						   " revenue_recorded, bid_amount, change_order_pay, bl.amount_billed, bl.rejected " +
						   " FROM contract_reports cr, job j, bid b, billings bl WHERE cr.gameid = ? AND cr.periodid = ? AND cr.companyid = ? AND j.jobid = ? " +
						   " AND cr.jobid = j.jobid AND b.jobid = cr.jobid AND cr.companyid = b.companyid AND cr.gameid = j.gameid AND cr.gameid = b.gameid " +
						   " AND j.jobid = bl.jobid AND cr.periodid = bl.periodid AND cr.gameid = bl.gameid  " +
						   " ORDER BY cr.jobid";

			pstmt.setInt(2, periodNum);

			pstmt.setInt(4, job.getJobID());

				int currentBillingAmount = 0;
				//if(!rset.getBoolean("rejected") )
				//{

										   rset.getInt("est_cost_to_complete"),
										   rset.getBoolean("big_computes_est_cost"),
										   rset.getInt("billed_to_date") + currentBillingAmount,
										   rset.getInt("current_cost"),
										   rset.getInt("previous_cost"),
										   rset.getInt("revenue_recorded")
				                          );

			    contract.setPeriodID(periodNum);

			//done with the result set
			rset.close();
			pstmt.close();

			//Check if this job is negotiated
			String negotiatedCheck = "SELECT jobid FROM negotiated_job WHERE jobid = ? AND gameid = ?";

			}
			else
			{
				//System.out.println("DefaultContractsReportDBHandler.getContract: Job " + contract.getJobID() + " is negotiated");

		}

    public CompletedContractsReportForm getCompletedContractsReport(Game game, int periodNum, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		CompletedContractsReportForm completedContractsReport = null;
		Vector contracts = new Vector();
		try
		{
			String query = "SELECT j.period_completed, cr.jobid, j.job_type_id, j.type_modifier, previous_cost, current_cost, cr.billed_to_date, est_cost_to_complete, revenue_recorded, bid_amount, change_order_pay, bl.amount_billed, bl.rejected " +
						   " FROM contract_reports cr, job j, bid b, billings bl WHERE cr.gameid = ? AND cr.periodid = j.period_completed AND cr.companyid = ? " +
						   " AND cr.jobid = j.jobid AND b.jobid = cr.jobid AND cr.companyid = b.companyid AND cr.gameid = j.gameid AND cr.gameid = b.gameid AND " +
						   "  j.is_active = FALSE AND cr.periodid <= ? AND j.jobid = bl.jobid AND cr.periodid = bl.periodid AND cr.gameid = bl.gameid ORDER BY cr.jobid";


            //The last entry in the contract_reports table will be for the period the job completed
            //however there may be more billings for the job in the billings table.
            //This next query will gather all these billings and sum them to get the total
            //amount billed on the job after the final contract entry
			String pastCompletionBillingsQuery = "SELECT amount_paid, amount_billed, rejected FROM billings WHERE gameid = ? AND companyid = ? " +
			          							 " AND jobid = ? AND periodid > ? AND periodid <= ? ORDER BY periodid";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
	        pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, periodNum);

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				PreparedStatement innerStatement = connection.prepareStatement(pastCompletionBillingsQuery, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
				innerStatement.setInt(1, game.getGameID());
				innerStatement.setInt(2, company.getCompanyID());
				innerStatement.setInt(3, rset.getInt("jobid"));
				innerStatement.setInt(4, rset.getInt("period_completed"));
				innerStatement.setInt(5, periodNum);

				ResultSet billingSet = innerStatement.executeQuery();
				int billedSinceCompleted = 0;
				while(billingSet.next())
				{
					//On the last billing we need to record submitted bills
					//rather than paid bills
					if(billingSet.isLast())
					{
						billedSinceCompleted += billingSet.getInt("amount_billed");
					}
					else
					{
						billedSinceCompleted += billingSet.getInt("amount_paid");
					}
				}
				billingSet.close();

				int currentBillingAmount = 0;
				if(!rset.getBoolean("rejected") || periodNum == rset.getInt("period_completed") )
				{
					currentBillingAmount = rset.getInt("amount_billed");
				}

				//If the job completed the period of this report
				//current info is separate from previous info
				if(rset.getInt("period_completed") == periodNum)
				{
					Contract c = new Contract(rset.getInt("jobid"),
											   rset.getInt("job_type_id"),
											   rset.getInt("type_modifier"),
											   rset.getInt("bid_amount"),
											   rset.getInt("change_order_pay"),
											   rset.getInt("est_cost_to_complete"),
											   true, //Auto complete doesn't matter anymore
											   rset.getInt("billed_to_date") + currentBillingAmount + billedSinceCompleted,
											   rset.getInt("current_cost"),
											   rset.getInt("previous_cost"),
											   rset.getInt("revenue_recorded")
					                          );

			   		c.setPeriodID(periodNum);

			        //Check if this job is negotiated
					String negotiatedCheck = "SELECT jobid FROM negotiated_job WHERE jobid = ? AND gameid = ?";
					PreparedStatement pstmt2 = connection.prepareStatement(negotiatedCheck, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
					pstmt2.setInt(1, rset.getInt("jobid"));
					pstmt2.setInt(2, game.getGameID());

					ResultSet rset2 = pstmt2.executeQuery();

					// Check if no rows are returned
					// (rset.last() returns false if there are no rows in the result set)
					if(!rset2.last())
					{
					}
					else
					{
					    //System.out.println("DefaultContractsReportDBHandler.getCompletedContractsReport: Job " + rset.getInt("jobid") + " is negotiated");
						c.setNegotiated(true);
					    c.setNegotiatedJob((NegotiatedJob) (new edu.calpoly.its.mas.big.db.DefaultJobDBHandler(connection, null)).getJob(game, rset.getInt("jobid")));
					}
					rset2.close();
					pstmt2.close();

            		contracts.add(c);

				}
				//If the current period is greater than the period the job completed
				//all job info is previous.  Current cost info is 0 and the revenue
				//recorded it equal to the bid amount
				else
				{
					Contract c = new Contract(rset.getInt("jobid"),
											   rset.getInt("job_type_id"),
											   rset.getInt("type_modifier"),
											   rset.getInt("bid_amount"),
											   rset.getInt("change_order_pay"),
											   rset.getInt("est_cost_to_complete"),
											   true, //Auto complete doesn't matter anymore
											   rset.getInt("billed_to_date") + currentBillingAmount + billedSinceCompleted,
											   0,
											   rset.getInt("current_cost") +
											   rset.getInt("previous_cost"),
											   rset.getInt("bid_amount") + rset.getInt("change_order_pay")
					                          );
			   		c.setPeriodID(periodNum);

			        //Check if this job is negotiated
					String negotiatedCheck = "SELECT jobid FROM negotiated_job WHERE jobid = ? AND gameid = ?";

					PreparedStatement pstmt2 = connection.prepareStatement(negotiatedCheck, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
					pstmt2.setInt(1, rset.getInt("jobid"));
					pstmt2.setInt(2, game.getGameID());

					ResultSet rset2 = pstmt2.executeQuery();

					// Check if no rows are returned
					// (rset.last() returns false if there are no rows in the result set)
					if(!rset2.last())
					{
					}
					else
					{
						//System.out.println("DefaultContractsReportDBHandler.getCompletedContractsReport: Job " + rset.getInt("jobid") + " is negotiated");
						c.setNegotiated(true);
					    c.setNegotiatedJob((NegotiatedJob) (new edu.calpoly.its.mas.big.db.DefaultJobDBHandler(connection, null)).getJob(game, rset.getInt("jobid")));
					}
					rset2.close();
					pstmt2.close();

            		contracts.add(c);
				}
			}

			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		completedContractsReport = new CompletedContractsReportForm(company.getTitle(), contracts);

		return completedContractsReport;
    }



    public void addCurrentJobInfo(Game game, int periodNum, Company company, Job job, int estimatedCostToComplete, int revenueRecorded) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO contract_reports (gameid, companyid, periodid, jobid, previous_cost, current_cost, " +
		                                             " billed_to_date, est_cost_to_complete, big_computes_est_cost, revenue_recorded) " +
		                                             "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

		String previousCostQuery = "SELECT previous_cost, current_cost, big_computes_est_cost FROM contract_reports WHERE gameid = ? " +
		                           " AND companyid = ? AND jobid = ? AND periodid = ?";

		String previousBilledQuery = "SELECT billed_to_date, revenue_recorded FROM contract_reports WHERE gameid = ? " +
		                           " AND companyid = ? AND jobid = ? AND periodid = ?";

		String currentBilling = "SELECT amount_billed, rejected FROM billings WHERE gameid = ? AND jobid = ? AND periodid = ? ";

		try
		{

			int lastPeriodPreviousCost = 0;
			int lastPeriodCurrentCost = 0;
			int previousBilled = 0;
			int currentBill = 0;
			boolean autoCompute = true;


			//GET NEEDED COST INFO
			PreparedStatement pstmt = connection.prepareStatement(previousCostQuery, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, job.getJobID());
			pstmt.setInt(4, periodNum - 1);

			ResultSet rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
			}
			else
			{
				rset.first();
				lastPeriodPreviousCost = rset.getInt("previous_cost");
				lastPeriodCurrentCost = rset.getInt("current_cost");
				autoCompute = rset.getBoolean("big_computes_est_cost");

			}
			//done with the result set
			rset.close();

			//GET NEEDED BILLING INFO
			//First getting total of billing previous to the current period
			pstmt = connection.prepareStatement(previousBilledQuery, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, job.getJobID());
			pstmt.setInt(4, periodNum - 1);

			rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
			}
			else
			{
				rset.first();
				previousBilled = rset.getInt("billed_to_date");

			}
			//done with the result set
			rset.close();

			//Next get the billing for this period
			pstmt = connection.prepareStatement(currentBilling, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());
			pstmt.setInt(3, periodNum - 1);

			rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
			}
			else
			{
				rset.first();
				if(!rset.getBoolean("rejected"))
				{
					currentBill = rset.getInt("amount_billed");
				}

			}
			//done with the result set
			rset.close();

			//INSERT THE NEW INFO

			//The previous cost for this period is the previous period's previous cost
			//and the cost incurred in the pervious period
			int previousCost = lastPeriodPreviousCost + lastPeriodCurrentCost;

			pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, periodNum);
			pstmt.setInt(4, job.getJobID());
			pstmt.setInt(5, previousCost);
			pstmt.setInt(6, job.getCurrentCost());
			pstmt.setInt(7, previousBilled + currentBill);
			pstmt.setInt(8, estimatedCostToComplete);
			pstmt.setBoolean(9, autoCompute);
			pstmt.setInt(10, revenueRecorded);


			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

    }

			pstmt.setInt(2, companyID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)

			close();

		}
