    public List getEquipmentParameters(Game game)
    	throws ParametersNotFoundException, BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Starting method getEquipmentParameters");
		}

		List list = null;

	    try
	    {
			Map m = new HashMap();
			m.put("gameid" , game.getGameID());
			m.put("adminid" , game.getAdminID());

			list = (List) sqlMap.queryForList("getEquipmentParameters", m);
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Returning from method getEquipmentParameters");
		}
	    return list;
    }

    public void copyEquipmentParameters(int oldAdminID, int oldGameID, int newAdminID, int newGameID) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Starting method copyEquipmentParameters from adminid=" + oldAdminID + ", gameid=" + oldGameID + " to adminid=" + newAdminID + " , gameid=" + newGameID);
		}
	    try
	    {
			Map m = new HashMap();
			m.put("gameid" , oldGameID);
			m.put("adminid" , oldAdminID);

			List list = (List) sqlMap.queryForList("getEquipmentParameters", m);

			//list is a List of "resultMaps" (one Map for each record) - each Map contains six key/value pairs (one for each field)

			for (Object o : list)
			{
				Map mm = (Map) o;

				Map toAdd = new HashMap();
				toAdd.put("gameid", newGameID);
				toAdd.put("adminid", newAdminID);
				toAdd.put("equipmentTitle", mm.get("equipmentTitle"));
				toAdd.put("equipmentCost", mm.get("equipmentCost"));
				toAdd.put("equipmentDeployCost", mm.get("equipmentDeployCost"));
				toAdd.put("equipmentDecommissionCost", mm.get("equipmentDecommissionCost"));
				toAdd.put("equipmentPercentRaise", mm.get("equipmentPercentRaise"));
				toAdd.put("equipmentBasePoints", mm.get("equipmentBasePoints"));
				toAdd.put("equipmentId", mm.get("equipmentId"));

				sqlMap.insert("addEquipmentParameters", toAdd);
			}

			Map m1 = new HashMap();
			m1.put("gameId" , oldGameID);
			m1.put("adminId" , oldAdminID);

			list = (List) sqlMap.queryForList("getEquipNetworkDep", m1);

			for (Object o : list)
			{
				Map mm = (Map) o;

				Map toAdd = new HashMap();
				toAdd.put("gameId", newGameID);
				toAdd.put("adminId", newAdminID);
				toAdd.put("activityTypeId", mm.get("activityTypeId"));
				toAdd.put("jobTypeId", mm.get("jobTypeId"));
				toAdd.put("equipType", mm.get("equipType"));
				toAdd.put("quantityOfEquip", mm.get("quantityOfEquip"));
				toAdd.put("percentTime", mm.get("percentTime"));

				sqlMap.insert("addEquipNetworkDep", toAdd);
			}
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Returning from method copyEquipmentParameters");
		}
	}
