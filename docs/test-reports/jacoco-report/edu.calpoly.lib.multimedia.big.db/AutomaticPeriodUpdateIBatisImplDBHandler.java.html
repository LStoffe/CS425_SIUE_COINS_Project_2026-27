	public int getAutomaticUpdatePolicy(Game game) throws BIGDatabaseFacadeException
	{
		int policy = -1;
		try
		{
			session = sqlMap.openSession();
			session.startTransaction();
			Integer policyInt = (Integer) session.queryForObject("getAutomaticUpdatePolicy", game.getGameID());
			if(policyInt != null)
			{
				policy = policyInt.intValue();
			}
			session.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				session.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
			finally
			{
				session.close();
			}
		}

		return policy;
	}

	/**
	 * Defined by <code>AutomaticPeriodUpdateDBHandler</code>.
	 */
	public DaysUpdatePolicySettings getDaysPolicySettings(Game game) throws BIGDatabaseFacadeException
	{
		DaysUpdatePolicySettings daysUpdatePolicySettings = new DaysUpdatePolicySettings();

		try
		{
			session = sqlMap.openSession();
			session.startTransaction();
			List maps = (List) session.queryForList("getDaysUpdatePolicySettings", game.getGameID());

			for(Object o : maps)
			{
				Map m = (Map) o;
				int day = ((Integer) m.get("day")).shortValue();
				if(day == 0)
				{
					daysUpdatePolicySettings.setSunday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 1)
				{
					daysUpdatePolicySettings.setMonday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 2)
				{
					daysUpdatePolicySettings.setTuesday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 3)
				{
					daysUpdatePolicySettings.setWednesday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 4)
				{
					daysUpdatePolicySettings.setThursday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 5)
				{
					daysUpdatePolicySettings.setFriday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
				else if(day == 6)
				{
					daysUpdatePolicySettings.setSaturday(true);
					daysUpdatePolicySettings.setUpdateTime((Date) m.get("time"));
				}
			}
			session.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				session.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
			finally
			{
				session.close();
			}
		}

		return daysUpdatePolicySettings;
	}
