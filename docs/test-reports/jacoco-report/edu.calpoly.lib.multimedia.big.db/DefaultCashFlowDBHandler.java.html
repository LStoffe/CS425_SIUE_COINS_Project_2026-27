package edu.calpoly.lib.multimedia.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//BIG v3 imports
import edu.calpoly.lib.multimedia.big.reports.CashFlowForm;

//JDK imports
import java.util.Vector;
import java.util.Iterator;

//Database imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>CashFlowDBHandler</code>.
 *
 * @author Kevin Craig
 *
 */
public class DefaultCashFlowDBHandler extends DefaultDBHandler implements CashFlowDBHandler
{

    public DefaultCashFlowDBHandler(Connection connection)
    {
		super(connection);
	}

    public CashFlowForm getCashFlowReport(Game game, int periodNum, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		CashFlowForm cashFlowForm = new CashFlowForm();

		try
		{
			String query = "SELECT amount, source, inflow " +
						   " FROM cash_flow WHERE gameid = ? AND periodid = ? AND companyid = ? " ;

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());


			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				int amount = rset.getInt("amount");
				String source = rset.getString("source");
				boolean inflow = rset.getBoolean("inflow");

				if(inflow)
				{
					cashFlowForm.addInflow(amount, source);
				}
				else
				{
					cashFlowForm.addOutflow(amount, source);
				}


			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return cashFlowForm;
    }



    public void addCashFlowReport(Game game, Company company, CashFlowForm theReport) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO cash_flow (gameid, companyid, periodid, amount, inflow, source) " +
		                                           " VALUES (?,?,?,?,?,?)";


		theReport.reset();
		while(theReport.moreInflows())
		{
			theReport.nextInflow();
			try
			{
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, company.getCompanyID());
				pstmt.setInt(3, theReport.getPeriod());
				pstmt.setInt(4, theReport.getCurrentInflowAmount());
				pstmt.setBoolean(5, true);
				pstmt.setString(6, theReport.getCurrentInflowSource());

				int rowsChanged = pstmt.executeUpdate();
			}
			catch(SQLException sqle)
			{
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

		while(theReport.moreOutflows())
		{
			theReport.nextOutflow();
			try
			{
				PreparedStatement pstmt = connection.prepareStatement(insert);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, company.getCompanyID());
				pstmt.setInt(3, theReport.getPeriod());
				pstmt.setInt(4, theReport.getCurrentOutflowAmount());
				pstmt.setBoolean(5, false);
				pstmt.setString(6, theReport.getCurrentOutflowSource());

				int rowsChanged = pstmt.executeUpdate();
			}
			catch(SQLException sqle)
			{
				close();
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

    }




    public void removeCashFlowReports(int gameid, int companyid) throws BIGDatabaseFacadeException
    {
		String remove = "DELETE FROM cash_flow WHERE gameid = ? AND companyid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameid);
			pstmt.setInt(2, companyid);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}


    }



}
