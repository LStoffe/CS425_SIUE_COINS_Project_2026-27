package edu.calpoly.lib.multimedia.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//BIG v3 imports

//JDK imports
import java.util.Vector;
import java.util.Iterator;

//Database imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>RatiosDBHandler</code>.
 *
 * @author Kevin Craig
 *
 */
public class DefaultRatiosDBHandler extends DefaultDBHandler implements RatiosDBHandler
{

    public DefaultRatiosDBHandler(Connection connection)
    {
		super(connection);
	}


    //Note, -1000 is a flag that means this ratio has no value.  It would be near-impossible
    //for any ratio to legitimately have a value even close to -1000.
    public Float[][] getRatios(Game game, int periodNum, int pastPeriods, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
	    //Array where the value for all ratios is nonexistant (null)
		Float[] nulls = { null, null, null, null, null, null, null, null, null, null, null, null, null, null};

		//Create the array that will be returned.  If must be able to hold all past period
		//values and the currently requested values.
		Float[][] returnArray = new Float[pastPeriods + 1][];
		for(int i = 0; i < returnArray.length; i++)
		{
			returnArray[i] = nulls;
		}

		try
		{
			String query = "SELECT current_ratio, receivable_turns, payable_turns, cash_turns, " +
						   "cash_to_wip, earnings_to_interest, debt_to_equity, gross_profit, " +
						   "net_profit, return_on_equity, return_on_assets, costs_to_sales, " +
						   "ga_to_sales, op_expense_to_sales " +
						   "FROM ratios WHERE gameid = ? AND periodid <= ? AND periodid >= ? AND companyid = ? " +
						   "ORDER BY periodid DESC";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, periodNum - pastPeriods);
			pstmt.setInt(4, company.getCompanyID());


			ResultSet rset = pstmt.executeQuery();

			int i = 0;
			while(rset.next())
			{
				returnArray[i] = new Float[14];
				for(int j = 0; j < 14; j++)
				{
					if(rset.getFloat(j+1) < -999)
					{
						returnArray[i][j] = null;
					}
					else
					{
						returnArray[i][j] = new Float(rset.getFloat(j+1));
					}
				}
				i++;
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return returnArray;
    }



    public void addRatios(Game game, int periodNumber, Company company, Float[] ratios) throws BIGDatabaseFacadeException
    {

		String insert = "INSERT INTO ratios (gameid, companyid, periodid, current_ratio, receivable_turns, " +
						"payable_turns, cash_turns, cash_to_wip, earnings_to_interest, debt_to_equity, " +
						"gross_profit, net_profit, return_on_equity, return_on_assets, costs_to_sales, " +
						"ga_to_sales, op_expense_to_sales) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

		try
		{

			PreparedStatement pstmt = connection.prepareStatement(insert);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, periodNumber);

			for(int i = 0; i < ratios.length; i++)
			{
				if(ratios[i] != null)
				{
					pstmt.setFloat(i+4, ratios[i].floatValue());
					//+4 is because the ratios start on the 4th "?" in the sql statement
				}
				else
				{
					pstmt.setFloat(i+4, -1000.0f);
				}
			}

			pstmt.executeUpdate();

			pstmt.close();

			connection.commit();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

    }


    public void removeRatios(int gameid, int companyid) throws BIGDatabaseFacadeException
    {

		String remove = "DELETE FROM ratios WHERE gameid = ? AND companyid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameid);
			pstmt.setInt(2, companyid);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}


    }



}
