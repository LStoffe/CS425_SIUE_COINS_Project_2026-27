package edu.calpoly.lib.multimedia.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;


//JDK imports
import java.util.Hashtable;

//Database imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>NegotiationDBHandler</code>.
 *
 * @author Kevin Craig
 *
 */
public class DefaultNegotiationDBHandler extends DefaultDBHandler implements NegotiationDBHandler
{

    public DefaultNegotiationDBHandler(Connection connection)
    {
		super(connection);
	}

	public Hashtable getNegotiationValues(Game game, int jobID)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		Hashtable values = new Hashtable();


		try
		{
			String select = "SELECT agreed_cost, agreed_percent_profit, state, customer_target_profit, customer_haggling " +
							"FROM negotiated_job " +
							"WHERE gameid = ? AND jobid = ? ";

			PreparedStatement pstatement = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstatement.setInt(1, game.getGameID());
			pstatement.setInt(2, jobID);

			ResultSet rset = pstatement.executeQuery();
			if(rset.next())
			{
				values.put("agreedCost", new Long(rset.getLong("agreed_cost")));
				values.put("agreedProfit", new Integer(rset.getInt("agreed_percent_profit")));
				values.put("state", new Integer(rset.getInt("state")));
				values.put("customerTargetProfit", new Integer(rset.getInt("customer_target_profit")));
				values.put("customerHaggling", new Integer(rset.getInt("customer_haggling")));
			}
			else
			{
				throw new ObjectNotFoundException();
			}
			rset.close();
			pstatement.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return values;
	}


	public void setNegotationValues(Game game, int jobID, Hashtable values)
		throws ObjectNotFoundException, BIGDatabaseFacadeException
	{
		Long agreedCost = (Long) values.get("agreedCost");
		Integer agreedProfit = (Integer) values.get("agreedProfit");
		Integer state = (Integer) values.get("state");
		Integer customerTargetProfit = (Integer) values.get("customerTargetProfit");
		Integer customerHaggling = (Integer) values.get("customerHaggling");

		try
		{
			String update = "UPDATE negotiated_job SET agreed_cost = ?, agreed_percent_profit = ?, state =?, customer_target_profit = ?, customer_haggling = ? " +
							"WHERE gameid = ? AND jobid = ? ";

			PreparedStatement pstatement = connection.prepareStatement(update);
			pstatement.setLong(1, agreedCost.longValue());
			pstatement.setInt(2, agreedProfit.intValue());
			pstatement.setInt(3, state.intValue());
			pstatement.setInt(4, customerTargetProfit.intValue());
			pstatement.setInt(5, customerHaggling.intValue());
			pstatement.setInt(6, game.getGameID());
			pstatement.setInt(7, jobID);

			pstatement.executeUpdate();
			pstatement.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}
}
