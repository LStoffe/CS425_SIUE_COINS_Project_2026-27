    public Hashtable getCAMs(Company company, int period) throws BIGDatabaseFacadeException, ObjectNotFoundException
  	{

		Hashtable cams = new Hashtable();

		String select = "SELECT financial_liquidity, financial_success, bid_responsibility, " +
		 				"pace, ethics, name_recognition, apartments, schools, offices, " +
		 				"hospitals, industrial, highways, bridges, sitedevelopment, massexcavation, undergroundutilities FROM appraisal_metrics " +
		 				"WHERE companyid = ? AND periodid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();

			if(!rset.last())
			{
				throw new ObjectNotFoundException();
			}

			{
				rset.first();

				cams.put("FinancialLiquidity", new Integer(rset.getInt("financial_liquidity")));
				cams.put("FinancialSuccess", new Integer(rset.getInt("financial_success")));
				cams.put("BiddingResponsibility", new Integer(rset.getInt("bid_responsibility")));
				cams.put("Pace", new Integer(rset.getInt("pace")));
				cams.put("Ethics", new Integer(rset.getInt("ethics")));
				cams.put("NameRecognition", new Integer(rset.getInt("name_recognition")));
				cams.put("Apartments", new Integer(rset.getInt("apartments")));
				cams.put("Schools", new Integer(rset.getInt("schools")));
				cams.put("Offices", new Integer(rset.getInt("offices")));
				cams.put("Hospitals", new Integer(rset.getInt("hospitals")));

    public void updateCAM(Company company, int cam, int amount) throws BIGDatabaseFacadeException, ObjectNotFoundException
  	{
		String metric = "";

		if(cam == CAMUpdater.FINANCIAL_LIQUIDITY)
		{
			metric = " financial_liquidity ";
		}
		else if(cam == CAMUpdater.FINANCIAL_SUCCESS)
		{
			metric = " financial_success ";
		}
		else if(cam == CAMUpdater.BID_RESPONSIBILITY)
		{
			metric = " bid_responsibility ";
		}
		else if(cam == CAMUpdater.PACE)
		{
			metric = " pace ";
		}
		else if(cam == CAMUpdater.ETHICS)
		{
			metric = " ethics ";
		}
		else if(cam == CAMUpdater.NAME_RECOGNITION)
		{
			metric = " name_recognition ";
		}
		else if(cam == CAMUpdater.APARTMENTS)
		{
			metric = " apartments ";
		}
		else if(cam == CAMUpdater.SCHOOLS)
		{
			metric = " schools ";
		}
		else if(cam == CAMUpdater.OFFICES)
		{
			metric = " offices ";
		}
		else if(cam == CAMUpdater.HOSPITALS)
		{
			metric = " hospitals ";
		}
		else if(cam == CAMUpdater.INDUSTRIAL)
		{
			metric = " industrial ";
		}
		else if(cam == CAMUpdater.HIGHWAYS)
		{
			metric = " highways ";
		}
		else if(cam == CAMUpdater.BRIDGES)
		{
			metric = " bridges ";
		}
		else if(cam == CAMUpdater.SITEDEVELOPMENT)
		{
			metric = " sitedevelopment ";
		}
		else if(cam == CAMUpdater.MASSEXCAVATION)
		{
			metric = " massexcavation ";
		}
		else if(cam == CAMUpdater.UNDERGROUNDUTILITIES)
		{
			metric = " undergroundutilities ";
		}

		String query = "SELECT " + metric + " FROM appraisal_metrics_updates WHERE companyid = ?";
		String update = "UPDATE appraisal_metrics_updates SET " + metric + " = ? " +
		                "WHERE companyid = ? ";

		int currentAmount;

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();
			if(!rset.last())
			{
				close();
				throw new ObjectNotFoundException();
			}
			else
			{
				rset.first();
				currentAmount = rset.getInt(metric.trim());
			}

			currentAmount += amount;

			pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, currentAmount);
			pstmt.setInt(2, company.getCompanyID());

			int rows =  pstmt.executeUpdate();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

    public void addCAMs(Company company, int period, Hashtable cams) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO appraisal_metrics (companyid, periodid, financial_liquidity, financial_success, " +
		                "bid_responsibility, pace, ethics, name_recognition, " +
		                "apartments, schools, offices, hospitals, industrial, highways, bridges, sitedevelopment, massexcavation, undergroundutilities) " +
		                "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?) ";

		int financialLiquidity = ((Integer) cams.get("FinancialLiquidity")).intValue();
		int financialSuccess = ((Integer) cams.get("FinancialSuccess")).intValue();
		int biddingResponsibility = ((Integer) cams.get("BiddingResponsibility")).intValue();
		int pace = ((Integer) cams.get("Pace")).intValue();
		int ethics = ((Integer) cams.get("Ethics")).intValue();
		int nameRecognition = ((Integer) cams.get("NameRecognition")).intValue();
		int apartments = ((Integer) cams.get("Apartments")).intValue();
		int schools = ((Integer) cams.get("Schools")).intValue();

		int bridges = ((Integer) cams.get("Bridges")).intValue();
		int sitedevelopment = ((Integer) cams.get("SiteDevelopment")).intValue();
		int massexcavation = ((Integer) cams.get("MassExcavation")).intValue();
		int undergroundutilities = ((Integer) cams.get("UndergroundUtilities")).intValue();

		try

			pstmt.setInt(17, massexcavation);

			int rows =  pstmt.executeUpdate();
		}
