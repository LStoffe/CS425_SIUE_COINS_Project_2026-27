		super(connection);

  	public Vector getCustomers() throws BIGDatabaseFacadeException
  	{
		Vector customers = new Vector();

		String select = "SELECT customerid, " +
					    " customer_name, " +
					    " fin_liq_factor, " +
					    " fin_liq_level, " +
					    " fin_suc_factor, " +
					    " fin_suc_level, " +
					    " responsibility_factor, " +
					    " responsibility_level, " +
					    " pace_factor, " +
					    " pace_level, " +
					    " ethics_factor, " +
					    " ethics_level, " +
					    " name_rec_factor, " +
					    " name_rec_level, " +
					    " job_type_factor, " +
					    " job_type_level " +
					    " FROM customer ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				customers.add(new Customer(rset.getInt("customerid"),
										   rset.getString("customer_name"),
										   rset.getInt("fin_liq_factor"),
										   rset.getInt("fin_liq_level"),
										   rset.getInt("fin_suc_factor"),
										   rset.getInt("fin_suc_level"),
										   rset.getInt("responsibility_factor"),
										   rset.getInt("responsibility_level"),
										   rset.getInt("pace_factor"),
										   rset.getInt("pace_level"),
										   rset.getInt("ethics_factor"),
										   rset.getInt("ethics_level"),
										   rset.getInt("name_rec_factor"),
										   rset.getInt("name_rec_level"),
										   rset.getInt("job_type_factor"),
										   rset.getInt("job_type_level")
										  )
						     );
			}
			rset.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return customers;
	}

	public Customer getCustomer(int customerID) throws BIGDatabaseFacadeException
	{
		Customer customer = null;

		String select = "SELECT customerid, " +
					    " customer_name, " +
					    " fin_liq_factor, " +
					    " fin_liq_level, " +
					    " fin_suc_factor, " +
					    " fin_suc_level, " +
					    " responsibility_factor, " +
					    " responsibility_level, " +
					    " pace_factor, " +
					    " pace_level, " +
					    " ethics_factor, " +
					    " ethics_level, " +
					    " name_rec_factor, " +
					    " name_rec_level, " +
					    " job_type_factor, " +
					    " job_type_level " +
					    " FROM customer WHERE customerid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, customerID);

			ResultSet rset = pstmt.executeQuery();

			if(rset.next())
			{
				customer = new Customer(rset.getInt("customerid"),
										   rset.getString("customer_name"),
										   rset.getInt("fin_liq_factor"),
										   rset.getInt("fin_liq_level"),
										   rset.getInt("fin_suc_factor"),
										   rset.getInt("fin_suc_level"),
										   rset.getInt("responsibility_factor"),
										   rset.getInt("responsibility_level"),
										   rset.getInt("pace_factor"),
										   rset.getInt("pace_level"),
										   rset.getInt("ethics_factor"),
										   rset.getInt("ethics_level"),
										   rset.getInt("name_rec_factor"),
										   rset.getInt("name_rec_level"),
										   rset.getInt("job_type_factor"),
										   rset.getInt("job_type_level")
										);
			}
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return customer;
	}

}
