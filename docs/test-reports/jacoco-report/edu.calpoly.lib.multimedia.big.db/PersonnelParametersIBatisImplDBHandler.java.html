    public Map<String, Map<String, MosToCUsRatio>> getJobRequiredMospParameters(Game game) throws BIGDatabaseFacadeException
    {
		Map<String, Map<String, MosToCUsRatio>> results = new HashMap<String, Map<String, MosToCUsRatio>>();

	    try
	    {
			sqlMap.startTransaction();
			List list = (List) sqlMap.queryForList("getJobTypeReqMosParametersForGame", game.getGameID());

			for (Object o : list)
			{
				Map mm = (Map) o;

				String jobType = mm.get("jobType").toString();
				String skillCode = (String) mm.get("skillCode");
				Integer cuAmount = (Integer) mm.get("cuAmount");
				Integer mosAmount = (Integer) mm.get("mosAmount");

				//Get a reference to the Map for the current job type.
				Map<String, MosToCUsRatio> jobTypeMap = (Map<String, MosToCUsRatio>) results.get(jobType);

				//If no existing Map was found, create one and start using it as the Map
				//for the job type.
				if(jobTypeMap == null)
				{
					jobTypeMap = new HashMap<String, MosToCUsRatio>();
					results.put(jobType, jobTypeMap);
				}

				jobTypeMap.put(skillCode, new MosToCUsRatio(cuAmount, mosAmount));

			}
			sqlMap.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

		return results;
	}

    public EmployeeSkillConversionRatioLookup getEmployeeSkillConversionRatios(Game game) throws BIGDatabaseFacadeException
    {
		EmployeeSkillConversionRatioLookup lookup = new EmployeeSkillConversionRatioLookup();

	    try
	    {
			sqlMap.startTransaction();
			List list = (List) sqlMap.queryForList("getMosPointConversionParametersForGame", game.getGameID());

			for (Object o : list)
			{
				Map mm = (Map) o;

				String employeeTitle = (String) mm.get("employeeTitle");
				String skillCode = (String) mm.get("skillCode");
				Float ratio = (Float) mm.get("baseToMospRatio");

				lookup.addConversionRatio(employeeTitle, skillCode, ratio.floatValue());
			}
