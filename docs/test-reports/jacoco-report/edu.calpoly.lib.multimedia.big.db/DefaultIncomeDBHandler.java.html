package edu.calpoly.lib.multimedia.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//JDK imports
import java.util.Vector;
import java.util.Iterator;

//Database imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

/**
 * Default implementation of <code>IncomeDBHandler</code>.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.8 $ $Date: 2002/04/11 23:36:16 $
 */
public class DefaultIncomeDBHandler extends DefaultDBHandler implements IncomeDBHandler
{

    public DefaultIncomeDBHandler(Connection connection)
    {
		super(connection);
	}

    public int getRevenueAccountedFor(int gameid, int companyid, int periodid) throws BIGDatabaseFacadeException
    {
		int amount = 0;

		try
		{
			String query = "SELECT accounted_for_revenue "+
						   " FROM income WHERE gameid = ? AND companyid = ? AND periodid = ?" ;

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, gameid);
			pstmt.setInt(2, companyid);
			pstmt.setInt(3, periodid);
			//System.out.println(pstmt.toString());
			ResultSet rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
				amount = 0;
				rset.close();
				pstmt.close();
			}
			else
			{
				amount = rset.getInt("accounted_for_revenue");
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return amount;
    }





    public void removeRevenueAccountedFor(int gameid, int companyid) throws BIGDatabaseFacadeException
    {
		String remove = "DELETE FROM income WHERE gameid = ? AND companyid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameid);
			pstmt.setInt(2, companyid);

			int rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}


    }


    public void addRevenueAccountedFor(int gameid, int companyid, int periodid, int amount) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO income (gameid, companyid, periodid, accounted_for_revenue) " +
		                                           " VALUES (?,?,?,?)";


		try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1,gameid);
			pstmt.setInt(2, companyid);
			pstmt.setInt(3, periodid);
			pstmt.setLong(4, amount);


			int rowsChanged = pstmt.executeUpdate();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

	}

}
