  	public Vector getCCCs(Game game) throws BIGDatabaseFacadeException
  	{
		Vector cccs = new Vector();

		String select = "SELECT cccid, fitness, name FROM computer_controlled_contractors " +
						"WHERE gameid = ? AND current = TRUE";

		String selectPrefs = "SELECT preference FROM ccc_job_type_preferences " +
							 "WHERE cccid = ? ORDER BY job_type_id";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{
				PreparedStatement prefsPstmt = connection.prepareStatement(selectPrefs, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
				prefsPstmt.setInt(1, rset.getInt("cccid"));

				ResultSet prefRset = prefsPstmt.executeQuery();
				Vector prefs = new Vector();

				while(prefRset.next())
				{
					prefs.add(new Integer(prefRset.getInt("preference")));
				}
				prefRset.close();

				int[] prefInts = new int[prefs.size()];
				for(int i = 0; i < prefInts.length; i++)
				{
					prefInts[i] = ((Integer) prefs.elementAt(i)).intValue();
				}

				cccs.add(new ComputerControlledContractor(rset.getInt("cccid"),
														  rset.getString("name"),
														  rset.getInt("fitness"),
														  prefInts
														 )
						);
			}
			rset.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return cccs;
	}

  	public Vector setCCCs(Game game, Vector cccs) throws BIGDatabaseFacadeException
  	{
		Vector rejectedCCCs = new Vector();

		String getCCCID = "SELECT nextval('big_cccid_sequence') ";
		String insertPrefs = "INSERT INTO ccc_job_type_preferences (cccid, job_type_id, preference) " +
							 " VALUES (?, ?, ?) ";
		String insertCCC = "INSERT INTO computer_controlled_contractors (cccid, gameid, fitness, name, current)	" +
						   "VALUES (?, ?, ?, ?, ?) ";

		String markAllNoncurrent = "UPDATE computer_controlled_contractors SET current = FALSE WHERE gameid = ?";

		String markCCCCurrent = "UPDATE computer_controlled_contractors SET current = TRUE WHERE cccid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(markAllNoncurrent);
			pstmt.setInt(1, game.getGameID());
			pstmt.executeUpdate();

			for(int i = 0; i < cccs.size(); i++)
			{
				ComputerControlledContractor ccc = (ComputerControlledContractor) cccs.elementAt(i);

				//if (uniqueCCCName(game.getGameID(), ccc.getName()))
				//{
					if(ccc.getCCCID() < 1)
					{
						//Create a cccid for this ccc
						pstmt = connection.prepareStatement(getCCCID);
						ResultSet rset = pstmt.executeQuery();
						if(rset.next())
						{
							ccc.setCCCID(rset.getInt(1));
						}
						else
						{
							close();
							throw new BIGDatabaseFacadeException("Cannot get next value from big_cccid_sequence");
						}
						rset.close();

						//Insert this ccc
						pstmt = connection.prepareStatement(insertCCC);
						pstmt.setInt(1, ccc.getCCCID());
						pstmt.setInt(2, game.getGameID());
						pstmt.setInt(3, ccc.getFitness());
						pstmt.setString(4, ccc.getName());
						pstmt.setBoolean(5, true);

						pstmt.executeUpdate();


						//Insert the job type prefs for this ccc
						int[] prefs = ccc.getJobTypePreferences();

						pstmt = connection.prepareStatement(insertPrefs);
						for(int j = 0; j < prefs.length; j++)
						{
							pstmt.setInt(1, ccc.getCCCID());
							pstmt.setInt(2, j+1);
							pstmt.setInt(3, prefs[j]);

							pstmt.executeUpdate();
						}

					}
					else
					{
						pstmt = connection.prepareStatement(markCCCCurrent);
						pstmt.setInt(1, ccc.getCCCID());
						pstmt.executeUpdate();
					}
				//}
				//else
				//{
				//	rejectedCCCs.add(ccc);
				//}
			}
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return rejectedCCCs;
	}

	//Returns true if there is no row in the COMPUTER_CONTROLLED_CONTRACTORS table
	//with the gameid/name combination specified
	private boolean uniqueCCCName(int gameid, String name) throws SQLException
	{
		String select = "SELECT cccid FROM computer_controlled_contractor WHERE gameid = ? " +
						"AND lower(name) = lower(?)";

		PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
		pstmt.setInt(1, gameid);
		pstmt.setString(2, name);

		ResultSet rset = pstmt.executeQuery();

		if(rset.next())
		{
			rset.close();
			//System.out.println(name + " is NOT a unique name in " + gameid);
			return false;
		}
		else
		{
			rset.close();
			//System.out.println(name + " is a unique name in " + gameid);
			return true;
		}

	}

  	public void removeCCCs(int gameid) throws BIGDatabaseFacadeException
  	{
		String removeJobPrefs = "DELETE FROM ccc_job_type_preferences WHERE cccid = ANY " +
								"(SELECT cccid FROM computer_controlled_contractors " +
								"WHERE gameid = ?)";
		String removeCCCs = "DELETE FROM computer_controlled_contractors WHERE gameid = ?";

		try
		{
			//Remove all job type prefs
			PreparedStatement pstmt = connection.prepareStatement(removeJobPrefs);
			pstmt.setInt(1, gameid);
			pstmt.executeUpdate();

			//Remove all CCCs
			pstmt = connection.prepareStatement(removeCCCs);
			pstmt.setInt(1, gameid);
			pstmt.executeUpdate();

			pstmt.close();
		}
		catch (SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}
