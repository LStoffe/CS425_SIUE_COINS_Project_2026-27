package edu.calpoly.lib.multimedia.big.db;

//BIG v2 imports
import edu.calpoly.its.mas.big.users.company.Company;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//JDK imports
import java.util.Vector;
import java.util.Iterator;

//Database imports
import java.sql.Connection;
import java.sql.Statement;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

//BIG v3 imports
import edu.calpoly.lib.multimedia.big.game.Message;

/**
 * Default implementation of <code>InboxDBHandler</code>.
 *
 * @author Kevin Craig
 *
 */
public class DefaultInboxDBHandler extends DefaultDBHandler implements InboxDBHandler
{
	Log log;


    public DefaultInboxDBHandler(Connection connection)
    {
		super(connection);
		log = LogFactory.getLog(this.getClass());
	}

	public void storeMessage(Message message)
	   throws BIGDatabaseFacadeException
	{
		String insert = "INSERT INTO inbox (companyID, periodID, type, link, sender, message, subject) " +
		    			" VALUES (?, ?, ?, ?, ?, ?, ?) ";

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " adding new message: To: " + message.getTo().getCompanyID() + " From: " + message.getFrom() + " Subject: " + message.getSubject() + " Body: " + message.getBody() );
		}

		try
		{
			PreparedStatement pstatement = connection.prepareStatement(insert);
			pstatement.setInt(1, message.getTo().getCompanyID());
			pstatement.setInt(2, message.getPeriodNumber());
			pstatement.setInt(3, message.getType());
			pstatement.setInt(4, message.getLink());
			pstatement.setString(5, message.getFrom());
			pstatement.setString(6, message.getBody());
			pstatement.setString(7, message.getSubject());

			pstatement.executeUpdate();
			connection.commit();

			pstatement.close();
		}
		catch (SQLException sqle)
		{
			close();
			if(log.isErrorEnabled())
			{
				log.error("", sqle);
			}
			throw new BIGDatabaseFacadeException(sqle);
		}
	}


    public Vector retrieveMessages(Company company, int period)
       throws BIGDatabaseFacadeException
    {
		Vector messages = new Vector();

		String select = "SELECT type, link, sender, message, subject " +
		  				" FROM inbox WHERE companyid = ? AND periodid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, company.getCompanyID());
			pstmt.setInt(2, period);

			ResultSet rset = pstmt.executeQuery();
			while(rset.next())
			{
				messages.add(new Message(rset.getString("sender"),
										 company,
										 rset.getString("message"),
										 rset.getString("subject"),
										 period,
										 rset.getInt("type"),
										 rset.getInt("link")
										)
							);
			}

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return messages;
	}

	public Vector retrieveAllMessages(Company company)
	   throws BIGDatabaseFacadeException
	{
		Vector messages = new Vector();

		String select = "SELECT type, link, sender, message, periodid, subject " +
		  				" FROM inbox WHERE companyid = ? " +
		  				" ORDER BY periodid DESC";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(select, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();
			while(rset.next())
			{
				messages.add(new Message(rset.getString("sender"),
										 company,
										 rset.getString("message"),
										 rset.getString("subject"),
										 rset.getInt("periodid"),
										 rset.getInt("type"),
										 rset.getInt("link")
										)
							);
			}

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		return messages;
	}


    public void deleteMessages(int companyID)
    	throws BIGDatabaseFacadeException
    {
		String delete = "DELETE FROM inbox WHERE companyid = ? ";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(delete);
			pstmt.setInt(1, companyID);

			pstmt.executeUpdate();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
	}

}
