    public int[] getScheduleMethods(Game game, Job job) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method getScheduleMethods(Game game, Job job)");
		}

		int[] methods;
		Vector v = new Vector();
		try
		{
			String query = " SELECT methodid FROM schedule_estimated_methods WHERE gameid = ? AND jobid = ? ORDER BY activityid";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, job.getJobID());

			ResultSet rset = pstmt.executeQuery();
			while(rset.next())
			{
				v.add(new Integer(rset.getInt("methodid")));
			}

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		methods = new int[v.size()];
		for(int i = 0; i < v.size(); i++)
		{
			methods[i] = ((Integer) v.elementAt(i)).intValue();
		}

		if(log.isDebugEnabled())
		{
			log.debug("Returning from method getScheduleMethods(Game game, Job job)");
		}

		return methods;
	}
