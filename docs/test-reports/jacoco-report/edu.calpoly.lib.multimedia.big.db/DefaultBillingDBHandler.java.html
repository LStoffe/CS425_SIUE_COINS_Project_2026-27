    public BillingForm getBillingForm(Game game, int periodNum, Company company, Job job) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		BillingForm billingForm = null;
		int[] activityAmounts =  new int[9];

		try
		{
			String query = "SELECT amount FROM billing_breakdown " +
						   "WHERE gameid = ? AND periodid = ? AND companyid = ? AND jobid = ? " +
						   "ORDER BY activityid";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());
			pstmt.setInt(4, job.getJobID());


			ResultSet rset = pstmt.executeQuery();

			int i = 0;
			while(rset.next())
			{
			   activityAmounts[i] = rset.getInt("amount");
			   i++;
			}
			if (i != 9)
			{
				close();
				throw new ObjectNotFoundException();
			}

			rset.close();
			pstmt.close();

 			query = "SELECT amount_billed, amount_paid, rejected, old_amount_billed FROM billings " +
						   "WHERE gameid = ? AND periodid = ? AND companyid = ? AND jobid = ? ";

 			pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());
			pstmt.setInt(4, job.getJobID());

			rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
				close();
				throw new ObjectNotFoundException();
			}
			else
			{
				rset.first();


				billingForm = new BillingForm(job.getJobID(),
											  activityAmounts,
											  rset.getInt("amount_billed"),
											  rset.getInt("amount_paid"),
											  rset.getBoolean("rejected"),
											  rset.getInt("old_amount_billed")
											 );
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return billingForm;
    }

    public Vector getAllBillingForms(Game game, Company company, int periodNum) throws BIGDatabaseFacadeException
    {
		Vector billingForms = new Vector();
        int[] activityAmounts = new int[9];

		try
		{

 			String query = "SELECT jobid, amount_billed, amount_paid, rejected, old_amount_billed FROM billings " +
						   "WHERE gameid = ? AND periodid = ? AND companyid = ? AND (amount_billed > 0 OR rejected) " +
						   " ORDER BY jobid " ;

 			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());

			ResultSet rset = pstmt.executeQuery();

			while(rset.next())
			{

				billingForms.add(new BillingForm(rset.getInt("jobid"),
											  activityAmounts,
											  rset.getInt("amount_billed"),
											  rset.getInt("amount_paid"),
											  rset.getBoolean("rejected"),
											  rset.getInt("old_amount_billed")
											 ));
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return billingForms;
	}

    public void addBillingForm(Game game, Company company, int periodNum, Job job) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO billings (gameid, companyid, jobid, periodid, amount_billed, amount_paid, rejected, old_amount_billed) " +
						"VALUES(?, ?, ?, ?, ?, ?, ?, ?) ";

		String insertActivityAmounts = "INSERT INTO billing_breakdown (gameid, companyid, periodid, jobid, activityid, amount) " +
						"VALUES(?, ?, ?, ?, ?, ?) ";


		try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, job.getJobID());
			pstmt.setInt(4, periodNum);
			pstmt.setInt(5, 0);
			pstmt.setInt(6, 0);
			pstmt.setBoolean(7, false);
			pstmt.setInt(8, 0);

			int rowsChanged = pstmt.executeUpdate();


			for(int i = 1; i < 10; i++)
			{
 				pstmt = connection.prepareStatement(insertActivityAmounts);
				pstmt.setInt(1, game.getGameID());
				pstmt.setInt(2, company.getCompanyID());
				pstmt.setInt(3, periodNum);
				pstmt.setInt(4, job.getJobID());
				pstmt.setInt(5, i);
				pstmt.setInt(6, 0);

				rowsChanged = pstmt.executeUpdate();
			}

		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

    }

    public void updateBillingForm(Game game, Company company, int periodNum, BillingForm theReport) throws BIGDatabaseFacadeException
    {

		String update = "UPDATE billings SET amount_billed = ?, amount_paid = ?, rejected = ?, old_amount_billed = ? " +
						"WHERE gameid = ? AND companyid = ? AND jobid = ? AND periodid = ?";

		String updateActivityAmounts = "UPDATE billing_breakdown SET amount = ? " +
						"WHERE gameid = ? AND companyid = ? AND jobid = ? AND periodid = ? AND activityid = ?";


		try
		{
			PreparedStatement pstmt = connection.prepareStatement(update);
			pstmt.setInt(1, theReport.getBillingAmount());
			pstmt.setInt(2, theReport.getAmountPaid());
			pstmt.setBoolean(3, theReport.rejected());
			pstmt.setInt(4, theReport.getOldAmountBilled());
			pstmt.setInt(5, game.getGameID());
			pstmt.setInt(6, company.getCompanyID());
			pstmt.setInt(7, theReport.getJobID());
			pstmt.setInt(8, periodNum);

			int rowsChanged = pstmt.executeUpdate();


			for(int i = 1; i < 10; i++)
			{
 				pstmt = connection.prepareStatement(updateActivityAmounts);
				pstmt.setInt(1, theReport.getActivityAmounts()[i-1]);
				pstmt.setInt(2, game.getGameID());
				pstmt.setInt(3, company.getCompanyID());
				pstmt.setInt(4, theReport.getJobID());
				pstmt.setInt(5, periodNum);
				pstmt.setInt(6, i);

				rowsChanged = pstmt.executeUpdate();
			}

		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

    }


    public void removeBillingForms(int gameID, int companyID) throws BIGDatabaseFacadeException
    {
		String remove = "DELETE FROM billings WHERE gameid = ? AND companyid = ?";
		String remove2 = "DELETE FROM billing_breakdown WHERE gameid = ? AND companyid = ?";

		try
		{
			PreparedStatement pstmt = connection.prepareStatement(remove);
			pstmt.setInt(1, gameID);
			pstmt.setInt(2, companyID);

			int rowsChanged = pstmt.executeUpdate();

			pstmt = connection.prepareStatement(remove2);
			pstmt.setInt(1, gameID);
			pstmt.setInt(2, companyID);

			rowsChanged = pstmt.executeUpdate();

			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}


    }
