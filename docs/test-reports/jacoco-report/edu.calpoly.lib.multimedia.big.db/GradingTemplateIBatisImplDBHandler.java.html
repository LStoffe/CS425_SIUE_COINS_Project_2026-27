package edu.calpoly.lib.multimedia.big.db;

//JDK imports
import java.util.List;
import java.util.ArrayList;
import java.util.Map;
import java.util.HashMap;

//JDBC imports
import java.sql.SQLException;

//BIG v2 imports
import edu.calpoly.its.mas.big.users.admin.Admin;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.db.DefaultDBHandler;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//iBATIS imports
import com.ibatis.sqlmap.client.SqlMapClient;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;



/**
 * Default implementation of <code>GradingTemplateDBHandler</code>.
 *
 * @author Kevin Craig
 *
 */
public class GradingTemplateIBatisImplDBHandler implements GradingTemplateDBHandler
{
	private SqlMapClient sqlMap;
	private Log log;

    public GradingTemplateIBatisImplDBHandler(SqlMapClient sqlMap)
    {
		this.sqlMap = sqlMap;
		this.log = LogFactory.getLog(this.getClass());
	}

	public void createGradingTemplate(Admin admin, int gameid) throws BIGDatabaseFacadeException
	{
		Map map = new HashMap(6);
		map.put("adminid", admin.getAdminID());
		map.put("gameid", gameid);
		map.put("template",  "COMPANY_EQUITY");
		map.put("template1", "COMPANY_BIDS");
		map.put("template2", "COMPANY_DISMISSED_BIDS");
		map.put("template3", "COMPANY_JOBS");

		try
		{
			sqlMap.startTransaction();
			sqlMap.insert("createGradingTemplate", map);
			sqlMap.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}
	}

	public HashMap getGradingTemplate(Game game) throws BIGDatabaseFacadeException
	{
        HashMap map = new HashMap();
		try
		{
			sqlMap.startTransaction();
			map = (HashMap) sqlMap.queryForObject("getGradingTemplate", game.getGameID());
			sqlMap.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}

		return map;
	}

	public void updateGradingTemplate(Admin admin, Game game, Map map) throws BIGDatabaseFacadeException
	{
		try
		{
			sqlMap.startTransaction();
			sqlMap.update("updateGradingTemplate", map);
			sqlMap.commitTransaction();
		}
		catch(SQLException sqle)
		{
			throw new BIGDatabaseFacadeException(sqle);
		}
		finally
		{
			try
			{
				sqlMap.endTransaction();
			}
			catch(SQLException sqle)
			{
				throw new BIGDatabaseFacadeException(sqle);
			}
		}
	}
}
