    public BalanceSheetForm getBalanceSheet(Game game, int periodNum, Company company) throws ObjectNotFoundException, BIGDatabaseFacadeException
    {
		BalanceSheetForm balanceSheetForm = null;

		try
		{
			String query = "SELECT cash_on_hand, current_accounts_receivable, retention_receivable, " +
			               " construction_equipment, autos, office_equipment, acc_depreciation, " +
						   " land, current_accounts_payable, retention_payable, income_taxes_payable, " +
						   " loan_balances, equity "+
						   " FROM balance_sheet WHERE gameid = ? AND periodid = ? AND companyid = ? " ;

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);

			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, periodNum);
			pstmt.setInt(3, company.getCompanyID());


			ResultSet rset = pstmt.executeQuery();

			// Check if no rows are returned
			// (rset.last() returns false if there are no rows in the result set)
			if(!rset.last())
			{
				close();
				throw new ObjectNotFoundException();
			}
			else
			{
				rset.first();


				balanceSheetForm = new BalanceSheetForm(company.getTitle(),
														rset.getLong("cash_on_hand"),
														rset.getInt("current_accounts_receivable"),
														rset.getInt("retention_receivable"),
														rset.getInt("construction_equipment"),
														rset.getInt("autos"),
														rset.getInt("office_equipment"),
														rset.getInt("acc_depreciation"),
														rset.getInt("land"),
														rset.getInt("current_accounts_payable"),
														rset.getInt("retention_payable"),
														rset.getInt("income_taxes_payable"),
														rset.getInt("loan_balances"),
														rset.getInt("equity")
													   );
			}
			//done with the result set
			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			//close the connection if an exception occurs
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}
		return balanceSheetForm;
    }



    public void addBalanceSheet(Game game, Company company, BalanceSheetForm theReport) throws BIGDatabaseFacadeException
    {
		String insert = "INSERT INTO balance_sheet (gameid, companyid, periodid, cash_on_hand, current_accounts_receivable, " +
		                                           "retention_receivable, construction_equipment, autos, office_equipment, " +
		                                           "acc_depreciation, land, current_accounts_payable, retention_payable, " +
		                                           "income_taxes_payable, loan_balances, equity) " +
		                                           "VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)";


		try
		{
			PreparedStatement pstmt = connection.prepareStatement(insert);
			pstmt.setInt(1, game.getGameID());
			pstmt.setInt(2, company.getCompanyID());
			pstmt.setInt(3, theReport.getPeriod());
			pstmt.setLong(4, theReport.getCashOnHand());
			pstmt.setInt(5, theReport.getCurrentAccountsReceivable());
			pstmt.setInt(6, theReport.getRetentionReceivable());
			pstmt.setInt(7, theReport.getConstructionEquipment());
			pstmt.setInt(8, theReport.getAutos());
			pstmt.setInt(9, theReport.getOfficeEquipment());
			pstmt.setInt(10, theReport.getAccumulatedDepreciation());
			pstmt.setInt(11, theReport.getLand());
			pstmt.setInt(12, theReport.getCurrentAccountsPayable());
			pstmt.setInt(13, theReport.getRetentionPayable());
			pstmt.setInt(14, theReport.getTaxes());
			pstmt.setLong(15, theReport.getLoanBalances());
