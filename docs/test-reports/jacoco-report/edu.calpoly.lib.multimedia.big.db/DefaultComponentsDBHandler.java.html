    public boolean isEstimatingComponentOn(Game game) throws BIGDatabaseFacadeException
    {
		if(log.isDebugEnabled())
		{
			log.debug("Starting method isEstimatingComponentOn(Game game)");
		}

		boolean on = false;
		try
		{
			String query = " SELECT estimating FROM component WHERE gameid = ?";

			PreparedStatement pstmt = connection.prepareStatement(query, ResultSet.TYPE_SCROLL_INSENSITIVE, ResultSet.CONCUR_READ_ONLY);
			pstmt.setInt(1, game.getGameID());

			ResultSet rset = pstmt.executeQuery();
			if(rset.next())
			{
				int onInt = rset.getInt("estimating");
				if(onInt > 0)
				on = true;
			}

			rset.close();
			pstmt.close();
		}
		catch(SQLException sqle)
		{
			close();
			throw new BIGDatabaseFacadeException(sqle);
		}

		if(log.isDebugEnabled())
		{
			log.debug("Estimating component is on: " + on);
		}

		if(log.isDebugEnabled())
		{
			log.debug("Returning from method isEstimatingComponentOn(Game game)");
		}

		return on;
	}
