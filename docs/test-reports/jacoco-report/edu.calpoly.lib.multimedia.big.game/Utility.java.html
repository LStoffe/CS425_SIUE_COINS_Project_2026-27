package edu.calpoly.lib.multimedia.big.game;


/**
 * This class provides some basic functions needed by multiple
 * other classes in BIG.
 */
public class Utility
{

	public static synchronized double average(double[] values)
	{
		double sum = 0.0;
		double length = (double) values.length;
		for(int i=0; i < values.length; i++)
		{
			sum += values[i];
		}
		return sum / length;
	}

	public static synchronized long average(long[] values)
	{
		double sum = 0.0f;
		double length = (float) values.length;
		for(int i=0; i < values.length; i++)
		{
			sum += values[i];
		}
		return Math.round(sum / length);
	}

    public static synchronized double stdDev(double[] values)
    {
		double average = average(values);
		double sum = 0.0;
		double length = (double) values.length;
		for(int i=0; i < values.length; i++)
		{
			sum += (values[i] - average) * (values[i] - average);
		}
		double variance = sum / length;
		return Math.sqrt(variance);
	}

	public static synchronized double stdDev(long[] values)
	{
		double average = (double) average(values);
		double sum = 0.0;
		double length = (double) values.length;
		for(int i=0; i < values.length; i++)
		{
			sum += (((double) values[i]) - average) * (((double) values[i]) - average);
		}
		double variance = sum / length;
		return Math.sqrt(variance);
	}

	public static void main(String args[])
	{
		double[] test = {2.0, 2.0, 2.0, 2.0, 2.0};
		System.out.println(Utility.stdDev(test));
	}
}
