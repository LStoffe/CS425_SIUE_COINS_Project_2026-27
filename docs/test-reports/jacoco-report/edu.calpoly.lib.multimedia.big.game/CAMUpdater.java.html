	/**
	 *  Computes the value of the financial liquidity metric for this company
	 *  based on the number of jobs held by this company, the cash on hand for this company, and the current ratio, cash turns ratio,
	 *  and cash to wip ratio for this company.  If the company's cash on hand is negative this method will always return the lowest possible value
	 *  (<code>-CAMReportForm.rawMetricRange</code>).
	 *  If the company hold no jobs
	 *  the FinancialLiquidity CAM will be neutral so long as the company's cash on hand is positive.
	 *  Otherwise a contribution will be calculated for each ratio that has a value.
	 *  This contribution will be based on where the value of the ratio falls into
	 *  a predetermined hardcoded range for each ratio.  (These ranges are based on the probable values of the ratios
	 *  unlikely to ever change.)  If a company has more than 3 jobs a bonus is given because it is more
	 *  difficult to have liquid assets if the company has many jobs.
	 *  If the CashToWIP ratio is negative it means the company has
	 */
	private int computeFinancialLiquidityValue(Float currentRatio, Float cashTurns, Float cashToWIP, Company company)
	{
		//First see if the company's cash on hand is negative.
		//If so return immediately
		if(company.getCashOnHand() < 1)
		{
			return (- CAMReportForm.rawMetricRange);
		}

		int numberOfJobs = 0;

		//Find the number of jobs held by this company by going to the database.
		try
		{
			Vector jobs = dbFacade.getJobDBHandler().getActiveJobs(game, company);
			numberOfJobs = jobs.size();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}

		//If the number of jobs is 0 then we can return 0 immediately
		if(numberOfJobs == 0)
		{
			return 0;
		}


		int currentRatioContribution = 3;
		int cashTurnsContribution = 3;
		int cashToWIPContribution = 3;

		if(currentRatio != null)
		{
			if(currentRatio.floatValue() < 1)
			{
				currentRatioContribution = 1;
			}
			else if(currentRatio.floatValue() < 3)
			{
				currentRatioContribution = 2;
			}
			else if (currentRatio.floatValue() < 6)
			{
				currentRatioContribution = 3;
			}
			else if (currentRatio.floatValue() < 10)
			{
				currentRatioContribution = 4;
			}
			else
			{
				currentRatioContribution = 5;
			}
		}

		if(cashTurns != null)
		{
			if(cashTurns.floatValue() < .01)
			{
				cashTurnsContribution = 1;
			}
			else if(cashTurns.floatValue() < .1)
			{
				cashTurnsContribution = 2;
			}
			else if (cashTurns.floatValue() < .25)
			{
				cashTurnsContribution = 3;
			}
			else if (cashTurns.floatValue() < .5)
			{
				cashTurnsContribution = 4;
			}
			else
			{
				cashTurnsContribution = 5;
			}
		}

		if(cashToWIP != null)
		{
			//

			if(cashToWIP.floatValue() < .02)
			{
				cashToWIPContribution = 1;
			}
			else if(cashToWIP.floatValue() < .05)
			{
				cashToWIPContribution = 2;
			}
			else if (cashToWIP.floatValue() < .1)
			{
				cashToWIPContribution = 3;
			}
			else if (cashToWIP.floatValue() < .15)
			{
				cashToWIPContribution = 4;
			}
			else
			{
				cashToWIPContribution = 5;
			}
		}


/*		if(log.isDebugEnabled())
		{
			log.debug("computing FinancialLiquidity");
			log.debug("currentRatioContribution " + currentRatioContribution);
			log.debug("cashTurnsContribution " + cashTurnsContribution);
			log.debug("cashToWIPContribution " + cashToWIPContribution);
		} */

		int sum = currentRatioContribution + cashTurnsContribution + cashToWIPContribution;
		float average = (float) sum / 3.0f;

/*		if(log.isDebugEnabled())
		{
			log.debug("average " + average );
		} */

		int returnValue = Math.round(average * CAMReportForm.formula) - CAMReportForm.rawMetricRange;

/*		if(log.isDebugEnabled())
		{
			log.debug("intial return value " + returnValue );
		} */

		if(numberOfJobs >= 3)
		{
			returnValue += 30;
/*			if(log.isDebugEnabled())
			{
				log.debug("3 job bonus");
			} */

		}
		if(returnValue > CAMReportForm.rawMetricRange)
		{
			returnValue = CAMReportForm.rawMetricRange;
		}

/*		if(log.isDebugEnabled())
		{
			log.debug("final return value " + returnValue );
		} */

		return returnValue;
	}

	/**
	 *  Computes the value for a Financial Success metric based on the values of
	 *  the Return on Equity and Return on Assets ratios.
	 *  Each ratios is converted into a contribution for the value of the Financial Success metric.
	 *  This contribution will be based on where the value of the ratio falls into
	 *  a predetermined, hardcoded range for each ratio.  (These ranges are based on the probable values of the ratios
	 *  unlikely to ever change.)
	 */
	private int computeFinancialSuccessValue(Float returnOnEquity, Float returnOnAssets)
	{
		int returnOnEquityContribution = 3;
		int returnOnAssetsContribution = 3;

		if(returnOnEquity != null)
		{
			if(returnOnEquity.floatValue() < -.05f)
			{
				returnOnEquityContribution = 1;
			}
			else if(returnOnEquity.floatValue() < 0.0f)
			{
				returnOnEquityContribution = 2;
			}
			else if(returnOnEquity.floatValue() < .05f)
			{
				returnOnEquityContribution = 3;
			}
			else if(returnOnEquity.floatValue() < .12f)
			{
				returnOnEquityContribution = 4;
			}
			else
			{
				returnOnEquityContribution = 5;
			}
		}

		if(returnOnAssets != null)
		{
			if(returnOnAssets.floatValue() < -.05f)
			{
				returnOnAssetsContribution = 1;
			}
			else if(returnOnAssets.floatValue() < 0.0f)
			{
				returnOnAssetsContribution = 2;
			}
			else if(returnOnAssets.floatValue() < .05f)
			{
				returnOnAssetsContribution = 3;
			}
			else if(returnOnAssets.floatValue() < .12f)
			{
				returnOnAssetsContribution = 4;
			}
			else
			{
				returnOnAssetsContribution = 5;
			}
		}

/*		if(log.isDebugEnabled())
		{
			log.debug("computing financial success");
			log.debug("returnOnEquityContribution " + returnOnEquityContribution);
			log.debug("returnOnAssetsContribution " + returnOnAssetsContribution);
		} */

		int sum = returnOnEquityContribution + returnOnAssetsContribution;
		float average = (float) sum / 2.0f;

/*		if(log.isDebugEnabled())
		{
			log.debug("average " + average );
		} */

		int returnValue = Math.round(average * CAMReportForm.formula) - CAMReportForm.rawMetricRange;

/*		if(log.isDebugEnabled())
		{
			log.debug("return value " + returnValue );
		} */

		return returnValue;
	}
