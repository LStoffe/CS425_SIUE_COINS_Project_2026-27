package edu.calpoly.lib.multimedia.big.game;

import static edu.calpoly.lib.multimedia.big.game.NamedVariables.*;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * This class is a dummy implemention of the <code>NamedVariableDataFacade</code> interface
 * that returns defined values for named variables (as opposed to actually looking up the
 * correct values from a database).  The variable names defined in <code>edu.calpoly.lib.multimedia.big.game.NamedVariables</code>
 * still should be provided to the methods, because the subtype of the <code>Object</code>
 * returned from a method may depend on what variable was requested.  Client code using a
 * <code>NamedVariableDataFacade</code> may depend on these objects being castable to a certain type.
 * The default return values are 1 for <code>Integer</code>, <code>Long</code>, and <code>Short</code> and 1.0 for <code>Float</code>
 * and <code>Long</code>
 *
 */
public class NamedVariableDataFacadeTestImpl implements NamedVariableDataFacade
{
	private Log log;

	private short shortValue = 1;
	private int intValue = 1;
	private long longValue = 1;
	private float floatValue = 1.0f;
	private double doubleValue = 1.0d;

	public NamedVariableDataFacadeTestImpl()
	{
		log = LogFactory.getLog(this.getClass());
	}

	/**
	 * Use the given values as the return values for all returned objects of that type.
	 */
	public NamedVariableDataFacadeTestImpl(short shortVal, int intVal, long longVal, float floatVal, double doubleVal)
	{
		log = LogFactory.getLog(this.getClass());
		this.shortValue = shortVal;
		this.intValue = intVal;
		this.longValue = longVal;
		this.floatValue = floatVal;
		this.doubleValue = doubleVal;
	}

	/**
	 * Returns defined test value of a named variable.
	 * Variable that are valid inputs to this method are: TOTAL_JOBS
	 */
	public Object getNamedVariableValue(String variable)
	{
		if(log.isDebugEnabled())
		{
			log.debug("Looking up named variable " + variable);
		}

		if(variable.equals(TOTAL_JOBS))
		{
			return new Integer(intValue);
		}

		return null;
	}

	/**
	 * Returns defined test value of the named variable that would be associated
	 * with one particular company were it a real value.  The companyid parameter is ignored.
	 * Variable that are valid inputs to this method are: (list)....
	 */
	public Object getCompanyNamedVariableValue(String variable, int companyid)
	{
		if(log.isDebugEnabled())
		{
			log.debug("Looking up named variable " + variable);
		}

		if(variable.equals(COMPANY_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_JOBS_IN_PROGRESS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_JOBS_COMPLETED))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_PUBLIC_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_PRIVATE_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_HEAVY_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_CLOSED_BID_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_NEGOTIATED_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_AUTO_JOBS))
		{
			return new Integer(intValue);
		}
		else if(variable.equals(COMPANY_EQUITY))
		{
			return new Long(longValue);
		}
		else if(variable.equals(TOTAL_GROSS_MARGIN_JOBS_IN_PROGRESS))
		{
			return new Long(longValue);
		}
		else if(variable.equals(TOTAL_GROSS_MARGIN_JOBS_COMPLETED))
		{
			return new Long(longValue);
		}
		else if(variable.equals(COMPANY_BIDS))
		{
			return new Long(longValue);
		}
		else if(variable.equals(COMPANY_DISMISSED_BIDS))
		{
			return new Long(longValue);
		}
		else if(variable.equals(SUM_SIX_CAM_VALUES))
		{
			return new Integer(intValue);
		}

		return null;
	}


	public Object getJobNamedVariableValue(String variable, int jobid)
	{
		return null;
	}
}
