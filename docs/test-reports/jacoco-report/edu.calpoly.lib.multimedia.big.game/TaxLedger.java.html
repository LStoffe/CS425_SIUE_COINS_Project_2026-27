package edu.calpoly.lib.multimedia.big.game;

import edu.calpoly.lib.multimedia.big.reports.IncomeStatementForm;

/**
 * Computes the amount of income tax a company must pay on all income in the current year to date.
 *
 * @author Kevin Craig
 *
 */
public class TaxLedger
{

  private float taxRate;
  private IncomeStatementForm[] yearsStatements;


  /**
   * Constructor
   * @param taxRate The percentage of earnings that will be taken as income tax
   * @param yearsStatements A array of IncomeStatementForms where the earnings
   * on each form will be considered by this TaxLedger when computing income tax.
   */
  public TaxLedger(float taxRate, IncomeStatementForm[] yearsStatements)
  {
      this.taxRate = taxRate;
      this.yearsStatements = yearsStatements;
  }

  public int getTaxes()
  {
	  int yearsPreTaxEarningsToDate = 0;

	  for(int i = 0; i < yearsStatements.length; i++)
	  {
		  yearsPreTaxEarningsToDate += yearsStatements[i].getPreTaxEarnings();
		  //System.out.println("Revenue: " + yearsStatements[i].getConstructionRevenue());
		  //System.out.println("Current earnings: " + yearsStatements[i].getPreTaxEarnings() + " total earnings: " + yearsPreTaxEarningsToDate);

	  }

	  //Companies have to earn at least $10 to pay taxes.  This is just to avoid
	  //weird rounding
	  if( yearsPreTaxEarningsToDate < 10)
	  {
		  return 0;
	  }
	  else
	  {
		  return Math.round(taxRate * yearsPreTaxEarningsToDate);
	  }


  }

}
