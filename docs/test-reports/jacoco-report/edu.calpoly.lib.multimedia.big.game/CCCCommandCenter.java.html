package edu.calpoly.lib.multimedia.big.game;

/**
 * Creates, destroys, and handles tasks related to CCCs.
 * This class is responsible for maintaining the correct number of
 * CCC's in BIG each period.  The following takes place during
 * period updates:<br><br>
 * First, an optimal number of CCCs for the next period is chosen
 * randomly from the the range specified by game parameters.
 * <br><br>
 * Next the number of CCCs to be destroyed is calculated.  The number
 * of CCCs to be destroyed is a percentage of the total number of
 * CCCs equal to the value for CCC Turnover game parameters.  (e.g. if the
 * CCC Turnover value is 25, then 25% of the CCCs will be destroyed).
 * <br><br>
 * Next the individual CCCs that will be destroyed are selected based.  On the internal
 * fitness value of the CCC.  Fitness values range from [0,99].  A temporary
 * value called the inverse fitness is calculated by subtracting a CCC's fitness value
 * from 100.  A roulette wheel selection is used where each company has a slice
 * proportionate to its inverse fitness.  The wheel is repeatedly spun until the required number
 * of CCCs is destroyed.  If the wheel lands on the slice of a CCC that has previously been
 * destroyed this update, then the wheel is spun again.
 * <br><br>
 * After the required number of CCCs are gone, new CCCs are added to bring the total number
 * of CCC's up to the optimal number established earlier.  If there are already more
 * CCCs than the optimal number then no CCCs are added.
 * <br><br>
 * When a new CCC is created, it is given a name and randomly assigned a fitness value,
 * and preferences for each job type.
 * <br><br>
 * This class also contains the method to place CCC bids on jobs, <code>placeCCCBids</code>.
 *
 * @author Kevin Craig
 *
 */
public class CCCCommandCenter
{
	private int minNumberOfCCCs;
	private int maxNumberOfCCCs;
	private int minCCCBidsPerJob;
	private int maxCCCBidsPerJob;
	private int cccTurnover;
	private int lowerProfitPercent;
	private int upperProfitPercent;
	private int fieldOverheadFixed;
	private int fieldOverheadVariable;
	private int officeOverheadFixed;
	private int bidExpensePercent;
	private int workdaysPerMonth;
	private int jobTypes;
	private Random random;

    /**

     * @param maxCCCBidsPerJob Maximum number of bids by CCCs that may be on a job

     * @param lowerProfitPercent Minimum percentage of costs a CCC will add to costs as profit
     *		when computing a bid price
     * @param upperProfitPercent Maximum percentage of costs a CCC will add to costs as profit
     		when computing a bid price
     * @param jobTypes Total number of job types in the game.  A preference will be created
     *		for this many job types.

	public CCCCommandCenter(int minNumberOfCCCs,

		this.maxCCCBidsPerJob = maxCCCBidsPerJob;
		this.cccTurnover = cccTurnover;

	}

    /**
     * Returns a newly created <code>ComputerControlledContractor</code> object.
     */
	public ComputerControlledContractor createCCC()
	{
		int fitness = random.nextInt(100);
		int[] jobTypePreferences = new int[jobTypes];

		int bestType = random.nextInt(jobTypes);
		int secondBest = random.nextInt(jobTypes);


		for(int i = 0; i < jobTypes; i++)
		{
			jobTypePreferences[i] = 25;
		}

		jobTypePreferences[secondBest] = 50;
		jobTypePreferences[bestType] = 70;

		if(secondBest == bestType)
		   jobTypePreferences[bestType] = 90;

		return new ComputerControlledContractor(NameGenerator.generateComputerControlledContractorName(), fitness, jobTypePreferences);
	}

	/**
	 * Performs the updates described in the class overview.
	 * @param CCCs <code>Vector</code> of <code>ComputerControlledContractor</code>
	 *    that represent the CCCs currently in a game.
	 * @return <code>Vector</code> The input list of CCCs updated as described above.
	 */
	public Vector updateCCCs(Vector CCCs)
	{
		int optimalNumberOfCCCs = random.nextInt(maxNumberOfCCCs-minNumberOfCCCs+1) + minNumberOfCCCs;
		//System.out.println("OPT #" + optimalNumberOfCCCs);

		//Calculate the number of CCCs to be deleted
		int cccsToGo = Math.round(((float) CCCs.size()) * (((float) cccTurnover)/100f));

		//System.out.println("TO GO #" + cccsToGo);

		int totalInverseFitness = 0;

		//The wheel is an array the holds the bounds of the slice
		//for each CCC.  This array forms a parallel
		//array to the input Vector, CCCs.
        //The value at each index in rouletteWheel is the upper bound
        //of the slice belonging to the CCC in CCCs at matching array index.
        //The lower bound is 0 for the CCC at array index 0, or 1 greater than the value
        //at index - 1 for all other indices.
        //Example: if the value at index 3 is 231, and the value at index 2 is
        //189, that means the slice for the company at index 3 in CCCs goes
        //from 190-231
		int[] rouletteWheel = new int[CCCs.size()];

		//Populate the roulette wheel
		for(int i = 0; i < rouletteWheel.length; i++)
		{
			ComputerControlledContractor ccc = (ComputerControlledContractor) CCCs.elementAt(i);

			totalInverseFitness += (100 - ccc.getFitness());
			rouletteWheel[i] = totalInverseFitness;
		}

		//Array to mark off deleted CCCs
		Object[] scratchList = CCCs.toArray();
		while(cccsToGo > 0)
		{
			int spin = random.nextInt(totalInverseFitness+1);

			//Determine which slice this spin falls in
			int slice = -1;
			int j = 0;
			while((j < rouletteWheel.length) && (slice < 0))
			{
				if(spin <= rouletteWheel[j])
				{
					slice = j;
				}
				j++;
			}

			//Check if the ccc who owns the slice is still
			//there (it could have already been destroyed this turn)
			if(scratchList[slice] != null)
			{
				ComputerControlledContractor ccc = (ComputerControlledContractor) scratchList[slice];
				//System.out.println("Removing: " + ccc.getName() + "   " + ccc.getFitness());
				CCCs.remove(scratchList[slice]);
				scratchList[slice] = null;
				cccsToGo--;
			}
		}

        //Add CCCs until the optimum is reached
		while(CCCs.size() < optimalNumberOfCCCs)
		{
			ComputerControlledContractor ccc = createCCC();
			//System.out.println("Adding: " + ccc.getName() + "   " + ccc.getFitness());
			CCCs.add(ccc);
		}


		return CCCs;
	}

	/**
	 * Generates CCC bids on a job.
	 * Runs through the list of CCCs and chooses some (or all)
	 * to create bids on the job based on the limits on
	 * number of CCC bids per job and internal job type preferences
	 * of the CCCs.
	 * @param CCCs <code>Vector</code> of CCCs currently in the game.  <b>PRE:</b> This <code>Vector</code>
	 *      cannot be empty, it must contain at least as many <code>ComputerControlledContractor</code> objects
	 *		as the minimum number of CCC bids per job.  It must not be empty even if this value is zero.
	 * @param Job The job for which to generate bids.
	 * @return <code>Vector</code> of <code>Bid</code> that represents
	 * the CCC bids on the job.
	 */
    public Vector placeCCCBids(Vector CCCs, Job job)
    {
		//The min and max number of CCC bids per job is halved for private jobs
		int minCCCBids = minCCCBidsPerJob;
		int maxCCCBids = maxCCCBidsPerJob;

		if(job.getTypeModifier() == Job.PRIVATE)
		{
			minCCCBids = minCCCBids / 2;
			if(minCCCBids < 1)
			   minCCCBids = 1;

			maxCCCBids = maxCCCBids / 2;
		}

		Vector bids = new Vector();

		//Index to start examining CCC to give a chance to bid.
		//This ensures that each CCC gets a fair chance to bid overall.
		//If we started at the top of the list each time, the companies at
		//the bottom of the list would have less of a chance due to the max number
		//of ccc bids per job
		int startingPoint = random.nextInt(CCCs.size());
		int i = 0;
		int j = startingPoint;
		int bidsMade = 0;

		//Used to mark of CCCs that have already made a bid to ensure
		//that no one gets to bid twice
		boolean [] scratch = new boolean[CCCs.size()];

		do
		{
			if(!scratch[j])
			{
				ComputerControlledContractor ccc = (ComputerControlledContractor) CCCs.elementAt(j);

				//The chance this CCC will bid is based on its level of preference
				//of this job type
				if(ccc.getJobTypePreferences()[job.getJobType()-1] > random.nextInt(100))
				{
					//This CCC will bid
					scratch[j] = true;
					bidsMade++;
					bids.add(generateBid(job.getGame().getGame().getGameID(),
										 ccc.getCCCID(),
										 job.getJobID(),
										 job.getDirectCost(),
										 job.getEstimatedWorkdays()
										)
							);
				}

			}


			j++;
			if(j == CCCs.size())
			{
				j = 0;
			}
			i++;
		}
		while( (bidsMade < minCCCBids) ||
			   ( (bidsMade < maxCCCBids) && (i < CCCs.size()) && (j != startingPoint))
			 );

		return bids;
	}

	/**
	 *
	 * @param gameID ID of the game to which the eventual <code>Bid</code>
	 *        will belong.
	 * @param jobNumber identifies the job on which the eventual
	 *        <code>Bid</code> will be placed.
	 * @param directCost direct cost of the job on which the eventual
	 *        <code>Bid</code> will be placed.
	 * @param extimatedWorkdays the number of estimated workdays for the
	 *        job on which the eventual <code>Bid</code> will be placed
	 * @returns <code>Bid</code> object representing CompanyX's bid on
	 *          the <code>Job</code> that invoked this method.
	 */
	public Bid generateBid(int gameID, int cccid, int jobNumber, long directCost,
						   int estimatedWorkdays)
	{
		long bidAmount = directCost;
		int numberOfPeriods = calculateNumberOfPeriods(estimatedWorkdays,
													   workdaysPerMonth);

		// For now, assume bidExpensePercent is in tenths of percents
		long bidExpense = (directCost * bidExpensePercent) / 1000;

		// Calculate total field overhead
		long fieldOverhead = (fieldOverheadFixed * estimatedWorkdays) +
			(((fieldOverheadVariable * directCost) / 100) * numberOfPeriods);

		// Calculate total office overhead
		long officeOverhead = (officeOverheadFixed * numberOfPeriods);

		bidAmount += bidExpense + fieldOverhead + officeOverhead;

		//System.out.println(bidAmount);
		//bidAmount = Math.round((((float) (100 - (discount/10))) / 100f) * ((float) bidAmount));
		//System.out.println(bidAmount + "\n\n");

		// Determine random profit markup
		int percentMarkup = (lowerProfitPercent * 100) +
			random.nextInt((upperProfitPercent * 100) -
								   (lowerProfitPercent + 1) * 100);

								   //The times 100s are to make the perfect profit have two decimal
								   //places.  like 7.34% or 3.45%
								   //The upper and lower limits are integers but by mulitiplying
								   //by 100 we can still get 2 decimal place accuracy

		// Add random profit markup
		bidAmount += (bidAmount * percentMarkup) / 10000; //100 * 100, to undo the *100s above and to
														  //go from % to decimal

		int companyID = - cccid;  //The companyID of a CCC is the negation of its cccid

        double[] unitPrices = new double[9];
        double theBidPrice = 0;

		// gameID and jobNumber aren't required by Job.determinrBidWinner(),
		// they are included for completeness and consistency
		return (new Bid(gameID, companyID, jobNumber, bidAmount, theBidPrice, unitPrices, 0));

	}

	/**
	 *
	 * @param gameID ID of the game to which the eventual <code>Bid</code>
	 *        will belong.
	 * @param jobNumber identifies the job on which the eventual
	 *        <code>Bid</code> will be placed.
	 * @param directCost direct cost of the job on which the eventual
	 *        <code>Bid</code> will be placed.
	 * @param extimatedWorkdays the number of estimated workdays for the
	 *        job on which the eventual <code>Bid</code> will be placed
	 * @returns <code>Bid</code> object representing CompanyX's bid on
	 *          the <code>Job</code> that invoked this method.
	 */
	public long estimateDirectJobCost(int gameID, int jobNumber, long directCost,
						   int estimatedWorkdays)
	{
		long bidAmount = directCost;
		int numberOfPeriods = calculateNumberOfPeriods(estimatedWorkdays,
													   workdaysPerMonth);


		// Calculate total field overhead
		long fieldOverhead = (fieldOverheadFixed * estimatedWorkdays) +
			(((fieldOverheadVariable * directCost) / 100) * numberOfPeriods);

		bidAmount += + fieldOverhead ;

		// gameID and jobNumber aren't required by Job.determinrBidWinner(),
		// they are included for completeness and consistency
		return bidAmount;

	}

	/**
	 * Given an estimated number of workdays and the number of
	 * workdays per month, determines the associated number of
	 * whole periods.
	 *
	 * @param extimatedWorkdays the number of estimated workdays for the
	 *        job on which the eventual <code>Bid</code> will be placed
	 * @param workdaysPerMonth the number of workdays per month (a game
     *        parameter.)
	 * @returns an int which represents the number of whole periods.
	 */
	private int calculateNumberOfPeriods(int estimatedWorkdays,
										 int workdaysPerMonth)
	{
		int workdaysPerPeriod = workdaysPerMonth * 2;

		if (estimatedWorkdays <= workdaysPerMonth) {
			return 1;
		} else {

			/* Take into account the fact that work doesn't start
			   until the second month of the first period by subtracting
			   the first month and adding one whole period. */

			return 1 + (int) Math.ceil((float) (estimatedWorkdays - workdaysPerMonth) /
									   (float) workdaysPerPeriod);
		}
	}

		}
	}
