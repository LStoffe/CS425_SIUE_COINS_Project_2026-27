/**
 * Periodically monitors the database and performs period updates,
 * on those games that have updates scheduled, as according to the schedule.
 * <br><br>
 * Exactly how the updates are performed, depends on the update policy for the game.
 * All games are monitored for changes to their policy and changes to their update
 * schedule within the policy.  This is a state-based process and everytime the
 * database is examined, the current state is determined based of the data it contains.
 * <br><br>
 *
 * Each of the 4 update policies is handled in the following manner:
 * <ul>
 *   <li>
 *     Manual - Games with the manual period update policy are ignored by this class.
 *   </li>
 *   <li>
 *     Interval - The next scheduled update time of the game is checked.  If that time has
 *                occurred between the last time the database was checked and now,
 *				  the game is updated and the next scheduled update is set
 *				  according to the interval.  If for whatever reason
 *				  no next scheduled update is found, or the next schuled update is in the past,
 				  the next scheduled update
 *                is set according the interval, as if an update had just occurred.
 *   </li>
 *   <li>
 *     Days - If today is a day with an update scheduled, and the update time has occurred between
 *            the last time the database was checked and now, the game is updated.
 *   </li>
 *   <li>
 *     Dates - The next non-complete scheduled update in time is examined. Update scheduled
 *   		   to be prior to the last time the database was checked are not considered.
 *			   If the scheduled date/time
 *			   has occurred between the last time the database was checked and now,
 * 		       the game is updated, and that scheduled update is marked as complete.
 *   </li>
 * </ul>
 *
 */
public class AutoUpdateThread extends Thread
{
	public static int SECOND = 1000;      //1000 milliseconds = 1 sec
	public static int MINUTE = 60*SECOND; //60 seconds in a minute
	public static int HOUR = 60*MINUTE;   //60 minutes in an hour

	private Log log;
	private BIGDatabaseFacadeInterface dbFacade;
	private int waitTime = 10*MINUTE;  //Default to checking for every 10 minutes

	//Reference to the ServletContext for the BIG application
	private ServletContext bigAppContext;

	//Flag to indicate whether or not this is the first run since this
	//thread has been started.  No day policy updates are done during the first run
	//because it is possible this thread was previously running
	//and already did an update for a given day, then was stopped, then restarted
	//within the waitTime of this thread.  This would cause the update to be done
	//twice.  Not doing day updates on the first run might mean an update will be missed
	//but this is preferable to doing two back to back updates.
	//Date and interval policy have additional safeguards against double updates.
	private boolean firstRun = true;

	public AutoUpdateThread()
	{
		log = LogFactory.getLog(this.getClass());
		dbFacade = new BIGDatabaseFacade();
	}

    /**
     * Creates a new <code>AutoUpdateThread</code>
     * with a specified amount of time to wait between updates.
     * The wait time can be specified in milliseconds or
     * using the time constants defined in this class for convenience.
     */
	public AutoUpdateThread(int waitTime)
	{
		log = LogFactory.getLog(this.getClass());
		dbFacade = new BIGDatabaseFacade();
		this.waitTime = waitTime;
		if(log.isInfoEnabled())
		{
			log.info("Created thread with wait time of " + waitTime + " milliseconds");
		}
	}

    /**
     * Creates a new <code>AutoUpdateThread</code>
     * with a specified amount of time to wait between updates and a ServletContext.
     * The wait time can be specified in milliseconds or
     * using the time constants defined in this class for convenience.
     * When an update is to occur, the
     * <code>ServletContext</code> will be checked for a
     * a <code>LoginBroker</code> under the key <code>loginBroker</code>.
     * If this <code.LoginBroker</code> is found, it will be used to
     * logout companies/disable company login during the update
     * and reenable company login after the update completes.
     */
	public AutoUpdateThread(int waitTime, ServletContext bigAppContext)
	{
		log = LogFactory.getLog(this.getClass());
		dbFacade = new BIGDatabaseFacade();
		this.waitTime = waitTime;
		if(log.isInfoEnabled())
		{
			log.info("Created thread with wait time of " + waitTime + " milliseconds");
		}
		this.bigAppContext = bigAppContext;
	}


	public void run()
	{
		try
		{
			//Keep running updates as scheduled as long as this thread exists
			while(true)
			{
/*  Took this out because the log file was filled with hundreds of lines for this entry
				if(log.isDebugEnabled())
				{
					log.debug("Checking for games with scheduled updates");
				}
*/
				List<Map> autoUpdateGames = dbFacade.getAutomaticUpdatePolicyDBHandler().getGamesWithAutomaticUpdatePolicy();

				for(Map map : autoUpdateGames)
				{
					int gameid = ((Integer) map.get("gameid")).intValue();
					int policy = ((Integer) map.get("policy")).shortValue();

					switch(policy)
					{
						case INTERVAL     : {
												doIntervalUpdateCheck(gameid);
							                    break;
										    }
						case DAYS_OF_WEEK : {
							  					if(!firstRun)
							  					{
													doDaysUpdateCheck(gameid);
												}
							                    break;
										    }
						case DATES		  : {
												doDatesUpdateCheck(gameid);
							                    break;
										    }
						default		      : {
											    log.warn("Game " + gameid + " has unknown policy code " + policy);
											    break;
										    }
					}
				}

				firstRun = false;
				this.sleep(waitTime);
			}
		}
		catch(Exception e)
		{
			log.fatal("" , e);
		}
	}

	private void doIntervalUpdateCheck(int gameid)
	{
		if(log.isDebugEnabled())
		{
			log.debug("Game " + gameid + " is using interval policy.  Checking for scheduled updates");
		}

		try
		{

			IntervalUpdatePolicySettings settings = dbFacade.getAutomaticUpdatePolicyDBHandler().getIntervalPolicySettings(Game.mockGame(gameid));

			Date nextUpdate = settings.getNextScheduledUpdate();
			Date updateTime = settings.getUpdateTime();

			//This will store the update date with the time portions being irrelevant
			Calendar cal = Calendar.getInstance();

			//This will store the update time with the date portions being irrelevant
			Calendar timeCal = Calendar.getInstance();

			cal.setTime(nextUpdate);
			timeCal.setTime(updateTime);

			//One set the correct time into the date calendar so we have a calendar
			//with the correct date and time
			cal.set(Calendar.HOUR_OF_DAY, timeCal.get(Calendar.HOUR_OF_DAY));
			cal.set(Calendar.MINUTE, timeCal.get(Calendar.MINUTE));
			cal.set(Calendar.SECOND, timeCal.get(Calendar.SECOND));

			//If the next scheduled date/time has passed since the last update check
			//do an update.  Otherwise do nothing.
			if(hasOccurred(cal.getTime()))
			{
				//Update the game
				updateGame(gameid);

				//Determine when the next update should occur
				cal.add(Calendar.DAY_OF_MONTH, settings.getInterval());
				Date nextScheduled = cal.getTime();
				dbFacade.getAutomaticUpdatePolicyDBHandler().deleteIntervalPolicySettings(Game.mockGame(gameid));
				dbFacade.getAutomaticUpdatePolicyDBHandler().addIntervalPolicySettings(Game.mockGame(gameid), settings.getInterval(), settings.getUpdateTime(), nextScheduled);
			}
			else if(cal.getTime().getTime() < (new Date()).getTime()) //Check if the next update time is in the past
			{
				//Set when the next update should occur.  Use right now as the starting point
				//and update after the interval has passed.
				Calendar now = Calendar.getInstance();
				now.setTime(new Date());
				now.add(Calendar.DAY_OF_MONTH, settings.getInterval());
				now.setTime(updateTime);
				Date nextScheduled = now.getTime();

				if(log.isDebugEnabled())
				{
					log.debug("Update time was in past, resetting next update time to " + nextScheduled);
				}

				dbFacade.getAutomaticUpdatePolicyDBHandler().deleteIntervalPolicySettings(Game.mockGame(gameid));
				dbFacade.getAutomaticUpdatePolicyDBHandler().addIntervalPolicySettings(Game.mockGame(gameid), settings.getInterval(), settings.getUpdateTime(), nextScheduled);
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug("Not time for update based on interval policy settings.");
				}
			}
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
	}

	private void doDaysUpdateCheck(int gameid)
	{
		if(log.isDebugEnabled())
		{
			log.debug("Game " + gameid + " is using days policy.  Checking for scheduled updates");
		}

		try
		{

			DaysUpdatePolicySettings settings = dbFacade.getAutomaticUpdatePolicyDBHandler().getDaysPolicySettings(Game.mockGame(gameid));

			Date updateTime = settings.getUpdateTime();

			//This will be the current time
			Calendar cal = Calendar.getInstance();

			//This will store the update time with the date portions being irrelevant
			Calendar timeCal = Calendar.getInstance();
			timeCal.setTime(updateTime);

			cal.set(Calendar.HOUR_OF_DAY, timeCal.get(Calendar.HOUR_OF_DAY));
			cal.set(Calendar.MINUTE, timeCal.get(Calendar.MINUTE));
			cal.set(Calendar.SECOND, timeCal.get(Calendar.SECOND));

			//cal now has the today's date, with the time set from the stored settings for the DAYS update policy
			//If an update is scheduled for this day of the week, we'll pass cal on to the
			//hasOccurred method to check if the update time has passed
			//There are 7 possible cases, one for each day of the week

			//Flag to be set by the switch statement
			boolean updateToday = false;

			switch(cal.get(Calendar.DAY_OF_WEEK))
			{
				case Calendar.SUNDAY    : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is SUNDAY");
											  }
											  if(settings.getSunday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.MONDAY    : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is MONDAY");
											  }
											  if(settings.getMonday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.TUESDAY   : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is TUESDAY");
											  }
											  if(settings.getTuesday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.WEDNESDAY : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is WEDNESDAY");
											  }
											  if(settings.getWednesday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.THURSDAY  : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is THURSDAY");
											  }
											  if(settings.getThursday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.FRIDAY    : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is FRIDAY");
											  }
											  if(settings.getFriday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				case Calendar.SATURDAY  : {
										      if(log.isDebugEnabled())
										      {
												  log.debug("Today is SATURDAY");
											  }
											  if(settings.getSaturday())
											  {
												  updateToday = true;
											  }
							                  break;
								          }

				default		            : {
										      log.warn(this.getClass() + ": Unexpected default case");
										      break;
									      }
			}

			//If today is a day to do an update, and the update time has passed
			//since the last check, do an update.  Otherwise do nothing
			if(updateToday && hasOccurred(cal.getTime()))
			{
				updateGame(gameid);
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug("Not time for update based on days policy settings.");
				}
			}
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
	}

	private void doDatesUpdateCheck(int gameid)
	{
		if(log.isDebugEnabled())
		{
			log.debug("Game " + gameid + " is using dates policy.  Checking for scheduled updates");
		}

		try
		{
			List<ScheduledDate> scheduledDates =  dbFacade.getAutomaticUpdatePolicyDBHandler().getDatePolicyScheduledUpdates(Game.mockGame(gameid));
			for(ScheduledDate sd : scheduledDates)
			{
				if(!sd.getComplete())
				{
					if(log.isDebugEnabled())
					{
						log.debug("Scheduled date " + sd.toString() + " has not yet completed.  Checking to see if update should occur now.");
					}

					//This will store the update date with the time portions being irrelevant
					Calendar cal = Calendar.getInstance();

					//This will store the update time with the date portions being irrelevant
					Calendar timeCal = Calendar.getInstance();

					cal.setTime(sd.getDate());
					timeCal.setTime(sd.getTime());

					//One set the correct time into the date calendar so we have a calendar
					//with the correct date and time
					cal.set(Calendar.HOUR_OF_DAY, timeCal.get(Calendar.HOUR_OF_DAY));
					cal.set(Calendar.MINUTE, timeCal.get(Calendar.MINUTE));
					cal.set(Calendar.SECOND, timeCal.get(Calendar.SECOND));

					//Check if the update date/time has passed since the last run
					if(hasOccurred(cal.getTime()))
					{
						updateGame(gameid);

						dbFacade.getAutomaticUpdatePolicyDBHandler().markDatePolicyUpdateComplete(Game.mockGame(gameid), sd.getDateHashCode());
					}
					else
					{
						if(log.isDebugEnabled())
						{
							log.debug("Not time for update based on dates policy settings.");
						}
					}
				}
				else
				{
					if(log.isDebugEnabled())
					{
						log.debug("Scheduled date " + sd.toString() + " has already completed.");
					}
				}
			}
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
	}

	//Return true if the time nextUpdate
	//has occurred between right now, and right now minus the wait time between runs.
	private boolean hasOccurred(Date nextUpdate)
	{
		boolean occurred = false;

		long now = (new Date()).getTime();
		long then = now - waitTime;

		long update = nextUpdate.getTime();

		if(log.isDebugEnabled())
		{
			log.debug("Then : " + then);
			log.debug("Now : " + now);
			log.debug("Update : " + update);
		}

		if((then < update) && (update <= now))
		{
			occurred = true;
		}

		return occurred;
	}

	//Creates an ExpandedGame of the game
	//and updates the period
	private void updateGame(int gameid)
	{
		if(log.isInfoEnabled())
		{
			log.info("Updating game " + gameid);
		}

		try
		{
			//We need the actual game now, not just an id
			Game game = dbFacade.getGameDBHandler().getGame(gameid);

			//DO NOT ATTEMPT TO UPDATE A GAME THAT HAS NOT STARTED
			if(game.getCurrentPeriodNumber() > 0)
			{
				LoginBroker loginBroker = null;

				//If the ServletContext was set, check it for a LoginBroker
				if(bigAppContext != null)
				{
					loginBroker = (LoginBroker) bigAppContext.getAttribute("loginBroker");
				}

				//If a LoginBroker was found, disable logins to this game
				if(loginBroker != null)
				{
				    loginBroker.disableGameLogins(game, bigAppContext);
				}

				//Create an ExpandedGame from the game and use it to update the period
				ExpandedGame xGame = new ExpandedGame(game, dbFacade);
                xGame.setEmailValues("localhost",
									 "25",
									 "bigupdates@conindedu.org",
									 "BIG Period Update Notification",
									 "This message is to inform you that the period in your game of BIG50 has been updated.\n\nDo not reply to this e-mail because it was automatically generated. To contact your instructors, please e-mail them directly."
									);
				xGame.updatePeriod();

				//Reallow company login
				//If a LoginBroker was found, disable logins to this game
				if(loginBroker != null)
				{
					if(log.isInfoEnabled())
					{
						log.info("game logins have been reenabled, period has been updated");
					}
        	    	loginBroker.initializePeriod(game, bigAppContext);
				}
			}
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
	}
