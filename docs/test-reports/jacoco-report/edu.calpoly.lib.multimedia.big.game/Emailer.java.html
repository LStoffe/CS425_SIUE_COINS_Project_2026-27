package edu.calpoly.lib.multimedia.big.game;

import java.net.Socket;
import java.net.InetAddress;
import java.net.UnknownHostException;
import java.io.PrintWriter;
import java.io.OutputStreamWriter;
import java.io.InputStreamReader;
import java.io.BufferedReader;
import java.io.IOException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Used to send e-mails.
 * Once an instance of this class has been created it will
 * use a socket to connect to an SMTP server.  The connection will remain
 * until the <code>finalize()</code> method is called.
 * After instanciation the <code>send</code> method can be used to send e-mails.
 * <br><br>
 * This class does not currently support attachments or CCs.
 *
 * @author Kevin Craig
 *
 */
public class Emailer
{
	private String SMTPhost;
	private String SMTPport = "25";
	private Socket SMTPServer;
	private PrintWriter out;
	private BufferedReader in;
	private Log log;


    /**
     * Creates a new instance using a given SMTP host and connecting on the default
     * SMTP port of 25.
     *
     */
	public Emailer(String SMTPhost)
	{
	    log = LogFactory.getLog(this.getClass());
		this.SMTPhost = SMTPhost;
		setupConnection();
	}

    /**
     * Creates a new instance using a given SMTP host
     * and port
     *
     */
	public Emailer(String SMTPhost, String SMTPport)
	{
	    log = LogFactory.getLog(this.getClass());
		this.SMTPhost = SMTPhost;
		this.SMTPport = SMTPport;
	    setupConnection();
	}



    //Sets up the connection to SMTP server
    private void setupConnection()
    {
		try
		{
			if(log.isDebugEnabled())
			{
				log.debug("Attempting to connect to SMTP server " + SMTPhost + " on port " + SMTPport);
			}
			SMTPServer = new Socket(InetAddress.getByName(SMTPhost), Integer.parseInt(SMTPport));
			out = new PrintWriter((new OutputStreamWriter(SMTPServer.getOutputStream())), true);
			in = new BufferedReader((new InputStreamReader(SMTPServer.getInputStream())));

			if(log.isDebugEnabled())
			{
				log.debug("Connection established to SMTP server " + SMTPhost + " on port " + SMTPport);
			}
			//System.out.println(in.readLine());
			out.println("helo " + SMTPhost);
			//System.out.println(in.readLine());
		}
		catch(Exception e)
		{
			log.error("", e);
		}
	}


	/**
	 * Sends out an e-mail message to all given addresses
	 * @param toAddrs A <code>String</code> array of e-mail addresses.  The
	 *                message will be send to all addresses in this list.
	 * @param fromAddr The e-mail address the e-mail messages will be sent from
	 * @param subject The Subject line of the e-mail message
	 * @param message The content of the e-mail
	 */
	public void send(String[] toAddrs, String fromAddr, String subject, String message)
	{
		try
		{
			for(int i = 0; i < toAddrs.length; i++)
			{
				if(log.isDebugEnabled())
				{
					log.debug("Sending email to " + toAddrs[i]);
				}

				out.println("mail from: " + fromAddr);
				//System.out.println(in.readLine());
				out.println("rcpt to: " + toAddrs[i]);
				//System.out.println(in.readLine());
				out.println("data");
				//System.out.println(in.readLine());
				out.println("To: " + toAddrs[i]);
				out.println("Subject: " + subject);
				out.println(message);
				out.println(".");
				//System.out.println(in.readLine());
			}
		}
		catch (Exception e)
		{
			log.error("", e);
		}
	}

    /**
     *  Kills the connection to the SMTP server.  This object now becomes useless.
     */
    public void finalize()
    {
		try
		{
			out.println("quit");
			//System.out.println(in.readLine());
			SMTPServer.close();
		}
		catch(Exception e)
		{
			log.error("", e);
		}
	}


    /**
     * Unit test.  Send e-mails to yourself and see if they got there okay.
     */
	public static void main(String args[])
	{
//		String host = "cmweb.caed.calpoly.edu";
        String host = "smtp.att.yahoo.com";
//		String port = "25";
		String port = "465";
//		String from = "big@cmweb.caed.calpoly.edu";
		String from = "jborland77@gmail.com";
		String subject= "BIG Email Test";
		String message = "This is a test message";
		String[] to = { "jborland77@gmail.com"};
		Emailer emailer = new Emailer(host, port);
		emailer.send(to, from, subject, message);
		emailer.finalize();
	}
}
