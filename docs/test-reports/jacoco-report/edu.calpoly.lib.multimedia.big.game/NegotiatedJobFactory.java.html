package edu.calpoly.lib.multimedia.big.game;


//JDK imports
import java.util.Vector;
import java.util.Hashtable;
import java.util.Random;
import java.util.Date;

//BIGv2 imports
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.JobFactory;
import edu.calpoly.its.mas.big.game.Method;
import edu.calpoly.its.mas.big.users.company.ExpandedCompany;
import edu.calpoly.its.mas.big.game.ExpandedGame;

//BIGv3 imports
import edu.calpoly.lib.multimedia.big.game.Message;


/**
 * Extends functionality of the <code>Job</code> class
 * to handle aspects unique to jobs that are negotiated.
 */
public class NegotiatedJobFactory extends JobFactory
{
	private Vector customers;
	private Hashtable negotiationParameters;
	private Vector companies;
	private int chanceOfNegotiations; //The percent chance that a job open for closed bid will become a negotiated job
	private Random r;

	public NegotiatedJobFactory(ExpandedGame game,
								Hashtable[] jobSizeParams,
           			            int[][] percentTakeoffs,
     						    Method[][] methods,
           			            Hashtable[][] networkDependencies,
           			            Hashtable liquidatedDamagesParameters,
								int currentPeriod,
							    Date periodDate,
							    int workdaysPerMonth,
							    int stdHours,
							    int overtimeHours,
							    Vector customers,
							    Hashtable negotiationParameters,
							    Vector companies
							    )
	{
		super(game,
			  jobSizeParams,
           	  percentTakeoffs,
     		  methods,
           	  networkDependencies,
           	  liquidatedDamagesParameters,
			  currentPeriod,
			  periodDate,
			  workdaysPerMonth,
			  stdHours,
			  overtimeHours,
			  0,
			  0,
			  1);

		this.customers = customers;
		this.negotiationParameters = negotiationParameters;
		this.companies = companies;
		r = new Random();
		chanceOfNegotiations = ((Integer) negotiationParameters.get("PercentOfClosedBidsBecomingNegotiations")).intValue();

	}

	/**
	 * Creates new <code>Job</code> or <code>NegotiatedJob</code> objects.
	 * All <code>Job</code> objects created will have a limited number of companies
	 * allowed to bid on them.
	 */
	public Job createJob(int jobID)
	{
		Job job = super.createJob(jobID);

		//Choose a customer for this job
		int customerID = r.nextInt(10); //10 is a magic number for now

		job.setTypeModifier(Job.PRIVATE);
		job.setCustomer((Customer) customers.elementAt(customerID));

		//System.out.println("Set customer " + job.getCustomer().getCustomerID() + " for job " + job.getJobID() );
		job.setCompaniesAllowedToBid(job.getCustomer().getAllowedCompanies(companies, job.getJobType()));

		int spin = r.nextInt(100);  //percent chance is spin out of 100

		//If there is only 1 company eligible to bid, or
		//by random chance, this job will become a negotiated job
		if ( ((spin < chanceOfNegotiations) && (job.getCompaniesAllowedToBid().size() > 0) ) || (job.getCompaniesAllowedToBid().size() == 1) )
		{
			//System.out.println("Negotiated job.  size: " + job.getCompaniesAllowedToBid().size());
			job = new NegotiatedJob(job, 0, 0, NegotiatedJob.NEW);
			if(job.getCompaniesAllowedToBid().size() > 1)
			{
				int companyChosen = r.nextInt(job.getCompaniesAllowedToBid().size());
				Vector newCompaniesAllowedToBid = new Vector();
				newCompaniesAllowedToBid.add(job.getCompaniesAllowedToBid().elementAt(companyChosen));
				job.setCompaniesAllowedToBid(newCompaniesAllowedToBid);
				//System.out.println("Negotiated job by chance  size: " + newCompaniesAllowedToBid.size());
			}

			//System.out.println("Negotiated job.  size: " + job.getCompaniesAllowedToBid().size());
			if(job.getCompaniesAllowedToBid().size() > 0)
			{
				ExpandedCompany ec = game.getExpandedCompany( ((Integer) job.getCompaniesAllowedToBid().elementAt(0)).intValue());  //The first company
				game.sendInGameMessageToCompany(new Message(job.getCustomer().getCustomerName(),
				 											ec.getCompany(),
				   											"We have a new job coming up and we would like your company to work it for us.  If you are interested in this job, please send us an estimate of your price.  We hope you will consider our request.",
				   											"Upcoming job for your company",
				   											game.getGame().getCurrentPeriodNumber(),
				   											Message.NEGOTATION_REQUEST,
				   											jobID
				   										   )
					   						   );
			}
		}
		else
		{
		   //If the job did not become negotiated then it remains a closed bid job
		   //Send an invitation to bid to each eligible company

		   for(int i = 0; i < job.getCompaniesAllowedToBid().size(); i++)
		   {
			   ExpandedCompany ec = game.getExpandedCompany( ((Integer) job.getCompaniesAllowedToBid().elementAt(i)).intValue());
			   game.sendInGameMessageToCompany(new Message(job.getCustomer().getCustomerName(),
			   											   ec.getCompany(),
			   											   "We have a new job coming up and we are planning to find a company to work it through bidding.  We feel your company might be a good candidate for this job and we invite you to submit a bid.",
			   											   "Invitation to bid",
			   											   game.getGame().getCurrentPeriodNumber(),
			   											   Message.CLOSED_BID_INVITATION,
			   											   jobID
			   											  )
				   							  );
		   }
		}

		return job;
	}


}
