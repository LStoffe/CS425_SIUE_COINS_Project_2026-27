    /**
     * Creates a fictitious contractor name
     */
	public static String generateComputerControlledContractorName()
	{
		String name = "Computer Controlled Contractor";

		if(random.nextInt(100) < 40) //40% chance of first part being 3-letter combo
		{
			name = threeLetterCombo();
		    //100% chance of having a business title
		    name += " " + contractorBusinessPart[random.nextInt(contractorBusinessPart.length)];

			//75% chance of having business type in name
			if(random.nextInt(100) > 25)
			{
				name += " " + contractorBusinessTypePart[random.nextInt(contractorBusinessTypePart.length)];
			}
		}
		else
		{
			if(random.nextInt(100) < 30) //% chance of first part being fictitious name
			{
				name = contractorFictitiousNamePart[random.nextInt(contractorFictitiousNamePart.length)];

				//100% chance of having a business title
				name += " " + contractorBusinessPart[random.nextInt(contractorBusinessPart.length)];

				//75% chance of having business type in name
				if(random.nextInt(100) > 25)
				{
					name += " " + contractorBusinessTypePart[random.nextInt(contractorBusinessTypePart.length)];
				}
			}
			else
			{
				//% chance of first part being single personal name
				name = contractorPersonalNamePart[random.nextInt(contractorPersonalNamePart.length)];

				if(random.nextInt(100) < 50) //% chance of first part having associations
				{

					name += " " + contractorAssociatesPart[random.nextInt(contractorAssociatesPart.length)];

					//100% chance of having a business title
					name += " " + contractorBusinessPart[random.nextInt(contractorBusinessPart.length)];
				}
				else
				{

					//30% chance of getting another personal name
					if(random.nextInt(100) < 50)
					{
						name += " & " + contractorPersonalNamePart[random.nextInt(contractorPersonalNamePart.length)];
					}

					//100% chance of having a business title
					name += " " + contractorBusinessPart[random.nextInt(contractorBusinessPart.length)];

					//75% chance of having business type in name
					if(random.nextInt(100) > 25)
					{
						name += " " + contractorBusinessTypePart[random.nextInt(contractorBusinessTypePart.length)];
					}
				}
			}
		}


		return name;
	}

    //Returns a <code>String</code> of 3 capital letters, or 2 capital letter separated by an ampersand
	private static String threeLetterCombo()
	{
		String letters = "";

		//25% chance String will contain the ampersand
		if( random.nextInt(100) < 25)
		{
			letters += alphabet[random.nextInt(alphabet.length)];
			letters += " & ";
			letters += alphabet[random.nextInt(alphabet.length)];
		}
		else
		{
			letters += alphabet[random.nextInt(alphabet.length)];
			letters += alphabet[random.nextInt(alphabet.length)];
			letters += alphabet[random.nextInt(alphabet.length)];


		}

		//Before returning the String check it against well-known offensive words
		//or acronyms to ensure nothing offensive is returned.
		//Add offensive 3-letter sequences here if need be
		if(letters.equals("KKK") ||
		   letters.equals("ASS") ||
		   letters.equals("SEX") ||
		   letters.equals("BAD") ||
		   letters.equals("S & M")
		  )
		{
			letters = threeLetterCombo();
		}
		return letters;
	}
