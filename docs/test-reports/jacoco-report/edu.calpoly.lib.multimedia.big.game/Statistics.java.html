package edu.calpoly.lib.multimedia.big.game;

import java.util.Random;

public class Statistics
{
	/**
	 * Returns a float from the normal curve, given the mean and stddev
	 * of the range.  The value returned will be no less than min and no greater
	 * than max.
	 */
	public static float normalFloat(float mean, float stddev, float min, float max)
	{
		Random r = new Random();
		float value;
		float normal = (float) r.nextGaussian();
		value = (normal * stddev) + mean;
		if (value > max)
		    value = max;
		if (value < min)
		    value = min;

		return value;
	}

	/**
	 * Returns an int from the normal curve, given the mean and stddev
	 * of the range.  The value returned will be no less than min and no greater
	 * than max.
	 */
	public static int normalInt(int mean, float stddev, int min, int max)
	{
		return Math.round(normalFloat((float) mean, stddev, (float) min, (float) max));
	}
}
