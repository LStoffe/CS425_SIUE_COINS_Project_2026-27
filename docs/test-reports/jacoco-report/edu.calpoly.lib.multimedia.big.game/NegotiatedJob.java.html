package edu.calpoly.lib.multimedia.big.game;


//BIGv2 imports
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.users.company.Bid;

//BIGv3 imports
import edu.calpoly.lib.multimedia.big.reports.Contract;

//JDK imports
import java.util.Date;
import java.util.Iterator;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Extends functionality of the <code>Job</code> class
 * to handle aspects unique to jobs that are negotiated.
 */
public class NegotiatedJob extends Job
{
	public static final int NEW = 100;
	public static final int NEGOTIATION_FAILED = 102;
	public static final int NEGOTIATION_SUCCEEDED = 103;
	public static final int NEGOTIATION_CANCELLED = 104;
	public static final int COST_REJECTED = 105;
	public static final int PERCENT_PROFIT_REJECTED	= 106;

	private int agreedProfit;
	private long agreedCost;
	private int state;

	private Log log;

	public NegotiatedJob(Job j, int agreedProfit, long agreedCost, int state)
	{
		super(j.getJobID(),
		      j.getLiquidatedDamages(),
		      j.getCompletionDate(),
		      j.getCompletionDeadline(),
		      j.getJobSize(),
		      j.getPercentRetained(),
		      j.getPercentOfJob(),
		      j.getDirectCost(),
		      j.getRetainedAmount(),
		      j.getLusRemaining(),
		      j.getPeriodCompleted(),
		      j.getPeriodCreated(),
		      j.getIsActive(),
		      j.getIsAuto(),
		      j.getJobType(),
		      j.getEstimatedWorkdays(),
		      j.getMethodScalingFactor(),
		      j.getChangeOrderPrice(),
		      j.getChangeOrderLus(),
		      j.getFixedFieldOverhead(),
		      j.getVariableFieldOverhead(),
		      j.getAmountRetainedFromSubcontractors(),
		      j.getBilledToDate(),
		      j.getTypeModifier()
			 );

		setRetentionID(j.getRetentionID());
		setActivities(j.getActivities());
		setCustomer(j.getCustomer());
		setCompaniesAllowedToBid(j.getCompaniesAllowedToBid());
		this.agreedProfit = agreedProfit;
		this.agreedCost = agreedCost;
		this.state = state;

		log = LogFactory.getLog(this.getClass());
	}


	public int getAgreedProfit()
	{
		return agreedProfit;
	}

	public String getAgreedProfitString()
	{
		Integer wholePart = new Integer(agreedProfit / 10);
		Integer partialPart = new Integer(agreedProfit % 10);
		return wholePart.toString() + "." + partialPart.toString();
	}

	public void setAgreedProfit(int agreedProfit)
	{
		this.agreedProfit = agreedProfit;
	}

	public long getAgreedCost()
	{
		return agreedCost + getChangeOrderPrice();
	}

	public void setAgreedCost(long agreedCost)
	{
		this.agreedCost = agreedCost;
	}

	public int getState()
	{
		return state;
	}

	public void setState(int state)
	{
		this.state = state;
	}

	/**
	 * Returns the maximum amount of gross income
	 * the company may get from this job.
	 * This amount is equal to the agreed cost of the
	 * job plus the agreed profit.
	 */
	public long getMaxPrice()
	{
		return getCostPlusProfit(getAgreedCost());
	}


	public long getCostPlusProfit(long cost)
	{
		double profit = (double) getAgreedProfit();
		profit = profit /1000; //The percent profit value is 10x the actual value.
							   //Divide by 1000 (10*100) for this factor of 10 and 100 to convert from percent to decimal
		double percent = 1 + profit;  //The total amount is 100 percent plus the profit percent

		return Math.round(((double) cost) * percent);
	}


  /**
	* Overrides method from Job.  There should be a Vector of
	* up to one player submitted Bid and some CCC bids, set through the setBids method, that is the
	* Bid from the company that negotiated this job.  This method
	* simply declares the player Bid the winner, or a CCC bid the winner if
	* no player submitted bid exists.
	*/
  public void determineBidWinner()
  {
	  boolean companyBidFound = false;
	  Bid lowBid = null;
	  Iterator bidIterator = bids.iterator();
	  while(bidIterator.hasNext())
	  {
		  Bid theBid = (Bid) bidIterator.next();

		  if (theBid.getCompanyID() > -1)
		  {
			  //System.out.println("Bid on negotiated job found for company " + theBid.getCompanyID());
			  //This is the chosen company... the only one who's not a CCC
			  companyBidFound = true;
			  theBid.setOutcomeCode(1);
			  lowBid = theBid;
		  }
		  else
		  {
			  theBid.setOutcomeCode(7);
		  }

	  }
	  //If the company didn't bid, just set the winner to be the first CCC
	  if(!companyBidFound)
	  {
		  lowBid = (Bid) bids.iterator().next();
		  lowBid.setOutcomeCode(1);
	  }



	  // Associate the winning company with this job, leaving <code>company</code>
	  // null if a CCC placed the winning bid;
	  if (lowBid.getCompanyID() > -1)
	  {
		  company = game.getExpandedCompany(lowBid.getCompanyID());
	      //System.out.println("Company for negotiated job set to " + lowBid.getCompanyID());
          game.getExpandedCompany(lowBid.getCompanyID()).awardJob(this, lowBid);

          //Give this company name recognition credit for getting this job
          int nameRecognitionBonus = Math.round(((float) jobSize) / ((float) CAMUpdater.NAME_RECOGNITION_GET_JOB_MODIFIER));
          game.modifyCAMUpdate(company, CAMUpdater.NAME_RECOGNITION, nameRecognitionBonus);

          //Give this company job type credit for getting this job
		  giveJobTypeCredit(CAMUpdater.JOB_TYPE_AWARD_BONUS);

		  bidAmount = getMaxPrice();
		  lowBid.setBidAmount(getMaxPrice());
	  }
	  game.storeBids(this, bids);

    }



    // do nothing for negotiated jobs
    public void chargeBiddingExpense(int expense)
    {
    }

    // do nothing for negotiated jobs
    public void determineResponsibility()
    {
    }

    public void jobCompleted(Contract contract)
    {
		long shareOfSavings = getShareOfSavings(contract.getCostIncurred() + getCurrentCost());
		long newAmount = getCostPlusProfit(Math.min(contract.getCostIncurred() + getCurrentCost(), getAgreedCost() ) );

		if(log.isDebugEnabled())
		{
			log.debug("negotiated job completed");
			log.debug("agreed cost " + getAgreedCost());
			log.debug("costIncurred" + contract.getCostIncurred() + getCurrentCost());
			log.debug("new amount " + newAmount);
		}

		newAmount += shareOfSavings;

		if(log.isDebugEnabled())
		{
			log.debug("new amount + share of savings = " + newAmount);
		}

		game.updateBid(company.getCompany(), this, newAmount);
    }

	/**
	 * Returns the share of savings on this negotiated job.
	 * Share of savings is calculated in the following manner:
	 * If the job is not complete, or the cost of the job
	 * exceeds the maximum agreed cost of the job,
	 * then the share of savings is $0.
	 * If the cost of the job is less than the maximum agreed cost,
	 * and the job is complete, then the share of savings is equals
	 * to 25% of the maximum cost of the job minus the actual cost of the job.
	 *
	 * @param cost Total cost of the job so far.
	 */
    public long getShareOfSavings(long cost)
    {
		if(getIsActive())
		{
			return 0;
		}
		else if(getAgreedCost() <= cost)
		{
			return 0;
		}
		else
		{
			long savings = getAgreedCost() - cost;
			long shareOfSavings = Math.round(.25d * ((double) savings) );

			if(log.isDebugEnabled())
			{
				log.debug("Agreed Cost: " + getAgreedCost());
				log.debug("Cost Incurred: " + cost);
				log.debug("Savings: " + savings);
				log.debug("Share of Savings: " + shareOfSavings);
			}

			return shareOfSavings;
		}

	}
}
