package edu.calpoly.lib.multimedia.big.game;

//Servlet imports
import javax.servlet.ServletOutputStream;
import javax.servlet.ServletContext;
import javax.servlet.ServletException;
import javax.servlet.RequestDispatcher;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

//Commons logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Servlet that upon creation, creates an AutoUpdateThread, starts it, and places
 * it in the servlet context.  The thread will continue to live in the servlet context and
 * periodically check to see if there is a game ready to be updated.
 *
 * @author Kevin Craig
 */
public class AutoUpdateServlet extends HttpServlet
{
    public void init() throws ServletException
    {
		Log log = LogFactory.getLog(this.getClass());
		if(log.isInfoEnabled())
		{
			log.info(this.getClass() + ": Creating AutoUpdateThread");
		}

		int waitTime = 300;

		try
		{
			waitTime = Integer.parseInt(getServletConfig().getInitParameter("waitTime"));
		}
		catch(NumberFormatException nfe)
		{
			log.warn(getServletConfig().getInitParameter("waitTime") + " is an invalid value for waitTime parameter.  Using default value of 10 minutes");
		}

		//Wait time as an init parameter is to be specified in minutes.
		waitTime = waitTime * AutoUpdateThread.MINUTE;

		try
		{
			AutoUpdateThread aut = new AutoUpdateThread(900000, getServletContext());
			aut.start();

		}
		catch(Exception e)
		{
			log.fatal("", e);
		}
	}
}
