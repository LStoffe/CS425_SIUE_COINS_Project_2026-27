package edu.calpoly.lib.multimedia.big.game;

import java.util.StringTokenizer;

/**
 * Represents a non-player contractor in BIG.
 * Non-player contractors, or Computer Controlled Contractors (CCCs)
 * add fairness and increase the realism of BIG.
 * Their most important purpose is to keep companies honest
 * by putting a ceiling on the amount a single job can go for.
 * At least one CCC will bid on every job, with a fair bid that has a
 * greedy profit margin.  This ensures both that companies have a chance
 * to beat the CCCs and get jobs, and that a company can never get
 * too much money on a single job, because the CCCs' bid(s) would be lower.
 * Without a CCC bid on each job, a company could make an exaggerated bid
 * and still be awarded the job because theirs was the only bid.
 * <br><br>
 * CCCs also add realism because it means there are companies out there
 * who the player companies have no knowledge of.
 * <br><br>
 *
 * A CCC's bid is calculated by adding the field/office overhead
 * (fixed and variable) and bid expense (a percentage of the direct cost
 * set as a game parameter) to the Job's direct cost.  This value is then
 * increased by a random profit percentage (the boundaries of which
 * are set as game parameters) to arrive at the final bid price.
 * <br><br>
 * Note: companyID for a CCC is its cccId negated.  All player companyIDs
 * are positive values.
 *
 *
 * @author Kevin Craig
 *
 */
public class ComputerControlledContractor implements Contractor
{
	private int cccID = 0;
	private int fitness;
	private int[] jobTypePreferences;  //Index + 1 is job type, value is % chance this CCC would bid on a job of that type
	private String name;

	public ComputerControlledContractor(String name, int fitness, int[] jobTypePreferences)
	{
		this.name = name;
		this.fitness = fitness;
		this.jobTypePreferences = jobTypePreferences;
	}

	public ComputerControlledContractor(int cccID, String name, int fitness, int[] jobTypePreferences)
	{
		this.cccID = cccID;
		this.name = name;
		this.fitness = fitness;
		this.jobTypePreferences = jobTypePreferences;
	}

	public int getCCCID()
	{
		return cccID;
	}

	public int getFitness()
	{
		return fitness;
	}

	public int[] getJobTypePreferences()
	{
		return jobTypePreferences;
	}

	public String getName()
	{
		return name;
	}

    /**
     *  Only an instance of <code>CCCDBHandler</code> should modify the the cccID.
     *  If this method is called by some other class the database could become confused
     *  and throw exception for this or other CCCs.
     *
     */
	public void setCCCID(int cccID)
	{
		this.cccID = cccID;
	}

	public void setFitness(int fitness)
	{
		this.fitness = fitness;
	}

	public void setJobTypePreferences(int[] jobTypePreferences)
	{
		this.jobTypePreferences = jobTypePreferences;
	}

	public void setName(String name)
	{
		this.name = name;
	}

	public String getTitle()
	{
		return name;
	}

	public int getContractorID()
	{
		return - cccID;
	}

    /**
     * Used to sort <code>ComputerControlledContractor</code> by title
     */
    public int compareTo(Object o)
    {
  	    Contractor c = (Contractor) o;

		String string1 = "";
		String string2 = "";

		StringTokenizer st1Tok = new StringTokenizer(c.getTitle(), " ");

		while (st1Tok.hasMoreTokens())
		{
			string1 += st1Tok.nextToken();
		}

		StringTokenizer st2Tok = new StringTokenizer(name, " ");

		while (st2Tok.hasMoreTokens())
		{
			string2 += st2Tok.nextToken();
		}

	    return string2.compareToIgnoreCase(string1);
    }
}
