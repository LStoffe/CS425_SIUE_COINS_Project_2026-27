package edu.calpoly.lib.multimedia.big.game;

import edu.calpoly.its.mas.big.users.company.Company;

/**
 * Represents a message to a company in BIG.  The message body can
 * contain be any String of less than 8129 characters.
 * Each message has a type associated with it.  A list of possible
 * types is defined as static ints.  The basic type is a message containing only
 * text.  Other types may contain links or other information.  In this
 * case the link field is used to hold the database ID of some other
 * database info that holds the extended information for that type.
 * The jsp that displays messages can use the type field of a message
 * to determine how to display
 * that message.
 *
 * @author Kevin Craig
 *
 */
public class Message
{
	/**
	 * Message type for a message that contains only a text String
	 */
	public static final int TEXT_ONLY = 100;

	/**
	 * Message type for a message that is a negotiation request.
	 * The link field should contain the database ID of the
	 * request information.
	 */
	public static final int NEGOTATION_REQUEST = 101;

	/**
	 * Message type for a message that is a closed bid invitation.
	 */
	public static final int CLOSED_BID_INVITATION = 102;

	private Company to;
	private String from;
	private String message;
	private String subject;
	private int periodNumber;
	private int type;
	private int link;

	/**
	 *  Constructor.
	 *  @param from If length is greater than 8129, only the first 8128
	 *  characters will be saved.
	 */
	public Message(String from,
						Company to,
						String message,
						String subject,
						int periodNumber,
						int type,
						int link)
	{
		if (from.length() < 8129)
		{
			this.from = from;
		}
		else
		{
			this.from = from.substring(0, 8128);
		}

		this.to = to;

		this.message = message;
		this.periodNumber = periodNumber;
		this.type = type;
		this.link = link;

		this.subject = "No subject";
		if((subject != null) && (subject.length() > 0))
		{
			this.subject = subject;
		}
	}

    public Company getTo()
    {
		return to;
	}

	public String getFrom()
	{
		return from;
	}

	public String getBody()
    {
		return message;
	}

	public String getSubject()
	{
		return subject;
	}

	/**
	 * Return the period this message arrived in the company's inbox
	 */
	public int getPeriodNumber()
	{
		return periodNumber;
	}

	/**
	 * Returns the type of this message
	 */
	public int getType()
	{
		return type;
	}

	/**
	 * Returns the value of the link field.
	 */
	public int getLink()
	{
		return link;
	}
}
