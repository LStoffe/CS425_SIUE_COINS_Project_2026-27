	public NamedVariableDataFacadeImpl(Admin admin, Game game, BIGDatabaseFacadeInterface dbFacade)
	{
		this.admin=admin;
		this.game=game;
		this.dbFacade=dbFacade;

		log = LogFactory.getLog(this.getClass());
	}

	/**
	 * Returns the value of a variable that is not associated with any one company.
	 * Variable that are valid inputs to this method are: TOTAL_JOBS
	 */
	public Object getNamedVariableValue(String variable)
	{
/*		if(log.isDebugEnabled())
		{
			log.debug("Looking up named variable " + variable);
		} */

		if(variable.equals(TOTAL_JOBS))
		{
			int value = 0;

			try
			{
				value = dbFacade.getJobDBHandler().getJobCountForGame(game.getGameID());
			}
			catch(BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
			}

			return new Integer(value);
		}

		return null;
	}

	public Object getJobNamedVariableValue(String variable, int jobid)
	{

		if(variable.equals(JOB_DAYS_LATE))
		{
			int value = 0;

			try
			{
				value = dbFacade.getJobDBHandler().getJobDaysLate(game.getGameID(), jobid);
			}
			catch(BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
			}
			return new Integer(value);
		}

		return null;
	}
