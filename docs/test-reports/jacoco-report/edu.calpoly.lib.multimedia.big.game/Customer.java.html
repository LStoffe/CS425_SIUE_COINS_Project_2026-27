	/**
	 * Job customer processes a bill and decides whether or not to pay it.
	 * This class is intended to be expanded in future versions of BIG.
	 * A call to this method results in two possibilities
	 * <ol>
	 *    <li>Bill Accepted - The amount of the bill would put the total amount billed on the job
	 *        within an acceptable level of the amount earned.
	 *    </li>
	 *    <li>
	 *        Bill Rejected - The amount of the bill would either put the total amount billed on the job
	 *        at more than the contract price or at an unacceptably high level above the amount earned.
	 *    </li>
	 * </ol>
	 * <br><br><b>For Bid Jobs:</b><br>
	 * The amount earned on a contract is calculated by multiplying the percent work complete,
	 * in this case using the value from the progress report (architect's estimate),
	 * by the contract price of the job including change orders.
	 * The acceptable level is determined as follows:<br>
	 * The max overbilling percent parameter set by the administrator is
	 * reduced by an amount equal to the max overbilling percent parameter * percent work complete.
	 * (e.g. if the max overbilling percent is 30%, when the job is 0% complete the acceptable level will
	 * be 30%, when the job is 100% complete the acceptable level will be 0%, when the job is 50% complete
	 * the acceptable level will be 15%).
	 * The percent work complete is increased by the acceptable level to compute the
	 * max customer job % complete belief.  This amount is multiplied by the contract price
	 * to get the maximum amount the customer will pay on the job.  If this amount is
	 * exceeded by the current bill amount plus the amount billed to date, the bill is rejected.
	 * <br><br>
	 * <b>For Negotiated Jobs:</b>
	 * The amount of the bill will be accepted only if it is no greater than
	 * the lesser of the cost of work done on the job plus profit and the maximum amount of the job.
	 * If the job is complete and the cost plus profit is lower, then share of savings is also included.
	 * @return BillingForm The same <code>BillingForm</code> that was passed in
	 * with the <code>amountPaid</code> and <code>rejected</code> fields set correctly.
	 * If the bill is rejected the amountPaid will be 0, otherwise it will be equal to
	 * to the billing amount.
	 */
	public BillingForm processBill(BillingForm bill, Job job)
	{
	    long maxWillPay = 0;
		if(! (job instanceof NegotiatedJob))
		{
		    //System.out.println("Actual amount complete " + job.getPercentOfJobComplete());
		    float acceptableLevel = (((float) maxOverbillingPercent) * (1 - (job.getPercentOfJobComplete() /100))) /100;
		    //System.out.println("Acceptable level is " + acceptableLevel);
		    float customerMaxJobPercentCompleteBelief = (job.getPercentOfJobComplete() / 100) * (1 + acceptableLevel);
		    //System.out.println("customer belief: " + customerMaxJobPercentCompleteBelief);

		    maxWillPay = Math.round(((float) job.getTotalPrice()) * customerMaxJobPercentCompleteBelief);
		    //System.out.println("customer will pay up to " + maxWillPay);
	    }
	    else
	    {
			NegotiatedJob nj = (NegotiatedJob) job;
			if(job.isComplete())
			{
				maxWillPay = job.getTotalPrice();
				//System.out.println("Job is complete.  Max will pay equal to end price of job: " + maxWillPay);
			}
			else
			{
				maxWillPay = NumberConverter.longValue(bill.getNegotiatedJobAmountEarnedSoFar());
				//System.out.println("Job is NOT complete.  Max will pay equal to : " + maxWillPay);
			}
	    }

		if((job.getBilledToDate() + bill.getBillingAmount()) > maxWillPay)
		{
		    bill.setAmountPaid(0);
			bill.setRejected(true);
		}
		else
		{
			bill.setAmountPaid(bill.getBillingAmount());
			bill.setRejected(false);
		}


		return bill;
	}


    /**
     * Returns true if the amount of the bill is ethical
     * for the amount of work complete on job.
     * A bill is considered unethical if the bill
     * plus the total billed to date is greater than
     * the percentage of the job complete times the price of the job
     * raised by a percentage equal to the game parameter for ethical overbilling.
     */
	public boolean ethicalBill(BillingForm bill, Job job)
	{
		long totalBilled = job.getBilledToDate();
		long totalEarned = Math.round( ( ((float) job.getPercentOfJobComplete() / 100)) * ((float) job.getTotalPrice()));
		long ethicalBillingLimit = Math.round(  (((float) overbillingPercentAffectingEthics / 100) + 1) * ((float) totalEarned));
		//System.out.println("totalBilled = " + totalBilled);
		//System.out.println("ethical limit = " + ethicalBillingLimit);
		if(totalBilled < ethicalBillingLimit)
		{
			return true;
		}
		else
		{
			return false;
		}
	}

	public Vector getAllowedCompanies(Vector companies, int jobType)
	{
		Vector allowedCompanies = new Vector();
		int thresholdForConsideration
			= computeWeightedAverageOfCamValues(financialLiquidityLevel,
												financialSuccessLevel,
												responsibilityLevel,
												paceLevel,
												ethicsLevel,
												nameRecognitionLevel,
												jobTypeLevel
											   );

		//System.out.println("Threshold is " + thresholdForConsideration);
		//System.out.println("Companies.size is " + companies.size());
		for(int i = 0; i < companies.size(); i++)
		{
			ExpandedCompany ec =  (ExpandedCompany) companies.elementAt(i);
			Hashtable cams = ec.getCAMs();

			int companyCAMWeightedAverage = computeWeightedAverageOfCamValues( ((Integer) cams.get("FinancialLiquidity")).intValue(),
											       							   ((Integer) cams.get("FinancialSuccess")).intValue(),
											       							   ((Integer) cams.get("BiddingResponsibility")).intValue(),
											      							   ((Integer) cams.get("Pace")).intValue(),
											       							   ((Integer) cams.get("Ethics")).intValue(),
											      							   ((Integer) cams.get("NameRecognition")).intValue(),
											      							   getJobTypeCAMValue( jobType,
																				   				   ((Integer) cams.get("Apartments")).intValue(),
											       												   ((Integer) cams.get("Schools")).intValue(),
											       					 							   ((Integer) cams.get("Offices")).intValue(),
											       												   ((Integer) cams.get("Hospitals")).intValue(),
											       					  							   ((Integer) cams.get("Industrial")).intValue(),
											       					  							   ((Integer) cams.get("Highways")).intValue(),
																								   ((Integer) cams.get("Bridges")).intValue(),
																								   ((Integer) cams.get("SiteDevelopment")).intValue(),
																								   ((Integer) cams.get("MassExcavation")).intValue(),
																								   ((Integer) cams.get("UndergroundUtilities")).intValue()
											                         							 )
												 							 );

			//System.out.println("Values for " + ec.getCompany().getTitle() + " is " + companyCAMWeightedAverage);

			if(companyCAMWeightedAverage  >= thresholdForConsideration)
			{
				allowedCompanies.add(new Integer(ec.getCompany().getCompanyID()));
				//System.out.println(ec.getCompany().getTitle() + " is allowed to bid");
			}
			else
			{
				//System.out.println(ec.getCompany().getTitle() + " is NOT allowed to bid");
			}
		}


		return allowedCompanies;
	}

	//Creates a weighted average of the cam values
	//passed as parameters.
	//The weights are always the values of the
	//"Factor" variables, (e.g. financialLiquidityFactor, paceFactor).
	//The cam values can come from the Level variables
	//or be the cam values for the company of interest.
	private int computeWeightedAverageOfCamValues(int financialLiquidity,
	  								   		      int financialSuccess,
	  							 			      int responsibility,
	  							 				  int pace,
	  							 				  int ethics,
	  							 				  int nameRecognition,
	  							 				  int jobType
											     )
	{
		int average = 0;

		average += Math.round((( (float) financialLiquidityFactor) /100f) * ((float) financialLiquidity));
		average += Math.round((( (float) financialSuccessFactor) /100f) * ((float) financialSuccess));
		average += Math.round((( (float) responsibilityFactor) /100f) * ((float) responsibility));
		average += Math.round((( (float) paceFactor) /100f) * ((float) pace));
		average += Math.round((( (float) ethicsFactor) /100f) * ((float) ethics));
		average += Math.round((( (float) nameRecognitionFactor) /100f) * ((float) nameRecognition));
		average += Math.round((( (float) jobTypeFactor) /100f) * ((float) jobType));

		return average;
	}


	//Figures out what the value of jobType metric should be by returning the
	//metric for the job type that is the correct type for the job type of the job in question.
	//e.g. If the job is of type apartments, the apartments metric is returned.
	//This method assumes the jobType id values are permanent forever and uses
	//hardcoding, so this entire method will need to be written for future updates to BIG
	//if jobTypes become dynamic
	private int getJobTypeCAMValue(int jobType, int apartments, int schools, int offices, int hospitals, int industrial, int highways, int bridges, int sitedevelopment, int massexcavation, int undergroundutilities)
	{
		if(jobType == 1)
		{
			return apartments;
		}
		else if (jobType == 2)
		{
			return schools;
		}
		else if (jobType == 3)
		{
			return offices;
		}
		else if (jobType == 4)
		{
			return hospitals;
		}
		else if (jobType == 5)
		{
			return industrial;
		}
		else if(jobType == 6)
		{
			return highways;
		}
		else if (jobType == 7)
		{
			return bridges;
		}
		else if (jobType == 8)
		{
			return sitedevelopment;
		}
		else if (jobType == 9)
		{
			return massexcavation;
		}
		else if (jobType == 10)
		{
			return undergroundutilities;
		}
		else
		{
			return 0;
		}
	}
