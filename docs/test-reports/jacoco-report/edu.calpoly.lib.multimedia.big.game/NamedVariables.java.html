package edu.calpoly.lib.multimedia.big.game;

/**
 * Defines some official names for some variables in BIG.
 *
 */
public final class NamedVariables
{
	/**
	 * The total number of jobs that have been created in the game so far.
	 */
	public static final String TOTAL_JOBS = "TOTAL_JOBS";

	/**
	 * The total number of bids made by a company so far.
	 */
	public static final String COMPANY_BIDS = "COMPANY_BIDS";

	/**
	 * The total number of bids made by a company so far that have been dismissed.
	 * Dismissed refers to a bid that did not win the job, because
	 * of some reason other than price.  Bids that do not win the job only
	 * because another bid is lower, are not "dismissed".
	 */
	public static final String COMPANY_DISMISSED_BIDS = "COMPANY_DISMISSED_BIDS";

	/**
	 * The total number of jobs awarded to a company.
	 */
	public static final String COMPANY_JOBS = "COMPANY_JOBS";

	/**
	 * The total number of public jobs awarded to a company.
	 */
	public static final String COMPANY_PUBLIC_JOBS = "COMPANY_PUBLIC_JOBS";

	/**
	 * The total number of private jobs awarded to a company.
	 */
	public static final String COMPANY_PRIVATE_JOBS = "COMPANY_PRIVATE_JOBS";

	/**
	 * The total number of heavy jobs awarded to a company.
	 */
	public static final String COMPANY_HEAVY_JOBS = "COMPANY_HEAVY_JOBS";

	/**
	 * The total number of private jobs awarded to a company through closed bid.
	 */
	public static final String COMPANY_CLOSED_BID_JOBS = "COMPANY_CLOSED_BID_JOBS";

	/**
	 * The total number of private jobs awarded to a company through negotiation.
	 */
	public static final String COMPANY_NEGOTIATED_JOBS = "COMPANY_NEGOTIATED_JOBS";

	/**
	 * The total number of jobs automatically awarded to a company by the admin.
	 */
	public static final String COMPANY_AUTO_JOBS = "COMPANY_AUTO_JOBS";

	/**
	 * The total number of private jobs awarded to a company through negotiation.
	 */
	public static final String COMPANY_EQUITY = "COMPANY_EQUITY";

	/**
	 * The total numbers of days a job was worked past its due day.
	 * If the job is currently active, then this refers to the number of days the
	 * job is currently late.
	 */
	public static final String JOB_DAYS_LATE = "JOB_DAYS_LATE";

    /**
	 * Added by me
	 */
	public static final String SUM_SIX_CAM_VALUES = "SUM_SIX_CAM_VALUES";
	public static final String COMPANY_JOBS_COMPLETED = "COMPANY_JOBS_COMPLETED";
	public static final String COMPANY_JOBS_IN_PROGRESS = "COMPANY_JOBS_IN_PROGRESS";
	public static final String TOTAL_GROSS_MARGIN_JOBS_COMPLETED = "TOTAL_GROSS_MARGIN_JOBS_COMPLETED";
	public static final String TOTAL_GROSS_MARGIN_JOBS_IN_PROGRESS = "TOTAL_GROSS_MARGIN_JOBS_IN_PROGRESS";

}
