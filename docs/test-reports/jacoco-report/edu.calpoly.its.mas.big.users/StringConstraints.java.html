package edu.calpoly.its.mas.big.users;



/**
 * Contains constants describing upper and lower limit for string lengths.
 * Any changes that need to be made to string length boundaries can be made here.  <br><br>
 * <b>Note:</b>Error messages which display bounds on string lengths (such as in response to a
 * user submitting a string that is too long) do not use constants from this class and their values must also
 * be updated if values here are changed.  String lengths in database and jsp may also need to be changed.
 *
 * @author Kevin Craig
 *
 * @version $Revision: 1.2 $ $Date: 2001/10/03 20:23:50 $
 */
public class StringConstraints
{

  /**
   * Minimum length of a username.
   */
  public static final int USERNAME_LOWER_LIMIT = 6;

  /**
   * Maximum length of a username.
   */
  public static final int USERNAME_UPPER_LIMIT = 10;

  /**
   * Minimum length of a password.
   */
  public static final int PASSWORD_LOWER_LIMIT = 6;

  /**
   * Maximum length of a password.
   */
  public static final int PASSWORD_UPPER_LIMIT = 10;

  /**
   * Minimum length of a game name.
   */
  public static final int GAMENAME_LOWER_LIMIT = 1;

  /**
   * Maximum length of a game name.
   */
  public static final int GAMENAME_UPPER_LIMIT = 20;

  /**
   * Minimum length of a company title.
   */
  public static final int COMPANY_TITLE_LOWER_LIMIT = 1;

  /**
   * Maximum length of a company title.
   */
  public static final int COMPANY_TITLE_UPPER_LIMIT = 80;



}
