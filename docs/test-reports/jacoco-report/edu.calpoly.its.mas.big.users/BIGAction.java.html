package edu.calpoly.its.mas.big.users;

import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;

import org.apache.struts.Globals;
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessages;

import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;

/**
 * Base class for BIG Actions that access the database.
 * Supports dependency injection of BIGDatabaseFacadeInterface for unit testing.
 * In production, dbFacade is null and getDbFacade() creates a real facade.
 * In tests, setDbFacade(mock) before execute() to use a mock instead.
 *
 * When the ActionServlet is not set (e.g. in unit tests), isCancelled, saveErrors,
 * and saveToken use test-friendly implementations that do not require Struts
 * infrastructure.
 */
public abstract class BIGAction extends Action {

    /** Injected for tests; null in production. */
    protected BIGDatabaseFacadeInterface dbFacade;

    /**
     * Set the database facade (for unit testing). When set, getDbFacade() returns this
     * instead of creating a new BIGDatabaseFacade. Production code never calls this.
     */
    public void setDbFacade(BIGDatabaseFacadeInterface facade) {
        this.dbFacade = facade;
    }

    /**
     * Returns the database facade. If a facade was injected via setDbFacade(), returns it.
     * Otherwise creates a new BIGDatabaseFacade using the request's DataSource.
     */
    protected BIGDatabaseFacadeInterface getDbFacade(HttpServletRequest request) {
        return getDbFacade(request, false);
    }

    /**
     * Returns the database facade, optionally holding the connection.
     * If a facade was injected via setDbFacade(), returns it (holdConnection ignored for mocks).
     */
    protected BIGDatabaseFacadeInterface getDbFacade(HttpServletRequest request, boolean holdConnection) {
        if (dbFacade != null) {
            return dbFacade;
        }
        return new BIGDatabaseFacade(getDataSource(request), holdConnection);
    }

    /**
     * When servlet is null (unit tests), checks the cancel parameter directly.
     * Otherwise delegates to Struts Action.
     */
    @Override
    protected boolean isCancelled(HttpServletRequest request) {
        if (getServlet() == null) {
            return request.getParameter("org.apache.struts.taglib.html.CANCEL") != null;
        }
        return super.isCancelled(request);
    }

    /**
     * When servlet is null (unit tests), stores errors in request attribute directly.
     * Otherwise delegates to Struts Action.
     */
    @Override
    protected void saveErrors(HttpServletRequest request, ActionMessages errors) {
        if (getServlet() == null) {
            request.setAttribute(Globals.ERROR_KEY, errors);
            return;
        }
        super.saveErrors(request, errors);
    }

    /**
     * When servlet is null (unit tests), stores a token in session so saveToken does not throw.
     * Otherwise delegates to Struts Action.
     */
    @Override
    protected void saveToken(HttpServletRequest request) {
        if (getServlet() == null) {
            HttpSession session = request.getSession(false);
            if (session != null) {
                session.setAttribute(Globals.TRANSACTION_TOKEN_KEY, java.util.UUID.randomUUID().toString());
            }
            return;
        }
        super.saveToken(request);
    }
}
