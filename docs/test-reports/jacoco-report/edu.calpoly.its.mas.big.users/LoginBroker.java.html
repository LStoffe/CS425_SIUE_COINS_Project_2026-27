/**
 * <code>LoginBroker</code> Determines whether or not a company may login to a game, and
 * whether or not the company may continue to be logged in whenever the company accesses
 * a BIG jsp page or attempts to submit a BIG form.
 * Reasons why a company is not allowed to be logged in:
 *   1.  Company's Game has not been updated from period 0 to period 1 (i.e. has not yet started).
 *   2.  Game's administrator may be in the process of updating a game from one period to another.
 *   3.  Game may have been updated from one period to another between the time the company made its
 *       last request and the time it makes the current request.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.9 $ $Date: 2002/05/02 18:56:48 $
 *
 */
 public class LoginBroker
 {
   ServletContext context;

  /**
   * Puts a GameLoginInfo object in the ServletContext when a new game is created.  Also used by a
   * servlet called upon from BIG's web.xml <load-on-startup> line to load GameLoginInfo objects
   * into the ServletContext for all games in the database.
   */
  public void addGameLoginInfoToServletContext(Game game, ServletContext context)
  {
	  this.context = context;
	  GameLoginInfo gameLoginInfo = new GameLoginInfo(game);
	  context.setAttribute(Integer.toString(game.getGameID()), gameLoginInfo);
  }


  /**
   * Used when an administrator starts a game, and just after an Administrator has updated a game's period.
   * Updates a game's GameLoginInfo currentperiod count by one, sets the GameLoginInfo's allow logins value
   * to "true," and sets GameLoginInfo's number of companies currently logged in value to zero.
   */
  public synchronized void initializePeriod(Game game, ServletContext context)
  {
	  GameLoginInfo gameLoginInfo = (GameLoginInfo) context.getAttribute(Integer.toString(game.getGameID()));
      gameLoginInfo.gameUpdated();
      GameLoginInfo testGameLoginInfo = (GameLoginInfo) context.getAttribute(Integer.toString(game.getGameID()));

	  Log log = LogFactory.getLog(this.getClass());
	  if(log.isDebugEnabled())
	  {
          log.debug("LoginBroker.initializePeriod(): Logins have been enabled, Period has been updated, numLoggedIn==0");
      	  log.debug("LoginBroker.initializePeriod(): testGameLoginInfo values are:  period=" + testGameLoginInfo.getCurrentPeriod() + ", numLoggedIn=" + testGameLoginInfo.getNumberCurrentlyLoggedIn() + ", loginAllowed=" + testGameLoginInfo.getLoginAllowed());
	  }
  }


  /**

	  }
  }


  /**
   * Returns the number of companies currently logged in to the specified game.
   */
  public synchronized int getNumberOfLoggedInCompanies(Game game, ServletContext context)
  {
	  GameLoginInfo gameLoginInfo = (GameLoginInfo) context.getAttribute(Integer.toString(game.getGameID()));

	  int i = -1;
	  i = gameLoginInfo.getNumberCurrentlyLoggedIn();

	  //System.out.println("LoginBroker: Number of companies logged in is " + i);
	  if(i==-1)
	    context.log("LoginBroker: Error retrieving number of companies logged from the ServletContext for game# " + game.getGameID());
      return i;
  }

  /**
   *  Verify if logins are allowed for a particular company for this game.  This method should be called at the
   *  beginning of every JSP page, and at the beginning of every ActionServlet accessible by companies.
   *  This method checks for the following:
   *    1. Company has been authenticated based on username/password
   *    2. Game for this company is available
   *       A game will not be available if the game has not yet been created
   *    3. Logins are currently allowed for this game
   *       Logins will not be allowed if the admin is currently updating the game's period,
   *       or the game has not yet been started (advanced to period 1).
   *    4. Period number recorded in company's session matches the current game number.
   *       These values may not be equal if the admin has updated the game's period in
   *       between a company's access of the BIG web pages.
   */
  public synchronized boolean verifyGameLoginAllowed(Game game, ServletContext context, HttpServletRequest request)
  {
	 boolean loginAllowed = false;
	 ActionErrors errors = new ActionErrors();
	 HttpSession session= request.getSession();

 	 Log log = LogFactory.getLog(this.getClass());

     // Verify that the user has been authenticated.  If the company has not been authenticated,
     // there will be no game object in the company's session.
     if (game == null)
     {
        //System.out.println("LoginBroker.verifyGameLoginAllowed(): No Game object in the user's session.");
		errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.userNotLoggedIn"));
		loginAllowed=false;
	 }

	 else
	 {
	    // Retrieve the GameLoginInfo object for this game from the ServletContext
		GameLoginInfo gameLoginInfo = (GameLoginInfo) context.getAttribute(Integer.toString(game.getGameID()));


        // If there is no gameLoginInfo in the ServletContext for this Game, the company is not allowed to log in or
        // access BIG web pages. This means the game does not exist, or there was a problem in the LoginInitizationServlet
        // in loading this information from the BIG database upon startup of BIG.
        if (gameLoginInfo == null)  // Invalid Game
        {
           loginAllowed = false;

		   if(log.isDebugEnabled())
		   {
		   	   log.debug("LoginBroker.verifyGameLoginAllowed(): No GameLoginInfo in ServletContext for this Game.");
		   }

           // Add error message to be displayed on "login alert" page
		   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.invalidGame"));
	    }
        else  // Game exists
        {
           if (!gameLoginInfo.getLoginAllowed())  // Logins not allowed for this game at this time
           {
			   if(log.isDebugEnabled())
			   {
				   log.debug("LoginBroker.verifyGameLoginAllowed(): Game Login not allowed for this game at this time");
			   }
               loginAllowed = false;


               // Add error message to be displayed on "login alert" page
               switch (game.getCurrentPeriodNumber())
               {
                  case 0: // Game period = 0: Game has not yet started
				     errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.gameNotStarted"));
                  break;
                  default: // Administrator is updating the period
				     errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.gameBeingAdvanced"));
		       }
	       }
           else // Game has started, and admin is not updating the period
           {
			   // Verify that the period recorded in the user's session equals the period of the game.
			   if (gameLoginInfo.getCurrentPeriod() != game.getCurrentPeriodNumber())  // Company and Game Periods are not equal
			   {
			      loginAllowed = false;

			      if(log.isDebugEnabled())
			   	  {
				      log.debug("gameLoginInfoPeriod= " + gameLoginInfo.getCurrentPeriod() + " and Game.period= " +game.getCurrentPeriodNumber());
				      log.debug("LoginBroker.verifyGameLoginAllowed(): Company and Game Periods are not equal");
			  	  }

                  // Add error message to be displayed on "login alert" page
				  errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("loginError.gameHasBeenAdvanced"));
			   }

			   else  // Company has passed all tests and is allowed to log in!
			   {
			      loginAllowed = true;
			   }
	       }
	    }
     }

     if(errors.size() > 0)
     {
		 request.setAttribute(Globals.ERROR_KEY, errors);
	 }

     return loginAllowed;
  }


  /**
   * Returns the Total number of companies currently logged in to all games.
   */
  private int getTotalNumberOfLoggedInCompanies()
  {
	 Log log = LogFactory.getLog(this.getClass());
	 BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade( (DataSource) context.getAttribute(Globals.DATA_SOURCE_KEY) );
	 int count = 0;

     try
     {
		 // Get all existing games from the database
		 Vector gameidVector = dbFacade.getGameDBHandler().getAllGames();

         //If any games exist, check each to see how many companies are currently logged in
         if((gameidVector != null) && (gameidVector.size() != 0))
         {
             for (Enumeration e = gameidVector.elements() ; e.hasMoreElements() ;)
             {
		        //Game game = dbFacade.getGameDBHandler().getGame( (int) e );
		        int numLogged = getNumberOfLoggedInCompanies( ( (Game) e.nextElement() ), context);
                count = count + numLogged;
             }
	      }
	 }
	 catch (BIGDatabaseFacadeException bdbe)
	 {
  	    log.fatal(bdbe);
	 }
	 finally
	 {
	    try
		{
			dbFacade.close();
		}
	    catch (BIGDatabaseFacadeException bdbe)
	    {
  	       	log.fatal(bdbe);
	    }
	 }
	 return count;
  }
