package edu.calpoly.its.mas.big.users;

//JDK imports
import java.util.Hashtable;

import javax.servlet.ServletContext;

//Struts imports
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;

/**
 * Logs a user in to BIG.
 * If the username/password combination is recognized <code>LoginAuthenticationAction</code> automatically determines if the user is an admin,
 * company, or head admin, and authenticates them.  An <code>Admin</code> or <code>Company</code> is added to the
 * session and the appropriate menu (admin or company) is displayed to the user.
 * <br><br>
 *
 * If the username/password combination is not recognized the user is returned to the login screen.
 *
 * @author Kevin Craig
 *  Modified 4/17/02 Dara Manker
 *
 * @version $Revision: 1.2 $ $Date: 2002/04/19 16:35:57 $
 *
 */
public class LoginAuthenticationAction extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
		           				 ActionForm form,
		           				 HttpServletRequest request,
		           				 HttpServletResponse response)
		{

			BIGLog log =  BIGLogFactory.getLog();
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " beginning execute method");
			}

            ActionMessages errors = new ActionMessages();

           //Get username and password from incoming ActionForm
           //Convert username to lower case for case insensitivty.  All usernames in database are lower case.  Passwords are case sensitive.
	       String username = ((LoginForm) form).getUsername().toLowerCase();
	       String password = ((LoginForm) form).getPassword();

	       User user = null;
	       ActionForward returnValue = mapping.findForward("failure");

		   BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

		   try
		   {

			   user = dbFacade.getAdminDBHandler().getAdmin(username);

			   if (user == null)
			   {
				   user = dbFacade.getCompanyDBHandler().getCompany(username);
			   }

	           if((user != null) && (user.getPassword().equals(password)))
	           {
			      //Set the user as logged in and add it to the session
    		      user.setIsLoggedIn(true);
		          HttpSession session = request.getSession();
		 	      session.setAttribute("user", user);

		 	      if (user instanceof Admin)
		 	      {
					  if(log.isInfoEnabled())
					  {
						  log.info("Admin " + username + " logged in");
					  }

				      //Return the admin menu
				      returnValue = mapping.findForward("admin");
			      }
			      else if (user instanceof Company)
			      {
					  //Add the game this company is in to the session
					  Game game = dbFacade.getGameDBHandler().getGame((Company) user);
					  session.setAttribute("game", game);

                      //Prepare for display of Mission Statement and Core Values
                      Company theCompany = (Company) user;
                      String missionStatement = theCompany.getMissionStatement();
                      String coreValues = theCompany.getCoreValues();
                      String companyUrl = theCompany.getCompanyUrl();

                      ((LoginForm) form).setMissionStatement(missionStatement);
                      ((LoginForm) form).setCoreValues(coreValues);
                      ((LoginForm) form).setCompanyUrl(companyUrl);
					  session.setAttribute("loginform", form );
					  session.setAttribute("company", theCompany);
					  session.setAttribute("isAdminAsCompany", new Integer(0));

					  //Adds to the session a list of current messages for this company
					  //These will be used by the inbox page, immediately after login
					  Vector messages = dbFacade.getInboxDBHandler().retrieveMessages(((Company) user), game.getCurrentPeriodNumber());
					  session.setAttribute("messages", messages);

                      // Put an object in the company's session for use if the company's session times out
                      ServletContext context = servlet.getServletContext();
                      TimeoutCompanySession timeoutSession = new TimeoutCompanySession(context, game.getGameID());
                      session.setAttribute("timeoutSession", timeoutSession);

				      //Set an attribute so that the company's main menu jsp page will know
				      // to increment the number of companies logged in
      		          request.setAttribute("newLogin", "true");

				      //Return the company main menu page
				      returnValue = mapping.findForward("company");


					  if(BIGLogFactory.isInfoEnabled())
					  {
						  log.info("Company " + username + " logged in to game " + game.getGameID());
						  BIGLog gameLog = BIGLogFactory.getGameLog(game.getGameID());

						  gameLog.info("Company " + username + " logged in");
						  BIGLog companyLog = BIGLogFactory.getCompanyLog(((Company) user).getCompanyID());
						  companyLog.info("Company " + username + " logged in");
					  }
			      }

		       }
		       else
		   	   {
				  if(log.isInfoEnabled())
				  {
				  	 log.info("Login failure for username " + username);
				  }
		          errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.login.password.mismatch"));
	           }

               dbFacade.close();

		   }
		   catch (BIGDatabaseFacadeException bdbe)
		   {
			   log.fatal("", bdbe);
			   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		   }
		   catch(ObjectNotFoundException onfe) //possibly thrown from call to dbFacade.getGame method
		   {
			   log.fatal("", onfe);
			   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.game.notFound"));
		   }

	       if (!errors.isEmpty())
		   {
			  if(log.isDebugEnabled())
			  {
			      log.debug(this.getClass() + " request returned with errors");
			  }

		   	  // Report any errors we have discovered back to the original form
		   	  saveErrors(request, errors);
		   	  saveToken(request);
		   	  return (new ActionForward(mapping.getInput()));
		   }
		   else
		   {
			  if(log.isDebugEnabled())
			  {
			      log.debug(this.getClass() + " request completed.  Forwarding to " + returnValue);
			  }
 			  return (returnValue);
		   }

	   } // end execute()

}
