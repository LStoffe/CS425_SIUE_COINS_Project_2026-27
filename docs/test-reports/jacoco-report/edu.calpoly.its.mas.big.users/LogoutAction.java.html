/**
 * Logs a user out of BIG.
 * Invalidates the user's session.
 *
 * @version $Revision: 1.8 $ $Date: 2002/04/19 14:44:32 $
 */
public class LogoutAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
			           			 ActionForm form,
			           			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{

		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");
        User user = (User) session.getAttribute("user");

		if (user == null)  // user object no longer exists because the session has timed out
		   return (mapping.findForward("loginFailure"));

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

		if( user instanceof Company )
		{
		   //Decrement the number of companies currently logged in
		   game = (Game) session.getAttribute("game");
		   context = servlet.getServletContext();
           loginBroker = (LoginBroker) context.getAttribute("loginBroker");
           loginBroker.decrementNumberOfLoggedInCompanies(game, context);

           if(log.isDebugEnabled())
           {
               log.debug("LogoutAction: number of logged-in companies decremented.");
		   }
           session.removeAttribute("timeoutSession");
		}

        session.removeAttribute("user");
        session.removeAttribute("game");
        session.invalidate();


 	    if(log.isInfoEnabled())
	    {
			log.info("LogoutAction: User " + user.getUsername() + " has logged out");
	    }


		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
		}
        return (mapping.findForward("success"));
    }

}
