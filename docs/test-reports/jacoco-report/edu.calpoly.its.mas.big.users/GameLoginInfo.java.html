package edu.calpoly.its.mas.big.users;

import edu.calpoly.its.mas.big.game.Game;

/**
 * <code>GameLoginInfo</code> is used to determine whether or not companies should be
 * allowed to login to a game, and how many companies are currently logged in to a game.
 * One GameLoginInfo object for each game is stored in BIG's ServletContext.
 * The ability of companies to log in is based on the loginAllowed=true property and
 * the condition that the currentPeriod stored in the company's session matches the
 * currentPeriod in this object.  The value of loginAllowed is set to true if the
 * Administrator has started the game (period > 0), and if the administrator is not
 * in the process of advancing the period of a game.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.3 $ $Date: 2002/04/15 20:10:41 $
 *
 */
 public class GameLoginInfo
 {

    private int gameID;
    private int currentPeriod;
    private boolean loginAllowed;
    private int numberCurrentlyLoggedIn;

       public GameLoginInfo(Game game)
       {
	      this.gameID = game.getGameID();
	      this.currentPeriod = game.getCurrentPeriodNumber();

	      // Determine if users should be allowed to log in to this game
	      if(game.getCurrentPeriodNumber() == 0)
		     this.loginAllowed = false;
		  else
		     this.loginAllowed = true;

	      this.numberCurrentlyLoggedIn = 0;
       }


       /**
       * Returns the GameID number.
       */
       public synchronized int getGameID()
       {
		   return this.gameID;
	   }

       /**
       * Sets the GameID number.
       */
       public synchronized void setGameID(int i)
       {
		   this.gameID = i;
	   }


       /**
       * Returns the current period.
       */
       public synchronized int getCurrentPeriod()
       {
		   return this.currentPeriod;
	   }

       /**
       * Sets the current period.
       */
       public synchronized void setCurrentPeriod(int i)
       {
		   this.currentPeriod = i;
	   }

       /**
       * Increments the current period by one.
       */
       public synchronized void incrementCurrentPeriod()
       {
          this.currentPeriod++;
       }



       /**
       * Returns status of loginAllowed.
       */
       public synchronized boolean getLoginAllowed()
       {
		   return this.loginAllowed;
	   }

       /**
       * Sets the value of loginAllowed.
       */
       public synchronized void setLoginAllowed(boolean loginValue)
       {
		   this.loginAllowed = loginValue;
	   }


       /**
       * Enables ability of companies to login.
       */
       public synchronized void enableLogins()
       {
          this.loginAllowed = true;
       }

       /**
       * Disables ability of companies to login.
       */
       public synchronized void disableLogins()
       {
          this.loginAllowed = false;
       }


       /**
       * Returns the number of companies currently logged in.
       */
       public synchronized int getNumberCurrentlyLoggedIn()
       {
		   return this.numberCurrentlyLoggedIn;
	   }

       /**
       * Sets the number of companies currently logged in.
       */
       public synchronized void setNumberCurrentlyLoggedIn(int i)
       {
		   if(i < 0)
		   {
		      this.numberCurrentlyLoggedIn = 0;
		   }
		   else
		   {
		      this.numberCurrentlyLoggedIn = i;
		   }
	   }

	   /**
       * Increments the number of companies currently logged in by one.
       */
       public synchronized void incrementNumberCurrentlyLoggedIn()
       {
          this.numberCurrentlyLoggedIn++;
       }

       /**
       * Decrements the number of companies currently logged in by one.
       */
       public synchronized void decrementNumberCurrentlyLoggedIn()
       {
          this.numberCurrentlyLoggedIn--;
          if(this.numberCurrentlyLoggedIn < 0)
          {
			  this.numberCurrentlyLoggedIn = 0;
		  }
       }

       /**
       * Increments the current period by one, sets the number of companies logged in to zero,
       * and enables company logins for this game.
       */
       public synchronized void gameUpdated()
       {
          this.currentPeriod++;
	      this.numberCurrentlyLoggedIn = 0;
          this.loginAllowed = true;
       }

  }
