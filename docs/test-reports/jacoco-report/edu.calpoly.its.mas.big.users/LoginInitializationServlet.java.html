package edu.calpoly.its.mas.big.users;

//JDK imports
import java.util.Enumeration;
import java.util.Vector;

//Serlvet imports
import javax.servlet.ServletException;
import javax.servlet.UnavailableException;
import javax.servlet.http.HttpServlet;
import javax.servlet.ServletContext;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import javax.sql.DataSource;

//Struts imports
import org.apache.struts.Globals;

//BIG imports
import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.game.Game;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * <code>LoginInitializationServlet</code> is used to load a GameLoginInfo object in to the
 * servlet context for each game found in the BIG database.  GameLoginInfo objects are used
 * to determine whether or not a company may log in, and to determine how many companies are
 * currently logged in to a game.  LoginInitializationServlet is run upon BIG startup by inclusion
 * in the web.xml's <load-on-startup> line.
 *
 * @author Dara Manker
 *
 * @version $Revision: 1.2 $ $Date: 2002/04/12 17:44:08 $
 */

public class LoginInitializationServlet extends HttpServlet
{


    /**
     * Loads one GameLoginInfo object into the servlet context for each game found in the
     * BIG database.
     */
    public void init() throws ServletException
    {
        // select log4J as the logging implementation for iBatis -- method must preceed any calls to iBatis
        com.ibatis.common.logging.LogFactory.selectLog4JLogging();

		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " initializing LoginInitializationServlet");
		}

        BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade( (DataSource) getServletContext().getAttribute(Globals.DATA_SOURCE_KEY) );

        // Instantiate the one and only LoginBroker for BIG
        LoginBroker loginBroker = new LoginBroker();
        ServletContext context = getServletContext();
        context.setAttribute("loginBroker", loginBroker);

        try
	    {
		    // Get all existing games from the database
		    Vector gameidVector = dbFacade.getGameDBHandler().getAllGames();
            dbFacade.close();

            //If any games exist, have LoginBroker put the GameLoginInfo objects into the ServletContext
            if((gameidVector != null) && (gameidVector.size() != 0))
            {

                for (Enumeration e = gameidVector.elements() ; e.hasMoreElements() ;)
                {
		  	        loginBroker.addGameLoginInfoToServletContext(((Game) e.nextElement()), context);
                }
	         }

	    }
	    catch (BIGDatabaseFacadeException bdbe)
	    {
  	       log.fatal(bdbe);
	    }
	    finally
	    {
			try
			{
				dbFacade.close();
			}
	    	catch (BIGDatabaseFacadeException bdbe)
	    	{
  	       		log.fatal(bdbe);
	    	}
		}
    }
}
