package edu.calpoly.lib.multimedia.big.users.company;

/**
 * Bond limits must have a minimum value in order to prevent
 * a company being shut out from bidding because
 * they're bond limits are too low for any possible job.
 * This class define the values PER_JOB_LIMIT_MIN and WIP_LIMIT_MIN
 * that are the minimum possible values for the per job bond limit and the
 * wip bond limit respectively.  Additionally two methods are defined
 * that will apply these minimums to a calculated bond limit and return
 * the appropriate value.  As of 4/15/04, PER_JOB_LIMIT_MIN is $3,000,000
 * and WIP_LIMIT_MIN is $5,000,000.
 */
public class BondLimitConstraints
{
	public static final long PER_JOB_LIMIT_MIN = 3000000;
    public static final long WIP_LIMIT_MIN = 5000000;

    public static long applyPerJobLimitMinimum(long perJobLimit)
    {
		return Math.max(PER_JOB_LIMIT_MIN, perJobLimit);
	}

	public static long applyWIPLimitMinimum(long wipLimit)
	{
		return Math.max(WIP_LIMIT_MIN, wipLimit);
	}
}
