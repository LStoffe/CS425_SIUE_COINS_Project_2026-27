package edu.calpoly.lib.multimedia.big.users.company;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

public class SubmitBillAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
					       		 ActionForm form,
					       		 HttpServletRequest request,
				           		 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
	    if(!isCancelled(request))
	    {

	        ActionMessages errors = new ActionMessages();

	        // Check for existance of Game object in user's session
            ServletContext context = servlet.getServletContext();
            HttpSession session = request.getSession();
	        Game game = (Game) session.getAttribute("game");
	        Company company = (Company) session.getAttribute("company");

            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

            // If for any reason the company should not be allowed to login, route the
            // company to an error message page.
	        if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		      return (mapping.findForward("loginFailure"));

	        BillingForm billingForm = (BillingForm) form;
	        BillingForm oldBillingForm = (BillingForm) session.getAttribute("oldBillingForm");
	        Job job = (Job) session.getAttribute("job");


     		//Check which submit button was pressed
			if(	request.getParameter("calculate") != null )
			{
				int billingAmount = 0;
				int[] activityAmounts = billingForm.getActivityAmounts();
				for(int i = 0; i < activityAmounts.length; i++)
				{
					billingAmount += activityAmounts[i];
				}

				billingForm.setBillingAmount(billingAmount);
				session.setAttribute("billingForm", billingForm);

				if (!errors.isEmpty())
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request returned with errors");
					}

					// Report any errors we have discovered back to the original form
					saveErrors(request, errors);
	 	    		saveToken(request);
	 	    		return (new ActionForward(mapping.getInput()));
				}
				else
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request completed.  Forwarding to \"calculated\"");
					}
					return mapping.findForward("calculated");
				}
			}
			else
			{
				BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));
            	try
            	{
					int prevRetention = job.calculateRetainedAmount(oldBillingForm.getBillingAmount());
					int currRetention = job.calculateRetainedAmount(billingForm.getBillingAmount());

					BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, game.getCurrentPeriodNumber() -1, company);
         		    ContractsInProgressReportForm contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, game.getCurrentPeriodNumber() -1, company);
 		 	        CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, game.getCurrentPeriodNumber() -1, company);

				    balanceSheet.setUnderbilledAmount(contractsInProgress.getTotalUnderbilling() + oldBillingForm.getBillingAmount() - billingForm.getBillingAmount());  //This new value won't be saved, it's only for this object
				    balanceSheet.setUnbilledAmount(completedContracts.getTotalUnbilled());

				    balanceSheet.setOverbilledAmount(contractsInProgress.getTotalOverbilling() );
					balanceSheet.setCurrentAccountsReceivable(balanceSheet.getCurrentAccountsReceivable() + (billingForm.getBillingAmount() - currRetention) - (oldBillingForm.getBillingAmount() - prevRetention));

					balanceSheet.setRetentionReceivable(balanceSheet.getRetentionReceivable() + currRetention - prevRetention);
					balanceSheet.setPeriod(game.getCurrentPeriodNumber() - 1);

					billingForm.setOldAmountBilled(oldBillingForm.getBillingAmount() );

					if(balanceSheet.assertBalance())
					{
						dbFacade.getBillingDBHandler().updateBillingForm(game, company, game.getCurrentPeriodNumber() -1, billingForm);
						dbFacade.getBalanceSheetDBHandler().updateBalanceSheet(game, company, balanceSheet);
					}
					else
					{
						if(log.isErrorEnabled())
						{
							log.error(this.getClass() + ": Balance Sheet does not balance for company id " + company.getCompanyID() + " in period " + game.getCurrentPeriodNumber() + " - " + balanceSheet.getTotalAssets() + " to " + balanceSheet.getTotalLiabilitiesAndEquity() );
						}
                        dbFacade.close();
						return mapping.findForward("error");
					}

					dbFacade.close();
				}
				catch (BIGDatabaseFacadeException bigdbe)
				{
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch (ObjectNotFoundException onfe)
				{
					log.fatal("", onfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}
				finally
				{
					try
					{
      	   				dbFacade.close();
					}
					catch (BIGDatabaseFacadeException e)
					{
						log.fatal(getClass().getName(), e);
 						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
					}
				}

				session.setAttribute("oldBillingForm", billingForm);

				if (!errors.isEmpty())
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request returned with errors");
					}

					// Report any errors we have discovered back to the original form
					saveErrors(request, errors);
	 	    		saveToken(request);
	 	    		return (new ActionForward(mapping.getInput()));
				}
				else
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
					}
					return (mapping.findForward("success"));
				}
			}
	    }
	    else
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}
}
