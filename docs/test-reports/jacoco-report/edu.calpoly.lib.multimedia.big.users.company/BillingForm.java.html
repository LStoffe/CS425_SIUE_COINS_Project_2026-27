/**
 * Holds data for a job billings.  The info associated with a billing is
 * the amount billed on each activity, and the total amount to bill.  The amount
 * billed on each activity is not used by BIG, but is stored as a convenience to companies
 * so they can recall the amounts they planned to bill on each activity.  The total billing
 * amount is used by BIG as the amount to attempt to bill.  If this number is not
 * outrageously high (as defined by game parameters) this amount will be paid to the company
 * as cash in the next period.
 * <br><br>
 * The billing form also makes use of the <code>Job</code> object and the Contracts in Progress Report
 * so values contained in these reports that relates to billing can be displayed in the
 * billing form.
 *
 * @author Kevin Craig
 */

	public void setContractsInProgressReport(ContractsInProgressReportForm contractsInProgress)
	{
		if(contractsInProgress != null)
		{
			boolean correctContract = false;
			while(contractsInProgress.moreContracts() && !correctContract)

				{
					correctContract = true;
					contract = contractsInProgress.getCurrentContract();
				}
			}
		}
	}

	/**
	 * If the underlying Job object is an instance of NegotiatedJob,
	 * this method returns the cost spend on the job so far
	 * plus the percent profit on the job, or if the
	 * cost spend so far is greater than the agreed cost
	 * of the job, returns the maximum amount the company may earn on this
	 * job.  If the Job is not
	 * a NegotiatedJob, this method returns "N/A".
	 */
	public String getNegotiatedJobAmountEarnedSoFar()
	{
		if(!negotiatedJob())
		{
			return "N/A";
		}
		else
		{
			NegotiatedJob nj = (NegotiatedJob) currentJob;
			if(contract.getCostIncurred() > nj.getAgreedCost())
			{
				return NumberConverter.stringValue(nj.getMaxPrice());
			}
			else
			{
				return NumberConverter.stringValue(nj.getCostPlusProfit(contract.getCostIncurred()));
			}
		}
	}

	/**
	 * If the underlying Job object is an instance of NegotiatedJob,
	 * and the job is complete,
	 * this method returns maximum amount that the company
	 * can bill for.  If the Job is not
	 * a NegotiatedJob, this method returns "N/A".
	 */
	public String getNegotiatedJobCompleteAmount()
	{
		if(!negotiatedJob())
		{
			return "N/A";
		}
		else if(!isJobComplete())
		{
		    return "N/A";
		}
		else
		{
			NegotiatedJob nj = (NegotiatedJob) currentJob;
			if(contract.getCostIncurred() > nj.getAgreedCost())
			{
				if(log.isDebugEnabled())
				{
					log.debug("BillingForm.getNegotiatedJobCompleteAmount: " + contract.getCostIncurred() + " " + nj.getAgreedCost() + " (max price)");
				}
				return NumberConverter.stringValue(nj.getMaxPrice() + getShareOfSavings());
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug("BillingForm.getNegotiatedJobCompleteAmount: " + contract.getCostIncurred() + " " + nj.getAgreedCost() + " (current price)");
				}
				return NumberConverter.stringValue(nj.getCostPlusProfit(contract.getCostIncurred()) + getShareOfSavings());
			}
		}
	}

	public String getNegotiatedJobEstimatedBillingAmount()
	{
		if(!negotiatedJob())
		{
			return "N/A";
		}
		else if(!isJobComplete())
		{
			NegotiatedJob nj = (NegotiatedJob) currentJob;
			if(contract.getCostIncurred() > nj.getAgreedCost())
			{
				return NumberConverter.stringValue(nj.getMaxPrice() - contract.getBilledToDate());
			}
			else
			{
				return NumberConverter.stringValue(nj.getCostPlusProfit(contract.getCostIncurred()) - contract.getBilledToDate());
			}
		}
		else
		{
			NegotiatedJob nj = (NegotiatedJob) currentJob;
			if(contract.getCostIncurred() > nj.getAgreedCost())
			{
				if(log.isDebugEnabled())
				{
					log.debug("BillingForm.getNegotiatedJobEstimatedBillingAmount: " + contract.getCostIncurred() + " " + nj.getAgreedCost() + " (max price)");
				}
				return NumberConverter.stringValue(nj.getMaxPrice() - contract.getBilledToDate() + getShareOfSavings());
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug("BillingForm.getNegotiatedJobEstimatedBillingAmount: " + contract.getCostIncurred() + " " + nj.getAgreedCost() + " (current cost)");
				}
				return NumberConverter.stringValue(nj.getCostPlusProfit(contract.getCostIncurred()) + getShareOfSavings() - contract.getBilledToDate());
			}
		}
	}

	/**
	 * If the underlying Job object is an instance of NegotiatedJob,
	 * and the job is complete, this method returns the savings on the job.
	 * If the Job is not
	 * a NegotiatedJob, this method returns 0.
	 */
	public long getSavings()
	{
		if(!negotiatedJob())
		{
			return 0;
		}
		else if (!isJobComplete())
		{
			return 0;
		}
		else
		{
			NegotiatedJob nj = (NegotiatedJob) currentJob;
			if(contract.getCostIncurred() < nj.getAgreedCost())
			{
				return nj.getAgreedCost() - contract.getCostIncurred();
			}
			else
			{
				return 0;
			}
		}
	}

	private void intsToStrings()
	{
		for(int i = 0; i < 9; i++)
		{
			activityAmountStrings[i] = NumberConverter.stringValueCommasOnly(activityAmounts[i]);
		}
	}

	private void stringsToInts()
	{
		for(int i = 0; i < 9; i++)
		{
			if(!NumberConverter.isValidMoneyString(activityAmountStrings[i]))
			{
				throw new NumberFormatException();
			}
			activityAmounts[i] = NumberConverter.intValue(activityAmountStrings[i]);
		}
	}

	/**
	 * Ensures all values are non-null, all number strings contain only numeric characters,
     * all number are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{

        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;

        //First make sure none of the Strings are null
        for(int i = 0; i < 9; i++)
        {
			if((activityAmountStrings[i] == null) || (activityAmountStrings[i].trim().length() < 1))
				nullParams = true;
			if((billingAmountString == null ) || (billingAmountString.trim().length() < 1))
			    nullParams = true;
		}

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the Strings contain only numeric characters.
			for(int i = 0; i < 9; i++)
			{
				//Check that the strings are valid numbers by trying to convert them to ints and checking for a NumberFormatException
				//Note that this also populated the int arrays
				try
				{
					activityAmounts[i] = NumberConverter.intValue(activityAmountStrings[i]);
				}
				catch(NumberFormatException nfe)
				{
					validNumbers = false;
				}

			}
			try
			{
				billingAmount = NumberConverter.intValue(billingAmountString);
			}
			catch(NumberFormatException nfe)
			{
				validNumbers = false;
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			for(int i = 0; i < 9; i++)
			{
				if(activityAmounts[i] < 0)
					positiveNumbers = false;
			}
			if(billingAmount < 0)
			{
				positiveNumbers = false;
			}
		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		return errors;
	}
