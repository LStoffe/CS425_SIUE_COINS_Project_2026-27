/**
 * Updates the currently selected method for how a loan is to be repaid.
 * BIG v3.2 supports only two methods for loan repayment: Periodic payments,
 * as implemented in previous versions of BIG, and immediate repayments.
 * This action sets the repayment method for a loan currently contained
 * in a <code>Vector</code> in the session, under the name "loans".  The query string must
 * contain the index of the correct loan within the <code>Vector</code> and the
 * option ("immediate" or "periodic") for repayment.
 *
 * @author Kevin Craig
 */
public class SetLoanRepaymentOptionAction extends Action
{

	public ActionForward execute(ActionMapping mapping,
					             ActionForm form,
					             HttpServletRequest request,
				                 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		    return (mapping.findForward("loginFailure"));

        Company company = (Company) session.getAttribute("company");
		Vector loans = (Vector) session.getAttribute("loans");

		String option = request.getParameter("option");
		int index = -1;

		try
		{
			index = Integer.parseInt(request.getParameter("loanIndex"));
		}
		catch(NumberFormatException nfe)
		{
			//do nothing, the error will be discovered in a moment when the index is checked
			//against the array bounds
		}

		//If anything required is missing, return the error page
		if((loans == null) || (index < 0) || (index >= loans.size()) || (option == null) )
		{
			return mapping.findForward("error");
		}

		//Get the loan from the Vector
		Loan loan = (Loan) loans.elementAt(index);

		//Set the correct option in the loan, or return the error page
		//if an invalid option was selected
		if(option.equals("immediate"))
		{
			loan.setRepayImmediately(true);
		}
		else if(option.equals("periodic"))
		{
			loan.setRepayImmediately(false);
		}
		else
		{
			return mapping.findForward("error");
		}


        //Now that the Loan object has been updated, save the changes to the database
        try
        {
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));
			dbFacade.getLoanDBHandler().updateLoan(loan);

			dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}


		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}

			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}
	}
}
