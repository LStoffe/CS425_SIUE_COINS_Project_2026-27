public class ConsiderNegotiationOfferAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

		Company company = (Company) session.getAttribute("company");
		NegotiatedJob job = (NegotiatedJob) session.getAttribute("jobUnderNegotiation");

		//If no job is set simply return to the main page.
		//No need for an error message, as this is only possible if someone
		//follows a bookmark, in which case they will simply be sent to
		//where they need to be to choose a job.
		if(job == null)
		{
	 	    return (new ActionForward(mapping.getInput()));
		}

		NegotiationOfferForm negotiationOfferForm = (NegotiationOfferForm) form;
		int percentProfit = negotiationOfferForm.getPercentProfit();
		long cost = negotiationOfferForm.getCost();

		//Hashtable to store the current negotiation values
		Hashtable negotiationValues = null;
		Hashtable negotiationParameters = null;
		Hashtable cccParameters = null;
		Hashtable misc = null;

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
		try
		{
			negotiationValues = dbFacade.getNegotiationDBHandler().getNegotiationValues(game, job.getJobID());
			negotiationParameters = dbFacade.getGameParametersDBHandler().getNegotiationParameters(game);
		    cccParameters = dbFacade.getGameParametersDBHandler().getComputerControlledContractorParameters(game);
		    misc = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
			dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ParametersNotFoundException pnfe)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ObjectNotFoundException onfe)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}

		//Check if the target profit and haggling values have already been set.  If not compute them and set them
        int targetProfit = ((Integer) negotiationValues.get("customerTargetProfit")).intValue();
        int customerHaggling = ((Integer) negotiationValues.get("customerHaggling")).intValue();

        if(targetProfit < 0)
        {
			targetProfit = computeCustomerTargetProfit(game, negotiationParameters);
			negotiationValues.put("customerTargetProfit", new Integer(targetProfit));
		}
		if(customerHaggling < 0)
		{
			customerHaggling = computeCustomerHaggling(game, negotiationParameters);
			negotiationValues.put("customerHaggling", new Integer(customerHaggling));
		}

        if(log.isDebugEnabled())
        {
        	log.debug("Company " + company.getCompanyID() + ": Customer Haggling: " + customerHaggling + " Target Profit: " + targetProfit);
		}

	    Random r = new Random();
		//First check if cost is below max cost customer will pay for this job

		long maxDirectCost = computeMaxDirectCost(game, job, cccParameters, misc);

		if(cost > maxDirectCost)
		{
			//Negotiation will continue based on the customer haggling value
			if(r.nextInt(100) < customerHaggling)
			{
				job.setState(NegotiatedJob.NEGOTIATION_FAILED);
			}
			else
			{
			    job.setState(NegotiatedJob.COST_REJECTED);
			}
		}
		else
		{
			//Check profit
			int maxPercentProfit = ((Integer) negotiationParameters.get("MaxPercentProfit")).intValue() * 10;
			//System.out.println("PP " + percentProfit + " mPP " + maxPercentProfit);

			//First make sure profit offer is not more than max profit for negotiated work in BIG
			if(percentProfit > maxPercentProfit)
			{
			    //Negotiation will continue based on the customer haggling value
			    if(r.nextInt(100) < customerHaggling)
				{
					job.setState(NegotiatedJob.NEGOTIATION_FAILED);
				}
				else
				{
				    job.setState(NegotiatedJob.COST_REJECTED);
				}
			}
			else
			{
				//This is the amount the customer will suggest if this offer is
				//rejected.  Go ahead and use it for comparison to avoid cases
				//where customer rejects offer then suggests a higher amount
				targetProfit += r.nextInt(6);
			    if(percentProfit > targetProfit)
			    {
					//System.out.println("percent profit greater than target profit");

			    	//Get random value between 0 and 100.  If greater than
			    	//customer haggling, customer will accept
			    	if(r.nextInt(100) > customerHaggling)
			    	{
		                job.setState(NegotiatedJob.NEGOTIATION_SUCCEEDED);
					    //System.out.println("accepted on haggling");
					}
					else
			    	{
						negotiationValues.put("customerTargetProfit", new Integer(targetProfit));
						((NegotiationOfferForm) form).setPercentProfit(targetProfit);

			    		//Negotiation will continue based on the customer haggling value
			    		if(r.nextInt(100) < customerHaggling)
						{
							job.setState(NegotiatedJob.NEGOTIATION_FAILED);
						}
						else
						{
						    job.setState(NegotiatedJob.COST_REJECTED);
						}
			       	    //System.out.println("rejected on haggling new target profit :" + targetProfit);
				    }
			    }
			    else
			    {
					//System.out.println("percent profit less than target profit");
		            job.setState(NegotiatedJob.NEGOTIATION_SUCCEEDED);
			    }
			}
		}


		job.setAgreedCost(cost);
		job.setAgreedProfit(percentProfit);


		session.setAttribute("jobUnderNegotiation", job);

		//Update the latest negotiation values in the database
		try
		{
			negotiationValues.put("agreedCost", new Long(cost));
			negotiationValues.put("agreedProfit", new Integer(percentProfit));
			negotiationValues.put("state", new Integer(job.getState()));

			dbFacade.getNegotiationDBHandler().setNegotationValues(game, job.getJobID(), negotiationValues);
			dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ObjectNotFoundException onfe)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}

		if (!errors.isEmpty())
		{
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);

	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(job.getState() == NegotiatedJob.NEGOTIATION_SUCCEEDED || job.getState() == NegotiatedJob.NEGOTIATION_FAILED || job.getState() == NegotiatedJob.NEGOTIATION_CANCELLED)
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"complete\"");
				}
			    return (mapping.findForward("complete"));
		    }
		    else
		    {
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"continue\"");
				}
			    return (mapping.findForward("continue"));
			}
		}
	}

    //Computes the maximum direct cost this company will pay
    private long computeMaxDirectCost(Game game, Job job, Hashtable cccParameters, Hashtable misc)
    {
		long maxCost = setupCCCCommandCenter(cccParameters, misc).estimateDirectJobCost(game.getGameID(), job.getJobID(), job.getDirectCost(), job.getEstimatedWorkdays());
		maxCost += Math.round(.5d * ((double) maxCost));
		//System.out.println("max direct cost is " + maxCost);
		return maxCost;
	}

	//Computes the customer target profit based on the negotiation parameters from
	//the database
	private int computeCustomerTargetProfit(Game game, Hashtable negotiationParameters)
	{
		int retValue = 0;

		//Negotiation parameter values.  times 10 because company's profit choices are stored as 10 times actual value
		int meanCustomerTargetProfit = ((Integer) negotiationParameters.get("MeanPercentProfit")).intValue() * 10;
		float stddevCustomerTargetProfit = ((Float) negotiationParameters.get("StdDevPercentProfit")).floatValue() * 10;
		int minCustomerTargetProfit = ((Integer) negotiationParameters.get("MinPercentProfit")).intValue() * 10;
		int maxCustomerTargetProfit = ((Integer) negotiationParameters.get("MaxPercentProfit")).intValue() * 10;

		retValue = Statistics.normalInt(meanCustomerTargetProfit, stddevCustomerTargetProfit, minCustomerTargetProfit, maxCustomerTargetProfit);

		return retValue;
	}

	//Computes the customer haggling value based on the negotiation parameters from
	//the database
	private int computeCustomerHaggling(Game game, Hashtable negotiationParameters)
	{
		int retValue = 0;

		//Negotiation parameter values.
		int meanCustomerHaggling = ((Integer) negotiationParameters.get("MeanCustomerHaggling")).intValue();
		float stddevCustomerHaggling = ((Float) negotiationParameters.get("StdDevCustomerHaggling")).floatValue();
		int minCustomerHaggling = ((Integer) negotiationParameters.get("MinCustomerHaggling")).intValue();
		int maxCustomerHaggling = ((Integer) negotiationParameters.get("MaxCustomerHaggling")).intValue();

		retValue = Statistics.normalInt(meanCustomerHaggling, stddevCustomerHaggling, minCustomerHaggling, maxCustomerHaggling);

		return retValue;
	}

	private CCCCommandCenter setupCCCCommandCenter(Hashtable cccParameters, Hashtable misc)
	{
		  int minNumberCCCs = ((Integer) cccParameters.get("minNumberOfCCCs")).intValue();
		  int maxNumberCCCs = ((Integer) cccParameters.get("maxNumberOfCCCs")).intValue();
		  int minCCCBidsPerJob = ((Integer) cccParameters.get("minNumberOfCCCBidsPerJob")).intValue();
		  int maxCCCBidsPerJob = ((Integer) cccParameters.get("maxNumberOfCCCBidsPerJob")).intValue();
		  int cccTurnover = ((Integer) cccParameters.get("CCCTurnover")).intValue();
		  int lowerPercentProfit = ((Integer) cccParameters.get("lowerPercentJobMarkup")).intValue();
		  int upperPercentProfit = ((Integer) cccParameters.get("upperPercentJobMarkup")).intValue();
		  int fieldOverheadFixed = ((Integer) misc.get("fieldOverheadFixed")).intValue();
		  int fieldOverheadVariable = ((Integer) misc.get("fieldOverheadVariable")).intValue();
		  int officeOverheadFixed = ((Integer) misc.get("officeOverheadFixed")).intValue();
		  int bidExpensePercent = ((Integer) misc.get("bidExpensePercent")).intValue();
		  int workdaysPerMonth = ((Integer) misc.get("workdaysPerMonth")).intValue();


		  return new CCCCommandCenter(minNumberCCCs,
		  										  maxNumberCCCs,
		  										  minCCCBidsPerJob,
		  										  maxCCCBidsPerJob,
		  										  cccTurnover,
		  										  lowerPercentProfit,
		  										  upperPercentProfit,
		  										  fieldOverheadFixed,
		  										  fieldOverheadVariable,
		  										  officeOverheadFixed,
		  										  bidExpensePercent,
		  										  workdaysPerMonth,
		  										  10 //right now there are 10 hardcoded job types.
			  									 );
	}
