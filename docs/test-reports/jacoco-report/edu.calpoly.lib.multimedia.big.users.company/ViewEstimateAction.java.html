package edu.calpoly.lib.multimedia.big.users.company;

import org.apache.struts.action.Action;

/**
 */
public class ViewEstimateAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}


	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

		Company company = (Company) session.getAttribute("company");

		String jobidString = (String) request.getParameter("jobid");
		int jobid = -1;

		try
		{
			jobid = Integer.parseInt(jobidString);
		}
		catch(Exception e)
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + ": could not determine id of job");
			}
	 	   	return (new ActionForward(mapping.getInput()));
		}

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
		CCCCommandCenter ccccommandCenter = null;
		int[] activityMethods = null;
		boolean estimateAlreadyAvailable = false;
		int estimatesAvailablePerPeriod = 0;
		int estimatesThisPeriod = 0;
		Job job = null;

		try
		{
		    Hashtable cccParameters = dbFacade.getGameParametersDBHandler().getComputerControlledContractorParameters(game);
		    Hashtable misc = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
		    int minNumberCCCs = ((Integer) cccParameters.get("minNumberOfCCCs")).intValue();
		    int maxNumberCCCs = ((Integer) cccParameters.get("maxNumberOfCCCs")).intValue();
		    int minCCCBidsPerJob = ((Integer) cccParameters.get("minNumberOfCCCBidsPerJob")).intValue();
		    int maxCCCBidsPerJob = ((Integer) cccParameters.get("maxNumberOfCCCBidsPerJob")).intValue();
		    int cccTurnover = ((Integer) cccParameters.get("CCCTurnover")).intValue();
		    int lowerPercentProfit = ((Integer) cccParameters.get("lowerPercentJobMarkup")).intValue();
		    int upperPercentProfit = ((Integer) cccParameters.get("upperPercentJobMarkup")).intValue();
		    int fieldOverheadFixed = ((Integer) misc.get("fieldOverheadFixed")).intValue();
		    int fieldOverheadVariable = ((Integer) misc.get("fieldOverheadVariable")).intValue();
		    int officeOverheadFixed = ((Integer) misc.get("officeOverheadFixed")).intValue();
		    int bidExpensePercent = ((Integer) misc.get("bidExpensePercent")).intValue();
		    int workdaysPerMonth = ((Integer) misc.get("workdaysPerMonth")).intValue();


		    ccccommandCenter = new CCCCommandCenter(minNumberCCCs,
		  										    maxNumberCCCs,
		  										    minCCCBidsPerJob,
		  										    maxCCCBidsPerJob,
		  										    cccTurnover,
		  										    lowerPercentProfit,
		  										    upperPercentProfit,
		  										    fieldOverheadFixed,
		  										    fieldOverheadVariable,
		  										    officeOverheadFixed,
		  										    bidExpensePercent,
		  										    workdaysPerMonth,
		  										    10 //right now there are 10 hardcoded job types.
			  									   );


			int[] oldJobIDs = dbFacade.getEstimatingComponentDBHandler().estimatesAvailableForCompany(game, company);

			for(int i = 0; i < oldJobIDs.length; i++)
			{
				Job j = dbFacade.getJobDBHandler().getJob(game, oldJobIDs[i]);
				if(j.getPeriodCreated() == game.getCurrentPeriodNumber())
				{
					estimatesThisPeriod++;
				}

				if(jobid == j.getJobID())
				{
					estimateAlreadyAvailable = true;
				}
			}

			estimatesAvailablePerPeriod = dbFacade.getEstimatingComponentDBHandler().getEstimatesPerPeriod(game);

			job = dbFacade.getJobDBHandler().getJob(game, jobid);
			activityMethods = dbFacade.getEstimatingComponentDBHandler().getScheduleMethods(game, job);

			if(!estimateAlreadyAvailable)
			{
				dbFacade.getEstimatingComponentDBHandler().makeEstimateAvailableForCompany(game, company, job);
			}

			dbFacade.close();

		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
	    catch(ObjectNotFoundException onfe)
	    {
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
	    }
	    catch(ParametersNotFoundException pnfe)
	    {
			log.fatal("", pnfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
	    }

	    boolean b = estimatesAvailablePerPeriod > estimatesThisPeriod;
	    if(estimateAlreadyAvailable)
	    {
			b = true;
		}

		session.setAttribute("estimateAvailable", new Boolean(b));
		session.setAttribute("estimatedCost", new Long(ccccommandCenter.estimateDirectJobCost(game.getGameID(), job.getJobID(), job.getDirectCost(), job.getEstimatedWorkdays())));
		session.setAttribute("activityMethods", activityMethods);
		session.setAttribute("job", job);

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}
    }
}
