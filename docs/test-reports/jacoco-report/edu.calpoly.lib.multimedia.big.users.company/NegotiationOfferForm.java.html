package edu.calpoly.lib.multimedia.big.users.company;


import java.util.Vector;
import java.util.Iterator;
import java.text.NumberFormat;

//BIG v2 imports
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.NumberConverter;

import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionMapping;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;



/**
 * Holds data for a negotiation offer.  This data is submitted
 * by a company in response to a negotiation offer by a customer.
 * There are only two values included in this form,
 * the total cost of the job, and a profit percentage.
 *
 * @author Kevin Craig
 */
public class NegotiationOfferForm extends ActionForm
{
	private int percentProfit;
	private String percentProfitString;
	private long cost;
	private String costString;

    public int getPercentProfit()
    {
		float percentProfitFloat = Float.parseFloat(percentProfitString);
		this.percentProfit = Math.round(percentProfitFloat * 10);
		return percentProfit;
	}

	public void setPercentProfit(int percentProfit)
	{
		this.percentProfit = percentProfit;
		Integer wholePart = new Integer(percentProfit / 10);
		Integer partialPart = new Integer(percentProfit % 10);
		this.percentProfitString = wholePart.toString() + "." + partialPart.toString();
	}

	public String getPercentProfitString()
	{
		return percentProfitString;
	}

	public void setPercentProfitString(String percentProfitString)
	{
		this.percentProfitString = percentProfitString;
	}

	public long getCost()
	{
		this.cost = NumberConverter.longValue(costString);
		return cost;
	}

	public void setCost(long cost)
	{
		this.cost = cost;
		this.costString = NumberConverter.stringValue(cost);
	}

	public String getCostString()
	{
		return this.costString;
	}

	public void setCostString(String costString)
	{
		this.costString = costString;
	}


	/**
	 * Ensures all values are non-null, all number strings contain only numeric characters,
     * all number are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{

        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;
        float f = 0f;
        long l = 0;

        //First make sure none of the Strings are null
        for(int i = 0; i < 9; i++)
        {
			if((percentProfitString == null) || (percentProfitString.trim().length() < 1))
				nullParams = true;
			if((costString == null ) || (costString.trim().length() < 1))
			    nullParams = true;
		}

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the Strings contain only numeric characters.

			//Check that the strings are valid numbers by trying to convert them to longs or floats and checking for a NumberFormatException
			try
			{
				f = Float.parseFloat(percentProfitString);
				l = NumberConverter.longValue(costString);
			}
			catch(NumberFormatException nfe)
			{
				validNumbers = false;
			}

		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero

			if((l < 0) || (f < 0))
			{
				positiveNumbers = false;
			}
		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		return errors;
	}
}
