package edu.calpoly.lib.multimedia.big.users.company;

//JDK imports
import java.util.Vector;
import java.util.ArrayList;
import javax.sql.DataSource;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletContext;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;


//BIG v2 Imports
import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Method;
import edu.calpoly.its.mas.big.game.Activity;
import edu.calpoly.its.mas.big.users.company.Company;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//BIG v3 imports
import edu.calpoly.lib.multimedia.big.reports.CompletedContractsReportForm;
import edu.calpoly.lib.multimedia.big.reports.ContractsInProgressReportForm;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


public final class ViewCurrentBillAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
					   			 ActionForm form,
					   			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{

		   log.debug(this.getClass() + " beginning execute method");
		}


	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");
	    Company company = (Company) session.getAttribute("company");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

        Job job = null;

        job = (Job) session.getAttribute("job");

        //Get the parameter off the end of the URL
		String index = (String) request.getParameter("jobIndex");

		int jobIndex = (new Integer(index)).intValue();

        Vector jobs = (Vector) session.getAttribute("jobs");

        if (jobs != null)
        {
	        job = (Job) jobs.elementAt(jobIndex);
			session.setAttribute("job", job);
			session.removeAttribute("jobs");
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"noJobFound\"");
			}
			return (mapping.findForward("noJobFound"));
		}

        BillingForm billingForm = new BillingForm();
		ContractsInProgressReportForm contractsInProgress = null;
		CompletedContractsReportForm completedContracts = null;

		//Make the calls to the database
		try
		{
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
			billingForm = dbFacade.getBillingDBHandler().getBillingForm(game, game.getCurrentPeriodNumber() - 1, company, job);


			Activity[] activities = new Activity[9];

			try
			{
				activities = dbFacade.getActivityDBHandler().getActivities(game, job, game.getCurrentPeriodNumber() -1);
			}
			catch(ObjectNotFoundException onfe)
			{

					activities[0] = new Activity(1,
												 job.getTotalJobSize() - 8, //make sure the total number of cus matches the job size
												 job.getTotalJobSize() - 8, //At this point the breakdown amongst activities is unimportant
												 0,
												 0,
												 0,
												 0,
												 false,
												 0,
												 0,
												 0,
												 0,
												 0,
												 0,
									  		     0);
				//This means the job has already finished so activity updates for the job are no longer
				//saved.  We'll use an empty set of activies.
				for(int i = 1; i < activities.length; i++)
				{
					activities[i] = new Activity(i+1,
												 1,
												 1,
												 0,
												 0,
												 0,
												 0,
												 false,
												 0,
												 0,
												 0,
												 0,
												 0,
												 0,
									  		     0);
				}

			}
			job.setActivities(activities);

			contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, game.getCurrentPeriodNumber() -1, company);
			completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, game.getCurrentPeriodNumber() -1, company);

            dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}


        //Place a BillingForm action form into the session.  Struts will use it to populate the values to edit
        billingForm.setJob(job);
		billingForm.setContractsInProgressReport(contractsInProgress);
		billingForm.setCompletedContractsReport(completedContracts);

        session.setAttribute("jobTitle", (new Integer(job.getJobID())).toString());
        session.setAttribute("billingForm", billingForm);

        BillingForm oldBillingForm = new BillingForm();
        oldBillingForm.setBillingAmount(billingForm.getBillingAmount());

        session.setAttribute("oldBillingForm", oldBillingForm);

        //List of companies is no longer needed
        session.removeAttribute("jobs");
        session.removeAttribute("jobTitles");


		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
			saveToken(request);
			return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}

	}
}
