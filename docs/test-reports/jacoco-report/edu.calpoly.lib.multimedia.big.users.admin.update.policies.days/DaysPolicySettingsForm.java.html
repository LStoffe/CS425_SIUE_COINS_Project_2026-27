package edu.calpoly.lib.multimedia.big.users.admin.update.policies.days;

//Java imports
import javax.servlet.http.HttpServletRequest;
import java.util.Date;
import java.text.DateFormat;
import java.text.SimpleDateFormat;

//Struts imports
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * ActionForm used to hold days update policy settings.
 *
 *
 * @author Kevin Craig
 *
 */
public final class DaysPolicySettingsForm  extends ActionForm
{
	private boolean sunday;
	private boolean monday;
	private boolean tuesday;
	private boolean wednesday;
	private boolean thursday;
	private boolean friday;
	private boolean saturday;

	private String updateTime;

	private Log log;

	public DaysPolicySettingsForm()
	{
		super();

		log = LogFactory.getLog(this.getClass());
	}

    //**************** Public Methods ******************


	/**
	 * All String parameters can be correctly converted to their expected type.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

        boolean error = false;

		DateFormat df = new SimpleDateFormat("h:mm a" );

		try
		{
			df.parse(updateTime);
		}
		catch(Exception e)
		{
			error = true;
		}

		if(error)
		{
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.invalid"));
		}

		return errors;
	}


	public void reset (ActionMapping mapping, HttpServletRequest request)
	{
		sunday = false;
		monday = false;
		tuesday = false;
		wednesday = false;
		thursday = false;
		friday = false;
		saturday = false;
	}


//**************************** Set & Get Methods ************************

	public boolean getSunday()
	{
		return sunday;
	}

	public void setSunday(boolean sunday)
	{
		this.sunday = sunday;
	}

	public boolean getMonday()
	{
		return monday;
	}

	public void setMonday(boolean monday)
	{
		this.monday = monday;
	}

	public boolean getTuesday()
	{
		return tuesday;
	}

	public void setTuesday(boolean tuesday)
	{
		this.tuesday = tuesday;
	}

	public boolean getWednesday()
	{
		return wednesday;
	}

	public void setWednesday(boolean wednesday)
	{
		this.wednesday = wednesday;
	}

	public boolean getThursday()
	{
		return thursday;
	}

	public void setThursday(boolean thursday)
	{
		this.thursday = thursday;
	}

	public boolean getFriday()
	{
		return friday;
	}

	public void setFriday(boolean friday)
	{
		this.friday = friday;
	}

	public boolean getSaturday()
	{
		return saturday;
	}

	public void setSaturday(boolean saturday)
	{
		this.saturday = saturday;
	}

	public String getUpdateTime()
	{
		return updateTime;
	}

	public void setUpdateTime(String updateTime)
	{
		this.updateTime = updateTime;
	}

	public Date getUpdateTimeAsDate()
	{
		Date time = null;
		DateFormat df = new SimpleDateFormat("h:mm a" );

		try
		{
			time = df.parse(updateTime);
		}
		catch(Exception e)
		{
			log.error("", e);
		}

		return time;
	}

	public void setUpdateTimeAsDate(Date updateTime)
	{
		DateFormat df = new SimpleDateFormat("h:mm a" );
		this.updateTime = df.format(updateTime);
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": updateTime is " + updateTime);
		}

	}

}
