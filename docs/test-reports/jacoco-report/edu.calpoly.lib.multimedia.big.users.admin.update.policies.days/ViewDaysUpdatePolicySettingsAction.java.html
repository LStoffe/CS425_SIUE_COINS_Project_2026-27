package edu.calpoly.lib.multimedia.big.users.admin.update.policies.days;

import org.apache.struts.action.Action;

import edu.calpoly.its.mas.big.game.Game;

import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;

//Logging imports
import org.apache.commons.logging.Log;

/**
 * @author Kevin Craig
 */
public class ViewDaysUpdatePolicySettingsAction extends Action
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}


		ActionMessages errors = new ActionMessages();

		HttpSession session = request.getSession();
		Game game = (Game) session.getAttribute("game");
		Admin admin = (Admin) session.getAttribute("user");

		//Create a new BIGDatabaseFacadeInterface which will hold connections
		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

		DaysUpdatePolicySettings settings = null;

		try
		{
			settings = dbFacade.getAutomaticUpdatePolicyDBHandler().getDaysPolicySettings(game);
			dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}

		if(log.isDebugEnabled())
		{
			if(settings != null)
			{
				log.debug(this.getClass() + ": settings retrieved");
			}
			else
			{
				log.debug(this.getClass() + ": no settings found");
			}
		}

		DaysPolicySettingsForm newForm = new DaysPolicySettingsForm();
		newForm.setSunday(settings.getSunday());
		newForm.setMonday(settings.getMonday());
		newForm.setTuesday(settings.getTuesday());
		newForm.setWednesday(settings.getWednesday());
		newForm.setThursday(settings.getThursday());
		newForm.setFriday(settings.getFriday());
		newForm.setSaturday(settings.getSaturday());

		newForm.setUpdateTimeAsDate(settings.getUpdateTime());

		session.setAttribute("daysUpdatePolicySettingsForm", newForm);


		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
			saveToken(request);
			return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}

	}
}
