 * NOTE -- This form is used ONLY when adding a new company.  The last item in ActionErrors causes a bug if using this form

 */
public class CompanyForm2 extends ActionForm

  private String username;
  private String password;
  private String confirmPassword;

  private String wipLimitString;
  private int perJobLimitAsWIPPercent = -1;
  private String perJobLimitAsWIPPercentString;

  private boolean aaLicense = false;
  private boolean bbLicense = false;
  int license;

  public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
  {
    ActionErrors errors = new ActionErrors();

    //Check username is not null and is acceptable length
    if ((username == null) || (username.length() < 1))
    {
        errors.add("username", new ActionMessage("error.company.username.required"));
    }
    else if ((username.length() < StringConstraints.USERNAME_LOWER_LIMIT) || (username.length() > StringConstraints.USERNAME_UPPER_LIMIT))
    {
		errors.add("username_length", new ActionMessage("error.username.length"));
	}

	//Check company title is not null and is acceptable length
    if ((title == null) || (title.length() < 1 ))
    {
        errors.add("title", new ActionMessage("error.company.title.required"));
    }
    else if ((title.length() < StringConstraints.COMPANY_TITLE_LOWER_LIMIT) || (title.length() > StringConstraints.COMPANY_TITLE_UPPER_LIMIT))
    {
		errors.add("title_length", new ActionMessage("error.company.title.length"));
	}

	//Check password is not null, and is acceptable length.  Check that the passwords match.
    if ((password == null) || (password.length() < 1))
    {
    	errors.add("password", new ActionMessage("error.company.password.required"));
	}
	else if ((password.length() < StringConstraints.PASSWORD_LOWER_LIMIT) || (password.length() > StringConstraints.PASSWORD_UPPER_LIMIT))
	{
		errors.add("password_length", new ActionMessage("error.password.length"));
	}
	else if (!(password.equals(confirmPassword)))
	{
		errors.add("confirmPassword", new ActionMessage("error.password.mismatch"));
	}

	//Check cash-on-hand is not null and is a number.  Convert cash-on-hand from String to int in the process.
    if ((cashOnHandString == null) || (cashOnHandString.length() < 1) || cashOnHandString.equals("$"))
    {
        errors.add("cashOnHandString", new ActionMessage("error.company.cashOnHand.required"));
    }
    else
    {
	    if (!NumberConverter.isValidMoneyString(cashOnHandString))
	    {
	       errors.add("cashOnHand", new ActionMessage("error.company.cashOnHand.type"));
	    }
	}

	//Check wip limit is not null and is a number.  Convert wip limit from String to int in the process.
    if ((wipLimitString == null) || (wipLimitString.length() < 1) || wipLimitString.equals("$"))
    {
        errors.add("wipLimit", new ActionMessage("error.company.wipLimit.required"));
    }
    else
    {
	    if (!NumberConverter.isValidMoneyString(wipLimitString))
	    {
	       errors.add("wipLimit", new ActionMessage("error.company.wipLimit.type"));
	    }
	    else if (wipLimit < 5000000)  //Check wip limit is greater than 5 million
	    {
			setWIPLimitString("$5,000,000");
	    	errors.add("negativeWipLimit", new ActionMessage("error.company.wipLimit.negative"));
		}
		else if (cashOnHand > wipLimit)
		{
			errors.add("wipLimit", new ActionMessage("error.company.cashOnHand.wipLimit"));
		}

	}

	//Check per job limit percent is not null and is a number.
    if ((perJobLimitAsWIPPercentString == null) || (perJobLimitAsWIPPercentString.length() < 1) )
    {
        errors.add("perJobLimit", new ActionMessage("error.company.perJobLimitPercent.required"));
    }
    else
    {
		try
		{
			perJobLimitAsWIPPercent = Integer.parseInt(perJobLimitAsWIPPercentString);

	    	if((perJobLimitAsWIPPercent < 30) || (perJobLimitAsWIPPercent > 100))
	    	{
	      	 errors.add("wipLimit", new ActionMessage("error.company.perJobLimitPercent.type"));
			}
		}
	    catch (NumberFormatException nfe)
	    {
	       errors.add("wipLimit", new ActionMessage("error.company.perJobLimitPercent.type"));
	    }
	}

	//Make sure at least one of the two license checkboxes have been checked
	HttpSession session = request.getSession();
    Game game = (Game) session.getAttribute("game");
    int gametype = game.getGametype();
	if( gametype == 3 && !aaLicense && !bbLicense )
    {
        errors.add("noLicense", new ActionMessage("error.company.noLicense"));
	}

    return errors;
  }
