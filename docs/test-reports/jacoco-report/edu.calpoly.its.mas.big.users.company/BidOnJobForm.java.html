	/**
	 * Checks that a bid amount was entered and that it is a
	 * valid amount of money.
	 */
    public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
    {
        ActionErrors errors = new ActionErrors();

        if(job.getTypeModifier() != Job.HEAVY)
        {
           boolean nullParams = false;
           boolean integerParseError = false;
           boolean negativeFound = false;

		   if ((bidAmountString == null) || (bidAmountString.trim().length() < 1)) nullParams = true;
		   else
		   {
			  try
			  {
				 if (!NumberConverter.isValidMoneyString(bidAmountString)) integerParseError = true;
				 else bidAmount = NumberConverter.longValue(bidAmountString);

				 if (bidAmount < 0) negativeFound = true;
			  }
			  catch (NumberFormatException nfe)
			  {
				 integerParseError = true;
			  }
		   }

           if (nullParams == true) {
               errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.null"));
           }

           if (integerParseError == true) {
               errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.format"));
           }

           if (negativeFound == true) {
               errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.negative"));
           }
	    }

	    if(job.getTypeModifier() == Job.HEAVY)
        {
			boolean nullParams = false;

			for(int i = 0; i < 9; i++)
			{
				if (quantities[i] != 0 && unitPrices[i] == 0) nullParams = true;
			}
			if (nullParams == true) {
               errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.unitprice"));
		    }
        }
		return errors;
	}
