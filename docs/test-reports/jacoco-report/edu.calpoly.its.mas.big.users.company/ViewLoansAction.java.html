/**
 * Action class that gathers necessary info for the jsp page that accepts users requests for loans.
 * The jsp page that displays the loan request html form will also need to display the maximum number of
 * loans per team, the interest rate on loans, the maximum money amount of a single loan, and the number of periods in loan repayment.
 * The number of loans currently held by the company is also needed so the ActionForm can know whether not
 * the company is requesting more loans than is allowed.
 * As a convenience to the companies this page can also display information about existing loans.
 * This information would be the initial loan amount, the current balance, and the number of payments left in the loan.
 * <br><br>
 * <code>ViewLoansAction</code> puts all this information in the session for the jsp page to use as needed.
 */
public class ViewLoansAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));


		Company company = (Company) session.getAttribute("company");

		Integer numberOfLoans;
		String maxLoanAmount;
		String maxLoans;
		String interestRate;
		String repaymentPeriods;

		Hashtable loanParams = null;
		Vector loans = null;

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

		try
		{
			loanParams = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
			loans = dbFacade.getLoanDBHandler().getLoans(company, game.getCurrentPeriodNumber());
            dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ParametersNotFoundException pnfe)
		{
			log.fatal("", pnfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}

        numberOfLoans = new Integer(loans.size());

        maxLoanAmount = NumberConverter.stringValue( ( (Integer) loanParams.get("maxLoanAmount") ).intValue());
        maxLoans =  ((Integer) loanParams.get("maxLoansPerTeam")).toString();
        interestRate = ((Integer) loanParams.get("loanInterestPerYear")).toString();
        repaymentPeriods = ((Integer) loanParams.get("loanRepaymentPeriods")).toString();

		session.setAttribute("numberOfLoans", numberOfLoans);
		session.setAttribute("loans", loans);
		session.setAttribute("maxLoanAmount", maxLoanAmount);
		session.setAttribute("maxLoans", maxLoans);
		session.setAttribute("interestRate", interestRate);
		session.setAttribute("repaymentPeriods", repaymentPeriods);

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}
    }
}
