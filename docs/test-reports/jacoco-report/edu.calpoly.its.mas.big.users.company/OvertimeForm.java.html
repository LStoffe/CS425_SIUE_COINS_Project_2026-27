package edu.calpoly.its.mas.big.users.company;

import java.util.Hashtable;
import java.util.TreeMap;
import java.util.Vector;
import java.util.Iterator;
import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.game.Activity;
import edu.calpoly.its.mas.big.game.Job;

/**
 * <code>ActionForm</code> which holds the overtime days requested for
 * each of the nine activities for one of a company's active jobs.
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.12 $ $Date: 2002/03/14 20:10:06 $
 */
public class OvertimeForm extends ActionForm {
	private Job job;

	private int[] overtime = new int[9];
	private String[] overtimeStrings = new String[9];

	/**
	 * Set the <code>Job</code> for which this <code>OvertimeForm</code>
	 * contains overtime values.
	 *
	 * @param job the job associated with this form
	 */
	public void setJob(Job job)
	{
		this.job = job;
	}

	/**
	 * Get the <code>Job</code> for which this <code>OvertimeForm</code>
	 * contains overtime values.
	 *
	 * @returns the job associated with this form
	 */
	public Job getJob()
	{
		return job;
	}

	/**
	 * Set all nine overtime values at once, populating both the int
	 * and String arrays of overtime values
	 *
	 * @param overtime int array of overtime values (9 elements)
	 */
	public void setOvertimeArray(int[] overtime)
	{
		this.overtime = overtime;
		for (int i = 0; i < 9; i++) {
			overtimeStrings[i] = String.valueOf(overtime[i]);
		}
	}

	/**
	 * Return the overtime values as an int array.
	 *
	 * @returns int[] of overtime values (indexed by activityId - 1)
	 */
	public int[] getOvertimeArray()
	{
		return overtime;
	}

	/**
	 * Get the number of overtime days requested for a particular
	 * activity.
	 *
	 * @param avtivityID int activity number (1-9)
	 * @returns String representation of overtime value for the
	 *          specified activity.
	 */
	public String getOvertime(int activityID)
	{
		return overtimeStrings[activityID-1];
	}

	/**
	 * Set a single overtime value.  Overtime value is represented
	 * as a String for the benefit of HTML form validation.
	 *
	 * @param activityID activity identifier (1-9)
	 * @param overtimeDays the number of overtime days as a <code>String</code>
	 */
	public void setOvertime(int activityID, String overtimeDays)
	{
		overtimeStrings[activityID-1] = overtimeDays;
	}


	/**
	 * Ensure that all overtime day values for the job held within this
	 * <code>OvertimeForm</code> are integer values greater than 0.
	 * During this verification, the integer array is populated with
	 * the integer values, for later use in <code>RequestOvertimesAction</code>.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
		boolean integerParseError = false;
		boolean negativeFound = false;

		for (int i = 0; i < 9; i++) {
			String overtimeDays = (String) overtimeStrings[i];

			if ((overtimeDays == null) ||
				(overtimeDays.trim().length() < 1)) {
				nullParams = true;
			} else {
				try {
					overtime[i] = Integer.parseInt(overtimeDays);
					if (overtime[i] < 0) {
						negativeFound = true;
					}
				} catch (NumberFormatException nfe) {
					integerParseError = true;
				}
			}
		}


		if (nullParams == true) {
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.null"));
		}

		if (integerParseError == true) {
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.format"));
		}

		if (negativeFound == true) {
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.number.negative"));
		}

		return errors;
	}

}
