package edu.calpoly.its.mas.big.users.company;

//JDK imports
import javax.sql.DataSource;
import java.util.Vector;
import java.util.ArrayList;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletContext;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG imports
import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Method;
import edu.calpoly.its.mas.big.users.admin.LabelValueBean;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

public class ViewCurrentMethodsAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
					   			 ActionForm form,
					   			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

        //Get the parameter off the end of the URL
		String index = (String) request.getParameter("jobIndex");

		int jobIndex = (new Integer(index)).intValue();

        Vector jobs = (Vector) session.getAttribute("jobs");

        Job job = null;

        if (jobs != null)
        {
	        job = (Job) jobs.elementAt(jobIndex);
			session.setAttribute("job", job);
			session.removeAttribute("jobs");
		}
		else
		{
			return (mapping.findForward("noJobFound"));
		}
		int[] currentMethods = null;
		Method[][] possibleMethods = new Method[9][];

		//Make the calls to the database
		try
		{
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);
			currentMethods = dbFacade.getCurrentMethodDBHandler().getCurrentMethods(game, job);
/*
Note: getCurrentMethods() retrieves the correct method numbers for the methods curently scheduled for use on an activity,
but the wrong names are displayed (building instead of heavy)

In tables: schedule_estimated_methods and column current_method_used in table activity all use numbers 1-5 for methods, even if Heavy job

getAllMethods() - retrieves a list of all possible methods to use on a certain activity type.
needs to be 10-18 if a heavy job
*/

            if (job.getTypeModifier() == Job.PUBLIC || job.getTypeModifier() == Job.PRIVATE)
            {
			   possibleMethods[0] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 1);
			   possibleMethods[1] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 2);
			   possibleMethods[2] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 3);
			   possibleMethods[3] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 4);
			   possibleMethods[4] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 5);
			   possibleMethods[5] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 6);
			   possibleMethods[6] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 7);
			   possibleMethods[7] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 8);
			   possibleMethods[8] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 9);
		    }
            if (job.getTypeModifier() == Job.HEAVY)
            {
			   possibleMethods[0] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 10);
			   possibleMethods[1] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 11);
			   possibleMethods[2] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 12);
			   possibleMethods[3] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 13);
			   possibleMethods[4] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 14);
			   possibleMethods[5] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 15);
			   possibleMethods[6] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 16);
			   possibleMethods[7] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 17);
			   possibleMethods[8] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 18);
		    }

            dbFacade.close();
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}

        //Place a ChangeMethodsForm action form into the session.  Struts will use it to populate the values to edit
        ChangeMethodsForm changeMethodsForm = new ChangeMethodsForm();

        changeMethodsForm.setActivity1((new Integer(currentMethods[0])).toString());
        changeMethodsForm.setActivity2((new Integer(currentMethods[1])).toString());
        changeMethodsForm.setActivity3((new Integer(currentMethods[2])).toString());
        changeMethodsForm.setActivity4((new Integer(currentMethods[3])).toString());
        changeMethodsForm.setActivity5((new Integer(currentMethods[4])).toString());
        changeMethodsForm.setActivity6((new Integer(currentMethods[5])).toString());
        changeMethodsForm.setActivity7((new Integer(currentMethods[6])).toString());
        changeMethodsForm.setActivity8((new Integer(currentMethods[7])).toString());
        changeMethodsForm.setActivity9((new Integer(currentMethods[8])).toString());


        ArrayList[] options = new ArrayList[9];

        for (int i = 0; i < 9; i++)
        {
            options[i] = new ArrayList();
	        options[i].add(new LabelValueBean(possibleMethods[i][0].getMethodName(), "1" ));
	        options[i].add(new LabelValueBean(possibleMethods[i][1].getMethodName(), "2" ));
	        options[i].add(new LabelValueBean(possibleMethods[i][2].getMethodName(), "3" ));
	        options[i].add(new LabelValueBean(possibleMethods[i][3].getMethodName(), "4" ));
	        options[i].add(new LabelValueBean(possibleMethods[i][4].getMethodName(), "5" ));
		}

		changeMethodsForm.setActivity1TypeArray(options[0]);
		changeMethodsForm.setActivity2TypeArray(options[1]);
		changeMethodsForm.setActivity3TypeArray(options[2]);
		changeMethodsForm.setActivity4TypeArray(options[3]);
		changeMethodsForm.setActivity5TypeArray(options[4]);
		changeMethodsForm.setActivity6TypeArray(options[5]);
		changeMethodsForm.setActivity7TypeArray(options[6]);
		changeMethodsForm.setActivity8TypeArray(options[7]);
		changeMethodsForm.setActivity9TypeArray(options[8]);

        session.setAttribute("jobTitle", (new Integer(job.getJobID())).toString());
        session.setAttribute("methodSchedulingForm", changeMethodsForm);

        //List of companies is no longer needed
        session.removeAttribute("jobs");
        session.removeAttribute("jobTitles");


		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}

			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
			saveToken(request);
			return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("success"));
		}

	}
}
