package edu.calpoly.its.mas.big.users.company;

//Servlet imports
import javax.servlet.ServletContext;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;

//BIG imports
import edu.calpoly.lib.multimedia.big.game.NegotiatedJob;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.users.LoginBroker;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;
import edu.calpoly.its.mas.big.db.CurrentMethodDBHandler;


//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


public class BidOnJobAction extends Action
{
    public ActionForward execute(ActionMapping mapping,
								 ActionForm form,
								 HttpServletRequest request,
								 HttpServletResponse response)
    {
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		// Will be either: 'Retract Bid' or 'Submit Bid'
		String action = request.getParameter("submit");

        // Process request if it was not cancelled
        if(!isCancelled(request)) {

	        ActionMessages errors = new ActionMessages();

            ServletContext context = servlet.getServletContext();
            HttpSession session = request.getSession();
	        Game game = (Game) session.getAttribute("game");

            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");


            // If for any reason the company should not be allowed to login, route the
            // company to an error message page.
	        if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		       return (mapping.findForward("loginFailure"));

            Company company = (Company) session.getAttribute("company");

			BidOnJobForm bidOnJobForm = (BidOnJobForm) form;

     		//If the user pressed the "Calculate Totals" button
			if(	request.getParameter("calculate") != null )
			{
                double[] totals = new double[9];
                int[] quantities = bidOnJobForm.getQuantities();
                double[] unitPrices = bidOnJobForm.getUnitPrices();

                for(int i = 0; i < 9; i++)
				{
					totals[i] = quantities[i] * unitPrices[i];
				}
				bidOnJobForm.setTotals(totals);
				session.setAttribute("bidOnJobForm", bidOnJobForm);

				if (!errors.isEmpty())
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request returned with errors");
					}

					// Report any errors we have discovered back to the original form
					saveErrors(request, errors);
	 	    		saveToken(request);
	 	    		return (new ActionForward(mapping.getInput()));
				}
				else
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " request completed.  Forwarding to \"calculated\"");
					}
					return mapping.findForward("calculated");
				}
			}

			Job job = null;
            BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);

            try
            {
      			job = dbFacade.getJobDBHandler().getJob(game, bidOnJobForm.getJobID());

				//If the job is private two checks must be done
				//before accepting the bid.
				//First if the job is negotiated, then the
				//state of the job must be NegotiatedJob.NEGOTIATION_SUCCEEDED
				//Addtionally the company making this request must be on the allowed
				//company list for the job.
				//If either of this cases fail this Action has been
				//accessed illegitimately and an error can be returned right away
				if(job.getTypeModifier() == Job.PRIVATE)
				{

					if(job instanceof NegotiatedJob)
					{
						if(log.isDebugEnabled())
						{
							log.debug(this.getClass() + " Job " + job.getJobID() + " is negotiated");
						}
						NegotiatedJob nj = (NegotiatedJob) job;
						if(nj.getState() != NegotiatedJob.NEGOTIATION_SUCCEEDED)
						{
							if(log.isDebugEnabled())
							{
								log.debug(this.getClass() + "Negotiated not succeedded, company may not bid");
							}
							dbFacade.close();
							errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.bid.notAllowed"));
            	        	saveErrors(request, errors);
            	        	saveToken(request);
            	        	return (new ActionForward(mapping.getInput()));
						}
						else
						{
							if(log.isDebugEnabled())
							{
								log.debug(this.getClass() + " Negotiation succeeded, company can bid");
							}
						}
					}
					else
					{
						if(log.isDebugEnabled())
						{
							log.debug(this.getClass() + " Job " + job.getJobID() + " is not negotiated");
						}
					}

					if(!job.isCompanyAllowedToBid(company.getCompanyID()))
					{
						if(log.isDebugEnabled())
						{
							log.debug(this.getClass() + " company " + company.getCompanyID() + " may NOT bid on job " + job.getJobID());
						}
						dbFacade.close();
						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.bid.notAllowed"));
            	        saveErrors(request, errors);
            	        saveToken(request);
            	        return (new ActionForward(mapping.getInput()));
					}
					else
					{
						if(log.isDebugEnabled())
						{
							log.debug(this.getClass() + " company " + company.getCompanyID() + " may bid on job " + job.getJobID());
						}
					}
				}
			}
			catch(Exception e)
			{
				log.fatal("Unable to get job information for jobid " + bidOnJobForm.getJobID(), e);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
			finally
			{
				try
				{
 	       	 		dbFacade.close();
				}
				catch (BIGDatabaseFacadeException e)
				{
					log.fatal(getClass().getName(), e);
 					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
			}

            try
            {
				if(job.getTypeModifier() == Job.HEAVY)
				{
				    //In case the user does not do "Calculate Totals" before "Submitting" the Bid
                    double[] totals = new double[9];
                    int[] quantities = bidOnJobForm.getQuantities();
                    double[] unitPrices = bidOnJobForm.getUnitPrices();

                    for(int i = 0; i < 9; i++)
				    {
				        totals[i] = quantities[i] * unitPrices[i];
				    }
				    bidOnJobForm.setTotals(totals);

                    double theTotalBid = 0;
                    for (int i = 0; i < 9; i++)
                    {
			            theTotalBid = theTotalBid + totals[i];
		            }
		            bidOnJobForm.setTheTotalBid(theTotalBid);
		            bidOnJobForm.setBidAmountLong( (long) theTotalBid);
			    }

				Bid bid = new Bid(game.getGameID(), company.getCompanyID(), bidOnJobForm.getJobID(), bidOnJobForm.getBidAmountLong(), bidOnJobForm.getTheTotalBid(), bidOnJobForm.getUnitPrices(), 0);
				bid.setMethods(bidOnJobForm.getMethods());

				if (action.equals("Retract Bid"))
				{

				    if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " Removing Bid: " + game.getGameID() + ", " + job.getJobID() + ", " + company.getCompanyID());
					}
					dbFacade.getBidDBHandler().removeBid(game.getGameID(), job.getJobID(), company.getCompanyID());

					// TODO: bounce to an informative page...
				}
				else if (action.equals("Submit Bid") || action.equals("Submit"))
				{
					// If the bid is not there, this will throw an ObjectNotFoundException which will be caught below
					dbFacade.getBidDBHandler().updateBid(game, job, company, bid);
				}

                dbFacade.close();
            }
            catch(ObjectNotFoundException onfe)
            {

				// If the bid wasn't there, try adding it
				try
				{
					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + " Adding Bid: " + game.getGameID() + ", " + job.getJobID() + ", " + company.getCompanyID());
					}

					Bid bid = new Bid(game.getGameID(), company.getCompanyID(), bidOnJobForm.getJobID(), bidOnJobForm.getBidAmountLong(), bidOnJobForm.getTheTotalBid(), bidOnJobForm.getUnitPrices(), 0);
					bid.setMethods(bidOnJobForm.getMethods());

					dbFacade.getBidDBHandler().addBid(game, job, company, bid);

					dbFacade.close();

				}
				catch (BIGDatabaseFacadeException bigdbe)
				{
					// Now we have a problem
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}

            }
            catch(BIGDatabaseFacadeException bigdbe)
            {
                // Problem with the database
				log.fatal("", bigdbe);
                errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
            }
			finally
			{
				try
				{
 	       	 		dbFacade.close();
				}
				catch (BIGDatabaseFacadeException e)
				{
					log.fatal(getClass().getName(), e);
 					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
			}

            if (!errors.isEmpty())
            {
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

                // Report any errors we have discovered back to the original form
                saveErrors(request, errors);
                saveToken(request);
                return (new ActionForward(mapping.getInput()));
            }
            else
            {
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
                return (mapping.findForward("success"));
            }
        }
        else
        {  // Request was cancelled

			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
            return (mapping.findForward("cancelled"));
        }

    }
}
