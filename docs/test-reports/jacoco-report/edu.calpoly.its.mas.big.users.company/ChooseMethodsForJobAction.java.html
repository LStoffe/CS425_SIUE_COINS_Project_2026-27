package edu.calpoly.its.mas.big.users.company;

/**
 * This <code>Action</code> builds a list of method names
 * (specific to each game)  If the user has already placed a bid
 * on the job identified by the jobID request parameters, the
 * previously chosen methods and bid amount are retrieved and added
 * to the <code>ActionForm</code>  for display by chooseMethodsForJob.jsp
 *
 * @author Andrew Von Dollen
 *
 * @version $Revision: 1.14 $ $Date: 2002/04/12 17:44:29 $
 */
public class ChooseMethodsForJobAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

		Company company = (Company) session.getAttribute("company");

		// Grab jobID parameter, which was sent by chooseJobForBid.jsp
		String jobIDString = request.getParameter("jobID");
		int jobID = 0;
		if (jobIDString != null && !jobIDString.equals(""))
		{
			jobID = Integer.parseInt(jobIDString);
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " cannot determine job id");
			}
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}

		Method[] methods = null;
		Bid bid = null;

		BidOnJobForm bidOnJobForm = new BidOnJobForm();
		bidOnJobForm.setJobID(jobID);

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request), true);

		Job dummyJob = new Job(bidOnJobForm.getJobID(), -1, null,
							   null, -1, -1, -1, -1, -1, -1, -1, -1, false, false, -1, -1, -1, -1, -1, -1, -1, -1, -1, Job.PUBLIC);

		try
		{
			// Ensure that the jobId passed in as a request parameter
			// corresponds to a current job, if not throw an exception
			Vector jobs = dbFacade.getJobDBHandler().getJobs(game, game.getCurrentPeriodNumber());
			Iterator jobsIter = jobs.iterator();
			boolean jobFound = false;
			while(jobsIter.hasNext())
			{
				Job job = (Job) jobsIter.next();
				if (job.getJobID() == jobID)
				{
					bidOnJobForm.setJob(job);
					jobFound = true;
				}
			}
			if (!jobFound)
				throw new ObjectNotFoundException();


			for (int activityNum = 1; activityNum <= 18; activityNum++)
			{
				methods = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, activityNum);

				String[] methodNames = new String[5];
				for (int methodNum = 0; methodNum < 5; methodNum++)
				{
					methodNames[methodNum] = methods[methodNum].getMethodName();
				}

				bidOnJobForm.setMethodNames(activityNum, methodNames);
			}

			try
			{
			    //Get quantities for Activities if this is a Heavy job
			    Job job = bidOnJobForm.getJob();
			    if(job.getTypeModifier() == Job.HEAVY)
			    {
				   Activity[] jobsActivities = dbFacade.getActivityDBHandler().getActivities(game, job, game.getCurrentPeriodNumber() - 1 );
                   int[] quantities = new int[9];
                   for (int i = 0; i < 9; i++)
				   {
					  quantities[i] = jobsActivities[i].getLusTotal();
		           }
                   bidOnJobForm.setQuantities(quantities);
			    }

			    // If the user has previously placed a bid on this job,
			    // populate the action form with that bid's details
				bid = dbFacade.getBidDBHandler().getBid(game, dummyJob, company);

				if (bid != null)
				{
					bidOnJobForm.setMethods(bid.getMethods());
				    bidOnJobForm.setBidAmount(Long.toString(bid.getBidAmount()));

                    if(job.getTypeModifier() == Job.HEAVY)
			        {
			            bidOnJobForm.setTheTotalBid(bid.getBidAmountUP());
			            bidOnJobForm.setUnitPrices(bid.getUnitPrices());

                        double[] totals = new double[9];
                        int[] quantities = bidOnJobForm.getQuantities();
                        double[] unitPrices = bidOnJobForm.getUnitPrices();

                        for(int i = 0; i < 9; i++)
				        {
					        totals[i] = quantities[i] * unitPrices[i];
				        }
				        bidOnJobForm.setTotals(totals);
                    }

				}
			}
			catch (ObjectNotFoundException onfe)
			{
				// This isn't a problem, it just means the user hasn't previously bid on this job
			}


			dbFacade.close();
		}
		catch(ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}
		finally
		{
			try
			{
         		dbFacade.close();
			}
			catch (BIGDatabaseFacadeException e)
			{
				log.fatal(getClass().getName(), e);
 				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
		}

		session.setAttribute("bidOnJobForm", bidOnJobForm);

		if (!errors.isEmpty())
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);

			// TODO: Change this to something sane?
			//   2/4/2005: This is sane for Struts, this is where ActionForms return to when validate fails.
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			}

			Job job = bidOnJobForm.getJob();
			if(job.getTypeModifier() == Job.HEAVY)
			{
				return (mapping.findForward("successHC"));
			}
			else
			{
			    return (mapping.findForward("success"));
            }
		}
    }
}
