package edu.calpoly.its.mas.big.users.company;

import javax.servlet.ServletContext;
import javax.servlet.http.HttpSessionBindingEvent;
import javax.servlet.http.HttpSessionBindingListener;

import edu.calpoly.its.mas.big.users.LoginBroker;

/**
 * When a company's session times out due to inactivity, this class will be notified
 * via an HttpSessionBindingEvent.  The valueUnbound method will decrement the number of
 * companies logged in, and will invalidate the user's session.
 *
 * @author Dara Manker
 * @version $Revision: 1.1 $ $Date: 2002/04/16 00:37:28 $
 * @param context The servlet context in which this company's session was running
 * @param gameID  The BIG Game ID for this company.
 */

public class TimeoutCompanySession implements HttpSessionBindingListener
{

    private ServletContext context;
    private int gameID;

    public TimeoutCompanySession(ServletContext sc, int gameid)
    {
		this.gameID = gameid;
		this.context = sc;
	}

    /**
     * This method is run when a company's session is timed out.  This method will
     * decrement the number of companies logged in, remove the game and user objects
     * from the company's session, and then kill the session.
     */
    public void valueUnbound(HttpSessionBindingEvent event)
    {
        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
        loginBroker.decrementNumberOfLoggedInCompanies(gameID, context);
    }

    /**
     * This method is empty: no processing is necessary by this object when
     * it is bound to the company's session.
     */
    public void valueBound(HttpSessionBindingEvent event)
    {
	}

}
