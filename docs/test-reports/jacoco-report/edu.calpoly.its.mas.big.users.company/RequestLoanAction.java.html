package edu.calpoly.its.mas.big.users.company;

//Struts imports
import org.apache.struts.action.Action;

/**
 * Creates a new loan and stores it to the database.
 * <br><br>
 * <b>Note:</b> It is assumed the loan approved if it passes the RequestLoanForm.validate method.
 * <code>RequestLoanAction</code> does no further checking to see if this loan should be
 * allowed to be in the database.  It does, however, check if a loan request for this period already exists and if so
 * overwrites that request with the current request rather than adding a new one.
 *
 * @author Kevin Craig
 * @version $Revision: 1.18 $ $Date: 2002/04/26 18:04:16 $
 */
public class RequestLoanAction extends Action
{

	public ActionForward execute(ActionMapping mapping,
					     		 ActionForm form,
					     		 HttpServletRequest request,
				         		 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
	    if(!isCancelled(request))
	    {
	        ActionMessages errors = new ActionMessages();

            ServletContext context = servlet.getServletContext();
            HttpSession session = request.getSession();
	        Game game = (Game) session.getAttribute("game");

            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

            // If for any reason the company should not be allowed to login, route the
            // company to an error message page.
	        if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		      return (mapping.findForward("loginFailure"));

			String forward = "error";

            RequestLoanForm requestLoanForm = (RequestLoanForm) form;
			long amount = requestLoanForm.getAmount();

            Loan loan = new Loan(amount, requestLoanForm.getAPR(), requestLoanForm.getRepaymentPeriods());

            Company company = (Company) session.getAttribute("company");

            String retract = (String) request.getParameter("retract");

			//Get rid of the old form
			session.removeAttribute("requestLoanForm");

			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

			if (retract != null)  //"Retract" submit button was pushed.  Remove loan for this period from database
			{
	    		try
	    		{
					dbFacade.getLoanDBHandler().removeLoan(company.getCompanyID(), game.getCurrentPeriodNumber());
					dbFacade.close();
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				finally
				{
					try
					{
        		 		dbFacade.close();
					}
					catch (BIGDatabaseFacadeException e)
					{
						log.fatal(getClass().getName(), e);
 						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
					}
				}

				forward = "retracted";
			}
			else
			{
	    		try
	    		{

					boolean rerequest = false;
					Vector loans = dbFacade.getLoanDBHandler().getLoans(company, game.getCurrentPeriodNumber());
					Iterator i = loans.iterator();
					while(i.hasNext())
					{
						Loan l = (Loan) i.next();
						if (!l.isApproved())
						{
							rerequest = true;
							loan.setLoanID(l.getLoanID());
						}
					}

					if(rerequest)
					{
						dbFacade.getLoanDBHandler().updateLoan(loan);
					}
					else
					{
            	    	dbFacade.getLoanDBHandler().addLoan(company, game.getCurrentPeriodNumber(), loan);
					}

                    dbFacade.close();
            	    session.setAttribute("amount", NumberConverter.stringValue(amount));
            	    session.setAttribute("rerequest", new Boolean(rerequest));
            	    session.setAttribute("totalAmount", NumberConverter.stringValue(loan.getTotalLoanValue()));
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ObjectNotFoundException onfe)
				{
					log.fatal("", onfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}

				forward = "success";
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to " + forward);
				}
				return (mapping.findForward(forward));
			}

	    }
	    else
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}

}
