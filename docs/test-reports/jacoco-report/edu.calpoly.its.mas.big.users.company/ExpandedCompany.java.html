  /**
   * Examines each bill submitted by the company this period rejecting those that
   * are too high and paying those that are reasonable.
   */
  public void processIncomingPayments(Customer customer)
  {
	  try
	  {
		  //This is the list of all jobs this company is currently working on or
		  //has ever worked on in which the amount of cash collected for the job
		  //is less than the total contract price for the job
	  	  Vector unbilledJobs = dbFacade.getJobDBHandler().getUnbilledJobs(game.getGame(), company);
	  	  Iterator i = unbilledJobs.iterator();
	  	  while(i.hasNext())
	  	  {
			  //Get the billing form associated with each job
			  Job j = (Job) i.next();

              BillingForm bf = dbFacade.getBillingDBHandler().getBillingForm(game.getGame(), game.getGame().getCurrentPeriodNumber() - 1, company, j);
              bf.setJob(j);
              bf.setContract(dbFacade.getContractsReportsDBHandler().getContract(game.getGame(), game.getGame().getCurrentPeriodNumber() - 1, j, company));
              bf = customer.processBill(bf, j);
              //System.out.println(bf.getBillingAmount() + " billing amount");
              int retainedAmount = j.calculateRetainedAmount(bf.getBillingAmount());
              balanceSheet.setCurrentAccountsReceivable(balanceSheet.getCurrentAccountsReceivable() - (bf.getBillingAmount() - retainedAmount));
              if(!bf.rejected())
              {
                  //Pay the amount
              	  increaseCashOnHand(bf.getBillingAmount() - retainedAmount, "PAYMENTS");

              	  //Set the new values into the job
              	  j.increaseBilledToDate(bf.getBillingAmount());
              	  j.setRetainedAmount(((int) j.getRetainedAmount()) + retainedAmount);
          	  	  j.setCompany(this);
          	  	  j.setGame(game);

			  	  //System.out.println("retained amount: " + retainedAmount);

			  	  dbFacade.getJobDBHandler().updateJob(game.getGame(), j);

			  	  //Checks if the amount of this bill is ethical or unethical
			  	  if (customer.ethicalBill(bf, j))
			  	  {
				      modifyCAMUpdate(CAMUpdater.ETHICS, CAMUpdater.ETHICS_ETHICAL_BILL);
				  }
				  else
				  {
				      modifyCAMUpdate(CAMUpdater.ETHICS, CAMUpdater.ETHICS_UNETHICAL_BILL);
				  }

		  	  }
		  	  else  // bill was rejected
		  	  {
                  balanceSheet.setRetentionReceivable(balanceSheet.getRetentionReceivable() - retainedAmount);

                  // reverse entries in previous period's balance sheet that were revised at time of billing

				  int amountBilled = bf.getBillingAmount();
				  int oldAmountBilled = bf.getOldAmountBilled();

				  int prevRetention = j.calculateRetainedAmount(oldAmountBilled);
				  int currRetention = j.calculateRetainedAmount(amountBilled);

				  BalanceSheetForm balanceSheet2 = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game.getGame(), game.getGame().getCurrentPeriodNumber() -1, company);
         		  ContractsInProgressReportForm contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game.getGame(), game.getGame().getCurrentPeriodNumber() -1, company);
 		 	      CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game.getGame(), game.getGame().getCurrentPeriodNumber() -1, company);

                  balanceSheet2.setUnderbilledAmount(contractsInProgress.getTotalUnderbilling() - oldAmountBilled + amountBilled );  //This new value won't be saved, it's only for this object
				  balanceSheet2.setUnbilledAmount(completedContracts.getTotalUnbilled());

				  balanceSheet2.setOverbilledAmount(contractsInProgress.getTotalOverbilling() );
			      balanceSheet2.setCurrentAccountsReceivable(balanceSheet2.getCurrentAccountsReceivable() - (amountBilled - currRetention) + (oldAmountBilled - prevRetention));

				  balanceSheet2.setRetentionReceivable(balanceSheet2.getRetentionReceivable() - currRetention + prevRetention);
				  balanceSheet2.setPeriod(game.getCurrentPeriodNumber() - 1);

                  dbFacade.getBalanceSheetDBHandler().updateBalanceSheet(game.getGame(), company, balanceSheet2);

				  bf.setOldAmountBilled(bf.getBillingAmount() );
                  bf.setBillingAmount(0);

			  }
			  dbFacade.getBillingDBHandler().updateBillingForm(game.getGame(), company, game.getGame().getCurrentPeriodNumber() -1, bf);
		  }
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("", bigdbe);
	  }
	  catch(ObjectNotFoundException onfe)
	  {
		  log.fatal("", onfe);
	  }
	  //This amount is be retention for jobs just completed
	  increaseCashOnHand(balanceSheet.getCurrentAccountsReceivable(), "RETENTION");
	  balanceSheet.setCurrentAccountsReceivable(0); //All payments have been received, zero out accounts receivable
  }

  /**
   * Updates all the loans held by this company.
   *
   * If a loan is marked as approved, normal payments will be made on it.
   * If a loan is not marked as approved it means the loan is new this period.
   * It will be marked as approved and payments on it will begin next period.
   */
  public void processLoans()
  {
	  Loan loan = null;
	  Iterator i = loans.iterator();
	  while(i.hasNext())
	  {
		  loan = (Loan) i.next();
		  if(loan.isApproved())
		  {
			  //Current loan - make a payment on it
              loan.makePayment();

              //Update the financial report form to show interest and principle payments, and the current balance
	          financialReportForm.increaseInterestPaidOnLoans(loan.getInterestPaid());
	          financialReportForm.increaseLoanPrinciplePayments(loan.getPaymentAmount() - loan.getInterestPaid());
	      	  balanceSheet.setLoanBalances(balanceSheet.getLoanBalances() + loan.getBalance());
              decreaseCashOnHand(loan.getPaymentAmount(), "LOAN_PAYMENT");
		  }
		  else
		  {
			  long amount = loan.getAmount();
			  increaseCashOnHand(amount, "LOAN");
			  loan.setApproved(true);
	      	  balanceSheet.setLoanBalances(balanceSheet.getLoanBalances() + amount);
		  }

		  //Store loans for the next period
		  if((loan != null) && (loan.getPaymentsRemaining() > 0))
		  {
			  try
			  {
				  //Store the state this loan will be in during the next period
				  dbFacade.getLoanDBHandler().addLoan(company, game.getCurrentPeriodNumber() + 1, loan);
			  }
			  catch(BIGDatabaseFacadeException bigdbe)
			  {
				  log.fatal(bigdbe);
			  }
		  }
	  }
  }

			  if(c == null)
			  {
			  }
			  else
			  {
				  revenueRecorded = c.getRevenuePreviouslyRecorded() + c.getCurrentRevenueRecorded();
				  bigComputesEstCostToComplete = c.getAutoComputeEstCost();
				  estCostToComplete = c.getEstimatedCostToComplete();
			  }

			  if(bigComputesEstCostToComplete)
			  {
				  //If BIG is to compute the estimated cost to complete use the JobCostEstimator now
		  	      estCostToComplete = JobCostEstimator.getEstimatedCost(game.getGame().getCurrentPeriodNumber() + 1, //1 is added because this estimate should not take into account this period, since work has just been completed
																		game.getGame().convertDateToPeriod(job.getCompletionDeadline()),
																		game.getGame().getEstimatedPercentageOfLastPeriodWorked(job.getCompletionDeadline()),
																		job.getActivities(),
																		game.getWorkdaysPerMonth() * 2,
																		job.getFixedFieldOverhead(),
																		job.getVariableFieldCharge(),
																		job.getLiquidatedDamages()
		  															   );
			  }
			  else
			  {
				  //If the players are computing the est cost to complete themselves
				  //their estimate will be automatically updated by substracting the costs incurred this period
				  if(estCostToComplete - job.getCurrentCost() > 0)
				  {
					  estCostToComplete = estCostToComplete - job.getCurrentCost();
				  }
			  }


			  if(log.isDebugEnabled())
			  {
				  log.debug("Company " + company.getCompanyID() + ": " + job.getJobID() + ", % complete: " + job.getPercentOfJobComplete());
				  log.debug("is active: " + job.getIsActive());
			  }

              //Record retention if the job is complete
              //Also clear the est cost to complete, since it has to be zero now that the job is complete
              if(!job.getIsActive())
              {
				  if(log.isDebugEnabled())
				  {
					  log.debug("Job " + job.getJobID() + " is complete");
				  	  log.debug("PROCESSING RETENTION");
				  	  log.debug("Job retained amout " + job.getRetainedAmount());
				  	  log.debug("Company " + company.getCompanyID() + ": " + " Bal sheet Retention Receivable" + balanceSheet.getRetentionReceivable());
				  }
				  balanceSheet.setRetentionReceivable(balanceSheet.getRetentionReceivable() - (int) job.getRetainedAmount());
				  balanceSheet.setCurrentAccountsReceivable(balanceSheet.getCurrentAccountsReceivable() + (int) job.getRetainedAmount());
				  job.clearRetainedAmount();

				  if(log.isDebugEnabled())
				  {
				  	  log.debug("Company " + company.getCompanyID() + ": " + " After updating Bal sheet, Retention Receivable " + balanceSheet.getRetentionReceivable());
				  }

				  //Job is already complete, no more cost to complete it
				  estCostToComplete = 0;
    		  }
    		  else
    		  {
				  if(!(job instanceof NegotiatedJob))
				  {
				      assignDaysLatePenalties(job);
			  	  }
			  }

			  //Save the current contract info
		  	  dbFacade.getContractsReportsDBHandler().addCurrentJobInfo(game.getGame(),
		  																	 game.getGame().getCurrentPeriodNumber(),
		  																	 company,
		  																	 job,
		  																	 estCostToComplete,
		  																	 revenueRecorded
  																	   );
  			 //We now have the contract current.
  			 //If this is a Negotiated Job that just completed,
  			 //use the contract cost info to compute share of savings
  			 //and store that in the bid.  call jobComplete method on NegotiatedJob to do this
  			 if(!job.getIsActive() && job instanceof NegotiatedJob)
  			 {
				 try
				 {
				     job.jobCompleted(dbFacade.getContractsReportsDBHandler().getContract(game.getGame(), game.getGame().getCurrentPeriodNumber() -1, job, company) );
				 }
				 catch(Exception e)
				 {
					 log.fatal("", e);
				 }
			 }
		  }
		  catch(BIGDatabaseFacadeException bigdbe)
		  {
			  log.fatal("", bigdbe);
		  }

          game.storeJob(job);


          //Remove the job from the list if it is finished.  This is needed for when the doWorkOnJobs method is
          //called when the game is being finalized because work will be done on all jobs in the list
          //11/20/02 I don't think this is needed.  It can't hurt but it's probably just wasting clock cycles - K.C.
          if(!job.getIsActive())
          {
			  i.remove();
		  }

	  }  // end "while(i.hasNext())"  --> on to the next job if any
  }

  /**
   * Gathers and prepares the activities for a job.
   * Retrieves from the database the list of activities for the job and the list of methods that should be
   * used on those activities.  A <code>Method</code> object will be created for each method and an <code>Activity</code>
   * object for each activity.  The <code>Activity</code> objects will be loaded with the <code>Method</code> objects
   * and network dependency parameters will be pulled from the <code>ExpandedGame</code> and loaded into the <code>Activity</code>
   * as well.
   * @param job The <code>Job</code> for which to prepate activities.
   * @return Activity[] An array of <code>Activity</code> objects for the job exactly as they were at the end of the last period
   */
  public Activity[] setupActivities(Job job)
  {
	  Activity[] activities = null;
	  int[] currentMethods = null;

      try
	  {
	      activities = dbFacade.getActivityDBHandler().getActivities(game.getGame(), job, game.getGame().getCurrentPeriodNumber() - 1);
	  }
      catch(ObjectNotFoundException onfe)
      {
		  log.fatal("", onfe);
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("", bigdbe);
	  }

	  try
	  {
		  //Retrieve the current method id's to be used on the activities
	      currentMethods = dbFacade.getCurrentMethodDBHandler().getCurrentMethods(game.getGame(), job);
	  }
      catch(ObjectNotFoundException onfe)
      {
		  log.fatal("", onfe);
	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("", bigdbe);
	  }

      //Load the activities with the values they need
      Method[][] methods = game.getMethods();
      Hashtable[][] networkDependencies = game.getNetworkDependencies();

      int jobType = job.getJobType();
      if (jobType > 5) // it's a Heavy job
      {
		  jobType = jobType - 5;
		  methods = game.getMethodsHC();
		  networkDependencies = game.getNetworkDependenciesHC();
	  }

	  int stsActivity, stsPercent, ftfActivity, ftfPercent;
      for(int j = 0; j < 9; j++)
      {
          //Changes the effect of uncertainty each time a method is used
          methods[j][currentMethods[j]-1].reRollCosts();
	  	  activities[j].setMethod(methods[j][currentMethods[j]-1]);

		  stsActivity = ((Integer) networkDependencies[jobType - 1][j].get("STSActivity")).intValue();
		  stsPercent = ((Integer) networkDependencies[jobType - 1][j].get("STSPercent")).intValue();
		  ftfActivity = ((Integer) networkDependencies[jobType - 1][j].get("FTFActivity")).intValue();
		  ftfPercent = ((Integer) networkDependencies[jobType - 1][j].get("FTFPercent")).intValue();

		  activities[j].setStartToStartActivity(stsActivity);
		  activities[j].setStartToStartPercent(stsPercent);
		  activities[j].setFinishToFinishActivity(ftfActivity);
		  activities[j].setFinishToFinishPercent(ftfPercent);

	  }
	  return activities;
  }

  private void addBillingFormsForUnbilledJobs() throws BIGDatabaseFacadeException
  {
	  Vector unbilledJobs = dbFacade.getJobDBHandler().getUnbilledJobs(game.getGame(), company);
	  Vector activeJobs = dbFacade.getJobDBHandler().getActiveJobs(game.getGame(), company);

	  //Search the list of active jobs for this company and add to the unbilled
	  //list any negotiated jobs that are incomplete, even if all billing is complete.
	  //Even though the company can no longer bill, the billing form is needed so it is recorded
	  //that 0 was billed.
	  for(Object o : activeJobs)
	  {
		  Job j = (Job) o;
		  if(j instanceof NegotiatedJob)
		  {
			  boolean found = false;
			  for(Object o2 : unbilledJobs)
			  {
				  Job j2 = (Job) o2;
				  if(j.getJobID() == j2.getJobID())
				  {
					  found = true;
				  }
			  }
			  if(!found)
			  {
				  unbilledJobs.add(j);
			  }
		  }
	  }

	  Iterator i = unbilledJobs.iterator();
	  while(i.hasNext())
	  {
		  Job j = (Job) i.next();
		  dbFacade.getBillingDBHandler().addBillingForm(game.getGame(), company, game.getGame().getCurrentPeriodNumber(), j);
	  }
  }

  /**
   * Process this company's consulting reports purchases this period.
   * Subtract each purchase amount from the company's cash on hand and
   * increase the consulting costs in <code>financialReportForm</code>
   * accordingly.  This method is invoked during period updates and
   * game finalization.
   */
  public void processConsultingCosts()
  {
	  Hashtable consultingCosts = null;

	  try
	  {
		  consultingCosts = dbFacade.getGameParametersDBHandler().getConsultingCostsParameters(game.getGame());
	  }
	  catch (Exception pnfe)
	  {
		  log.fatal("Error retrieving consulting costs parameters: " + company.getCompanyID());
		  log.fatal("", pnfe);
	  }


	  try
	  {
		  /*
			purchaseMap is indexed by report ID, its entries are the number
			of reports (which might be zero) purchased of each type during
			this period.

			Report IDs are:
			COMPLETE_LIST_OF_BIDS = 0;
			CONSTRUCTION_DEMAND = 1;
			LABOR_AVAILABILITY = 2;
			MATERIAL_COST_INDEX = 3;
			WEATHER_FORECAST = 4;

			If this company purchased 3 Complete List of Bid Reports (each for a
			different period) and the Weather Forecast this period, purchaseMap
			looks like this: {3,0,0,0,1}
		  */
		  int[] purchaseMap = dbFacade.getReportLedgerDBHandler().getPurchasesForPeriod(company.getCompanyID(), game.getCurrentPeriodNumber());

		  for (int i = 0; i < purchaseMap.length; i++)
		  {
			  if (null != purchaseMap && 0 != purchaseMap[i])
			  {
				  int reportCost = getReportCost(i, consultingCosts);

				 // System.out.println("    Period " + game.getCurrentPeriodNumber() +
					//				 ", Spent $" + reportCost * purchaseMap[i] + " on Report " + i);

				  decreaseCashOnHand(reportCost * purchaseMap[i], "CONSULTING");
				  dbFacade.getCompanyDBHandler().updateCompany(game.getGame(), company);
				  financialReportForm.increaseConsultingExpenses(reportCost * purchaseMap[i]);
			  }
		  }

	  }
	  catch(BIGDatabaseFacadeException bigdbe)
	  {
		  log.fatal("Error processing consulting costs for company " + company.getCompanyID());
		  log.fatal("", bigdbe);
	  }
	  catch(ObjectNotFoundException onfe)
	  {
		  log.fatal("", onfe);
	  }
  }


	/**
	 * Determine the cost of a report, identified by <code>reportID</code>
	 *
	 * @param reportId the ID of the report
	 * @param costs the <code>Hashtable</code> containing the
	 *        report costs from the database.
	 */
    private int getReportCost(int reportID, Hashtable costs)
    {
        if (reportID == ConsultingReportsAction.COMPLETE_LIST_OF_BIDS)
        {
            return ((Integer) costs.get("Complete_List_Of_Bids")).intValue();
        }
        else if (reportID == ConsultingReportsAction.CONSTRUCTION_DEMAND)
        {
            return ((Integer) costs.get("Market_Demand")).intValue();
        }
        else if (reportID == ConsultingReportsAction.LABOR_AVAILABILITY)
        {
            return ((Integer) costs.get("Labor_Availability")).intValue();
        }
        else if (reportID == ConsultingReportsAction.MATERIAL_COST_INDEX)
        {

	/**
	 * Reevaluates bond limits for this company.
	 * Consult requirements doc for specifics on how bond limits are recomputed
	 */
	public void reevaluateBondLimits()
	{
		Vector cashOnHandValues = null;
		CAMReportForm cams = null;
		try
		{
			cashOnHandValues = dbFacade.getUtilityDBHandler().getPreviousPeriodsCashOnHand(company.getCompanyID(), 3, game.getCurrentPeriodNumber()-1);
			cams = new CAMReportForm(dbFacade.getCAMDBHandler().getCAMs(company, game.getCurrentPeriodNumber() - 1));
		}
		catch (BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
		catch (ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
		}

		long currentCash;
		long prevCash;
		long prevCash2;
		double cashAvg = 0;
		long newWIPLimit;

		if(cashOnHandValues.size() < 1)
		{
			//If there's an error ignore the reeval
			log.error("Company " + company.getCompanyID() + ": Error: No cash-on-hand for computing bond limits");
			return;
		}
		else if(cashOnHandValues.size() == 1)
		{
			cashAvg = (double) ((Long) cashOnHandValues.elementAt(0)).longValue();
		}
		else if(cashOnHandValues.size() == 2)
		{
			currentCash = ((Long) cashOnHandValues.elementAt(0)).longValue();
			prevCash = ((Long) cashOnHandValues.elementAt(1)).longValue();

			cashAvg = (.4d *  (double) prevCash) + (.6d * (double) currentCash);
		}
		else if(cashOnHandValues.size() > 2)
		{
			currentCash = ((Long) cashOnHandValues.elementAt(0)).longValue();
			prevCash = ((Long) cashOnHandValues.elementAt(1)).longValue();
			prevCash2 = ((Long) cashOnHandValues.elementAt(2)).longValue();

			cashAvg = (.1d * (double) prevCash2) + (.3d *  (double) prevCash) + (.6d * (double) currentCash);
		}

		if(log.isDebugEnabled())
		{
			log.debug("Company " + company.getCompanyID() + ": Cash avg: " + cashAvg);
		}

		double loanBalanceAdjustment = (double)balanceSheet.getLoanBalances() * .9d;
		double loanAdjustedCashAvg = cashAvg - loanBalanceAdjustment;

		if(log.isDebugEnabled())
		{
			log.debug("Company " + company.getCompanyID() + ": Loan Adj Cash avg: " + loanAdjustedCashAvg);
		}

		newWIPLimit = Math.round((double) loanAdjustedCashAvg / ((double) company.getWIPLimitPercent() / 100d));
		//System.out.println("New WIP Limit: " + newWIPLimit);
		newWIPLimit = newWIPLimit - (newWIPLimit % 100000);
		//System.out.println("New WIP Limit: " + newWIPLimit);
		newWIPLimit = BondLimitConstraints.applyWIPLimitMinimum(newWIPLimit);

		if(log.isDebugEnabled())
		{
			log.debug("Company " + company.getCompanyID() + ": New WIP Limit: " + newWIPLimit);
		}

		company.setWIPLimit(newWIPLimit);


		int financialLiquidity = 3;
		int financialSuccess = 3;
		int pace = 3;

		if(cams == null)
		{
			//If there's an error ignore the reeval
			log.error("Company " + company.getCompanyID() + ": Error: No cash-on-hand for computing bond limits");
			return;
		}

		financialLiquidity = cams.getFinancialLiquidity();
		financialSuccess = cams.getFinancialSuccess();
		pace = cams.getPace();

		if(log.isDebugEnabled())
		{
			log.debug("Company " + company.getCompanyID() + ": financial liquiduty: " + financialLiquidity);
			log.debug("Company " + company.getCompanyID() + ": financialSuccess: " + financialSuccess);
			log.debug("Company " + company.getCompanyID() + ": pace: " + pace);
		}

		int newPerJobLimitPercent = company.getPerJobLimitAsWIPPercent();
		if(financialLiquidity < 2)
		{
			newPerJobLimitPercent -= 5;
		}
		else if(financialLiquidity == 2)
		{
			newPerJobLimitPercent -= 2;
		}
		else if(financialLiquidity == 3)
		{
			newPerJobLimitPercent += 1;
		}
		else if(financialLiquidity == 4)
		{
			newPerJobLimitPercent += 3;
		}
		else if(financialLiquidity > 4)
		{
			newPerJobLimitPercent += 5;
		}

		if(financialSuccess < 2)
		{
			newPerJobLimitPercent -= 5;
		}
		else if(financialSuccess == 2)
		{
			newPerJobLimitPercent -= 2;
		}
		else if(financialSuccess == 3)
		{
			newPerJobLimitPercent += 1;
		}
		else if(financialSuccess == 4)
		{
			newPerJobLimitPercent += 3;
		}
		else if(financialSuccess > 4)
		{
			newPerJobLimitPercent += 5;
		}

		if(pace < 2)
		{
			newPerJobLimitPercent -= 5;
		}
		else if(pace == 2)
		{
			newPerJobLimitPercent -= 2;
		}
		else if(pace == 3)
		{
			newPerJobLimitPercent += 1;
		}
		else if(pace == 4)
		{
			newPerJobLimitPercent += 3;
		}
		else if(pace > 4)
		{
			newPerJobLimitPercent += 5;
		}		log.debug("Company " + company.getCompanyID() + ": newPerJobLimitPercent: " + newPerJobLimitPercent);


		newPerJobLimitPercent = Math.min(100, newPerJobLimitPercent);
		newPerJobLimitPercent = Math.max(30, newPerJobLimitPercent);
		company.setPerJobLimitAsWIPPercent(newPerJobLimitPercent);

		log.debug("Company " + company.getCompanyID() + ": newPerJobLimitPercent: " + newPerJobLimitPercent);

		long newPerJobLimit = Math.round((double) newWIPLimit * ( ((double) newPerJobLimitPercent) / 100d));
		log.debug("Company " + company.getCompanyID() + ": newPerJobLimit: " + newPerJobLimit);
		newPerJobLimit = BondLimitConstraints.applyPerJobLimitMinimum(newPerJobLimit);
		company.setPerJobLimit(newPerJobLimit);

		log.debug("Company " + company.getCompanyID() + ": newPerJobLimit: " + newPerJobLimit);

		try
		{
			dbFacade.getCompanyDBHandler().updateCompany(game.getGame(), company);
		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
		}
		catch (ObjectNotFoundException onfe)
		{
			log.fatal("", onfe);
		}

        game.sendInGameMessageToCompany(new Message("TBC",
					  					  		    company,
													"We have recently reviewed your company and reset your bond limits as follows:  Your WIP Limit is now " + NumberConverter.stringValue(newWIPLimit) + ", and your Per Job Limit is now " + NumberConverter.stringValue(newPerJobLimit) + ".  We will re-evaluate your bond limits periodically based on the success of your company.",
													"Bond Limit Re-evaluation",
													game.getGame().getCurrentPeriodNumber(),
													Message.TEXT_ONLY,
													-1
												   )
									   );

	}
