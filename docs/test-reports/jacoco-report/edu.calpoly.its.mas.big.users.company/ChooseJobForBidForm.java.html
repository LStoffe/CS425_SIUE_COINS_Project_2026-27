package edu.calpoly.its.mas.big.users.company;

import java.util.Vector;
import java.util.ArrayList;
import java.util.TreeMap;
import java.util.Iterator;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;
import edu.calpoly.its.mas.big.game.Job;
import edu.calpoly.its.mas.big.users.company.Bid;

public class ChooseJobForBidForm extends ActionForm
{
	private int jobID;

	/* Both of these TreeMaps are indexed by jobID */
	private TreeMap jobs;
	private TreeMap bids = new TreeMap();

    public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
    {
        ActionErrors errors = new ActionErrors();

        if (jobs.get(new Integer(jobID)) == null) {
            errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
        }

		return errors;
	}


	public int getJobID()
	{
		return jobID;
	}

	public void setJobID(int jobID)
	{
		this.jobID = jobID;
	}

	public int getJobType()
	{
		return ((Job) jobs.get(new Integer(jobID))).getJobType();
	}

	public TreeMap getJobs()
	{
		return jobs;
	}

	public void setJobs(Vector jobs)
	{
		if (this.jobs == null)
			this.jobs = new TreeMap();

		// Walk through the Vector, and build the TreeMap
	    Iterator jobIter = jobs.iterator();
		while (jobIter.hasNext()) {
			Job job = (Job) jobIter.next();
			this.jobs.put(new Integer(job.getJobID()), job);
		}

	}

	public int getNumberOfJobs()
	{
		if (jobs == null)
			return 0;
		else
			return jobs.size();
	}

	public void setBid(int jobID, Bid bid)
	{
		bids.put(new Integer(jobID), bid);
	}

	public Bid getBid(int jobID)
	{
		return (Bid) bids.get(new Integer(jobID));
	}


	public long getBidAmount(int jobID)
	{

		Bid bid = (Bid) bids.get(new Integer(jobID));
		if (bid == null)
			return -1;
		else
			return bid.getBidAmount();
	}

	public double getBidAmountUP(int jobID)
	{

		Bid bid = (Bid) bids.get(new Integer(jobID));
		if (bid == null)
			return -1;
		else
			return bid.getBidAmountUP();
	}

}
