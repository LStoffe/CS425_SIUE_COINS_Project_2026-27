package edu.calpoly.its.mas.big.users.company;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * ActionForm used to hold the dollar amount of a loan request.
 *
 * @author Kevin Craig
 * @version $Revision: 1.10 $ $Date: 2002/04/26 18:03:36 $
 */
public class RequestLoanForm extends ActionForm
{
  private String loanAmount;
  private int APR;
  private int numberOfLoans;
  private int maxLoans;
  private long maxLoanAmount;
  private int repaymentPeriods;

	/**
     * Insures that this loan request is allowed and that the amount requested is a positive number
     * less than or equal to the maximum amount
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

		//First make sure the company doens't already have the maximum number of loans allowed
		if(numberOfLoans >= maxLoans) //it should never be greater than, but just in case
		{
			errors.add("maxLoansExceeded", new ActionMessage("error.loan.maxLoansExceeded"));
		}
		else
		{
			//Make sure the amount is not null
			if((loanAmount == null) || (loanAmount.length() < 1) || loanAmount.equals("$"))
			{
				errors.add("required", new ActionMessage("error.loan.required"));
			}
			else
			{
				//Make sure the String is a valid number
				if(!NumberConverter.isValidMoneyString(loanAmount))
				{
					errors.add("mismatch", new ActionMessage("error.loan.mismatch"));
				}
				else
				{
					long amount = NumberConverter.longValue(loanAmount);

					//Test if loan is within allowed boundaries ( 0 < amount <= maxLoanAmount)
					if (amount <= 0)
						errors.add("negative", new ActionMessage("error.loan.negative"));
					else if (amount > maxLoanAmount)
					    errors.add("maxLoanAmountExceeded", new ActionMessage("error.loan.maxLoanAmountExceeded"));

				}
			}
		}
		return errors;
	}


	/**
	 * Returns the amount of this loan as an long
	 */
	public long getAmount()
	{
 	   return NumberConverter.longValue(loanAmount);
	}

	/**
	 * Returns the dollar amount of this loan request
	 */
	public String getLoanAmount()
	{
	    return loanAmount;
	}

 	/**
  	 * Sets the dollar amount of this loan request
  	 */
  	public void setLoanAmount(String loanAmount)
  	{
  	    this.loanAmount = loanAmount;
  	}

  	/**
  	 * Returns the annual percentage interest rate that will be applied to this loan request
  	 */
  	public int getAPR()
  	{
		  return APR;
  	}

  	/**
  	 * Sets the annual percentage interest rate that will be applied to this loan request
  	 */
  	public void setAPR(int APR)
  	{
		  this.APR = APR;
  	}

  	/**
  	 * Sets the number of active loans held by the company making this request
  	 */
  	public void setNumberOfLoans(int numberOfLoans)
  	{
		  this.numberOfLoans = numberOfLoans;
  	}

  	/**
  	 * Sets the maximum number of active loans a company can have at any one time
  	 */
  	public void setMaxLoans(int maxLoans)
  	{
		  this.maxLoans = maxLoans;
  	}

  	/**
  	 * Sets the maximum dollar amount for any single loan
  	 */
  	public void setMaxLoanAmount(long maxLoanAmount)
  	{
		  this.maxLoanAmount = maxLoanAmount;
  	}

  	/**
  	 * Sets the number of periods in loan repayment
  	 */
	public void setRepaymentPeriods(int repaymentPeriods)
	{
		this.repaymentPeriods = repaymentPeriods;
	}

	/**
	 * Returns the number of periods in loan repayment
	 */
	public int getRepaymentPeriods()
	{
		return repaymentPeriods;
	}

}
