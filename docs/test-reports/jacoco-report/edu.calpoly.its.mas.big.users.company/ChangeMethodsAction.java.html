package edu.calpoly.its.mas.big.users.company;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

public class ChangeMethodsAction extends Action
{
	public ActionForward execute(ActionMapping mapping,
					     		 ActionForm form,
					     		 HttpServletRequest request,
				         	     HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
	    if(!isCancelled(request))
	    {

	        ActionMessages errors = new ActionMessages();

	        // Check for existance of Game object in user's session
            ServletContext context = servlet.getServletContext();
            HttpSession session = request.getSession();
	        Game game = (Game) session.getAttribute("game");
	        Company company = (Company) session.getAttribute("company");

            LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

            // If for any reason the company should not be allowed to login, route the
            // company to an error message page.
	        if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		      return (mapping.findForward("loginFailure"));

			Job job = (Job) session.getAttribute("job");

	        int[] methods = new int[9];

	        ChangeMethodsForm changeMethodsForm = (ChangeMethodsForm) form;

            methods[0] = Integer.parseInt(changeMethodsForm.getActivity1());
            methods[1] = Integer.parseInt(changeMethodsForm.getActivity2());
            methods[2] = Integer.parseInt(changeMethodsForm.getActivity3());
            methods[3] = Integer.parseInt(changeMethodsForm.getActivity4());
            methods[4] = Integer.parseInt(changeMethodsForm.getActivity5());
            methods[5] = Integer.parseInt(changeMethodsForm.getActivity6());
            methods[6] = Integer.parseInt(changeMethodsForm.getActivity7());
            methods[7] = Integer.parseInt(changeMethodsForm.getActivity8());
            methods[8] = Integer.parseInt(changeMethodsForm.getActivity9());

            BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));
            try
            {
				dbFacade.getCurrentMethodDBHandler().updateCurrentMethods(game, job, methods);

                //************************************************************************
				//********************* NEW FOR BIG v3 ***********************************
				//************************************************************************
				//Refigure the estimated cost to complete based on the new methods

				//Get workdays per month parameter
				Hashtable misc = dbFacade.getGameParametersDBHandler().getMiscParameters(game);
				int workdaysPerMonth = ((Integer) misc.get("workdaysPerMonth")).intValue();

				//Create an estimate for cost to complete
				int estCostToComplete = 0;
				estCostToComplete = JobCostEstimator.getEstimatedCost(game.getCurrentPeriodNumber(),
																	  game.convertDateToPeriod(job.getCompletionDeadline()),
				  													  game.getEstimatedPercentageOfLastPeriodWorked(job.getCompletionDeadline()),
																	  activitySetupLite(game, job, getDataSource(request)),
																	  workdaysPerMonth * 2,
																	  job.getFixedFieldOverhead(),
																	  job.getVariableFieldCharge(),
																	  job.getLiquidatedDamages()
						  											 );

				dbFacade.getContractsReportsDBHandler().updateEstimatedCostToComplete(game,
						  														company,
						  														job,
						  														estCostToComplete);

				//Change the taxes and equity based on the change in revenue that will
				//result from costs being changed (using cost-based method of % complete)
				IncomeStatementGetter getter = new IncomeStatementGetter(game, company, dbFacade);

				IncomeStatementForm incomeStatement = getter.getIncomeStatementComputeTaxes(game.getCurrentPeriodNumber() - 1);

				BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, game.getCurrentPeriodNumber() - 1, company);
				balanceSheet.setEquity(incomeStatement.getEndRetainedEarnings());

				BalanceSheetForm prevBalanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, game.getCurrentPeriodNumber() - 2, company);
				balanceSheet.setPeriod(game.getCurrentPeriodNumber() - 1); //update gets the period number from the report


				//This checks if taxes were paid in the period the report is for.
				//Taxes are paid at the start of a year.  Each year contains 6 periods,
				//(this is hardcoded for now).  Reports for a period are
				//shown the subsequent period.  But to determine if a report
				//should contains taxes the report previous to the report
				//period must be examined to determine if it is a year end.
				//The minus 2 is one period back for the report, then another period back
				//to check for year end.

				//Example: If this is period 8 then the current reports are for period 7.
				//Since period 7 is a year taxes should be paid at the start of the
				//period.  This means the only taxes at the end of period 7 should be those
				//incurred during period 7.  However if this were the report for a period
				//not at the start of a year, then the current balance sheet taxes should
				//be the running total of taxes for the current year.  To achieve this
				//the previous total for taxes must be added to the current tax adjustment.
		 		if(((game.getCurrentPeriodNumber() - 2) % 6) == 0)
		 		{
         			balanceSheet.setTaxes(incomeStatement.getIncomeTaxes());
		 		}
		 		else
		 		{
         			balanceSheet.setTaxes(prevBalanceSheet.getTaxes() + incomeStatement.getIncomeTaxes());
		 		}

				dbFacade.getBalanceSheetDBHandler().updateBalanceSheet(game, company, balanceSheet);


				//************************************************************************
				//************************************************************************
				//************************************************************************

				dbFacade.close();
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}
			catch (ObjectNotFoundException onfe)
			{
				log.fatal("", onfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch(ParametersNotFoundException pnfe)
			{
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
  	  		}
			finally
			{
				try
				{
        		 	dbFacade.close();
				}
				catch (BIGDatabaseFacadeException e)
				{
					log.fatal(getClass().getName(), e);
 					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}

				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}


	/**
	 * Returns Job j's activities as an array of Activity objects.
	 * This methods will load activity data and the current methods for each
	 * activity so this information can be accessed, however it does
	 * not setup network dependencies, so activities can not be worked.
	 *
	 * <br><br>
	 * New for BIG v3
	 */
	private Activity[] activitySetupLite(Game game, Job job, DataSource dataSource) throws BIGDatabaseFacadeException, ObjectNotFoundException
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

    	Activity[] activities = null;
		int[] currentMethods = null;

		BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(dataSource);

		activities = dbFacade.getActivityDBHandler().getActivities(game, job, game.getCurrentPeriodNumber() - 1);

	    //Retrieve the current method id's to be used on the activities
		currentMethods = dbFacade.getCurrentMethodDBHandler().getCurrentMethods(game, job);

		Method[][] methods = new Method[9][5];

        if (job.getTypeModifier() == Job.HEAVY)
        {
		   methods[0] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 10);
		   methods[1] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 11);
		   methods[2] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 12);
		   methods[3] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 13);
		   methods[4] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 14);
		   methods[5] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 15);
		   methods[6] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 16);
		   methods[7] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 17);
		   methods[8] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 18);
		}
		else
		{
		   methods[0] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 1);
		   methods[1] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 2);
		   methods[2] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 3);
		   methods[3] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 4);
		   methods[4] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 5);
		   methods[5] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 6);
		   methods[6] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 7);
		   methods[7] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 8);
		   methods[8] = dbFacade.getCurrentMethodDBHandler().getAllMethods(game, 9);
		}

		for(int j = 0; j < 9; j++)
      	{
		    activities[j].setMethod(methods[j][currentMethods[j]-1]);
		}

		dbFacade.close();

	    return activities;
	}

}
