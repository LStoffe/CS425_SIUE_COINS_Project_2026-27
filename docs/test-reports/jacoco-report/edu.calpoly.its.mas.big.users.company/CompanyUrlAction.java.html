package edu.calpoly.its.mas.big.users.company;

import javax.servlet.ServletContext;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.ObjectNotFoundException;

//Logging imports
import edu.calpoly.lib.multimedia.big.log.BIGLog;
import edu.calpoly.lib.multimedia.big.log.BIGLogFactory;

import edu.calpoly.its.mas.big.users.LoginBroker;
import edu.calpoly.its.mas.big.users.LoginForm;
import edu.calpoly.its.mas.big.game.Game;


public class CompanyUrlAction extends Action {

	public ActionForward execute(ActionMapping mapping,
		           				 ActionForm form,
		           				 HttpServletRequest request,
		           				 HttpServletResponse response)
	{
	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

        int isFirstURL = (Integer) session.getAttribute("isFirstURL");

        if(isFirstURL == 1)
        {
            isFirstURL = 0;
            session.setAttribute("isFirstURL", isFirstURL);

            //Tried this, it compiled but didn't solve the problem
            //ActionForm loginform = (ActionForm) session.getAttribute("loginform");
            //form = loginform;

            LoginForm loginform = (LoginForm) session.getAttribute("loginform");
            String mission =    loginform.getMissionStatement();
            String corevalues = loginform.getCoreValues();
            String companyurl = loginform.getCompanyUrl();

            session.setAttribute("oldCompanyurl", companyurl);

            //Note - A major headache here - This class does not find the existing LoginForm object and
            //creates it's own instance.  This is how I fixed that problem.
            ((LoginForm) form).setMissionStatement(mission);
            ((LoginForm) form).setCoreValues(corevalues);
            ((LoginForm) form).setCompanyUrl(companyurl);
		    session.setAttribute("loginform", form );

            //Tried to call JSP file directly, but it didn't work (compilation error)
            //Followed the example in file: LoginVerificationTag.java
            //PageContext pageContext = JspFactory.getPageContext();
            //pageContext.forward("/missionStatement.jsp");

            return (mapping.findForward("tojsp"));
        }

        else
        {
		   LoginForm theForm = (LoginForm) form;

	       //Process request if it was not cancelled
           if(!isCancelled(request))
           {
               //save the updated Company URL to the database
		       BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));
               game = (Game) session.getAttribute("game");
               Company theCompany = (Company) session.getAttribute("company");

               String missionStatement = theForm.getMissionStatement();
               String coreValues = theForm.getCoreValues();
               String companyUrl = theForm.getCompanyUrl();

               theCompany.setMissionStatement(missionStatement);
               theCompany.setCoreValues(coreValues);
               theCompany.setCompanyUrl(companyUrl);

               BIGLog log =  BIGLogFactory.getLog();
               errors = new ActionMessages();
               try
		       {
                   dbFacade.getCompanyDBHandler().updateCompany(game, theCompany);
			   }
			   catch(BIGDatabaseFacadeException BIGDbe)
			   {
			   	   log.fatal("", BIGDbe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			   }
			   catch(ObjectNotFoundException onfe)
			   {
			 	   log.fatal("", onfe);
				   errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			   }
			   finally
			   {
					try
					{
 	       		 		dbFacade.close();
					}
					catch (BIGDatabaseFacadeException e)
					{
						log.fatal(getClass().getName(), e);
 						errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
					}
			   }

               session.setAttribute("loginform", form);
		       return (mapping.findForward("success"));
	       }
	       else  //Request was cancelled
		   {
			   String oldCompanyurl = (String) session.getAttribute("oldCompanyurl");
			   theForm.setCompanyUrl(oldCompanyurl);
			   session.setAttribute("loginform", theForm);

			   return (mapping.findForward("cancelled"));
     	   }
        }
	}
}
