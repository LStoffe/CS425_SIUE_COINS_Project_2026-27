 * by requesting the current one from a <code>BalanceSheetDBHandler</code> then updating

  /**
   * Returns true if the balance sheet balances.
   * There is $5 of leeway for both assests and liabilitys/equity
   * to account for rounding errors.  This means true will be returned if these
   * values are within $10 of each other
   */
  public boolean assertBalance()
  {
	  boolean balance = false;
	  long l = getTotalAssets() - getTotalLiabilitiesAndEquity();
	  if ((l >= -5) && (l <= 5))
	  {
		  balance = true;
	  }

	  return balance;
  }

  public void setNewReportDate(int periodNumber, Date startYear)
  {
	  //Multiply 2 times the number of periods -1 to get the month id of the first month.  Month's are zero based in the Calendar class
	  int months = 2 * (periodNumber - 1);

	  int firstMonthOfThisPeriod = months;

      Calendar calendar = new GregorianCalendar();
	  calendar.setTime(startYear);

      // save startYear as an int -- available for calls using getStartYear method
      setStartYear(calendar.get(Calendar.YEAR));

	  //Add the number of months passed to the start date to get the date of the first month in this period
	  calendar.add(Calendar.MONTH, firstMonthOfThisPeriod);

	  //Add one to the month of the first month date to get the date of the second month of this period
	  //Calendar month will now be set to the second month
	  calendar.add(Calendar.MONTH, 1);
	  Date secondMonth = calendar.getTime();

	  SimpleDateFormat formatter = new SimpleDateFormat("MMMMM");

	  String secondMonthString = formatter.format(secondMonth);

	  formatter = new SimpleDateFormat("yyyy");

	  String yearString = formatter.format(secondMonth);

      // obtain the maximum value that the date field could have
	  int lastDate = calendar.getActualMaximum(Calendar.DATE);

      String newReportDate = secondMonthString + " " + lastDate + ", " + yearString;

      setReportDate(newReportDate);
  }
