/**
 *  Handles requests to view a income statement.
 *
 *  @author Kevin Craig
 */
public class IncomeStatementAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
		           				 ActionForm form,
		           				 HttpServletRequest request,
		           				 HttpServletResponse response)
		{
		   Log log = LogFactory.getLog(this.getClass());
		   if(log.isDebugEnabled())
		   {

		       log.debug(this.getClass() + " beginning execute method");
		   }

	       ActionMessages errors = new ActionMessages();

	       // Check for existance of Game object in user's session
           ServletContext context = servlet.getServletContext();

           HttpSession session = request.getSession();
           if(session == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

	       Game game = (Game) session.getAttribute("game");
           if(game == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

		   Company company = null;

           String periodNumber = (String) request.getParameter("period");
           User user = (User) session.getAttribute("user");

           if(user instanceof Company)
           {
              LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
              // If for any reason the company should not be allowed to login, route the
              // company to an error message page.
	          if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	          {
		        return (mapping.findForward("loginFailure"));
		      }
			  company = (Company) user;
 		   }
	  	   else if(user instanceof Admin)
	   	   {
				company = (Company) session.getAttribute("company");
           }


		   int reportPeriod;

           if (periodNumber == null)
           {
			   reportPeriod = game.getCurrentPeriodNumber() - 1;
		   }
		   else
		   {
			   reportPeriod = Integer.parseInt(periodNumber);
		   }

           //Dummy income statement.  If an IncomeStatement doesn't exist for a company/period
           //because the company was created after that period, this dummy will prevent
           //null pointer exceptions
		   IncomeStatementForm incomeStatement = new IncomeStatementForm();

		   if(reportPeriod > 0)
		   {

		   	  try
		   	  {
		   		  BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

				  IncomeStatementGetter getter = new IncomeStatementGetter(game, company, dbFacade);

				  //If this is the start of a year income statement the taxes must be computed
				  //each time the income statement is accessed.
				  //if(((reportPeriod - 1) % 6) == 0)
				  //{
				  	  incomeStatement = getter.getIncomeStatementComputeTaxes(reportPeriod);
				  //}
				  //else
				  //{
				  	 // incomeStatement = getter.getIncomeStatement(reportPeriod);
				  //}

				  dbFacade.close();
		   	  }
		      catch (BIGDatabaseFacadeException bigdbe)
		      {
				  log.fatal("", bigdbe);
				  errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		      }
		      catch (ObjectNotFoundException onfe)
		      {
				  //Do nothing.  This company must have been created after the report period.
		      }
		      catch (ParametersNotFoundException pnfe)
		      {
				  log.fatal("", pnfe);
				  errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
		      }
		   }
		   else
		   {
				//do nothing if there is no report for this period
		   }

		   incomeStatement.setCompanyName(company.getTitle());
		   incomeStatement.setReportDate(game.getDateString(reportPeriod));
		   incomeStatement.setPeriod(reportPeriod);

		   session.setAttribute("incomeStatement", incomeStatement);

	       if (!errors.isEmpty())
		   {
			  if(log.isDebugEnabled())
			  {
			      log.debug(this.getClass() + " request returned with errors");
			  }
		   	  // Report any errors we have discovered back to the original form
		   	  saveErrors(request, errors);
		   	  saveToken(request);
		   	  return (new ActionForward(mapping.getInput()));
		   }
		   else
		   {
			   if(log.isDebugEnabled())
			   {
			       log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			   }
 			   return (mapping.findForward("success"));
		   }

	   } // end execute()

}
