package edu.calpoly.lib.multimedia.big.reports;

import edu.calpoly.lib.multimedia.big.game.NegotiatedJob;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Holds info a single job in the contracts in progress report or completed contracts report.
 * The <code>get</code> method names correspond roughly with the fields required in the BIGv3 SRS,
 * requirement 1.2.9 Contracts in Progress Report, and 1.2.10 Completed Contracts Report.
 * (The fields in 1.2.10 are a subset of those in 1.2.9 with the word "Final" added before some of them.)
 *
 * @author Kevin Craig
 */
public class Contract
{
	private Log log;

	private int jobID;
	private int jobTypeID;
	private int periodID;  //period this contract is for
	private int originalContract;
	private int changeOrders;
	private int adjustedContract;
	private int estimatedCostToComplete;
	private int billedToDate;
	private int currentCost;
	private int previousCosts;
    private int revenueRecorded;
    private int jobTypeModifier;

	private int estimatedTotalCost;
	private boolean autoComputeEstCost;

	private boolean negotiated;

	private NegotiatedJob negotiatedJob;

	public Contract(int jobID,
					     int jobTypeID,
					     int jobTypeModifier,
					     int originalContract,
					     int changeOrders,
					     int estimatedCostToComplete,
					     boolean autoComputeEstCost,
					     int billedToDate,
					     int currentCost,
					     int previousCosts,
					     int revenueRecorded)
	{
		this.jobID = jobID;
		this.jobTypeID = jobTypeID;
		this.jobTypeModifier = jobTypeModifier;
		this.originalContract = originalContract;
		this.changeOrders = changeOrders;
		this.adjustedContract = changeOrders + originalContract;
		this.estimatedCostToComplete = estimatedCostToComplete;
		this.autoComputeEstCost = autoComputeEstCost;
		this.billedToDate = billedToDate;
		this.currentCost = currentCost;
		this.previousCosts = previousCosts;
		this.revenueRecorded = revenueRecorded;

		this.estimatedTotalCost = estimatedCostToComplete + currentCost + previousCosts;

		log = LogFactory.getLog(this.getClass());
	}

	public int getPeriodID()
	{
		return periodID;
	}

	public void setPeriodID(int periodID)
	{
		this.periodID = periodID;
	}

    public boolean getNegotiated()
    {
		return negotiated;
	}

	public void setNegotiated(boolean negotiated)
	{
		this.negotiated = negotiated;
	}

    public NegotiatedJob getNegotiatedJob()
    {
		return negotiatedJob;
	}

	public void setNegotiatedJob(NegotiatedJob negotiatedJob)
	{
		this.negotiatedJob = negotiatedJob;
	}

	public int getJobID()
	{
		return jobID;
	}

	public int getJobTypeID()
	{
		return jobTypeID;
	}

	public int getJobTypeModifier()
	{
		return jobTypeModifier;
	}

	public boolean getAutoComputeEstCost()
	{
		return autoComputeEstCost;
	}

	public int getOriginalContract()
	{
		return originalContract;
	}

	public int getChangeOrders()
	{
		return changeOrders;
	}

	public int getAdjustedContract()
	{
		return adjustedContract;
	}

	public int getCostIncurred()
	{
		return getCurrentCostIncurred() + getPreviousCostIncurred();
	}

	public int getEstimatedCostToComplete()
	{
		return estimatedCostToComplete;
	}

	public int getEstimatedTotalCost()
	{
		return estimatedTotalCost;
	}

    public int getEstimatedTotalGrossMargin()
    {
		return adjustedContract - estimatedTotalCost;
	}

    public float getEstimatedTotalGrossMarginPercent()
    {
		return ((float) getEstimatedTotalGrossMargin()) / ((float) adjustedContract);
	}

    public float getPercentWorkComplete()
    {
		return ((float) getCostIncurred()) / ((float) getEstimatedTotalCost());
	}

    public int getBilledToDate()
    {
		return billedToDate;
	}

	public int getUnbilledPortionOfContract()
	{
		return getAdjustedContract() - getBilledToDate();
	}

	public int getRevenueEarned()
	{
		return getCurrentRevenueEarned() + getPreviousRevenueEarned();
	}

	public int getRevenueToBeEarned()
	{
		return getAdjustedContract() - getRevenueEarned();
	}

	public int getOverbilling()
	{
		if (getBilledToDate() > getRevenueEarned())
		{
			return getBilledToDate() - getRevenueEarned();
		}
		else
		{
			return 0;
		}
	}

	public int getUnderbilling()
	{
		if (getBilledToDate() < getRevenueEarned())
		{
			return getRevenueEarned() - getBilledToDate();
		}
		else
		{
			return 0;
		}
	}

    public int getGrossMarginEarned()
    {
		return getRevenueEarned() - getCostIncurred();
	}

	public float getGrossMarginEarnedPercent()
	{
		return ((float) getGrossMarginEarned()) / ((float) getRevenueEarned());
	}

    public int getRevenuePreviouslyRecorded()
    {
		return revenueRecorded;
	}

	public int getCurrentRevenueRecorded()
	{
		return getRevenueEarned() - getRevenuePreviouslyRecorded();
	}

	public float getCurrentPercentWorkComplete()
	{
		return ((float) getCurrentCostIncurred()) / ((float) getEstimatedTotalCost());
	}

    public int getCurrentCostIncurred()
    {
		return currentCost;
	}

    public int getCurrentRevenueEarned()
    {
		if(getNegotiated())
		{
			if(getCostIncurred() > negotiatedJob.getAgreedCost())
			{
/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + ": Returning current revenue for at or over cost negotiated job: " + (int) (negotiatedJob.getMaxPrice() - getRevenuePreviouslyRecorded()));
				} */
				return (int) (negotiatedJob.getMaxPrice() - getRevenuePreviouslyRecorded());
			}
			else
			{
				//If this contract is for the period the job completed, add in the share of savings
				long shareOfSavings = negotiatedJob.getShareOfSavings(getCostIncurred());
				int currentRevenue = (int) (negotiatedJob.getCostPlusProfit(getCostIncurred()) - getRevenuePreviouslyRecorded());

				// log.debug(this.getClass() + " ContractPeriod : " + periodID + " PeriodCompleted: " + negotiatedJob.getPeriodCompleted());


				if((periodID >= negotiatedJob.getPeriodCompleted()) && (negotiatedJob.getPeriodCompleted() > 0))
				{
/*					if(log.isDebugEnabled())
					{
						log.debug(this.getClass() + ": Current revenue includes share of savings");
					} */
					currentRevenue += shareOfSavings;
				}

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + ": Returning current revenue for undercost negotiated job: " + currentRevenue);
				} */
			    return currentRevenue;
			}
		}
		else
		{
/*			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + ": Returning current revenue for big job: " + Math.round(getAdjustedContract() * getCurrentPercentWorkComplete()));
			} */
		    return Math.round(getAdjustedContract() * getCurrentPercentWorkComplete());
		}
	}

    public int getCurrentGrossMargin()
    {
		return getCurrentRevenueEarned() - getCurrentCostIncurred();
	}

	public float getCurrentGrossMarginPercent()
	{
		if (getCurrentRevenueEarned() < 1)
		{
			return 0;
		}
		else
		{
			return ((float) getCurrentGrossMargin()) / ((float) getCurrentRevenueEarned());
		}
	}

	public float getPreviousPercentWorkComplete()
	{
		return ((float) getPreviousCostIncurred()) / ((float) getEstimatedTotalCost());
	}

    public int getPreviousCostIncurred()
    {
		return previousCosts;
	}

    public int getPreviousRevenueEarned()
    {
		if(getNegotiated())
		{
			return revenueRecorded;
		}
		else
		{
		    return Math.round(getAdjustedContract() * getPreviousPercentWorkComplete());
		}
	}

    public int getPreviousGrossMargin()
    {
		return getPreviousRevenueEarned() - getPreviousCostIncurred();
	}

	public float getPreviousGrossMarginPercent()
	{
		if (getPreviousRevenueEarned() < 1)
		{
			return 0;
		}
		else
		{
			return ((float) getPreviousGrossMargin()) / ((float) getPreviousRevenueEarned());
		}
	}

}
