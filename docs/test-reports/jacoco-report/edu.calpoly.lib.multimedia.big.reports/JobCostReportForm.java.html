package edu.calpoly.lib.multimedia.big.reports;

import edu.calpoly.its.mas.big.reports.ReportForm;
import edu.calpoly.its.mas.big.reports.FinancialReportForm;

import java.util.Vector;
import java.util.Iterator;

/**
 * Contains cost info for all jobs worked during a period.
 * See BIGv3 SRS for a description of the current fields in
 * the Job Cost Report.
 *
 * This report is in effect a report of reports.
 * JSP files are intended to run through each job in the report and get the info for each
 * job.  All <code>get</code> methods in this class apply to the current job.
 *
 * @author Kevin Craig
 */
public class JobCostReportForm extends ReportForm
{

    private FinancialReportForm frf;

    /**
     * Creates a new <code>JobCostReportForm</code> to wrap around a
     * <code>FinancialReportForm</code>.
     */
    public JobCostReportForm(FinancialReportForm frf)
    {
		this.frf = frf;
	}

    /**
     * Returns the title of the company that is the subject of this report.
     * @return Title of company that is the subject of this report.
     */
    public String getCompanyName()
    {
		if(frf != null)
		{
	        return frf.getCompanyName();
	    }
	    else
	    {
			return null;
		}
    }


    /**
     * Sets the title of the company that is the subject of this report.
     * @param companyName Title of company that is the subject of this report
     */
    public void setCompanyName(String companyName)
    {
		if(frf != null)
		{
    		frf.setCompanyName(companyName);
		}
    }

    /**
     * Returns true if there are more jobs left in this report.  False o/w
     */
	public boolean jobsLeft()
	{
		if(frf != null)
		{
       		return frf.jobsLeft();
		}
		else
		{
			return false;
		}
	}

    /**
     * Advances the current job to the next job in this report.
     */
	public void nextJob()
	{
		if(frf != null)
		{
			frf.nextJob();
		}
	}

    /**
     * Returns the job id of the current job.  If there is no current job because the
     * underlying <code>FinancialReportForm</code> is <code>null</code>, -1 will be returned.
     */
	public int getJobID()
	{
		if (frf != null)
		{
			return frf.getJobID();
		}
		else
		{
			return -1;
		}
	}

	public String getJobType()
	{
		if( frf != null)
		{
			return frf.getJobType();
		}
		else
		{
			return null;
		}
	}

	public short getTypeModifier()
	{
		if( frf != null)
		{
			return frf.getTypeModifier();
		}
		else
		{
			return 0;
		}
	}

	public long getLaborCost()
	{
		if(frf != null)
		{
			return frf.getDirectLaborCost();
		}
		else
		{
			return 0;
		}
	}

	public long getMaterialsCost()
	{
		if(frf != null)
		{
			return frf.getDirectMaterialsCost();
		}
		else
		{
			return 0;
		}
	}

	public long getEquipmentCost()
	{
		if(frf != null)
		{
			return frf.getDirectEquipmentCost();
		}
		else
		{
			return 0;
		}
	}

	public long getSubcontractorCost()
	{
		if(frf != null)
		{
			return frf.getSubcontractorCost();
		}
		else
		{
			return 0;
		}
	}

	public long getFieldOverhead()
	{
		if(frf != null)
		{
			return frf.getFieldOverhead();
		}
		else
		{
			return 0;
		}
	}

	public int getDaysLate()
	{
		if(frf != null)
		{
			return frf.getDaysLate();
		}
		else
		{
			return 0;
		}
	}

	public long getLiquidatedDamagesPerDay()
	{
		if(frf != null)
		{
			return frf.getLiquidatedDamagesPerDay();
		}
		else
		{
			return 0;
		}
	}

	public long getLiquidatedDamages()
	{
		if(frf != null)
		{
			return frf.getLiquidatedDamages();
		}
		else
		{
			return 0;
		}
	}

	public long getTotalCosts()
	{
		return getLaborCost() + getMaterialsCost() + getEquipmentCost() + getSubcontractorCost() + getFieldOverhead() + getLiquidatedDamages();
	}

}
