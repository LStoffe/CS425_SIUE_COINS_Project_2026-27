package edu.calpoly.lib.multimedia.big.reports;

import java.util.Vector;
import edu.calpoly.its.mas.big.reports.ReportForm;

/**
 * Holds info for a the payments report.  This is really just a wrapper around
 * a vector of jobs that has fields for the standard report headers.
 *
 * @author Kevin Craig
 */
public class PaymentsReport extends ReportForm
{

  private String companyName;
  private Vector billingForms;

  /**
   * Creates an empty balance sheet.  An empty balance sheet can be used if a dummy balance sheet
   * is needed but should not be used for any real purpose because the data may be inconsistent with
   * data in other reports.
   */
  public PaymentsReport()
  {
  	  billingForms = new Vector();
  }

  public String getCompanyName()
  {
	  return companyName;
  }

  public void setCompanyName(String name)
  {
	  this.companyName = name;
  }

  public Vector getBillingForms()
  {
	  return billingForms;
  }

  public void setBillingForms(Vector billingForms)
  {
	  this.billingForms = billingForms;
  }


}
