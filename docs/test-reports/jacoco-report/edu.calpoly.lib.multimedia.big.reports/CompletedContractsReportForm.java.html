package edu.calpoly.lib.multimedia.big.reports;

import edu.calpoly.its.mas.big.reports.ReportForm;

import java.util.Vector;
import java.util.Iterator;

/**
 * Holds info for a Completed Contracts Report
 * <br><br> A <code>CompeltedContractsReportForm</code> contains each item appearing on the completed contracts
 * report in the BIGv3 SRS.<br><br>
 * This class can be given to a jsp file to display report data to the user.
 * Creation of new <code>CompletedContractsReportForm</code> objects should only be done by a <code>ContractsReportsDBHandler</code>
 * to ensure consistency with other reports.
 *
 * <br><br>
 * A <code>
 * ContractsReportDBHandler</code> can create a <code>CompletedContractsReportForm</code>
 * using data already stored in the database.
 *
 * <br><br>
 * A <code>ContractsInProgressReportForm</code> is comprised of data for many different contracts.
 * A pointer is maintained to the current contract and each get method where the returned value
 * is for a particular contract will return that value for the current contract.
 * The getTotalXXX methods return totals and are next affected by the current contract.
 *
 * @author Kevin Craig
 */
public class CompletedContractsReportForm extends ReportForm
{
  private String companyName;
  private Vector contracts;
  private Iterator contractsIterator;
  private Contract currentContract;

  private int totalCostIncurred = 0;
  private int totalUnbilled = 0;
  private int totalRevenueEarned = 0;

  private int totalOverbilled = 0;

  private int totalCurrentCostIncurred = 0;
  private int totalCurrentRevenueEarned = 0;
  private int totalCurrentRevenueRecordable = 0;

  private int totalPreviousCostIncurred = 0;
  private int totalPreviousRevenueEarned = 0;
  private int totalRevenuePreviouslyRecorded = 0;



  /**
   * Creates an empty contracts in progress report.  An empty contracts in progress report
   * can be used if a dummy contracts in progress report
   * is needed but should not be used for any real purpose because the data may be inconsistent with
   * data in other reports.
   */
  public CompletedContractsReportForm()
  {
      companyName = null;
  }

  /**
   * Constructor to be used by the database when creating a new <code>CompletedContractsReportForm</code>.
   * New <code>CompletedContractsReportForm</code>s should only ever be created from database data.
   * <br><br>
   * <b>Precondition:</b> The class instantiating <code>ContractsInProgressReportForm</code> implements
   * <code>edu.calpoly.lib.multimedia.big.db.ContractsReportsDBHandler</code>.
   */
  public CompletedContractsReportForm(String companyName, Vector contracts)
  {
	  this.companyName = companyName;
	  this.contracts = contracts;
	  contractsIterator = this.contracts.iterator();

	  while(contractsIterator.hasNext())
	  {
		  Contract c = (Contract) contractsIterator.next();

		  totalCostIncurred += c.getCostIncurred();
		  totalUnbilled += c.getUnbilledPortionOfContract();
		  totalOverbilled += c.getOverbilling();
		  totalRevenueEarned += c.getRevenueEarned();

		  totalCurrentCostIncurred += c.getCurrentCostIncurred();
		  totalCurrentRevenueEarned += c.getCurrentRevenueEarned();
		  totalCurrentRevenueRecordable += c.getCurrentRevenueRecorded();

		  totalPreviousCostIncurred += c.getPreviousCostIncurred();
		  totalPreviousRevenueEarned += c.getPreviousRevenueEarned();
		  totalRevenuePreviouslyRecorded += c.getRevenuePreviouslyRecorded();
	  }

	  //Reset the iterator
	  contractsIterator = this.contracts.iterator();

  }

  public String getCompanyName()
  {
	  return companyName;
  }

  public void setCompanyName(String name)
  {
	  this.companyName = name;
  }

  /**
   * Returns true if there is data for more contracts in this report.
   * Returns false if there are no more contracts in the current report.
   */
  public boolean moreContracts()
  {
	  return contractsIterator.hasNext();
  }

  /**
   * Advances the current contract to the next contract in the report.
   * If the current contract is the last contract in the report
   * this method will do nothing and the last contract will remain the current contract.
   */
  public void nextContract()
  {
  	  if(contractsIterator.hasNext())
  	     currentContract = (Contract) contractsIterator.next();
  }

  public Contract getCurrentContract()
  {
	  return currentContract;
  }

  public int getTotalCostIncurred()
  {
  	  return totalCostIncurred;
  }

  public int getTotalUnbilled()
  {
	  return totalUnbilled;
  }

  public int getTotalOverbilled()
  {
	  return totalOverbilled;
  }

  public int getTotalRevenueEarned()
  {
	  return totalRevenueEarned;
  }

  public int getTotalGrossMarginEarned()
  {
	  return getTotalRevenueEarned() - getTotalCostIncurred();
  }

  public float getTotalGrossMarginEarnedPercent()
  {
	  return ((float) getTotalGrossMarginEarned()) / ((float) getTotalRevenueEarned());
  }

  public int getTotalCurrentCostIncurred()
  {
	  return totalCurrentCostIncurred;
  }

  public int getTotalCurrentRevenueEarned()
  {
	  return totalCurrentRevenueEarned;
  }

  public int getTotalCurrentRevenueRecordable()
  {
	  return totalCurrentRevenueRecordable;
  }

  public int getTotalCurrentGrossMarginEarned()
  {
	  return getTotalCurrentRevenueEarned() - getTotalCurrentCostIncurred();
  }

  public float getTotalCurrentGrossMarginEarnedPercent()
  {
	  return ((float) getTotalCurrentGrossMarginEarned() ) / ((float) getTotalCurrentRevenueEarned());
  }

  public int getTotalPreviousCostIncurred()
  {
	  return totalPreviousCostIncurred;
  }

  public int getTotalPreviousRevenueEarned()
  {
	  return totalPreviousRevenueEarned;
  }

  public int getTotalRevenuePreviouslyRecorded()
  {
	  return totalRevenuePreviouslyRecorded;
  }

  public int getTotalPreviousGrossMarginEarned()
  {
	  return getTotalPreviousRevenueEarned() - getTotalPreviousCostIncurred();
  }

  public float getTotalPreviousGrossMarginEarnedPercent()
  {
	  if(getTotalPreviousRevenueEarned() < 1)
	  {
		  return 0;
	  }
	  else
  	  {
	  	  return ((float) getTotalPreviousGrossMarginEarned() ) / ((float) getTotalPreviousRevenueEarned());
	  }
  }


  //************** METHODS THAT APPLY TO THE CURRENT CONTRACT **************************

    /**
     * Returns the jobID of the current contract
     */
    public int getJobID()
	{
		return currentContract.getJobID();
	}

    /**
     * Returns the jobTypeID of the current contract
     */
	public int getJobTypeID()
	{
		return currentContract.getJobTypeID();
	}

	/**
	 * Returns the PUBLIC / PRIVATE modifier for the current job
	 */
	public int getJobTypeModifier()
	{
		return currentContract.getJobTypeModifier();
	}

    /**
     * Returns the total cost incurred of the current contract
     */
	public int getCostIncurred()
	{
		return currentContract.getCostIncurred();
	}

    /**
     * Returns the total amount unbilled on the current contract
     */
	public int getUnbilledPortionOfContract()
	{
		return currentContract.getUnbilledPortionOfContract();
	}

    /**
     * Returns the total revenue earned on the current contract
     */
	public int getRevenueEarned()
	{
		return currentContract.getRevenueEarned();
	}

    /**
     * Returns the gross margin earned on the current contract
     */
    public int getGrossMarginEarned()
    {
		return currentContract.getGrossMarginEarned();
	}

    /**
     * Returns the gross margin percent earned of the current contract
     */
	public float getGrossMarginEarnedPercent()
	{
		return currentContract.getGrossMarginEarnedPercent();
	}

    /**
     * Returns the cost incurred on the current contract, during period of this report
     */
    public int getCurrentCostIncurred()
    {
		return currentContract.getCurrentCostIncurred();
	}

    /**
     * Returns the revenue earned on the current contract, during period of this report
     */
    public int getCurrentRevenueEarned()
    {
		return currentContract.getCurrentRevenueEarned();
	}

    /**
     * Returns the revenue recordable on the current contract, during period of this report
     */
    public int getCurrentRevenueRecordable()
    {
		return currentContract.getCurrentRevenueRecorded();
	}

    /**
     * Returns the gross margin of the current contract, during period of this report
     */
    public int getCurrentGrossMargin()
    {
		return currentContract.getCurrentGrossMargin();
	}

    /**
     * Returns the gross margin percent of the current contract, during period of this report
     */
	public float getCurrentGrossMarginPercent()
	{
		return currentContract.getCurrentGrossMarginPercent();
	}

    /**
     * Returns the cost incurred on the current contract, during all periods previous to the period of this report
     */
    public int getPreviousCostIncurred()
    {
		return currentContract.getPreviousCostIncurred();
	}

    /**
     * Returns the revenue earned on the current contract, during all periods previous to the period of this report
     */
    public int getPreviousRevenueEarned()
    {
		return currentContract.getPreviousRevenueEarned();
	}

    /**
     * Returns the revenue recorded on the current contract, during all periods previous to the period of this report
     */
    public int getRevenuePreviouslyRecorded()
    {
		return currentContract.getRevenuePreviouslyRecorded();
	}

    /**
     * Returns the gross margin percent of the current contract, during all periods previous to the period of this report
     */
    public int getPreviousGrossMargin()
    {
		return currentContract.getPreviousGrossMargin();
	}

    /**
     * Returns the gross margin percent of the current contract, during all periods previous to the period of this report
     */
	public float getPreviousGrossMarginPercent()
	{
		return currentContract.getPreviousGrossMarginPercent();
	}
}
