/**
 *
 *  @author Kevin Craig
 */
public class PersonnelReportAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
		           				 ActionForm form,
		           				 HttpServletRequest request,
		           				 HttpServletResponse response)
		{
		   Log log = LogFactory.getLog(this.getClass());
		   if(log.isDebugEnabled())
		   {
		       log.debug(this.getClass() + " beginning execute method");
		   }

	       ActionMessages errors = new ActionMessages();

	       // Check for existance of Game object in user's session
           ServletContext context = servlet.getServletContext();

           HttpSession session = request.getSession();
           if(session == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

	       Game game = (Game) session.getAttribute("game");
           if(game == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

		   Company company = null;

           String periodNumber = (String) request.getParameter("period");
           User user = (User) session.getAttribute("user");

           if(user instanceof Company)
           {
              LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
              // If for any reason the company should not be allowed to login, route the
              // company to an error message page.
	          if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	          {
		        return (mapping.findForward("loginFailure"));
		      }
			  company = (Company) user;
 		   }
	  	   else if(user instanceof Admin)
	   	   {
				company = (Company) session.getAttribute("company");
           }

		   int reportPeriod;

           if (periodNumber == null)
           {
			   reportPeriod = game.getCurrentPeriodNumber() - 1;
		   }
		   else
		   {
			   reportPeriod = Integer.parseInt(periodNumber);
		   }

           //Exit before doing the work if the period requested is before the start of the game
		   if(reportPeriod < 1)
		   {
			   session.setAttribute("personnelReport", new PersonnelReportForm());
			   return mapping.findForward("success");
		   }


		   PersonnelReportForm personnelReport = new PersonnelReportForm();

		   try
		   {
		   	   BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

	  		   Map<String, Integer> skills = dbFacade.getCompanyPersonnelDBHandler().getUnfilledMosp(game, company, reportPeriod);

			   dbFacade.close();

			   List<String> messages = new ArrayList<String>();

			   try
			   {
				   //CL Advise
				   if(skills.get("CL") > 0)
				   {
					   if(skills.get("CL") > skills.get("OW"))
					   {
						   messages.add("Your company lacks vision and direction.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //PL Advise
				   if(skills.get("PL") > 0)
				   {
					   if(skills.get("PL") > skills.get("OW"))
					   {
						   messages.add("Your company lacks people who can take charge of jobs.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //BS Advise
				   if(skills.get("BS") > 0)
				   {
					   if(skills.get("BS") > skills.get("OW"))
					   {
						   messages.add("It's difficult for your company to keep track of all the schedule changes on jobs.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //BE Advise
				   if(skills.get("BE") > 0)
				   {
					   if(skills.get("BE") > skills.get("OW"))
					   {
						   messages.add("Your company's estimating needs are not being met.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //S Advise
				   if(skills.get("BSF") > 50)
				   {
					   messages.add("Safety coordination is becoming a full-time job at your company.");
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //GFA Advise
				   if(skills.get("GFA") > 0)
				   {
					   if(skills.get("GFA") > skills.get("OW"))
					   {
						   messages.add("Your company lacks people to manage accounts and finances.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //PC
				   if(skills.get("PC") > 0)
				   {
					   if(skills.get("PC") > skills.get("OW"))
					   {
						   messages.add("Too few people are trying to support too many job sites.");
					   }
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //AB
				   if(skills.get("AB") > 0)
				   {
					   messages.add("There are unfinished bills and invoices to be processed");
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   try
			   {
				   //OW
				   if(skills.get("OW") > 0)
				   {
					   messages.add("People are continuously being interrupted by phone calls.");
				   }
			   }
			   catch(NullPointerException npe)
			   {
			   }

			   personnelReport.setMessages(messages);

	   	   }
	   	   catch(BIGDatabaseFacadeException bigdbe)
	   	   {
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		   }

		   personnelReport.setCompanyName(company.getTitle());
		   personnelReport.setReportDate(game.getDateString(reportPeriod));
		   personnelReport.setPeriod(reportPeriod);

		   session.setAttribute("personnelReport", personnelReport);

	       if (!errors.isEmpty())
		   {
			   if(log.isDebugEnabled())
			   {
			       log.debug(this.getClass() + " request returned with errors");
			   }

		   	   // Report any errors we have discovered back to the original form
		   	   saveErrors(request, errors);
		   	   saveToken(request);
		   	   return (new ActionForward(mapping.getInput()));
		   }
		   else
		   {
		       if(log.isDebugEnabled())
		       {
		           log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
	           }
 		       return (mapping.findForward("success"));
		   }

	   } // end execute()

}
