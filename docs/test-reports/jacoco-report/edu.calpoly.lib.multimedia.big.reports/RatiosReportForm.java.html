    /**
     * Creates a new Ratios Report Form
     * The 3 other ReportForms are used to calculate the current values for each
     * ratio.  Ratios from past periods are provided in the <code>pastRatios</code> argument.
     * The first index is in the 2D array is the number of periods previous to the
     * period of the report.  The second index is the ratio number where 0 = Current Ratio,
     * 1 = Receviable Turns, 2 = Payable Turns,
     *         3 = Cash Turns, 4 = Cash to WIP, 5 = Earnings to Interest, 6 = Debt to Equity,
     *         7 = Gross Profit, 8 = Net Profit, 9 = Return on Equity, 10 = Return on Assets, 11 =  Costs to Sales,
     *	     12 = GA to Sales, 13 = Operating Expense to Sales.
     * So for example, pastRatios[2][7] should contain the value of the Gross Profit ratio from 2 periods ago.
     * @param pastRatios Cannot be <code>null</code> but can have a length of 0.
     *
     */
	public RatiosReportForm(IncomeStatementForm incomeStatement,
							BalanceSheetForm balanceSheet,
							ContractsInProgressReportForm contractsInProgress,
							CompletedContractsReportForm completedContracts,
							Float[][] pastRatios
						   )
	{

		//Set the values in the Balance Sheet that do not come from the database
		balanceSheet.setUnderbilledAmount(contractsInProgress.getTotalUnderbilling());
		balanceSheet.setUnbilledAmount(completedContracts.getTotalUnbilled());
		balanceSheet.setOverbilledAmount(contractsInProgress.getTotalOverbilling() + completedContracts.getTotalOverbilled());

        //*************  CURRENT RATIO  ****************
		currentRatios = new Float[pastRatios.length + 1];
		if(balanceSheet.getTotalCurrentLiabilities() != 0)
		{
			currentRatios[0] = new Float( ( (float) balanceSheet.getTotalCurrentAssets() ) / ( (float) balanceSheet.getTotalCurrentLiabilities()) );
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    currentRatios[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			currentRatios[i+1] = pastRatios[i][0];
		}

		currentRatioAvg = computeAverage(currentRatios);


        //*************  RECEIVABLE TURNS  ****************
		receivableTurns = new Float[pastRatios.length + 1];
		if((balanceSheet.getCurrentAccountsReceivable() + balanceSheet.getRetentionReceivable()) != 0)
		{
			receivableTurns[0] = new Float( ( (float) incomeStatement.getConstructionRevenue() ) / ( (float) (balanceSheet.getCurrentAccountsReceivable() + balanceSheet.getRetentionReceivable())) );
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    receivableTurns[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			receivableTurns[i+1] = pastRatios[i][1];
		}

        receivableTurnsAvg = computeAverage(receivableTurns);


        //*************  PAYABLE TURNS  ****************
		payableTurns = new Float[pastRatios.length + 1];
		if((balanceSheet.getCurrentAccountsPayable() + balanceSheet.getRetentionPayable()) != 0)
		{
			payableTurns[0] = new Float( ( (float) incomeStatement.getCostOfConstruction() ) / ( (float) (balanceSheet.getCurrentAccountsPayable() + balanceSheet.getRetentionPayable())) );
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    payableTurns[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			payableTurns[i+1] = pastRatios[i][2];
		}

        payableTurnsAvg = computeAverage(payableTurns);


        //*************  CASH TURNS  ****************
		cashTurns = new Float[pastRatios.length + 1];
		if(balanceSheet.getCashOnHand() != 0)
		{
			cashTurns[0] = new Float( ( (float) incomeStatement.getConstructionRevenue() ) / ( (float) balanceSheet.getCashOnHand()) );
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    cashTurns[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			cashTurns[i+1] = pastRatios[i][3];
		}

        cashTurnsAvg = computeAverage(cashTurns);


        //*************  CASH TO WIP  ****************
		cashToWIP = new Float[pastRatios.length + 1];
		if(contractsInProgress.getTotalRevenueToBeEarned() != 0)
		{
			cashToWIP[0] = new Float( ( (float) balanceSheet.getCashOnHand()) / ( (float) contractsInProgress.getTotalRevenueToBeEarned() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    cashToWIP[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			cashToWIP[i+1] = pastRatios[i][4];
		}

       cashToWIPAvg = computeAverage(cashToWIP);


        //*************  EARNINGS TO INTEREST  ****************
		earningsToInterest = new Float[pastRatios.length + 1];
		if(incomeStatement.getInterestExpense() != 0)
		{
			earningsToInterest[0] = new Float( ( (float) incomeStatement.getEarningsFromOperations()) / ( (float) Math.abs(incomeStatement.getInterestExpense()) ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    earningsToInterest[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			earningsToInterest[i+1] = pastRatios[i][5];
		}

        earningsToInterestAvg = computeAverage(earningsToInterest);


        //*************  DEBT TO EQUITY  ****************
		debtToEquity = new Float[pastRatios.length + 1];
		if(balanceSheet.getEquity() != 0)
		{
			debtToEquity[0] = new Float( ( (float) balanceSheet.getTotalCurrentLiabilities() ) / ( (float) balanceSheet.getEquity()));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    debtToEquity[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			debtToEquity[i+1] = pastRatios[i][6];
		}

        debtToEquityAvg = computeAverage(debtToEquity);


        //*************  GROSS PROFIT  ****************
		grossProfit = new Float[pastRatios.length + 1];
		if(incomeStatement.getConstructionRevenue() != 0)
		{
			grossProfit[0] = new Float( ( (float) incomeStatement.getGrossProfit() ) / ( (float) incomeStatement.getConstructionRevenue()));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    grossProfit[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			grossProfit[i+1] = pastRatios[i][7];
		}

        grossProfitAvg = computeAverage(grossProfit);


        //*************  NET PROFIT  ****************
		netProfit = new Float[pastRatios.length + 1];
		if(incomeStatement.getConstructionRevenue() != 0)
		{
			netProfit[0] = new Float( ( (float) incomeStatement.getNetEarnings() ) / ( (float) incomeStatement.getConstructionRevenue()));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    netProfit[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			netProfit[i+1] = pastRatios[i][8];
		}

        netProfitAvg = computeAverage(netProfit);


        //************* RETURN ON EQUITY ****************
		returnOnEquity = new Float[pastRatios.length + 1];
		if(incomeStatement.getEndRetainedEarnings() != 0)
		{
			returnOnEquity[0] = new Float( ( (float) incomeStatement.getPreTaxEarnings() ) / ( (float) incomeStatement.getEndRetainedEarnings() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    returnOnEquity[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			returnOnEquity[i+1] = pastRatios[i][9];
		}

        returnOnEquityAvg = computeAverage(returnOnEquity);


        //*************  RETURN ON ASSETS ****************
		returnOnAssets = new Float[pastRatios.length + 1];
		if(balanceSheet.getTotalAssets() != 0)
		{
			returnOnAssets[0] = new Float( ( (float) incomeStatement.getPreTaxEarnings() ) / ( (float) balanceSheet.getTotalAssets() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    returnOnAssets[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			returnOnAssets[i+1] = pastRatios[i][10];
		}

		returnOnAssetsAvg = computeAverage(returnOnAssets);


        //************* COSTS TO SALES ****************
		costsToSales = new Float[pastRatios.length + 1];
		if(incomeStatement.getConstructionRevenue() != 0)
		{
			costsToSales[0] = new Float( ( (float) incomeStatement.getCostOfConstruction() ) / ( (float) incomeStatement.getConstructionRevenue() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    costsToSales[0] = null;
		}

		for(int i = 0; i < pastRatios.length; i++)
		{
			costsToSales[i+1] = pastRatios[i][11];
		}

		costsToSalesAvg = computeAverage(costsToSales);


        //*************  GA TO SALES  ****************
		gaToSales = new Float[pastRatios.length + 1];
		if(incomeStatement.getConstructionRevenue() != 0)
		{
			gaToSales[0] = new Float( ( (float) incomeStatement.getTotalGAExpense() ) / ( (float) incomeStatement.getConstructionRevenue() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    gaToSales[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			gaToSales[i+1] = pastRatios[i][12];
		}

		gaToSalesAvg = computeAverage(gaToSales);


        //************* OPERATING EXPENSE TO SALES  ****************
		opExpenseToSales = new Float[pastRatios.length + 1];
		if(incomeStatement.getConstructionRevenue() != 0)
		{
			opExpenseToSales[0] = new Float( ( (float) (incomeStatement.getCostOfConstruction() + incomeStatement.getTotalGAExpense()) ) / ( (float) incomeStatement.getConstructionRevenue() ));
		}
		else
		{
			//Null signifies no value due to a divide by zero error
		    opExpenseToSales[0] = null;
		}
		for(int i = 0; i < pastRatios.length; i++)
		{
			opExpenseToSales[i+1] = pastRatios[i][13];
		}

		opExpenseToSalesAvg = computeAverage(opExpenseToSales);

	}


    private Float computeAverage(Float[] ratios)
    {
        int items = ratios.length;
	    float average = 0.0f;
	    for(int i = 0; i < ratios.length; i++)
	    {
	        if (ratios[i] != null)
	        {
		   	    average += ratios[i].floatValue();
		    }
		    else
		    {
			    //Ignore periods with no value when taking the average
			    items--;
		    }
	    }

	    //If all periods have no value, the average will also have no value (i.e. be null)
	    if(items == 0)
	    {
		    return null;
	    }
	    else
	    {
		    return new Float(average / ((float) items));
	    }

    }
