package edu.calpoly.lib.multimedia.big.reports;


import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * ActionForm to hold input for choice between using BIG to compute
 * the Estimated Cost to Complete field on the Contracts in Progress
 * report or letting the company do it and to hold dollar amount if company
 * chooses to calculate the value themselves.
 *
 *
 * @author Kevin Craig
 */
public class EstimatedCostToCompleteChangeForm extends ActionForm
{
	private String BIGorSelf = "BIG";
	private String amountIfSelf;

	public String getBIGorSelf()
	{
		return BIGorSelf;
	}

	public void setBIGorSelf(String BIGorSelf)
	{
		this.BIGorSelf = BIGorSelf;
	}

	public String getAmountIfSelf()
	{
		return amountIfSelf;
	}

	public void setAmountIfSelf(String amountIfSelf)
	{
		this.amountIfSelf = amountIfSelf;
	}


	/**
     * Insures that this loan request is allowed and that the amount requested is a positive number
     * less than or equal to the maximum amount
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
		ActionErrors errors = new ActionErrors();

		if(BIGorSelf.equals("self"))
		{
			//Make sure the amount is not null
			if((amountIfSelf == null) || (amountIfSelf.trim().length() < 1) || amountIfSelf.equals("$"))
			{
				errors.add("amount", new ActionMessage("error.estimatedCostToComplete.amountRequired"));
			}
			else
			{
				//Make sure the String is a valid number
				if(!NumberConverter.isValidMoneyString(amountIfSelf))
				{
					errors.add("amount", new ActionMessage("error.estimatedCostToComplete.mismatch"));
				}
				else
				{
					long amount = NumberConverter.longValue(amountIfSelf);

					//Test if amount is positive
					if (amount <= 0)
						errors.add("amount", new ActionMessage("error.estimatedCostToComplete.negative"));
				}
			}
		}
		return errors;
	}

}
