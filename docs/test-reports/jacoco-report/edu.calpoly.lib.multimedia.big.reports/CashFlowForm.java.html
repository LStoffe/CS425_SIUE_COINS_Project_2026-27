package edu.calpoly.lib.multimedia.big.reports;


import java.util.Vector;
import java.util.Iterator;
import java.util.Hashtable;
import edu.calpoly.its.mas.big.reports.ReportForm;

/**
 * Class representing a cash flow report.
 * Calls to ExpandedCompany.decreaseCashOnHand() or ExpandedCompany.increaseCashOnHand()
 * result in an entry being made in this report.  This report stores the
 * cash on hand amount added or subtracted, whether or not it was a cash inflow or
 * outflow, and the source.  This class is robust enough to handle negative cash inflows and
 * positive cash outflows if the need arises.  The source is passed to this report
 * as some code word.  It is the responisibility of this class to maintain a list
 * of all possible code words and translate them into human readable sources.
 * This translate will not occur until the report is actually viewed by a human
 * because .properties file info is needed, which is available from the CashFlowAction class.
 *
 * <br><br>
 * This class acts like an interator for inflows and outflows.  The nextInflow and nextOutflow
 * can be used to set the current inflow or outflow and then the getCurrentXXXX methods can
 * be used to retrieve info about that inflow or outflow.  Note that if an add method is called
 * the form is reset and the next inflow and outflow will be the first ones in the form.
 *
 * If the source field of an add method is the same as the source in a previous
 * add, both amounts will be summed an count as the same inflow or outflow.
 * (eg. addOutflow(1000, "LABOR_EXPENSE") addOutflow(2000, "LABOR_EXPENSE") will result in the
 * inflow: LABOR_EXPENSE: 3000).
 *
 * @author Kevin Craig
 */
public final class CashFlowForm extends ReportForm
{

	private class CashFlowFormEntry
	{
		public CashFlowFormEntry(int amount, String source)
		{
			this.source = source;
			this.amount = amount;
		}

		public String source;
		public int amount;
	}


	private String companyName;

	private Vector inflows;
	private Vector outflows;
	private Iterator inflowIterator;
	private Iterator outflowIterator;
	private CashFlowFormEntry currentInflow;
	private CashFlowFormEntry currentOutflow;

	private int startingCashOnHand;
	private int endingCashOnHand;

	private Hashtable sourceTitleMapping;

	public CashFlowForm()
	{
		startingCashOnHand = 0;
		inflows = new Vector();
		outflows = new Vector();
		sourceTitleMapping = new Hashtable();
	}


	public CashFlowForm(int cashOnHand)
	{
		startingCashOnHand = cashOnHand;
		inflows = new Vector();
		outflows = new Vector();
		sourceTitleMapping = new Hashtable();
	}

  //************************  Public Methods  ***********************************

    public String getCompanyName()
    {
	    return companyName;
    }

    public void setCompanyName(String name)
    {
	    this.companyName = name;
    }

    /**
     * Adds a cash inflow (an entry on the report) for some amount
     * from some source.  If there is already an entry on the form
     * from the given source, the amount will be added to the current
     * entry for that source rather than creating a new entry.
     * Reset() is called implicitly by this methods, so the current
     * cash inflow and outflow be set to the first ones in the report.
     * @param amount Cash amount.  If amount is less than 1 this method returns
     * without doing anything.
     * @param source Source code for the source
     *
     */
    public void addInflow(int amount, String source)
    {
		if(amount < 1)
		{
			return;
		}

		CashFlowFormEntry cashFlowFormEntry = null;
		reset();
		while(inflowIterator.hasNext() && (cashFlowFormEntry == null))
		{
			CashFlowFormEntry cfef = (CashFlowFormEntry) inflowIterator.next();
			if (cfef.source.equals(source))
			{
				cashFlowFormEntry = cfef;
			}
		}

		if(cashFlowFormEntry != null)  //already have an entry from this source
		{
			cashFlowFormEntry.amount += amount;
		}
		else
		{
			inflowIterator = null;
			inflows.add(new CashFlowFormEntry(amount, source));
		}
		reset();
	}

    /**
     * Adds a cash outflow (an entry on the report) for some amount
     * from some source.  If there is already an entry on the form
     * from the given source, the amount will be added to the current
     * entry for that source rather than creating a new entry.
     * Reset() is called implicitly by this methods, so the current
     * cash inflow and outflow be set to the first ones in the report.
     * @param amount Cash amount.  If this amount is less than 1 this method returns
     * without doing anything.
     * @param source Source code for the source
     *
     */
    public void addOutflow(int amount, String source)
    {
		if(amount < 1)
		{
			return;
		}

		CashFlowFormEntry cashFlowFormEntry = null;
		reset();
		while(outflowIterator.hasNext() && (cashFlowFormEntry == null))
		{
			CashFlowFormEntry cfef = (CashFlowFormEntry) outflowIterator.next();
			if (cfef.source.equals(source))
			{
				cashFlowFormEntry = cfef;
			}
		}

		if(cashFlowFormEntry != null)  //already have an entry from this source
		{
			cashFlowFormEntry.amount += amount;
		}
		else
		{
			outflowIterator = null;
			outflows.add(new CashFlowFormEntry(amount, source));
		}
		reset();
	}

    public void addInflow(long amount, String source)
    {
		addInflow((int) amount, source);
	}

    public void addOutflow(long amount, String source)
    {
		addOutflow((int) amount, source);
	}

	public void setStartingCashOnHand(int amount)
	{
		startingCashOnHand = amount;
	}

	public int getStartingCashOnHand()
	{
		return startingCashOnHand;
	}

	public void setEndingCashOnHand(int amount)
	{
		endingCashOnHand = amount;
	}

	public int getEndingCashOnHand()
	{
		return endingCashOnHand;
	}

    /**
     * Resets the iterators for current inflow and outflow
     */
	public void reset()
	{
		inflowIterator = inflows.iterator();
		outflowIterator = outflows.iterator();
	}

	public boolean moreInflows()
	{
		if(inflowIterator != null)
		{
			return inflowIterator.hasNext();
		}
		return false;
	}

	public boolean moreOutflows()
	{
		if(outflowIterator != null)
		{
			return outflowIterator.hasNext();
		}
		return false;
	}

	public void nextInflow()
	{
		currentInflow = (CashFlowFormEntry) inflowIterator.next();
	}

	public void nextOutflow()
	{
		currentOutflow = (CashFlowFormEntry) outflowIterator.next();
	}

	public int getCurrentInflowAmount()
	{
		return currentInflow.amount;
	}

	public int getCurrentOutflowAmount()
	{
		return currentOutflow.amount;
	}

    /**
     * Returns the source code for the current inflow
     */
	public String getCurrentInflowSource()
	{
		return currentInflow.source;
	}

    /**
     * Returns the source code for the current outflow
     */
	public String getCurrentOutflowSource()
	{
		return currentOutflow.source;
	}

    /**
     * Returns a title for the source specified by the current inflow source code.
     * This title is determined by the mapping set in the <code>setSourceTitleMapping()</code>
     * method.  If no mapping has been set via this method or the current source code
     * is not present in the mapping, the source code is returned.
     */
	public String getCurrentInflowSourceTitle()
	{
		String title = (String) sourceTitleMapping.get(currentInflow.source);
		if(title == null)
		{
			return currentInflow.source;
		}
		else
		{
			return title;
		}
	}

    /**
     * Returns a title for the source specified by the current outflow source code.
     * This title is determined by the mapping set in the <code>setSourceTitleMapping()</code>
     * method.  If no mapping has been set via this method or the current source code
     * is not present in the mapping, the source code is returned.
     */
	public String getCurrentOutflowSourceTitle()
	{
		String title = (String) sourceTitleMapping.get(currentOutflow.source);
		if(title == null)
		{
			return currentOutflow.source;
		}
		else
		{
			return title;
		}
	}

	/**
	 * Returns the current mapping of inflow/outflow source codes to titles.
	 */
	public Hashtable getSourceTitleMapping()
	{
		return sourceTitleMapping;
	}

	/**
	 * Sets the current mapping of inflow/outflow source codes to titles.
	 * This mapping should be in the form of a <code>Hashtable</code>
	 * where the keys are <code>String</code>s representing the
	 * inflow/outflow source codes and the values are <code>String</code>s
	 * representing the associated titles.
	 */
	public void setSourceTitleMapping(Hashtable sourceTitleMapping)
	{
		this.sourceTitleMapping = sourceTitleMapping;
	}
}
