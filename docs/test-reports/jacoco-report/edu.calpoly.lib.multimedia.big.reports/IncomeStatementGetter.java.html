/**
 *  Get an income statement, doing all the tax calculations needed.
 *
 *  @author Kevin Craig
 */
public class IncomeStatementGetter
{
	private Game game;
	private Company company;
	private BIGDatabaseFacadeInterface dbFacade;
	private float taxRate;
	private float depreciationRate;

	private Log log;

	public IncomeStatementGetter(Game game, Company company, BIGDatabaseFacadeInterface dbFacade) throws ParametersNotFoundException, BIGDatabaseFacadeException
	{
		this.game = game;
		this.company = company;
		this.dbFacade = dbFacade;

		Hashtable misc = dbFacade.getGameParametersDBHandler().getMiscParameters(game);

        this.taxRate = ((float) (((Integer) misc.get("incomeTaxPercent")).intValue())) / 100f; //divide by 100 to convert from % to decimal

        log = LogFactory.getLog(this.getClass());
	}

	/**
	 * Retrieves an income statement.  In this method the amount of income tax recorded in the income statement
	 * will be calculated based on the total revenue for the year.  Note that this method involves
	 * many database accesses and calculations.
	 */
    public IncomeStatementForm getIncomeStatementComputeTaxes(int periodNum) throws BIGDatabaseFacadeException, ObjectNotFoundException
    {

		 FinancialReportForm financialReportForm = dbFacade.getFinancialReportDBHandler().getFinancialReport(game, periodNum, company);
         ContractsInProgressReportForm contractsInProgress = contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, periodNum, company);
 		 CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, periodNum, company);
 		 BalanceSheetForm prevBalanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, periodNum - 1, company);

		 IncomeStatementForm incomeStatement = new IncomeStatementForm(financialReportForm,
				  										    		   prevBalanceSheet,
				  										    		   contractsInProgress,
				  										    		   completedContracts,
				  										    		   .10f, //FIX DEPRECIATION
				  										    		   0);

		 int taxes = handleTaxes(incomeStatement, periodNum);

		 if(((periodNum - 1) % 6) == 0)
		 {
			 if(log.isDebugEnabled())
			 {
			 	 log.debug("dont sub prev");
			 }
			 incomeStatement.setIncomeTaxes(taxes);
		 }
		 else
		 {
			 if(log.isDebugEnabled())
			 {
			     log.debug("sub prev " + taxes + " " + prevBalanceSheet.getTaxes());
		 	 }
		 	 incomeStatement.setIncomeTaxes(taxes - prevBalanceSheet.getTaxes());
		 }

		 return incomeStatement;
	}

	/**
	 * Retrieves an income statement.  The taxes recorded in this income statement will
	 * be based on the taxes recorded in the current and previous balance sheets.  Note
	 * that these values should have originally got into the balance sheets using
	 * the income tax values computed in calls to getIncomeStatementComputeTaxes.
	 *
	 */
    public IncomeStatementForm getIncomeStatement(int periodNum) throws BIGDatabaseFacadeException, ObjectNotFoundException
    {

	  FinancialReportForm financialReport = dbFacade.getFinancialReportDBHandler().getFinancialReport(game, periodNum, company);
      ContractsInProgressReportForm contractsInProgress = contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, periodNum, company);
      CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, periodNum, company);
 	  BalanceSheetForm prevBalanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, periodNum - 1, company);
 	  BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, periodNum, company);

	  return new IncomeStatementForm(financialReport,
				  					 prevBalanceSheet,
				  					 contractsInProgress,
				  					 completedContracts,
				  					 .10f, //FIX
				  					 balanceSheet.getTaxes() - prevBalanceSheet.getTaxes()
				  					);
	}

  	//Calculates a new estimate for the
  	//amount of taxes that must be paid for the current year
  	private int handleTaxes(IncomeStatementForm currentIncomeStatement, int periodNum) throws BIGDatabaseFacadeException, ObjectNotFoundException
  	{

  		  int periodsToConsider = periodNum % 6;  //magic number... 6 periods per year
  		  if(log.isDebugEnabled())
  		  {
		      log.debug("periods to consider: "+ periodsToConsider);
		  }

  		  boolean yearEnd = false;

  		  if (periodsToConsider == 0)
  		  {
			  periodsToConsider = 6;
		  }

		  IncomeStatementForm[] previousStatements = new IncomeStatementForm[periodsToConsider];

		  //Load the current income statement into the array
		  periodsToConsider--;
		  previousStatements[periodsToConsider] = currentIncomeStatement;

		  while(periodsToConsider > 0)
		  {
			  periodsToConsider--;
			  previousStatements[periodsToConsider] = getIncomeStatement(periodNum, periodsToConsider+1);
		  }

  	      //use the TaxLedger to computer the tax amount
  	      TaxLedger taxLedger = new TaxLedger(taxRate, previousStatements);
  	      int taxes = taxLedger.getTaxes();

          return taxes;
  }

  //Returns the income statement for a given number of period prior to the current period
  //Pre: periodsAgo > 0
  //Pre: currentPeriod - periodsAgo > 0
  //Note: The taxes, net earnings, and equity fields of the incomeStatement will not necessarily be correct
  private IncomeStatementForm getIncomeStatement(int periodNum, int periodsAgo) throws BIGDatabaseFacadeException, ObjectNotFoundException
  {

	  FinancialReportForm financialReport = dbFacade.getFinancialReportDBHandler().getFinancialReport(game, periodNum - periodsAgo, company);
      ContractsInProgressReportForm contractsInProgress = contractsInProgress = dbFacade.getContractsReportsDBHandler().getContractsInProgressReport(game, periodNum - periodsAgo, company);
      CompletedContractsReportForm completedContracts = dbFacade.getContractsReportsDBHandler().getCompletedContractsReport(game, periodNum - periodsAgo, company);
 	  BalanceSheetForm prevBalanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, (periodNum - periodsAgo) - 1, company);
 	  BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, (periodNum - periodsAgo) , company);

	  return new IncomeStatementForm(financialReport,
				  					 prevBalanceSheet,
				  					 contractsInProgress,
				  					 completedContracts,
				  					.10f, //FIX
				  					balanceSheet.getTaxes() - prevBalanceSheet.getTaxes()
				  					);
  }

}
