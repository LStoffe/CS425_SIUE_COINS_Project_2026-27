package edu.calpoly.lib.multimedia.big.reports;

import edu.calpoly.its.mas.big.reports.ReportForm;
import edu.calpoly.its.mas.big.reports.FinancialReportForm;

/**
 * Class representing income statement.  The data for this report
 * is found or derived from the contract reports, the current and previous balance sheets, and
 * the old BIG v2 financial report form.  This report acts as a wrapper around these four reports
 * and does not contain any information that cannot be found in or derived from data contained in
 * those reports.  Note that loading 5 other reports is ineffecient because many values from each of
 * reports will not be used, but is simple and a simple way to insure the values in the income
 * statement will be consist with values in the other reports.
 * <br><br>
 * If the time to load an income statement becomes unacceptably long, this design will need to be
 * redone to have the income statement pull the values it needs directly from the database rather
 * than use the other reports as intermediaries.
 * <br><br>
 * For a description of each field in the income statement please refer to the BIG v3 SRS.
 *
 * @author Kevin Craig
 */
public class IncomeStatementForm extends ReportForm
{
	private FinancialReportForm financialReport;
	private BalanceSheetForm prevBalanceSheet;
	private ContractsInProgressReportForm contractsInProgress;
	private CompletedContractsReportForm completedContracts;

	private String companyName;

	float depreciationRate = 0f;

	int taxes = 0;

	public IncomeStatementForm(FinancialReportForm financialReport,
							   BalanceSheetForm prevBalanceSheet,
							   ContractsInProgressReportForm contractsInProgress,
							   CompletedContractsReportForm completedContracts,
							   float depreciationRate,
							   int taxes)
	{
		this.financialReport = financialReport;
		this.prevBalanceSheet = prevBalanceSheet;
		this.contractsInProgress = contractsInProgress;
		this.completedContracts = completedContracts;
		this.depreciationRate = depreciationRate;
		this.taxes = taxes;

		if(contractsInProgress == null)
		{
		    //System.out.println("IncomeStatementForm(Constructor): contractsInProgress is null");
		}
	}



	public IncomeStatementForm()
	{
		financialReport = null;
		prevBalanceSheet = null;
		contractsInProgress = null;
		completedContracts = null;
	}

  //************************  Set and Get Methods  ***********************************

    public String getCompanyName()
    {
	    return companyName;
    }

    public void setCompanyName(String name)
    {
	    this.companyName = name;
    }


  	public int getConstructionRevenue()
  	{
		return contractsInProgress.getTotalCurrentRevenueRecorded() + completedContracts.getTotalCurrentRevenueRecordable();
  	}

  	public int getCostOfConstruction()
  	{
	    return contractsInProgress.getTotalCurrentCostIncurred() + completedContracts.getTotalCurrentCostIncurred();
  	}

  	public int getGrossProfit()
  	{
		return getConstructionRevenue() - getCostOfConstruction();
  	}

  	public int getOfficeOverhead()
  	{
		return (int) financialReport.getOfficeOverhead();
  	}

  	public int getBiddingExpense()
  	{
		return (int) financialReport.getBiddingExpenses();
  	}

  	public int getConsultingExpense()
  	{
		return (int) financialReport.getConsultingExpenses();
  	}

  	public int getSettlementExpense()
  	{
		return (int) financialReport.getSettlementExpenses();
  	}

  	public int getTotalGAExpense()
  	{
		return getOfficeOverhead() + getBiddingExpense() + getConsultingExpense() + getSettlementExpense();
  	}

  	public int getEarningsFromOperations()
  	{
		return getGrossProfit() - getTotalGAExpense();
  	}

  	public int getInterestExpense()
  	{
		return 0 - (int) financialReport.getInterestPaidOnLoans();
  	}


  	public int getDepreciation()
  	{
		return 0;
  	}

  	public int getTotalOtherIncome()
  	{
		return   getInterestExpense() +
		  		 getDepreciation();
	}

  	public int getPreTaxEarnings()
  	{
		return getEarningsFromOperations() + getTotalOtherIncome();
  	}

  	public int getIncomeTaxes()
  	{
		return taxes;
  	}

  	public void setIncomeTaxes(int taxes)
  	{
		this.taxes = taxes;
	}

  	public int getNetEarnings()
  	{
		return getPreTaxEarnings() - getIncomeTaxes();
  	}

  	public int getStartRetainedEarnings()
  	{
		return prevBalanceSheet.getEquity();
  	}

  	public int getEndRetainedEarnings()
  	{
		return getStartRetainedEarnings() + getNetEarnings();
  	}

}
