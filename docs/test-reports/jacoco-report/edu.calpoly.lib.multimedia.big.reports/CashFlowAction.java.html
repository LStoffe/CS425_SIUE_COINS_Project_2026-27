/**
 *  Handles requests to view a cash flow report.
 *
 *  @author Kevin Craig
 */
public class CashFlowAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
		           			     ActionForm form,
		           				 HttpServletRequest request,
		           				 HttpServletResponse response)
		{
		   Log log = LogFactory.getLog(this.getClass());
		   if(log.isDebugEnabled())
		   {

		       log.debug(this.getClass() + " beginning execute method");
		   }

	       ActionMessages errors = new ActionMessages();

	       // Check for existance of Game object in user's session
           ServletContext context = servlet.getServletContext();

           HttpSession session = request.getSession();
           if(session == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

	       Game game = (Game) session.getAttribute("game");
           if(game == null)
           {
		      return (mapping.findForward("loginFailure"));
		   }

		   Company company = null;

           String periodNumber = (String) request.getParameter("period");
           User user = (User) session.getAttribute("user");

           if(user instanceof Company)
           {
              LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");
              // If for any reason the company should not be allowed to login, route the
              // company to an error message page.
	          if(!loginBroker.verifyGameLoginAllowed(game, context, request))
	          {
		        return (mapping.findForward("loginFailure"));
		      }
			  company = (Company) user;
 		   }
	  	   else if(user instanceof Admin)
	   	   {
				company = (Company) session.getAttribute("company");
           }


		   int reportPeriod;

           if (periodNumber == null)
           {
			   reportPeriod = game.getCurrentPeriodNumber() - 1;
		   }
		   else
		   {
			   reportPeriod = Integer.parseInt(periodNumber);
		   }

           //Dummy income statement.  If an IncomeStatement doesn't exist for a company/period
           //because the company was created after that period, this dummy will prevent
           //null pointer exceptions
		   CashFlowForm cashFlowForm = new CashFlowForm();

		   if(reportPeriod > 0)
		   {

		   	  try
		   	  {
		   		  BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

				  cashFlowForm = dbFacade.getCashFlowDBHandler().getCashFlowReport(game, reportPeriod, company);
				  BalanceSheetForm prevBalanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, reportPeriod - 1, company);
				  BalanceSheetForm balanceSheet = dbFacade.getBalanceSheetDBHandler().getBalanceSheet(game, reportPeriod, company);

				  cashFlowForm.setStartingCashOnHand((int) prevBalanceSheet.getCashOnHand());
				  cashFlowForm.setEndingCashOnHand((int) balanceSheet.getCashOnHand());

				  dbFacade.close();
		   	  }
		      catch (BIGDatabaseFacadeException bigdbe)
		      {
				  log.fatal("", bigdbe);
				  errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		      }
		      catch (ObjectNotFoundException onfe)
		      {
				  //Do nothing.  This company must have been created after the report period.
		      }
		   }
		   else
		   {
				//do nothing if there is no report for this period
		   }

		   MessageResources resources = getResources(request);

		   Hashtable sourceTitleMapping = new Hashtable();
		   sourceTitleMapping.put("LABOR_EXPENSE", resources.getMessage("cashFlowReport.LABOR_EXPENSE"));
		   sourceTitleMapping.put("EQUIPMENT_EXPENSE", resources.getMessage("cashFlowReport.EQUIPMENT_EXPENSE"));
		   sourceTitleMapping.put("TAXES", resources.getMessage("cashFlowReport.TAXES"));
		   sourceTitleMapping.put("BID_EXPENSE", resources.getMessage("cashFlowReport.BID_EXPENSE"));
		   sourceTitleMapping.put("ACCOUNTS_PAYABLE", resources.getMessage("cashFlowReport.ACCOUNTS_PAYABLE"));
		   sourceTitleMapping.put("OFFICE_OVERHEAD", resources.getMessage("cashFlowReport.OFFICE_OVERHEAD"));
		   sourceTitleMapping.put("FIELD_OVERHEAD", resources.getMessage("cashFlowReport.FIELD_OVERHEAD"));
		   sourceTitleMapping.put("LIQ_DAM", resources.getMessage("cashFlowReport.LIQ_DAM"));
		   sourceTitleMapping.put("LOAN", resources.getMessage("cashFlowReport.LOAN"));
		   sourceTitleMapping.put("PAYMENTS", resources.getMessage("cashFlowReport.PAYMENTS"));
		   sourceTitleMapping.put("RETENTION", resources.getMessage("cashFlowReport.RETENTION"));
		   sourceTitleMapping.put("LOAN_PAYMENT", resources.getMessage("cashFlowReport.LOAN_PAYMENT"));
		   sourceTitleMapping.put("CONSULTING", resources.getMessage("cashFlowReport.CONSULTING"));
		   sourceTitleMapping.put("SETTLEMENT", resources.getMessage("cashFlowReport.SETTLEMENT"));

		   cashFlowForm.setSourceTitleMapping(sourceTitleMapping);
		   cashFlowForm.setCompanyName(company.getTitle());
		   cashFlowForm.setReportDate(game.getDateString(reportPeriod));
		   cashFlowForm.setPeriod(reportPeriod);

		   session.setAttribute("cashFlowForm", cashFlowForm);

	       if (!errors.isEmpty())
		   {
			  if(log.isDebugEnabled())
			  {
			      log.debug(this.getClass() + " request returned with errors");
			  }

		   	  // Report any errors we have discovered back to the original form
		   	  saveErrors(request, errors);
		   	  saveToken(request);
		   	  return (new ActionForward(mapping.getInput()));
		   }
		   else
		   {
			   if(log.isDebugEnabled())
			   {
			       log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
			   }
 			   return (mapping.findForward("success"));
		   }

	   } // end execute()

}
