package edu.calpoly.lib.multimedia.big.reports;

import edu.calpoly.its.mas.big.game.Activity;
import edu.calpoly.its.mas.big.game.Method;

/**
 * Estimates the cost to complete a job based on the required completion date of the job,
 * contruction units remaining on the job, methods chosen, and overtime assigned.
 *
 * @author Kevin Craig
 */
public class JobCostEstimator
{

	/**
	 * Makes an estimate for the total cost left to complete a job
	 * in the following manner:
	 * <ul>
	 *   <li>Do the following for each activity:</li>
	 *      <ul>
	 *        <li>Determine how many CU's are to be completed during overtime
	 *            by multipling CU's/hour for the method chosen by the number
	 *            of hours overtime for the activity.</li>
	 *        <li>Take the min of this number of CU's and the CU's left in the activity.  Call this OCUs</li>
	 *        <li>Computer a cost per CU by applying the overtime modifyer to labor and summing
	 *            the materials, subcontractor and modifyed labor costs per CU.  Call this OUC COST.</li>
	 *        <li>Subtract OCU's from the CU's left in the activity
	 *             (not in the actual activity just in our calculation).</li>
	 *        <li>Multiply OCU's by OCU COST and add this number to the estimate.</li>
	 *        <li>Compute a normal cost per CU by summing labor, materials and subcontractor costs
	 *            per CU</li>
	 *        <li>Multiply this cost by any CUs left in the activity after OUC's have been subtracted.</li>
	 *        <li>Add this amount to the estimate.</li>
	 *      </ul>
	 *    <li>The number of periods left to be worked is calculated by subtracting the
	 *        current period number from the period that the job's required completion dates falls.
	 *        If this number is negative, meaning the job has missed its required completion
	 *		  date, the number of periods left to be worked will be set to 1.  Additionally, the
	 *		  lastPeriodDaysWorkedPercent will be ignored.  The result of this is that
	 *		  jobs that have missed their deadlines will be assumed to complete on the last day
	 *		  of the next period (the period after whatever the current period is).
	 *    </li>
	 *
	 *    <li>Calculate fixed field overhead:  Multiply the number of workdays per period by
	 *        the number of periods left to be worked -1.  Add to this the number of workdays
	 *		  per period times the percentage of the last period worked.  This gives
	 *		  an estimate for the number of workdays remaining.  Multiply this figure
	 *		  by the fixed cost per day and add the result to the estimate.</li>
	 *    <li>Calculate variable field overhead : multiply the variableFieldOverheadCostPerPeriod
	 *        by the number of periods left to be worked.
	 *		  Add these amounts to the estimate.
	 *    </li>
	 *    <li>
	 *        If the job has missed it required completion date, add liquidated damages
	 *        for each day it will take the job to complete.  Since late jobs are assumed to
	 *		  finish on the last day of the next period, this means 2 full months of liquidated damamges
	 *		  must be added to the cost of the job.  In this case a month will be defined to have 30 days.
	 *    </li>
	 * </ul>
	 *
	 * @param currentPeriod The period number of the current period.
	 * @param completionPeriod The period number of the period in which the required
	 *	completion date for this job falls (This can be calculated using the Game.convertDateToPeriod method).
	 *  This refers to the original required completion date of the job, regardless of whether or not
	 *  the current period is before or after it,
	 *  so this parameter will never change between subsequent calls to this method for the same job.
	 * @param lastPeriodDaysWorkedPercent An estimate of how much of the job's final period is worked.
	 *  For example if the job finished on the first day of its final period this number would be
	 *  fairly low, around 1 or 2%.  If the job finished on the last day of the period it completed
	 *  this number would be 100%.
	 * @param activities Array of 9 <code>Activity</code> objects that make up the job.
	 *  No changes will be made to these objects.
	 * @param workdaysPerPeriod Game parameter that is the number of work days per period
	 * @param fixedFieldOverhead Game parameter that is the fixed field overhead cost per day
	 * @param variableFieldOverheadCostPerPeriod Game parameter that is the variable field overhead cost
	 *  per period for this job.  This number should be in dollars.
	 * @return An estimate in dollars for how much the job will cost to complete
	 *
	 */
	public static synchronized int getEstimatedCost(int currentPeriod,
								        			int completionPeriod,
								      				float lastPeriodDaysWorkedPercent,
									   				Activity[] activities,
   								       				int workdaysPerPeriod,
									   				int fixedFieldOverhead,
									   				int variableFieldOverheadCostPerPeriod,
									   				int liquidatedDamagesPerDay)
	{
		//Current estimate amount
		int estimate = 0;

		//make sure the job is not already complete
		if (!checkJobStillCurrent(activities))
		{
			return 0;
		}

		//Add the estimated direct cost to complete each activity
        for(int i = 0; i < 9; i++)
        {

			if(!activities[i].isComplete())
			{
            	//Calculate number of CU's to complete during overtime
            	int overtimeCUs = activities[i].getOvertimeDays() * activities[i].getOvertimeHours() * activities[i].getCurrentMethod().getProductivity();

            	//Calculate cost per overtime CU using this method
            	int costPerOvertimeCU = ((int)(activities[i].getCurrentMethod().getBaseLaborCost() * activities[i].getOvertimeModifier())) +
                                    activities[i].getCurrentMethod().getBaseMaterialsCost() +
                                    activities[i].getCurrentMethod().getSubcontractorCost() +
                                    activities[i].getCurrentMethod().getBaseEquipmentCost();

			    //If the number of CUs assigned to be worked during overtime
			    //is greater than the number of CU's left in the activity
			    //set it back to the CU's left in the activity
				if (overtimeCUs > activities[i].getLusRemaining())
				{
					overtimeCUs = activities[i].getLusRemaining();
				}

				//Add the overtime cost to the estimate
				estimate += overtimeCUs * costPerOvertimeCU;

				//Number of CUs in activity that will not be worked during overtime
				//Note that this this number is >= 0 because overtimeCUs <= lus remaining
				int nonOvertimeCUs = activities[i].getLusRemaining() - overtimeCUs;

				//Calculate cost per non-overtime CU
				int costPerNormalCU = activities[i].getCurrentMethod().getBaseLaborCost() +
                                      activities[i].getCurrentMethod().getBaseMaterialsCost() +
                                      activities[i].getCurrentMethod().getSubcontractorCost() +
                                      activities[i].getCurrentMethod().getBaseEquipmentCost();

                //Add the cost to complete the non overtime work to the estimate
                estimate += nonOvertimeCUs * costPerNormalCU;
			}
		}

        //Overhead
        int periodsLeft = (completionPeriod - currentPeriod);

		//System.out.println("Periods left: " + periodsLeft);

		if (periodsLeft < 0)  //Job has missed deadline
		{
			estimate += workdaysPerPeriod * fixedFieldOverhead;
			estimate += variableFieldOverheadCostPerPeriod;

			//Liquidated damages  (Assume 30 days in a month.  This will never change)
			estimate += 30 * 2 * liquidatedDamagesPerDay;
		}
		else
		{
			//Fixed field over head is cost per day times number of days
			//There are two parts, the number days in full periods
			//and the days left in the period the job completes
			int daysLeft = periodsLeft * workdaysPerPeriod + (int)(workdaysPerPeriod * lastPeriodDaysWorkedPercent);

			//System.out.println("Days left: " + daysLeft);

			estimate += daysLeft * fixedFieldOverhead;

            //Plus one to account for the current period
			estimate += (periodsLeft + 1) * variableFieldOverheadCostPerPeriod;

		}

		return estimate;
	}

	private static boolean checkJobStillCurrent(Activity[] activities)
	{
		boolean current = false;
		for(int i = 0; i < activities.length; i++)
		{
			if (activities[i].isComplete() == false)
			   current = true;
		}

		return current;
	}

}
