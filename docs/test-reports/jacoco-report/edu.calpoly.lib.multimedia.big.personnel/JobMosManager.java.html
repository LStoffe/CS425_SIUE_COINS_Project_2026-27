package edu.calpoly.lib.multimedia.big.personnel;

//BIG imports
import edu.calpoly.its.mas.big.game.Job;

//JDK imports
import java.util.List;
import java.util.Iterator;
import java.util.Map;
import java.util.HashMap;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Class to determine how many Mos points for each skill
 * a job requires, to keep track of how many of each skill
 * are completed, and to determine penalties for incomplete
 * mos points.
 */
public class JobMosManager
{
	//Map of Map.  Each inner Map has keys that are skill codes
	//and values that are the Mos to Cu ratio for a job type.
	//The key in the outer Map represents the job type id.  So key "1" is
	//for job type id 1, "2" for 2, and so on.
	private Map<String, Map<String, MosToCUsRatio>> jobTypeMosParams;

	//Map where each skill code is a key and the related value is
	//the penalty per each incomplete mosp of that skill.
	private Map<String,Integer> costPerMospParams;

	//The Jobs whose Mosp are being managed.
	private List<Job> jobs;

	private Map<String, Integer> mospCompletion;

	private Map<String, Integer> mospFailure;

	private Log log;

	/**
	 *
	 *
	 */
	public JobMosManager(Map<String, Map<String, MosToCUsRatio>> jobTypeMosParams,
					     Map<String,Integer> costPerMospParams,
					     List<Job> jobs
					    )
	{
		log = LogFactory.getLog(this.getClass());

		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": Creating JobMosManager ");
		}

		this.jobTypeMosParams = jobTypeMosParams;
		this.costPerMospParams = costPerMospParams;
		this.jobs = jobs;

		mospCompletion = new HashMap<String, Integer>();
		mospFailure = new HashMap<String, Integer>();
		setupMospCompletionTable();
	}

	private void setupMospCompletionTable()
	{
		for(Job job : jobs)
		{
/*			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " computing MOSP required for job " + job.getJobID());
			} */

			Map<String, MosToCUsRatio> params = jobTypeMosParams.get((new Integer(job.getJobType())).toString());

			for(String skillCode : params.keySet())
			{
				MosToCUsRatio mosRatio = params.get(skillCode);

				int mospReq = Math.round(((float) job.getJobSize()) * mosRatio.getRatio());

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + ": MOSP req for skill " + skillCode + " is " + mospReq);
				} */

				Integer currentMospForSkill = mospCompletion.get(skillCode);

				if(currentMospForSkill == null)
				{
					currentMospForSkill = new Integer(0);
				}

				Integer newMospForSkill = currentMospForSkill.intValue() + mospReq;

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + ": New total MOSP req for skill " + skillCode + " is " + newMospForSkill.toString());
				} */

				mospCompletion.put(skillCode, newMospForSkill);
			}
		}

		clearEmptySkillsFromTable();

	}

	private void clearEmptySkillsFromTable()
	{
		//Make a pass of the table removing all the skills that are zero or less
		Iterator<String> i = mospCompletion.keySet().iterator();
		while(i.hasNext())
		{
			String skillCode = i.next();
			Integer mosp = mospCompletion.get(skillCode);
			if(mosp.intValue() <= 0)
			{
				i.remove();

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " " + skillCode + " has been removed from job MOSPs");
				} */
			}
		}
	}

	public void reduceRequiredMOSP(String skillCode, int amount)
	{
		Integer currentAmount = mospCompletion.get(skillCode);

		if(currentAmount != null)
		{
			int newAmount = currentAmount.intValue() - amount;

			if(newAmount > 0)
			{
				mospCompletion.put(skillCode, new Integer(newAmount));
			}
			else
			{
				mospCompletion.remove(skillCode);

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " " + skillCode + " has been removed from job MOSPs");
				} */
			}
		}

	}

	/**
	 * This is used to indicate to the JobMosManager that
	 * the company is unable to complete this skill
	 * the the manager should treat it as failed and move on.
	 */
	public void failMosp(String skillCode)
	{
		Integer currentAmount = mospCompletion.get(skillCode);

		if(currentAmount != null)
		{
			mospCompletion.remove(skillCode);
			mospFailure.put(skillCode, currentAmount);
		}
	}

	/**
	 * Return true if at least one skill still requires MOSP, false if not.
	 */
	public boolean stillMOSPRequired()
	{
		return (mospCompletion.keySet().size() > 0);
	}


	/**
	 * Returns the skill code for the skill that is currently contributing the biggest
	 * dollar amount to the total penalty.
	 * <br><br>
	 * A valid question might be "Why is this the most damaging skill?  Shouldn't
	 * the skill with the highest cost per point <i>always</i> be the most damaging?"
	 * Technically this is true, but what we're really looking for here
	 * is the skill that will have the most benefit.  Different employee
	 * types can produce different points as different rates.  If we always
	 * take the skill with the biggest cost per point, that means each employee will
	 * work at that skill until it is complete, even if it would be far more efficient
	 * for them to work at a different skill.
	 * The skill contributing the most to the dollar penalty will change over time.
	 * Assuming a good distribution of employee skills, an employee who is efficient at the skill
	 * can be selected to work on the most damaging skill until a different skill becomes more damaging.
	 * Then a different employee (who is efficient at that skill) will likely take over to work on that skill.
	 * <br><br>
	 * Note this does not guarantee perfect efficiency.  If no one is efficient at the
	 * most damaging skill, everyone will still work at it until it is not the most damaging
	 * regardless of other skills that could be causing a fair amount of damage.
	 * The employee assignment algorithm may require a lot of work if efficiency problems
	 * appear.
	 */
	public String getMostDamagingMOS()
	{
		String skill = "";
		int damage = 0;
		Integer remainingMosp = null;
		for(String skillCode : mospCompletion.keySet())
		{
			remainingMosp = mospCompletion.get(skillCode);
			Integer penaltyPerMosp = costPerMospParams.get(skillCode);

			int skillDamage = (remainingMosp.intValue() * penaltyPerMosp.intValue());
			if(skillDamage > damage)
			{
				damage = skillDamage;
				skill = skillCode;
			}
		}

/*		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + ": returning next most damaging skill : " + skill + " @ $" + damage + "(" + remainingMosp.intValue() + ") left" );
		} */

		return skill;
	}


	public int getTotalMospPenalty()
	{
		int totalPenalty = 0;
		for(String skillCode : mospCompletion.keySet())
		{
			Integer remainingMosp = mospCompletion.get(skillCode);
			Integer penaltyPerMosp = costPerMospParams.get(skillCode);

			totalPenalty += (remainingMosp.intValue() * penaltyPerMosp.intValue());
		}
		for(String skillCode : mospFailure.keySet())
		{
			Integer remainingMosp = mospFailure.get(skillCode);
			Integer penaltyPerMosp = costPerMospParams.get(skillCode);

			totalPenalty += (remainingMosp.intValue() * penaltyPerMosp.intValue());
		}

		return totalPenalty;
	}

	/**
	 * Return a <code>List</code> of all MOS that
	 * have any points unfilled and the amount.
	 */
	public Map<String, Integer> getFailedMosp()
	{
		Map<String, Integer> skillMap = new HashMap<String,Integer>();
		for(String skillCode : mospCompletion.keySet())
		{
			Integer remainingMosp = mospCompletion.get(skillCode);

			skillMap.put(skillCode, remainingMosp);

		}
		for(String skillCode : mospFailure.keySet())
		{
			Integer remainingMosp = mospFailure.get(skillCode);

			skillMap.put(skillCode, remainingMosp);
		}

		return skillMap;
	}


}
