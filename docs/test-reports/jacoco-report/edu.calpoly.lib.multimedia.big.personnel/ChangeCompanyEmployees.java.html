package edu.calpoly.lib.multimedia.big.personnel;

public class ChangeCompanyEmployees extends BIGAction
{
	public ActionForward execute(ActionMapping mapping,
				     			 ActionForm form,
				     			 HttpServletRequest request,
			         			 HttpServletResponse response)
	{
		BIGLog log = BIGLogFactory.getLog();
		if(log.isTraceEnabled())
		{
			log.trace(this.getClass() + " beginning execute method");
		}


	    ActionMessages errors = new ActionMessages();

        ServletContext context = servlet.getServletContext();
        HttpSession session = request.getSession();
	    Game game = (Game) session.getAttribute("game");

        LoginBroker loginBroker = (LoginBroker) context.getAttribute("loginBroker");

        // If for any reason the company should not be allowed to login, route the
        // company to an error message page.
	    if(!loginBroker.verifyGameLoginAllowed(game, context, request))
		   return (mapping.findForward("loginFailure"));

		Company company = (Company) session.getAttribute("company");

		String action = request.getParameter("action");
		String employeeType = request.getParameter("employeeType");

		if(BIGLogFactory.isDebugEnabled())
		{
			BIGLog companyLog = BIGLogFactory.getCompanyLog(company.getCompanyID());
			companyLog.debug("Requested personnel change: Action = " + action + ", Employee Type =" + employeeType);
		}

		BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);

		try
		{
			EmployeeTypeRecord etr = dbFacade.getCompanyPersonnelDBHandler().getEmployeeTypeRecord(game, company, game.getCurrentPeriodNumber(), employeeType);

			if(etr == null)
			{
				//There is no mapping at all for this employee type.  Check the employeeType against
				//the list of available of employee types
				if(dbFacade.getCompanyPersonnelDBHandler().isValidEmployeeType(game, employeeType))
				{
					etr = new EmployeeTypeRecord();
					etr.setEmployeeType(employeeType);
				}
			}

			if(etr == null) //still null means this isn't a registered employee type
			{
				//If no EmployeeTypeRecord was found it's likely a bad parameter name was entered
				//Add an error message and log the error
				log.error(this.getClass() + " no employee type record found for employee type: " + employeeType + ", companyid: " + company.getCompanyID() + ", period: " + game.getCurrentPeriodNumber());
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.parameters.invalid"));
			}
			else
			{
				if(action.equalsIgnoreCase("hire"))
				{
					if(log.isDebugEnabled())
					{
						log.debug("Hire new " + employeeType);
					}
					doHire(etr);
				}
				else if(action.equalsIgnoreCase("fire"))
				{
					if(log.isDebugEnabled())
					{
						log.debug("Fire " + employeeType);
					}
					doFire(etr);
				}

				dbFacade.getCompanyPersonnelDBHandler().deleteEmployeeTypeRecord(game, company, game.getCurrentPeriodNumber(), employeeType);
				dbFacade.getCompanyPersonnelDBHandler().addEmployeeTypeRecord(game, company, game.getCurrentPeriodNumber(), employeeType, etr.getEmployeeTypeCount(), etr.getEmployeeTypeHires(), etr.getEmployeeTypeFires());
			}

			dbFacade.close();

		}
		catch(BIGDatabaseFacadeException bigdbe)
		{
			log.fatal("", bigdbe);
			errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
		}

		if(log.isDebugEnabled())
		{
		}


		if (!errors.isEmpty())
		{
			if(log.isTraceEnabled())
			{
				log.trace(this.getClass() + " request returned with errors");
			}
			// Report any errors we have discovered back to the original form
			saveErrors(request, errors);
	 	   	saveToken(request);
	 	   	return (new ActionForward(mapping.getInput()));
		}
		else
		{
			if(log.isTraceEnabled())
			{
				log.trace(this.getClass() + " request completed.  Forwarding to \"success\"");
			}
			return (mapping.findForward("success"));
		}
    }

    //If the amount to fire is zero, the amount to hire is increased by one.
    //Otherwise the amount to fire is decreased by one.
    private void doHire(EmployeeTypeRecord etr)
    {
		if(etr.getEmployeeTypeFires() > 0)
		{
			etr.setEmployeeTypeFires(etr.getEmployeeTypeFires() - 1);
		}
		else
		{
			etr.setEmployeeTypeHires(etr.getEmployeeTypeHires() + 1);
		}
	}

    //If the amount to hire is zero, the amount to fire is increased by one.
    //Otherwise the amount to hire is decreased by one.
	private void doFire(EmployeeTypeRecord etr)
	{
		if(etr.getEmployeeTypeHires() > 0)
		{
			etr.setEmployeeTypeHires(etr.getEmployeeTypeHires() - 1);
		}
		else
		{
			etr.setEmployeeTypeFires(etr.getEmployeeTypeFires() + 1);
		}
	}
}
