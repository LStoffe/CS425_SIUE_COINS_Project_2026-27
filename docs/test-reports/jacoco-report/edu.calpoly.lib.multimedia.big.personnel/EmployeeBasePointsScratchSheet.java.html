package edu.calpoly.lib.multimedia.big.personnel;

//JDK imports
import java.util.List;
import java.util.ArrayList;
import java.util.Map;
import java.util.HashMap;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Class to keep track of how many base points
 * each employee type has remaining during the calculations
 * that convert employee base points to Main Office Skill points for a
 * Main Office Skill.  As points are used up they can
 * be "scratched" off the sheet.
 *
 * TODO: Journal how points are used so that the results can be dumped
 * to the database later.
 */
public class EmployeeBasePointsScratchSheet
{
	private Map<String, Integer> scratchSheet;
	private Log log;

	public EmployeeBasePointsScratchSheet()
	{
		scratchSheet = new HashMap<String, Integer>();
		log = LogFactory.getLog(this.getClass());
	}

	/**
	 * Returns the current base points remaining
	 * for an employee type.  All points must previously been
	 * added through a call to <code>addBasePoints</code>.
	 * If not points have been added for an employee type
	 * a call to this method for that type will always return 0.
	 */
	public int getPointsRemaining(String employeeType)
	{
		Integer points = scratchSheet.get(employeeType);
		if(points == null)
		{
			return 0;
		}
		else
		{
			return points.intValue();
		}
	}

	/**
	 * Adds base points for an employee type.
	 */
	public void addBasePoints(String employeeType, int points)
	{
		Integer currentPoints = scratchSheet.get(employeeType);
		if(currentPoints == null)
		{
			currentPoints = new Integer(0);
		}

/*		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " marking " + points + " base points for employee type " + employeeType + " on the scratchsheet");
		} */

		scratchSheet.put(employeeType, new Integer(currentPoints.intValue() + points));
	}

	/**
	 * Reduces base points for an employee type.  If base points
	 * are reduced to 0 or lower, the employee type is removed from the
	 * list of employee type returned by the
	 * <code>getEmployeeTypesWithPoints</code> method.
	 */
	public void reduceBasePoints(String employeeType, int points)
	{
		Integer currentPoints = scratchSheet.get(employeeType);
		if(currentPoints == null)
		{
			currentPoints = new Integer(0);
		}

		int newPoints = currentPoints.intValue() - points;

/*		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " scratching off " + points + " base points for employee type " + employeeType + " from the scratchsheet");
			log.debug(this.getClass() + " new amount of points is " + newPoints);
		} */

		if(newPoints > 0)
		{
			scratchSheet.put(employeeType, new Integer(newPoints));
		}
		else
		{
			scratchSheet.remove(employeeType);
/*			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " employee type " + employeeType + " has been removed from scratchsheet");
			} */

		}
	}

	/**
	 * Returns a <code>List</code> of the employee types who still have base points remaining
	 */
	public List<String> getEmployeeTypesWithPoints()
	{
		return new ArrayList(scratchSheet.keySet());
	}

	/**
	 * return true if there is at least one employee left with base points, false if not
	 */
	public boolean employeesLeftWithBasePoints()
	{
		return (scratchSheet.keySet().size() > 0);
	}
}
