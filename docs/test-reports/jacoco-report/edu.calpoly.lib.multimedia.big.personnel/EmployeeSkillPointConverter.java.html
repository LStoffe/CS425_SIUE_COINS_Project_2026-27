package edu.calpoly.lib.multimedia.big.personnel;

//JDK imports
import java.util.List;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Class that does the actual work of determining how each employee types'
 * base points are converted into Main Office Skill points.
 * <br><br>
 * In the real world a construction company having jobs and staying in business
 * would cause overhead work to at the main office.  The company must hire employees
 * to do this work.  Additionally the work at a main office may require many different
 * skills: finance, marketing, leadership.  Not every employee is capable of
 * doing everyone else's job (however there is some overlap).
 * <br><br>
 * BIG models main office work in terms units called of points.  Each point is
 * associated with a certain skill that is required in the main office, giving
 * them the name Main Office Skill Points (MOSPs).  In the real world, the amount
 * of skills a company requires in different business areas probably depends
 * on their business plan and the goals of the company.  In BIG the number of MOSP
 * a company require for each different skill depends on the number, size, and type
 * of company's current jobs.  This is far easier to model at this stage in BIG.
 * <br><br>
 * In BIG jobs create the need for MOSPs.  Each MOSP can be though of as some amount of
 * work, requiring some skill, that must be completed for the company to function.
 * In BIG penalties are usually monetary, so each unfilled MOSP will cause the company to
 * be charged some dollar amount (with the reason given this is the cost to hire a consultant or temp).
 * To fill MOSPs without being charged a penalty companies can hire personnel.  Personnel are
 * paid a salary but this should be significantly less than the penalty.  Each
 * employee type comes with a set number of Base Points.  Base Points are useless
 * by themselves, but they can be converted into useful MOSPs.  The amount
 * of Base Point required to create a MOSP depends on the skill type of the MOSP
 * and the conversion rate for the employee type for that skill.  Some employee types
 * are far more efficient at turning Base Points into MOSPs for certain skills than other.
 * Not all employee types can even turn Base Points into MOSPs for all skills.
 * For many employees though there is overlap in the types of MOSP they can generate.
 * The purpose of this algorithm is to generate MOSPs from Base Points efficiently.
 * <br><br>
 * <pre>
   Example:
   Employee Type 1 has 10 Base Points and can turn 1 Base Point into 1 MOSP for Skill 1, or 2 Base Points into 1 MOSP for Skill 2.
   Employee Type 2 has 5 Base Points and can turn 3 Base Points into 1 MOSP for Skill 1, or 1 Base Point into 1 MOSP for Skill 2.

   Say 8 MOSP are required (based on the number of jobs) for Skill 1 and 6 are required for Skill 2.
   To be efficient, Employee 1 should convert 8 Base Points into 8 MOSP for Skill 1 and 2 Base Points into 1 MOSP for Skill 2
   and Employee 2 should turn all 5 Base Points into MOSP for Skill 2.
   </pre>
 *
 */
public class EmployeeSkillPointConverter
{
	private Log log;

	//This DEFINETLY needs to be brought out into configuration.  This is the number
	//of MOSPs generated at a time.  Currently Base points are stored as ints
	//but ratios are stored as floats.  This means signficant base points may
	//be lost to rounding.  The best solution is to convert base point storage to
	//float but for now we can increase the number of MOSPs generated at a time.
	private int MOSP_GEN_SIZE = 2;

	private float floatEqualRange = .02f; //Two float are considered equal if they are less than this close together.
										 //It's a wide range because small differents in float values are not very significant for this usage.

	private float efficiencyCutoff = 0.2f;

	public EmployeeSkillPointConverter()
	{
		this.log = LogFactory.getLog(this.getClass());
	}


	/**
	 * <code>convert</code> is the implementation
	 * for the employee base point to mos point generation algorithm.
	 * @param employeeSkillConversionRatioLookup Must contain all needed conversion ratios for employee types.
	 * @param scratchSheet Must already be loaded with available employee types and base points.
	 * When this method returns the scratchSheet will be used up.
	 * @param jobMosManager Generated MOSPs will be given to the JobMosManager
	 * and which should update its counters accordingly.  When this method returns
	 * the JobMosManager will contain the work done in the form that its MOSP counters should
	 * be lowered or zeroed out.
	 *
	 */
	public void convert(EmployeeSkillConversionRatioLookup employeeSkillConversionRatioLookup,
					    EmployeeBasePointsScratchSheet scratchSheet,
					    JobMosManager jobMosManager
					   )
	{
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " starting convert method");
			// log.debug(this.getClass() + " employee base points will be used to generate needed MOS points");
	 	}

		//Loop while we still have base points and still need mos points.
		while(scratchSheet.employeesLeftWithBasePoints() && jobMosManager.stillMOSPRequired())
		{
			//Pick the next most damaging skill
			String skill = jobMosManager.getMostDamagingMOS();

			//Get the employees that still have base points
			List<String> employees = scratchSheet.getEmployeeTypesWithPoints();

			//Pick the employee to generate the next points
			String employee = pickBestEmployeeForSkill(skill, employees, employeeSkillConversionRatioLookup);

			if(employee == null) //no one can complete skill
			{
				jobMosManager.failMosp(skill);
			}
			else
			{

				//Subtract the points from the job skill and (after conversion using the ratio) the employee's base points
				jobMosManager.reduceRequiredMOSP(skill, MOSP_GEN_SIZE);

				float mospPoints = (float) MOSP_GEN_SIZE;
				int basePoints = Math.round(mospPoints / employeeSkillConversionRatioLookup.getConversionRatio(employee, skill));

/*				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + ": reducing Job Required " + skill + " points by " + MOSP_GEN_SIZE + " and " + employee + " base points by " + basePoints);
				} */

				scratchSheet.reduceBasePoints(employee, basePoints);
			}

		}

 		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " ending convert method");
 		}

	}

	//Picks the employee type with the best ratio for the skill.
	//In the event there are multiple employee types with the same ratio,
	//picks the employee type with the lowest generalization value.
	private String pickBestEmployeeForSkill(String skill, List<String> employees, EmployeeSkillConversionRatioLookup lookup)
	{
		String bestEmployee = null;

		float bestRatio = 0.0f;

		for(String employee : employees)
		{
			float ratio = lookup.getConversionRatio(employee, skill);
			if(ratio > efficiencyCutoff)
			{
				if(equalFloats(ratio, bestRatio, floatEqualRange))
				{
					float gen1 = lookup.getEmployeeSkillGeneralization(employee);
					float gen2 = lookup.getEmployeeSkillGeneralization(bestEmployee);

					if(gen1 < gen2)
					{
						bestRatio = ratio;
						bestEmployee = employee;
					}
				}
				else if(ratio > bestRatio)
				{
					bestRatio = ratio;
					bestEmployee = employee;
				}
			}
		}

/*		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " picked " + bestEmployee + " as best employee for skill");
		} */

		return bestEmployee;
	}

	private boolean equalFloats(float float1, float float2, float distance)
	{
		if ( Math.abs(float1 - float2) < distance )
			return true;
		return false;
	}


}
