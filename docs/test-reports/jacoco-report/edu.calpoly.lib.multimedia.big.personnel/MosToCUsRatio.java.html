package edu.calpoly.lib.multimedia.big.personnel;

/**
 * Class representing the amount of Main Office Skill Points (mosp)
 * required for a certain about of Construction Units (cus).
 */
public class MosToCUsRatio
{
	private int cus;
	private int mosp;

	public MosToCUsRatio()
	{
		this.cus = 0;
		this.mosp = 0;
	}

	public MosToCUsRatio(int cus, int mosp)
	{
		this.cus = cus;
		this.mosp = mosp;
	}

	public int getCus()
	{
		return this.cus;
	}

	public void setCus(int cus)
	{
		this.cus = cus;
	}

	public int getMosp()
	{
		return mosp;
	}

	public void setMosp(int mosp)
	{
		this.mosp = mosp;
	}

	/**
	 * Returns the ratio as a float.  If Cus is 0, 0 is returned.
	 */
	public float getRatio()
	{
		if(cus == 0)  //would cause divide by 0 error
		{
			return 0;
		}
		else
		{
			return (((float) mosp) / ((float) cus));
		}
	}
}
