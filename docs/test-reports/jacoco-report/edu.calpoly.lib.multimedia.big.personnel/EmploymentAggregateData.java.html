package edu.calpoly.lib.multimedia.big.personnel;

//JDK imports
import java.util.List;

//BIG imports
import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * Analyzes a list of <code>EmploymentTypeRecord</code>s to produce aggregate employment data.
 */
public class EmploymentAggregateData
{
	private int totalEmployees;
	private long totalPayrollPerPeriod;
	private long newTotalPayrollPerPeriod;

	public EmploymentAggregateData(List<EmployeeTypeRecord> employeeTypeRecords)
	{
		analyze(employeeTypeRecords);
	}

	public int getTotalEmployees()
	{
		return totalEmployees;
	}

	public long getTotalPayroll()
	{
		return totalPayrollPerPeriod;
	}

	public String getFormattedTotalPayroll()
	{
		return NumberConverter.stringValue(totalPayrollPerPeriod);
	}

	public long getNewTotalPayroll()
	{
		return newTotalPayrollPerPeriod;
	}

	public String getFormattedNewTotalPayroll()
	{
		return NumberConverter.stringValue(newTotalPayrollPerPeriod);
	}


	private void analyze(List<EmployeeTypeRecord> employeeTypeRecords)
	{
		totalEmployees = 0;
		totalPayrollPerPeriod = 0;
		for(EmployeeTypeRecord etr : employeeTypeRecords)
		{
			totalEmployees += etr.getEmployeeTypeCount();
			totalPayrollPerPeriod += etr.getEmployeeTypeTotalCost();
			newTotalPayrollPerPeriod += etr.getNewEmployeeTypeTotalCost();
		}
	}
}
