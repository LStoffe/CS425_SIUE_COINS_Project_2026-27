package edu.calpoly.lib.multimedia.big.personnel;

//JDK imports
import java.util.Map;
import java.util.HashMap;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Class for easy lookup of employee mosp conversion ratios.
 */
public class EmployeeSkillConversionRatioLookup
{
	//Keys in first map are employee types, in second map are
	//skill codes.
	private Map<String, Map<String, Float>> lookupTable;

	private Log log;

	public EmployeeSkillConversionRatioLookup()
	{
		this.log = LogFactory.getLog(this.getClass());

		lookupTable = new HashMap<String, Map<String,Float>>();
	}

	/**
	 * Returns the base point to mosp ratio (base points * ratio = mosp)
	 * for an employee type and skill.
	 * @param employeeType The employee type for which to return the ratio
	 * @param skillCode The skill for which to return the ratio
	 * @return The requested ratio or 0.0f if no such employee type
	 * or skill code exist in the lookup table.
	 */
	public float getConversionRatio(String employeeType, String skillCode)
	{
		Map employeeMap = lookupTable.get(employeeType);

		if(employeeMap == null)
		{
			return 0.0f;
		}
		else
		{
			Float ratio = (Float) employeeMap.get(skillCode);
			if(ratio != null)
			{
				return ratio.floatValue();
			}
			else
			{
				return 0.0f;
			}
		}
	}

	/**
	 * Loads the lookup table.  In order for a conversion ratio
	 * to be returned by <code>getConversionRatio</code>, it must
	 * first have been set through this method.
	 *
	 */
	public void addConversionRatio(String employeeType, String skillCode, float ratio)
	{
		Map<String, Float> employeeMap = lookupTable.get(employeeType);

		if(employeeMap == null)
		{
			employeeMap = new HashMap<String, Float>();
			lookupTable.put(employeeType, employeeMap);
		}

		employeeMap.put(skillCode, new Float(ratio));
	}


	/**
	 * The generalization value for an employee type is the sum of their
	 * ratios for all skills.  A higher sum means that employee type
	 * is more efficient at converting base points to MOSP for more skills.
	 */
	public float getEmployeeSkillGeneralization(String employeeType)
	{
		float generalization = 0.0f;
		Map<String, Float> employeeMap = lookupTable.get(employeeType);

		if(employeeMap != null)
		{
			for(String skill : employeeMap.keySet())
			{
				Float ratio = (Float) employeeMap.get(skill);
				if(ratio != null)
				{
					generalization += ratio;
				}
			}
		}

		return generalization;
	}


}
