package edu.calpoly.lib.multimedia.big.personnel;

import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * JavaBean class representing the current status
 * of an employee type for a company during a period.
 * This includes, the employee type, the salary/wage per period
 * for that time, how many of employees of the type the company
 * has employeed, and changes to the number of that employee type to
 * occur in the period.
 *
 * EmployeeTypeHires/Fires are not included in employeeTypeCount for the current period,
 * but will be included in the employeeTypeCount for the period after this one.  The formula for
 * an employeeTypeCount is <code>employeeTypeCount[period n+1] = employeeTypeCount[period n] + employeeTypeHires[period n] - employeeTypeFires[period n]</code>
 * Realistically employeeTypeHires and employeeTypeFires will never both have a value greater than 0 in the same period,
 * but
 */
public class EmployeeTypeRecord
{
	private int companyID;
	private int periodNumber;
	private String employeeType;
	private int employeeCost;
	private int employeeHiringCost;
	private int employeeFiringCost;
	private int employeeTypeCount;
	private int employeeTypeHires;
	private int employeeTypeFires;
	private int employeeTypeBasePoints;
	private int category;

	protected String jobDescription = "Job description not found";

	public EmployeeTypeRecord()
	{
	}

	/**
	 * Returns the companyID of the company to which this record belongs.
	 */
	public int getCompanyID()
	{
		return this.companyID;
	}

	/**
	 * Sets the companyID of the company to which this record belongs.
	 */
	public void setCompanyID(int companyID)
	{
		this.companyID = companyID;
	}

	public void setCompanyIDInteger(Integer companyID)
	{
		if(companyID != null)
		{
			this.companyID = companyID.intValue();
		}
		else
		{
			this.companyID = 0;
		}
	}

	/**
	 * Returns the periodNumber of the period to which this record applies.
	 */
	public int getPeriodNumber()
	{
		return this.periodNumber;
	}

	/**
	 * Sets the periodNumber of period to which this record applies.
	 */
	public void setPeriodNumber(int periodNumber)
	{
		this.periodNumber = periodNumber;
	}


	public void setPeriodNumberInteger(Integer periodNumber)
	{
		if(periodNumber != null)
		{
			this.periodNumber = periodNumber.intValue();
		}
		else
		{
			this.periodNumber = 0;
		}
	}

	/**
	 * Returns the type of employee to which this record applies.
	 */
	public String getEmployeeType()
	{
		return this.employeeType;
	}

	/**
	 * Sets the type of employee to which this record applies.
	 */
	public void setEmployeeType(String employeeType)
	{
		this.employeeType = employeeType;
	}

	/**
	 * Returns the cost per period for each employee of the employee type
	 */
	public int getEmployeeCost()
	{
		return this.employeeCost;
	}

	public String getFormattedEmployeeCost()
	{
		return NumberConverter.stringValue(employeeCost);
	}

	/**
	 * Sets the cost per period for each employee of the employee type
	 */
	public void setEmployeeCost(int employeeCost)
	{
		this.employeeCost = employeeCost;
	}

	/**
	 * Returns the one time cost to hire an employee of the employee type
	 */
	public int getEmployeeHiringCost()
	{
		return this.employeeHiringCost;
	}

	public String getFormattedEmployeeHiringCost()
	{
		return NumberConverter.stringValue(employeeHiringCost);
	}

	/**
	 * Sets the one time cost to hire an employee of the employee type
	 */
	public void setEmployeeHiringCost(int employeeFiringCost)
	{
		this.employeeHiringCost = employeeHiringCost;
	}

	/**
	 * Returns the one time cost to fire an employee of the employee type
	 */
	public int getEmployeeFiringCost()
	{
		return this.employeeCost;
	}

	public String getFormattedEmployeeFiringCost()
	{
		return NumberConverter.stringValue(employeeFiringCost);
	}

	/**
	 * Sets the one time cost to fire an employee of the employee type
	 */
	public void setEmployeeFiringCost(int employeeFiringCost)
	{
		this.employeeCost = employeeCost;
	}

	/**
	 * Returns the number of employees of the employeeType that the company has employeed during the period.
	 */
    public int getEmployeeTypeCount()
    {
		return employeeTypeCount;
	}

	/**
	 * Sets the number of employees of the employeeType that the company has employeed during the period.
	 */
	public void setEmployeeTypeCount(int employeeTypeCount)
	{
		this.employeeTypeCount = employeeTypeCount;
	}

	public void setEmployeeTypeCountInteger(Integer employeeTypeCount)
	{
		if(employeeTypeCount != null)
		{
			this.employeeTypeCount = employeeTypeCount.intValue();
		}
		else
		{
			this.employeeTypeCount = 0;
		}
	}

	/**
	 * Returns the number of employees of the employeeType that the company will add this period.
	 */
	public int getEmployeeTypeHires()
	{
		return employeeTypeHires;
	}

	/**
	 * Sets the number of employees of the employeeType that the company will add this period.
	 */
	public void setEmployeeTypeHires(int employeeTypeHires)
	{
		this.employeeTypeHires = employeeTypeHires;
	}

	public void setEmployeeTypeHiresInteger(Integer employeeTypeHires)
	{
		if(employeeTypeHires != null)
		{
			this.employeeTypeHires = employeeTypeHires.intValue();
		}
		else
		{
			this.employeeTypeHires = 0;
		}
	}

	/**
	 * Returns the number of employees of the employeeType that the company will subtract this period.
	 */
	public int getEmployeeTypeFires()
	{
		return employeeTypeFires;
	}

	/**
	 * Sets the number of employees of the employeeType that the company will substract this period.
	 */
	public void setEmployeeTypeFires(int employeeTypeFires)
	{
		this.employeeTypeFires = employeeTypeFires;
	}

	public void setEmployeeTypeFiresInteger(Integer employeeTypeFires)
	{
		if(employeeTypeFires != null)
		{
			this.employeeTypeFires = employeeTypeFires.intValue();
		}
		else
		{
			this.employeeTypeFires = 0;
		}
	}

    /**
     * Returns the total amount spend on this employee type <code>employeeCost * employeeTypeCount</code>
     */
	public int getEmployeeTypeTotalCost()
	{
		return getEmployeeTypeCount() * getEmployeeCost();
	}

    /**
     * Returns the total amount spend on this employee type <code>employeeCost * employeeTypeCount</code>
     */
	public String getFormattedEmployeeTypeTotalCost()
	{
		return NumberConverter.stringValue(getEmployeeTypeTotalCost());
	}

	/**
	 * Returns a <code>String</code> representation of this object.
	 */
	public String toString()
	{
		return "CompanyID: " + companyID + ", Period Number: " + periodNumber + ", Employee Type: " + employeeType + ", Employee Cost: " + employeeCost + ", Employee Count: " + employeeTypeCount + ", Employee Hires: " + employeeTypeHires + ", Employee Fires: " + employeeTypeFires;
	}

	/**
	 * Returns the job description for this employee type.
	 */
	public String getJobDescription()
	{
		return jobDescription;
	}

    /**
     * Sets the job description for this employee type.
     */
    public void setJobDescription(String jobDescription)
    {
		this.jobDescription = jobDescription;
	}

	/**
	 * Returns the number of employees of the employeeType that the company will have employed after hires and fire are taken into account.
	 */
    public int getNewEmployeeTypeCount()
    {
		int count = employeeTypeCount;
		count = count + employeeTypeHires - employeeTypeFires;

		if(count < 0) count = 0; //Can't have negative employees

		return count;
	}

	public int getNewEmployeeTypeTotalCost()
	{
		return getNewEmployeeTypeCount() * getEmployeeCost();
	}

    /**
     * Returns the total amount spend on this employee type <code>employeeCost * employeeTypeCount</code>
     */
	public String getFormattedNewEmployeeTypeTotalCost()
	{
		return NumberConverter.stringValue(getNewEmployeeTypeTotalCost());
	}

	/**
	 * Gets the base points for a single employee of this type.
	 */
	public int getEmployeeBasePoints()
	{
		return employeeTypeBasePoints;
	}

	/**
	 * Sets the base points for a single employee of this type.
	 */
	public void setEmployeeBasePoints(int basePoints)
	{
		this.employeeTypeBasePoints = basePoints;
	}

	/**
	 * Gets the total base points for this employee type for the company.
	 * This is the total of the base points for each employee of this type employeed by
	 * the company.
	 */
	public int getTotalBasePointsForType()
	{
		return getEmployeeTypeCount() * getEmployeeBasePoints();
	}

	/**
	 * Gets the category of employee ("1" = Commercial, "2" = Heavy).
	 */
	public int getCategory()
	{
		return category;
	}

}
