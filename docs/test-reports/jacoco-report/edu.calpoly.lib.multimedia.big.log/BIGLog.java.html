package edu.calpoly.lib.multimedia.big.log;

//JDK imports
import java.io.PrintWriter;
import java.io.FileOutputStream;
import java.util.Date;
import java.text.SimpleDateFormat;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Class that writes log messages to a log file.
 */
public class BIGLog
{
	private String logFilePath;

	private PrintWriter out;

	private static String dateFormatString = "MMM-dd-yyyy kk:mm:ss";
	private SimpleDateFormat dateFormat;

	//We'll also add all entries to the master log file
	//using commons-logging
	private Log log;

	public BIGLog(String logFilePath)
	{
		this.logFilePath = logFilePath;

		try
		{
			out = new PrintWriter(new FileOutputStream(logFilePath));
		}
		catch(Exception e)
		{
			e.printStackTrace();
		}

		dateFormat = new SimpleDateFormat(dateFormatString);

		log = LogFactory.getLog(this.getClass());
	}

	private void writeToLog(String message, String level)
	{
		StringBuilder sb = new StringBuilder();
		sb.append(dateFormat.format(new Date()));
		sb.append(" - ");
		sb.append(level);
		sb.append(" - ");
		sb.append(message);
		out.println(sb.toString());
		out.flush();
	}

	private void writeToLog(Exception e, String level)
	{
		StringBuilder sb = new StringBuilder();
		sb.append(dateFormat.format(new Date()));
		sb.append(" - ");
		sb.append(level);
		out.println(sb.toString());
		e.printStackTrace(out);
		out.flush();
	}

	private void writeToLog(String message, Exception e, String level)
	{
		StringBuilder sb = new StringBuilder();
		sb.append(dateFormat.format(new Date()));
		sb.append(" - ");
		sb.append(level);
		sb.append(" - ");
		sb.append(message);
		out.println(sb.toString());
		e.printStackTrace(out);
		out.flush();
	}

	//************  TRACE *******************//
	public void trace(String message)
	{
		if(isTraceEnabled())
		{
			writeToLog(message, "TRACE");
			log.trace(message);
		}
	}

	public void trace(Exception e)
	{
		if(isTraceEnabled())
		{
			writeToLog(e, "TRACE");
			log.trace(e);
		}
	}

	public void trace(String message, Exception e)
	{
		if(isTraceEnabled())
		{
			writeToLog(message, e, "TRACE");
			log.trace(message, e);
		}
	}

	//************  DEBUG *******************//
	public void debug(String message)
	{
		if(isDebugEnabled())
		{
			writeToLog(message, "DEBUG");
			log.debug(message);
		}
	}

	public void debug(Exception e)
	{
		if(isDebugEnabled())
		{
			writeToLog(e, "DEBUG");
			log.debug(e);
		}
	}

	public void debug(String message, Exception e)
	{
		if(isDebugEnabled())
		{
			writeToLog(message, e, "DEBUG");
			log.debug(message, e);
		}
	}

	//************  INFO *******************//
	public void info(String message)
	{
		if(isInfoEnabled())
		{
			writeToLog(message, "INFO");
			log.info(message);
		}
	}

	public void info(Exception e)
	{
		if(isInfoEnabled())
		{
			writeToLog(e, "INFO");
			log.info(e);
		}
	}

	public void info(String message, Exception e)
	{
		if(isInfoEnabled())
		{
			writeToLog(message, e, "INFO");
			log.info(message, e);
		}
	}

	//************ WARN *******************//
	public void warn(String message)
	{
		if(isWarnEnabled())
		{
			writeToLog(message, "WARN");
			log.warn(message);
		}
	}

	public void warn(Exception e)
	{
		if(isWarnEnabled())
		{
			writeToLog(e, "WARN");
			log.warn(e);
		}
	}

	public void warn(String message, Exception e)
	{
		if(isWarnEnabled())
		{
			writeToLog(message, e, "WARN");
			log.warn(message, e);
		}
	}

	//************  ERROR *******************//
	public void error(String message)
	{
		if(isErrorEnabled())
		{
			writeToLog(message, "ERROR");
			log.error(message);
		}
	}

	public void error(Exception e)
	{
		if(isErrorEnabled())
		{
			writeToLog(e, "ERROR");
			log.error(e);
		}
	}

	public void error(String message, Exception e)
	{
		if(isErrorEnabled())
		{
			writeToLog(message, e, "ERROR");
			log.error(message, e);
		}
	}

	//************  FATAL *******************//
	public void fatal(String message)
	{
		if(isFatalEnabled())
		{
			writeToLog(message, "FATAL");
			log.fatal(message);
		}
	}

	public void fatal(Exception e)
	{
		if(isFatalEnabled())
		{
			writeToLog(e, "FATAL");
			log.fatal(e);
		}
	}

	public void fatal(String message, Exception e)
	{
		if(isFatalEnabled())
		{
			writeToLog(message, e, "FATAL");
			log.fatal(message, e);
		}
	}

	public boolean isTraceEnabled()
	{
		return BIGLogFactory.isTraceEnabled();
	}

	public boolean isDebugEnabled()
	{
		return BIGLogFactory.isDebugEnabled();
	}

	public boolean isInfoEnabled()
	{
		return BIGLogFactory.isInfoEnabled();
	}

	public boolean isWarnEnabled()
	{
		return BIGLogFactory.isWarnEnabled();
	}

	public boolean isErrorEnabled()
	{
		return BIGLogFactory.isErrorEnabled();
	}

	public boolean isFatalEnabled()
	{
		return BIGLogFactory.isFatalEnabled();
	}

	public void finalize()
	{
		out.flush();
		out.close();
	}
}
