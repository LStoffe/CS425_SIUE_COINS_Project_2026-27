public class BIGLogFactory
{
	private static boolean trace;
	private static boolean debug;
	private static boolean info;
	private static boolean warn;
	private static boolean error;
	private static boolean fatal;

	private static String[] possibleLevels = { "trace", "debug", "info", "warn", "error", "fatal" };
	private static String logDirectory;
	private static String logLevel;

	private static File logDirectoryAsFile;

	private static Map<String, BIGLog> logs;

	static
	{
		//Use commons-logging to get us started
		Log log = LogFactory.getLog("edu.calpoly.lib.multimedia.big.log.BIGLogFactory");
		if(log.isInfoEnabled())
		{
			log.info("Bringing up BIG logging system...");
		}

		try
		{
			//Use the log4j Loader to retrieve the URL of the BIGLog config file.
			URL configFileURL = Loader.getResource("BIGLog.properties");

			Properties BIGLogProps = new Properties();

			BIGLogProps.load(configFileURL.openStream());

			logDirectory = BIGLogProps.getProperty("log.directory");

			if(log.isDebugEnabled())
			{
				log.debug("Saving BIG logs in: " + logDirectory);
			}
			setLogLevel(BIGLogProps.getProperty("log.level"));

		}
		catch(Exception e)
		{
			log.fatal(e);
		}

		logs = new HashMap<String, BIGLog>();

		logDirectoryAsFile = new File(logDirectory);

		if(!logDirectoryAsFile.exists())
		{
			try
			{
				logDirectoryAsFile.mkdir();
			}
			catch(Exception e)
			{
				e.printStackTrace();
			}
		}
	}

	//Sets the log level to the specified level, or to level 'warn'
	//if no log level is specified or an invalid level is specified.
	private static void setLogLevel(String level)
	{
		if(level == null)
		{
			logLevel = "warn";
		}
		else if(inList(possibleLevels, level))
		{
			logLevel = level;
		}
		else
		{
			logLevel = "warn";
		}

		if(logLevel.equalsIgnoreCase("fatal"))
		{
			fatal = true;
		}
		else if(logLevel.equalsIgnoreCase("error"))
		{
			fatal = true; error = true;
		}
		else if(logLevel.equalsIgnoreCase("warn"))
		{
			fatal = true; error = true; warn = true;
		}
		else if(logLevel.equalsIgnoreCase("info"))
		{
			fatal = true; error = true; warn = true; info = true;
		}
		else if(logLevel.equalsIgnoreCase("debug"))
		{
			fatal = true; error = true; warn = true; info = true; debug = true;
		}
		else if(logLevel.equalsIgnoreCase("trace"))
		{
			fatal = true; error = true; warn = true; info = true; debug = true; trace = true;
		}
	}

	private static boolean inList(String[] list, String value)
	{
		boolean found = false;
		for(int i = 0; i < list.length; i++)
		{
			if(value.equalsIgnoreCase(list[i]))
			{
				found = true;
			}
		}
		return found;
	}

	private static BIGLog getLog(String logName)
	{
		BIGLog log = logs.get(logName);
		if(log == null)
		{
//			log = new BIGLog(logDirectoryAsFile.getPath() + "\\" + logName + ".log");
			log = new BIGLog(logDirectoryAsFile.getPath() + System.getProperty("file.separator") + logName + ".log");
			logs.put(logName, log);
		}
		return log;
	}

	/**
	 * Returns a general log
	 */
	public static BIGLog getLog()
	{
		return getLog("General");
	}

	/**
	 * Returns the log for the game with a given id
	 */
	public static BIGLog getGameLog(int gameID)
	{
		return getLog("Game_" + gameID);
	}

	/**
	 * Returns the log for the company with the given id
	 */
	public static BIGLog getCompanyLog(int companyID)
	{
		return getLog("Company_" + companyID);
	}

	/**
	 * Returns the log for the job in the given game with the given job number.
	 */
	public static BIGLog getJobLog(int gameID, int jobID)
	{
		return getLog("Game_" + gameID + "_Job_" + jobID);
	}

	public static boolean isTraceEnabled()
	{
		return trace;
	}

	public static boolean isDebugEnabled()
	{
		return debug;
	}

	public static boolean isInfoEnabled()
	{
		return info;
	}

	public static boolean isWarnEnabled()
	{
		return warn;
	}

	public static boolean isErrorEnabled()
	{
		return error;
	}

	public static boolean isFatalEnabled()
	{
		return fatal;
	}

}
