	public String modifyMasterEquipList()
	{

	public void validateCompanyPurchaseNewOrUsedEquipmentMarket()
	{
		// need to know how many rows are in the table
		HttpSession session = servletRequest.getSession();
        int numbEquip = (int) session.getAttribute("valueOfI");

        if (numbEquip != 0) {
           for (int i=0; i<numbEquip; i++) {

              // there is something in one of the fields
              if ( removeMe[i].equals("true") || !cash_paid[i].equals("") || !loan_amount[i].equals("") ) {

                 // forgot to check the box
                 if ( removeMe[i].equals("false") ) addActionError("You must check the box if you wish to purchase equipment");

                 // they didn't pay all cash
                 if ( !(cash_paid[i].equals(price_to_buy[i]) && loan_amount[i].equals("") ) ) {

                    // they left the cash amount blank
                    if ( cash_paid[i].equals("") ) {
                       cash_paid[i] = "0";
                       addActionError("You can't purchase equipment without making at least a 25% cash payment");
                    }
                    // they left the loan amount blank
                    if ( loan_amount[i].equals("") ) {
                       loan_amount[i] = "0";
                       addActionError("You need to pay all cash or borrow some money to complete this purchase");
                    }
                    // cash payment must be at least 25% of purchase price
                    if ( Integer.parseInt(cash_paid[i]) < (0.25 * Integer.parseInt(price_to_buy[i]) ) ) addActionError("Cash paid must be at least 25% of Price");

                    // cash plus loan must equal price
                    if ( Integer.parseInt(price_to_buy[i]) != Integer.parseInt(cash_paid[i]) + Integer.parseInt(loan_amount[i]) ) addActionError("Cash paid + Loan must equal Price");

			     } // end of "did not pay all cash
		      } // end of "there is something here"
           }  // end of --> for (int i=0; i<numbEquip; i++)
	    }  // end of --> if (numbEquip != 0)
	}

    // Change a Company's Dispatch page
	public String changeDispatch()
	{
		doInitial();
        HttpSession session = servletRequest.getSession();

        // total of all rows in the table (all 4 categories)
        int valueOfI = (int) session.getAttribute("valueOfI");

        // need to know how many rows in each category
        int endOfA = (int) session.getAttribute("endOfA");
        int endOfB = (int) session.getAttribute("endOfB");
        int endOfC = (int) session.getAttribute("endOfC");
        int endOfD = (int) session.getAttribute("endOfD");

        String failMe = (String) session.getAttribute("failMe");
        if ( failMe.equals("yes") ) return "fail";

        if ( valueOfI != 0 ) {  // there are some items on this page
           Map m = new HashMap();

           try
           {
              for (int i=0; i<endOfA; i++) {                   // new equipment on a job
                 if ( !projNumb[i].equals("") ) {
				    m.put("project_numb", Integer.parseInt(projNumb[i]) );
				    m.put("is_in_market", new Boolean(false) );
				    m.put("is_in_fleet",  new Boolean(true) );
				    m.put("is_on_project", new Boolean(true) );
				    m.put("equip_numb", equip_numb[i]);

				    sqlMap.update("updateNewDispatch", m);
                 }
		      }

              for (int i=endOfA; i<endOfB; i++) {              // used equipment on a job
                 if ( !projNumb[i].equals("") ) {
				    m.put("project_numb", Integer.parseInt(projNumb[i]) );
				    m.put("is_in_market", new Boolean(false) );
				    m.put("is_in_fleet",  new Boolean(true) );
				    m.put("is_on_project", new Boolean(true) );
				    m.put("equip_numb", equip_numb[i]);

				    sqlMap.update("updateUsedDispatch", m);
                 }
		      }

              for (int i=endOfB; i<endOfC; i++) {              // rental equipment on a job
                 if ( !projNumb[i].equals("") ) {
				    m.put("project_numb", Integer.parseInt(projNumb[i]) );
				    m.put("is_in_market", new Boolean(false) );
				    m.put("is_in_fleet",  new Boolean(true) );
				    m.put("is_on_project", new Boolean(true) );
				    m.put("equip_numb", equip_numb[i]);

				    sqlMap.update("updateRentalDispatch", m);
                 }
		      }

              for (int i=endOfC; i<endOfD; i++) {              // take equipment off job -- back to fleet
                 if ("true".equals( removeMe[i] )) {
				    m.put("project_numb", 0);
				    m.put("is_in_market", new Boolean(false) );
				    m.put("is_in_fleet",  new Boolean(true) );
				    m.put("is_on_project", new Boolean(false) );
				    m.put("equip_numb", equip_numb[i]);

                    String firstLetter = Character.toString(equip_numb[i].charAt(0) );  // extract the first character of string

                    if ("N".equals(firstLetter) ) sqlMap.update("updateNewDispatch", m);
                    if ("U".equals(firstLetter) ) sqlMap.update("updateUsedDispatch", m);
                    if ("R".equals(firstLetter) ) sqlMap.update("updateRentalDispatch", m);
                 }
		      }

           }  // end of try
		   catch(SQLException e)
		   {
			   e.printStackTrace();
			   throw new RuntimeException ("Error executing sqlMap update_New_Used_Or_Rental_Dispatch. Cause: " + e);
		   }

        }  // end of --> if ( valueOfI != 0 )

        return "success";
    }
