    // Display (view) the master list of crews for each method - ADMIN
	public String masterCrewList()
	{
        doInitial();
        // obtain data for the master list of crews
        List<Object> masterCrewList = new ArrayList<Object>();
		try
		{
			masterCrewList = (List<Object>) sqlMap.queryForList("getListOfCrews", null );
		}
		catch(SQLException e)
		{
			e.printStackTrace();
			throw new RuntimeException ("Error executing sqlMap query getListOfCrews. Cause: " + e);
		}

        int k = 0;
        int oldSecondDigit = 1;
        for (int i=0; i<45; i++) methodTotalCost[i] = Float.parseFloat("0");

        // get the total hourly cost for each individual item of equipment and save them to the bean "masterCrewList"
        iterator = masterCrewList.iterator();

        while (iterator.hasNext()) {
			EquipmentMethodsBean nextOne = (EquipmentMethodsBean) iterator.next();
			Float totalHourlyCost = nextOne.getCt_rate() * (nextOne.getPercent_ct_hour()/100) * nextOne.getNumber_required() * (nextOne.getPercent_time_used()/100);
			nextOne.setTotal_cost_item(totalHourlyCost);

            // Done! Now we need to work on calculating the GRAND TOTAL hourly cost of EQUIPMENT for each of the METHODS

            // extract the second digit of the method
            int secondDigit = Integer.parseInt(Integer.toString(nextOne.getMethod() ).substring(1, 2));

            if (secondDigit == oldSecondDigit) {     // we are still in the same method
               methodTotalCost[k] += totalHourlyCost;
            }
            else {   // method has changed and this is the first item in new method
               k++;
               oldSecondDigit = secondDigit;
               methodTotalCost[k] += totalHourlyCost;
            }
        }

        // holds data for the first eight columns
        sessionMap.put("masterCrewList", masterCrewList);

        // need to update table: "equip_method_prod_cost" with current values
        // also calculate & save the productivity of each METHOD
        Map m = new HashMap();

        k = 0;
        for (int i=0; i<9; i++) {
			for (int j=0; j<5; j++) {

		        int methodNumber = ( (i+1) * 10) + (j+1);
		        try
		        {
			        String calTransCode = (String) sqlMap.queryForObject("getPrimaryProductivity", methodNumber );
			        float productivity = (int) sqlMap.queryForObject("getMethodProductivity", calTransCode );
			        methodProductivity[k] = productivity;
			        m.put("methodNumber", methodNumber);
			        m.put("productivity", productivity);
			        m.put("totalCost", methodTotalCost[k] );
			        sqlMap.update("updateMethodProdCost", m);
			        k++;
		        }
		        catch(SQLException e)
		        {
		     	    e.printStackTrace();
			        throw new RuntimeException ("Error executing sqlMap query getPrimaryProductivity & getMethodProductivity & updateMethodProdCost. Cause: " + e);
		        }
            } // end of for j=0
        } // end of for i=0

        // need to save values to the session for columns nine and ten
        sessionMap.put("methodTotalCost", methodTotalCost);
        sessionMap.put("methodProductivity", methodProductivity);

        // add the new values for total cost and productivity to table method
        HttpSession session = servletRequest.getSession();
        Game g = (Game) session.getAttribute("game");
        int gameID = g.getGameID();
        int adminID = g.getAdminID();
        Map m1 = new HashMap();

        k = 0;
        for (int i=0; i<9; i++) {
			for (int j=0; j<5; j++) {
		        try
		        {
			       m1.put("activityType", i+10 );
			       m1.put("methodID", j+1 );
			       m1.put("gameID", gameID );
			       m1.put("adminID", adminID );
			       m1.put("productivity", Math.round(methodProductivity[k]) );
			       m1.put("equipCostPerCU", Math.round(methodTotalCost[k] / methodProductivity[k]) );
			       sqlMap.update("updateMethodTable", m1);
			       k++;
		        }
		        catch(SQLException e)
		        {
		     	    e.printStackTrace();
			        throw new RuntimeException ("Error executing sqlMap query updateMethodTable. Cause: " + e);
		        }
            } // end of for j=0
        } // end of for i=0


// Take the next two out -- They change admin's default and Game's default values

        Map m2 = new HashMap();
