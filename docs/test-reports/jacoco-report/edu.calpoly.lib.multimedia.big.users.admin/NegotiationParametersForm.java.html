	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * that all numbers are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;
        boolean minsValid = true;

		int minWordOfMouthRateInt = 0;
		int maxWordOfMouthRateInt = 0;
		int minWordOfMouthIntensityInt = 0;
		int maxWordOfMouthIntensityInt = 0;
		int percentOfClosedBidsBecomingNegotiationsInt = 0;
		int meanPercentProfitInt = 0;
		float stdDevPercentProfitFloat = 0.0f;
		int minPercentProfitInt = 0;
		int maxPercentProfitInt = 0;
		int likelihoodCustomerBreaksNegotiationsInt = 0;
		int overEstimationOfCostEffectingEthicsLimitInt = 0;
		int meanCustomerHagglingInt = 0;
		float stdDevCustomerHagglingFloat = 0.0f;
		int minCustomerHagglingInt = 0;
		int maxCustomerHagglingInt = 0;

		if((minWordOfMouthRate.trim() == null) || (minWordOfMouthRate.trim().length() < 1))
			nullParams = true;
		if((maxWordOfMouthRate.trim() == null) || (maxWordOfMouthRate.trim().length() < 1))
		    nullParams = true;
		if((minWordOfMouthIntensity.trim() == null) || (minWordOfMouthIntensity.trim().length() < 1))
			nullParams = true;
		if((maxWordOfMouthIntensity.trim() == null) || (maxWordOfMouthIntensity.trim().length() < 1))
			nullParams = true;
		if((percentOfClosedBidsBecomingNegotiations.trim() == null) || (percentOfClosedBidsBecomingNegotiations.trim().length() < 1))
			nullParams = true;
		if((meanPercentProfit.trim() == null) || (meanPercentProfit.trim().length() < 1))
			nullParams = true;
		if((stdDevPercentProfit.trim() == null) || (stdDevPercentProfit.trim().length() < 1))
			nullParams = true;
		if((minPercentProfit.trim() == null) || (minPercentProfit.trim().length() < 1))
			nullParams = true;
		if((maxPercentProfit.trim() == null) || (maxPercentProfit.trim().length() < 1))
			nullParams = true;
		if((likelihoodCustomerBreaksNegotiations.trim() == null) || (likelihoodCustomerBreaksNegotiations.trim().length() < 1))
			nullParams = true;
		if((overEstimationOfCostEffectingEthicsLimit.trim() == null) || (overEstimationOfCostEffectingEthicsLimit.trim().length() < 1))
			nullParams = true;
		if((meanCustomerHaggling.trim() == null) || (meanCustomerHaggling.trim().length() < 1))
			nullParams = true;
		if((stdDevCustomerHaggling.trim() == null) || (stdDevCustomerHaggling.trim().length() < 1))
			nullParams = true;
		if((minCustomerHaggling.trim() == null) || (minCustomerHaggling.trim().length() < 1))
			nullParams = true;
		if((maxCustomerHaggling.trim() == null) || (maxCustomerHaggling.trim().length() < 1))
			nullParams = true;


		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the number Strings contain only numeric characters.

			//Check that the strings are valid numbers by trying to convert them to ints and checking for a NumberFormatException
			try
			{
				minWordOfMouthRateInt = Integer.parseInt(minWordOfMouthRate);
				maxWordOfMouthRateInt = Integer.parseInt(maxWordOfMouthRate);
				minWordOfMouthIntensityInt = Integer.parseInt(minWordOfMouthIntensity);
				maxWordOfMouthIntensityInt = Integer.parseInt(maxWordOfMouthIntensity);
				percentOfClosedBidsBecomingNegotiationsInt = Integer.parseInt(percentOfClosedBidsBecomingNegotiations);
				meanPercentProfitInt = Integer.parseInt(meanPercentProfit);
				stdDevPercentProfitFloat = Float.parseFloat(stdDevPercentProfit);
				minPercentProfitInt = Integer.parseInt(minPercentProfit);
				maxPercentProfitInt = Integer.parseInt(maxPercentProfit);
				likelihoodCustomerBreaksNegotiationsInt = Integer.parseInt(likelihoodCustomerBreaksNegotiations);
				overEstimationOfCostEffectingEthicsLimitInt = Integer.parseInt(overEstimationOfCostEffectingEthicsLimit);
				meanCustomerHagglingInt = Integer.parseInt(meanCustomerHaggling);
				stdDevCustomerHagglingFloat = Float.parseFloat(stdDevCustomerHaggling);
				minCustomerHagglingInt = Integer.parseInt(minCustomerHaggling);
				maxCustomerHagglingInt = Integer.parseInt(maxCustomerHaggling);

			}
			catch(NumberFormatException nfe)
			{
				validNumbers = false;
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			if(minWordOfMouthRateInt < 0)
				positiveNumbers = false;
			if(maxWordOfMouthRateInt < 0)
				positiveNumbers = false;
			if(minWordOfMouthIntensityInt < 0)
				positiveNumbers = false;
			if(maxWordOfMouthIntensityInt < 0)
				positiveNumbers = false;
			if(percentOfClosedBidsBecomingNegotiationsInt < 0)
				positiveNumbers = false;
			if(meanPercentProfitInt < 0)
				positiveNumbers = false;
			if(stdDevPercentProfitFloat < 0)
				positiveNumbers = false;
			if(minPercentProfitInt < 0)
				positiveNumbers = false;
			if(maxPercentProfitInt < 0)
				positiveNumbers = false;
			if(likelihoodCustomerBreaksNegotiationsInt < 0)
				positiveNumbers = false;
			if(overEstimationOfCostEffectingEthicsLimitInt < 0)
				positiveNumbers = false;
			if(meanCustomerHagglingInt < 0)
				positiveNumbers = false;
			if(stdDevCustomerHagglingFloat < 0)
				positiveNumbers = false;
			if(minCustomerHagglingInt < 0)
				positiveNumbers = false;
			if(maxCustomerHagglingInt < 0)
				positiveNumbers = false;

		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		//Check that all min values are greater than or equal to their corresponding max value
		if(validNumbers && !nullParams)  //validNumbers == true AND nullParams == false
		{
			if(minWordOfMouthRateInt > maxWordOfMouthRateInt)
				minsValid = false;

			if(minWordOfMouthIntensityInt > maxWordOfMouthIntensityInt)
			    minsValid = false;

			if(minPercentProfitInt > maxPercentProfitInt)
			    minsValid = false;

			if(minCustomerHagglingInt > maxCustomerHagglingInt)
			    minsValid = false;
		}

		if(!minsValid && validNumbers && !nullParams)  //minsValid == false AND validNumbers == true AND nullParams == false
		{
			errors.add("minmax", new ActionMessage("error.parameters.minmax2"));
		}


		return errors;
	}
