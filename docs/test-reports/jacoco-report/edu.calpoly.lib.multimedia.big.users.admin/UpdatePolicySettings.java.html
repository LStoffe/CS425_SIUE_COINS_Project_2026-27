package edu.calpoly.lib.multimedia.big.users.admin;

import static edu.calpoly.lib.multimedia.big.game.AutomaticPeriodUpdatePolicies.*;

/**
 * Base class for classes which contain settings for a specific period update policy.
 * The extending classes should define whatever fields they need for their specific policy.
 *
 */
public abstract class UpdatePolicySettings
{
	protected int policyType;

	public int getPolicyType()
	{
		return this.policyType;
	}

	public void setPolicyType(int policyType)
	{
		this.policyType = policyType;
	}

}
