//Logging imports

public class ViewEquipmentParametersAction extends BIGAction
{
	private Log log;

	public ViewEquipmentParametersAction()
	{
		super();
		log = LogFactory.getLog(this.getClass());
	}

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();
			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			//Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			List parameters = null;

			try
			{
				parameters = dbFacade.getEquipmentParametersDBHandler().getEquipmentParameters(game);

			}
			catch(ParametersNotFoundException pnfe)
			{
				//All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Some problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

            //parameters is a List of "resultMaps" (one Map for each record) - each Map contains six key/value pairs (one for each field)

            Map mm = new HashMap();
            List equipParams = new ArrayList();

            //iterator.hasNext() returns true if the iteration (List) has more elements
            for (Iterator iterator = parameters.iterator(); iterator.hasNext(); )
	        {
	            //iterator.next() returns the next element (Map) in the List
			    mm = (Map) iterator.next();

				EquipmentParametersForm theForm = new EquipmentParametersForm();

				theForm.setGameId(game.getGameID());
				theForm.setAdminId(game.getAdminID());
				theForm.setTitle((String)mm.get("equipmentTitle"));
				theForm.setCostPerPeriod((Integer)mm.get("equipmentCost"));
				theForm.setDeployCost((Integer)mm.get("equipmentDeployCost"));
				theForm.setDecommissionCost((Integer)mm.get("equipmentDecommissionCost"));
				theForm.setPercentRaise((Integer)mm.get("equipmentPercentRaise"));
				theForm.setBasePoints((Integer)mm.get("equipmentBasePoints"));
				theForm.setEquipId((Integer)mm.get("equipmentId"));

				equipParams.add(theForm);
            }
			//equipParams is a List (collection) of theForm objects - one object for each record
			session.setAttribute("equipParams", equipParams);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}
}
