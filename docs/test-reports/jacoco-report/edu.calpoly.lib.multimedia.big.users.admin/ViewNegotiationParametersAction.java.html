package edu.calpoly.lib.multimedia.big.users.admin;

//JDK imports
import java.util.Hashtable;
import javax.sql.DataSource;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import edu.calpoly.its.mas.big.users.BIGAction;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIG v2 imports
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.users.admin.Admin;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;

/**
 * Handles use case of viewing the liquidated damage game parameters.
 * Uses the session attribute "user", which is an <code>Admin</code>, and the session attribute "game"
 * which is a <code>edu.calpoly.its.mas.big.Game</code>, to determine the current administrator
 * and game.  Retrieves from the database the number of jobs game parameters that go
 * with this admin and game and places them in a <code>NegotiationParametersForm</code> in the session
 * under the name "negotiationParametersForm".  The game parameters in this form can be displayed to the user
 * using jsp.
 *
 * @author Kevin Craig
 *
 *
 */
public class ViewNegotiationParametersAction extends BIGAction
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();
			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			//Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			Hashtable parameters = null;

			int minWordOfMouthRate = 0;
			int maxWordOfMouthRate = 0;
			int minWordOfMouthIntensity = 0;
			int maxWordOfMouthIntensity = 0;
			int percentOfClosedBidsBecomingNegotiations = 0;
			int meanPercentProfit = 0;
			float stdDevPercentProfit = 0;
			int minPercentProfit = 0;
			int maxPercentProfit = 0;
			int likelihoodCustomerBreaksNegotiations = 0;
			int overEstimationOfCostEffectingEthicsLimit = 0;
			int meanCustomerHaggling = 0;
			float stdDevCustomerHaggling = 0;
			int minCustomerHaggling = 0;
			int maxCustomerHaggling = 0;

			try
			{
				parameters = dbFacade.getGameParametersDBHandler().getNegotiationParameters(game);

				if(parameters != null)
				{
					minWordOfMouthRate = ((Integer) parameters.get("MinWordOfMouthRate")).intValue();
					maxWordOfMouthRate = ((Integer) parameters.get("MaxWordOfMouthRate")).intValue();
					minWordOfMouthIntensity = ((Integer) parameters.get("MinWordOfMouthIntensity")).intValue();
					maxWordOfMouthIntensity = ((Integer) parameters.get("MaxWordOfMouthIntensity")).intValue();
					percentOfClosedBidsBecomingNegotiations = ((Integer) parameters.get("PercentOfClosedBidsBecomingNegotiations")).intValue();
					meanPercentProfit = ((Integer) parameters.get("MeanPercentProfit")).intValue();
					stdDevPercentProfit = ((Float) parameters.get("StdDevPercentProfit")).floatValue();
					minPercentProfit = ((Integer) parameters.get("MinPercentProfit")).intValue();
					maxPercentProfit = ((Integer) parameters.get("MaxPercentProfit")).intValue();
					likelihoodCustomerBreaksNegotiations = ((Integer) parameters.get("LikelihoodCustomerBreaksNegotiations")).intValue();
					overEstimationOfCostEffectingEthicsLimit = ((Integer) parameters.get("OverEstimationOfCostEffectingEthicsLimit")).intValue();
					meanCustomerHaggling = ((Integer) parameters.get("MeanCustomerHaggling")).intValue();
					stdDevCustomerHaggling = ((Float) parameters.get("StdDevCustomerHaggling")).floatValue();
					minCustomerHaggling = ((Integer) parameters.get("MinCustomerHaggling")).intValue();
					maxCustomerHaggling = ((Integer) parameters.get("MaxCustomerHaggling")).intValue();
				}
				else
				{
					//Some problem occurred getting the parameters and wasn't caught earlier
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}

				dbFacade.close();

			}
			catch(ParametersNotFoundException pnfe)
			{
				//All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Some problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			NegotiationParametersForm negotiationForm = new NegotiationParametersForm();
			negotiationForm.setMinWordOfMouthRate(minWordOfMouthRate);
			negotiationForm.setMaxWordOfMouthRate(maxWordOfMouthRate);
			negotiationForm.setMinWordOfMouthIntensity(minWordOfMouthIntensity);
			negotiationForm.setMaxWordOfMouthIntensity(maxWordOfMouthIntensity);
			negotiationForm.setPercentOfClosedBidsBecomingNegotiations(percentOfClosedBidsBecomingNegotiations);
			negotiationForm.setMeanPercentProfit(meanPercentProfit);
			negotiationForm.setStdDevPercentProfit(stdDevPercentProfit);
			negotiationForm.setMinPercentProfit(minPercentProfit);
			negotiationForm.setMaxPercentProfit(maxPercentProfit);
			negotiationForm.setLikelihoodCustomerBreaksNegotiations(likelihoodCustomerBreaksNegotiations);
			negotiationForm.setOverEstimationOfCostEffectingEthicsLimit(overEstimationOfCostEffectingEthicsLimit);
			negotiationForm.setMeanCustomerHaggling(meanCustomerHaggling);
			negotiationForm.setStdDevCustomerHaggling(stdDevCustomerHaggling);
			negotiationForm.setMinCustomerHaggling(minCustomerHaggling);
			negotiationForm.setMaxCustomerHaggling(maxCustomerHaggling);

			session.setAttribute("negotiationForm", negotiationForm);

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}
}
