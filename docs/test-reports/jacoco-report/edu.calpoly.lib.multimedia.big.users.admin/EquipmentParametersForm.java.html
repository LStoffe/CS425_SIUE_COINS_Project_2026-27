package edu.calpoly.lib.multimedia.big.users.admin;

import javax.servlet.http.HttpServletRequest;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionMapping;

import edu.calpoly.its.mas.big.game.NumberConverter;

/**
 * ActionForm used to hold equipment parameters.
 *
 *
 * @author Jim Borland
 *
 */
public final class EquipmentParametersForm  extends ActionForm
{
	private boolean makeDefaults = false;

	private int gameId;
	private int adminId;
	private int costPerPeriod;
	private int deployCost;
	private int decommissionCost;
	private int percentRaise;
	private int basePoints;
	private int equipId;

	private String title;
	private String description;

	private String costPerPeriodString;
	private String deployCostString;
	private String decommissionCostString;


	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

		return errors;
	}


//**************************** Set & Get Methods ************************

    public boolean getMakeDefaults()
    {
		return makeDefaults;
	}

	public void setMakeDefaults(boolean makeDefaults)
	{
		this.makeDefaults = makeDefaults;
	}

	public int getGameId() {
		return gameId;
	}

	public void setGameId(int gameId) {
		this.gameId = gameId;
	}

    public int getAdminId() {
		return adminId;
	}

	public void setAdminId(int adminId) {
		this.adminId = adminId;
	}

	public int getCostPerPeriod() {
		return costPerPeriod;
	}

	public void setCostPerPeriod(int costPerPeriod) {
		this.costPerPeriod = costPerPeriod;
	}

	public int getDeployCost() {
		return deployCost;
	}

	public void setDeployCost(int deployCost) {
		this.deployCost = deployCost;
	}

	public int getDecommissionCost() {
		return decommissionCost;
	}

	public void setDecommissionCost(int decommissionCost) {
		this.decommissionCost = decommissionCost;
	}

	public int getPercentRaise() {
		return percentRaise;
	}

	public void setPercentRaise(int percentRaise) {
		this.percentRaise = percentRaise;
	}

	public int getBasePoints() {
		return basePoints;
	}

	public void setBasePoints(int basePoints) {
		this.basePoints = basePoints;
	}

	public int getEquipId() {
		return equipId;
	}

	public void setEquipId(int equipId) {
		this.equipId = equipId;
	}

	public String getTitle() {
		return title;
	}

	public void setTitle(String title) {
		this.title = title;
	}

	public String getDescription() {
		return description;
	}

	public void setDescription(String description) {
		this.description = description;
	}

	public String getCostPerPeriodString() {
		return (NumberConverter.stringValue(costPerPeriod));
	}

	public void setCostPerPeriodString(String costPerPeriodString) {
		this.costPerPeriodString = costPerPeriodString;
	}

	public String getDeployCostString() {
		return (NumberConverter.stringValue(deployCost));
	}

	public void setDeployCostString(String deployCostString) {
		this.deployCostString = deployCostString;
	}

	public String getDecommissionCostString() {
		return (NumberConverter.stringValue(decommissionCost));
	}

	public void setDecommissionCostString(String decommissionCostString) {
		this.decommissionCostString = decommissionCostString;
	}
}
