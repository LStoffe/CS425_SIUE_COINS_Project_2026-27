/**
 * @author Kevin Craig
 */
public class ModifyGradingTemplateAction extends Action
{

    private String[] template = new String[4];

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();
			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			template[0] = (String) ((DynaActionForm) form).get("templateString");
			template[1] = (String) ((DynaActionForm) form).get("templateString1");
			template[2] = (String) ((DynaActionForm) form).get("templateString2");
			template[3] = (String) ((DynaActionForm) form).get("templateString3");

            for (int i=0; i<4; i++) {

			   NamedVariableDataFacade namedVariableDataFacade = new NamedVariableDataFacadeTestImpl();
			   BIGFormulaInterpreter interpreter = new BIGFormulaInterpreter(namedVariableDataFacade);

			   //Test out the formula using variable facade test interface.  If nothing is returned
			   //then the formula did not compile
			   String testFormulaResult = interpreter.interpretFormula(template[i], 0);

			   if(log.isDebugEnabled())
			   {
				   log.debug(this.getClass() + ": Template test results = " + testFormulaResult);
			   }

			   if(testFormulaResult.trim().length() < 1)
			   {
				   errors.add("templateError", new ActionMessage("error.gradingTemplate.syntax.error"));
			   }
			   else
			   {
				   if(log.isDebugEnabled())
				   {
					   log.debug(this.getClass() + ": grading template looks okay.");
				   }
			   }

		    }  // end of for loop

			Map map = new HashMap(6);
			map.put("adminid", admin.getAdminID());
			map.put("gameid", game.getGameID());
			map.put("template",  template[0]);
			map.put("template1", template[1]);
			map.put("template2", template[2]);
			map.put("template3", template[3]);

			try
			{
				BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));
				dbFacade.getGradingTemplateDBHandler().updateGradingTemplate(admin, game, map);
				dbFacade.close();
			}
			catch(BIGDatabaseFacadeException bigdbe)
			{
				log.fatal("", bigdbe);
			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}
	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}
}
