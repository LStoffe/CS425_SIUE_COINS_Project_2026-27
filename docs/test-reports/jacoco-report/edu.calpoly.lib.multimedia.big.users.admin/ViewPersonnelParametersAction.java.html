package edu.calpoly.lib.multimedia.big.users.admin;

/**
 * Handles use case of viewing the liquidated damage game parameters.
 * Uses the session attribute "user", which is an <code>Admin</code>, and the session attribute "game"
 * which is a <code>edu.calpoly.its.mas.big.Game</code>, to determine the current administrator
 * and game.  Retrieves from the database the number of jobs game parameters that go
 * with this admin and game and places them in a <code>NegotiationParametersForm</code> in the session
 * under the name "negotiationParametersForm".  The game parameters in this form can be displayed to the user
 * using jsp.
 *
 * @author Kevin Craig
 *
 *
 */
public class ViewPersonnelParametersAction extends BIGAction
{
	private Log log;

	public ViewPersonnelParametersAction()
	{
		super();
		log = LogFactory.getLog(this.getClass());
	}

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();
			Game game = (Game) session.getAttribute("game");
			Admin admin = (Admin) session.getAttribute("user");

			//Create a new BIGDatabaseFacadeInterface which will hold connections
			BIGDatabaseFacadeInterface dbFacade = getDbFacade(request);
			Map parameters = null;

			try
			{
				parameters = dbFacade.getPersonnelParametersDBHandler().getPersonnelParameters(game);
				dbFacade.close();

			}
			catch(ParametersNotFoundException pnfe)
			{
				//All games should have parameters.  If there are no parameters found for a game it is an error.
				log.fatal("", pnfe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
			}
			catch (BIGDatabaseFacadeException bigdbe)
			{
				//Some problem with the database
				log.fatal("", bigdbe);
				errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
			}

			//The Map returned from the database has employee types as keys and
			//costs as values.  Before returning to the view the costs should be converted
			//from integers to properly formatted currency strings.
			try
			{
				Map formattedParameters = parameters.getClass().newInstance();
				for(Object o : parameters.keySet())
				{
					Integer i = (Integer) parameters.get(o);
					formattedParameters.put(o, NumberConverter.stringValue(i.intValue()));
				}
				parameters = formattedParameters;
			}
			catch(Exception e)  //Formatting is not critical, so log the error and continue
			{
				log.error("", e);
			}


			session.setAttribute("personnelParameters", parameters);
		    session.setAttribute("personnelParametersForm", prepopulateActionForm(mapping, parameters));

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"success\"");
				}
				return (mapping.findForward("success"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}
	}

	//Initialized index property array to correct length and prepopulates
	//with the values
	private ActionForm prepopulateActionForm(ActionMapping mapping, Map parameters)
	{
		try
		{
			FormBeanConfig fbc = mapping.getModuleConfig().findFormBeanConfig("personnelParametersForm");
			FormPropertyConfig[] configs = fbc.findFormPropertyConfigs();

			//Create an duplicate FormBeanConfig that's not frozen so we change change the array size
			FormBeanConfig newfbc = new FormBeanConfig();
			newfbc.setName(fbc.getName());
			newfbc.setType(fbc.getType());
			for(int i = 0; i < configs.length; i++)
			{
				if(configs[i].getName().equals("employeeCosts"))
				{
					newfbc.addFormPropertyConfig(new FormPropertyConfig("employeeCosts", "java.util.HashMap", null, parameters.size()));
				}
				else
				{
					log.debug(configs[i].toString());
					newfbc.addFormPropertyConfig(configs[i]);
				}
			}
			DynaActionForm form = (DynaActionForm) newfbc.createActionForm(servlet);

			for(Object o : parameters.keySet())
			{
				String key = (String) o;
				String value = (String) parameters.get(key);

				form.set("employeeCosts", key, value);
			}

			return form;
		}
		catch(Exception e)
		{
			log.error("", e);
		}
		return null;
	}

	//Creates a string array of the employee costs per period, as properly formatted currency
	private String[] createCostsStringArray(Map parameters)
	{
		String[] s = new String[parameters.size()];
		int i = 0;
		for(Object o : parameters.keySet())
		{
			s[i] = (String) parameters.get(o);
			i++;
		}

		return s;
	}

}
