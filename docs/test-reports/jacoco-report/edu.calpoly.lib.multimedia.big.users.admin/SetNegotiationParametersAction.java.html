package edu.calpoly.lib.multimedia.big.users.admin;

//JDK imports
import java.util.Hashtable;
import javax.sql.DataSource;

//Servlet imports
import javax.servlet.RequestDispatcher;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpSession;
import javax.servlet.http.HttpServletResponse;

//Struts imports
import org.apache.struts.action.Action;
import org.apache.struts.action.ActionMessage;
import org.apache.struts.action.ActionMessages;
import org.apache.struts.action.ActionErrors;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;
import org.apache.struts.action.ActionServlet;
import org.apache.struts.util.MessageResources;

//BIGv2 imports
import edu.calpoly.its.mas.big.game.Game;
import edu.calpoly.its.mas.big.users.admin.Admin;

//Database imports
import edu.calpoly.its.mas.big.db.BIGDatabaseFacade;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeException;
import edu.calpoly.its.mas.big.db.BIGDatabaseFacadeInterface;
import edu.calpoly.its.mas.big.db.ParametersNotFoundException;

//Logging imports
import org.apache.commons.logging.Log;
import org.apache.commons.logging.LogFactory;


/**
 * Determines whether the administrator wishes to set parameters or restore defaults for number of jobs parameters then performs the appropriate action.
 * If the administrator wishes to store his current parameters settings to the database s/he would have pushed the set parameters button
 * (which must have the name "submit").  In this case the values currently in the NegotiationParametersForm in the session will be
 * stored to the database.  These parameters will also be made the admin's defaults if the getMakeDefaults method of the form returns true.
 * <br><br>
 * If the administrator pressed some other submit button (any name other than "submit" or "cancel") the parameters in the NegotiationParametersForm
 * will be overwritten with a copy of the admin's defaults for the current parameters.  The defaults will also be re-copied into the database overwriting the current
 * game parameters.
 *
 * @author Kevin Craig
 *
 *
 */
public class SetNegotiationParametersAction extends Action
{

	public ActionForward execute(ActionMapping mapping,
				       			 ActionForm form,
				       			 HttpServletRequest request,
			           			 HttpServletResponse response)
	{
		Log log = LogFactory.getLog(this.getClass());
		if(log.isDebugEnabled())
		{
			log.debug(this.getClass() + " beginning execute method");
		}

		//Process request if it was not cancelled
        if(!isCancelled(request))
        {
			ActionMessages errors = new ActionMessages();

			HttpSession session = request.getSession();

			//Retrieve the current game and admin from the session
			Admin admin = (Admin) session.getAttribute("user");
			Game game = (Game) session.getAttribute("game");

			//Get the game parameters from the action form
			NegotiationParametersForm negotiationParametersForm = (NegotiationParametersForm) form;

            //Determine which submit button was pressed by the Administrator. If the SetParameters button is null the other RestoreDefaults button had to be pushed
            String testSetParametersSubmitButton = (String) request.getParameter("submit");

			//Create a database facade that will hold the connection since it will be used repeatedly
			BIGDatabaseFacadeInterface dbFacade = new BIGDatabaseFacade(getDataSource(request));

			//Hashtable used to transfer the parameters to the database facade
			Hashtable parameters = null;


			if (testSetParametersSubmitButton != null)  //"Set Parameters" submit button was pushed.
			{
				//Store the current parameters values from the form into the database
				try
				{
					parameters = new Hashtable();
					parameters.put("MinWordOfMouthRate", new Integer(negotiationParametersForm.getMinWordOfMouthRate()));
					parameters.put("MaxWordOfMouthRate", new Integer(negotiationParametersForm.getMaxWordOfMouthRate()));
					parameters.put("MinWordOfMouthIntensity", new Integer(negotiationParametersForm.getMinWordOfMouthIntensity()));
					parameters.put("MaxWordOfMouthIntensity", new Integer(negotiationParametersForm.getMaxWordOfMouthIntensity()));
					parameters.put("PercentOfClosedBidsBecomingNegotiations", new Integer(negotiationParametersForm.getPercentOfClosedBidsBecomingNegotiations()));
					parameters.put("MeanPercentProfit", new Integer(negotiationParametersForm.getMeanPercentProfit()));
					parameters.put("StdDevPercentProfit", new Float(negotiationParametersForm.getStdDevPercentProfit()));
					parameters.put("MinPercentProfit", new Integer(negotiationParametersForm.getMinPercentProfit()));
					parameters.put("MaxPercentProfit", new Integer(negotiationParametersForm.getMaxPercentProfit()));
					parameters.put("LikelihoodCustomerBreaksNegotiations", new Integer(negotiationParametersForm.getLikelihoodCustomerBreaksNegotiations()));
					parameters.put("OverEstimationOfCostEffectingEthicsLimit", new Integer(negotiationParametersForm.getOverEstimationOfCostEffectingEthicsLimit()));
					parameters.put("MeanCustomerHaggling", new Integer(negotiationParametersForm.getMeanCustomerHaggling()));
					parameters.put("StdDevCustomerHaggling", new Float(negotiationParametersForm.getStdDevCustomerHaggling()));
					parameters.put("MinCustomerHaggling", new Integer(negotiationParametersForm.getMinCustomerHaggling()));
					parameters.put("MaxCustomerHaggling", new Integer(negotiationParametersForm.getMaxCustomerHaggling()));

					dbFacade.getGameParametersDBHandler().updateNegotiationParameters(game, parameters);

					if(negotiationParametersForm.getMakeDefaults())
					{
						dbFacade.getGameParametersDBHandler().updateDefaultNegotiationParameters(admin, parameters);
					}

					dbFacade.close();
				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					//Problem with the database
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
					//The parameters for this game weren't there.  There should never be a game without parameters
					log.fatal("", pnfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}
			}
			else //"Restore defaults" submit button was pushed
			{
				try
				{

					//Retrieve this admins default parameters from the database and re-store them for the current game
					parameters = dbFacade.getGameParametersDBHandler().getDefaultNegotiationParameters(admin);
					dbFacade.getGameParametersDBHandler().updateNegotiationParameters(game, parameters);
					dbFacade.close();

					//The parameter vaules also need to be put in the Action Form so the user can see them
					negotiationParametersForm.setMinWordOfMouthRate(((Integer) parameters.get("MinWordOfMouthRate")).intValue());
					negotiationParametersForm.setMaxWordOfMouthRate(((Integer) parameters.get("MaxWordOfMouthRate")).intValue());
					negotiationParametersForm.setMinWordOfMouthIntensity(((Integer) parameters.get("MinWordOfMouthIntensity")).intValue());
					negotiationParametersForm.setMaxWordOfMouthIntensity(((Integer) parameters.get("MaxWordOfMouthIntensity")).intValue());
					negotiationParametersForm.setPercentOfClosedBidsBecomingNegotiations(((Integer) parameters.get("PercentOfClosedBidsBecomingNegotiations")).intValue());
					negotiationParametersForm.setMeanPercentProfit(((Integer) parameters.get("MeanPercentProfit")).intValue());
					negotiationParametersForm.setStdDevPercentProfit(((Float) parameters.get("StdDevPercentProfit")).floatValue());
					negotiationParametersForm.setMinPercentProfit(((Integer) parameters.get("MinPercentProfit")).intValue());
					negotiationParametersForm.setMaxPercentProfit(((Integer) parameters.get("MaxPercentProfit")).intValue());
					negotiationParametersForm.setLikelihoodCustomerBreaksNegotiations(((Integer) parameters.get("LikelihoodCustomerBreaksNegotiations")).intValue());
					negotiationParametersForm.setOverEstimationOfCostEffectingEthicsLimit(((Integer) parameters.get("OverEstimationOfCostEffectingEthicsLimit")).intValue());
					negotiationParametersForm.setMeanCustomerHaggling(((Integer) parameters.get("MeanCustomerHaggling")).intValue());
					negotiationParametersForm.setStdDevCustomerHaggling(((Float) parameters.get("StdDevCustomerHaggling")).floatValue());
					negotiationParametersForm.setMinCustomerHaggling(((Integer) parameters.get("MinCustomerHaggling")).intValue());
					negotiationParametersForm.setMaxCustomerHaggling(((Integer) parameters.get("MaxCustomerHaggling")).intValue());

				}
				catch(BIGDatabaseFacadeException bigdbe)
				{
					//Problem with the database
					log.fatal("", bigdbe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.database"));
				}
				catch(ParametersNotFoundException pnfe)
				{
					//The parameters for this game weren't there.  There should never be a game without parameters
					log.fatal("", pnfe);
					errors.add(ActionMessages.GLOBAL_MESSAGE, new ActionMessage("error.unknown"));
				}

			}

			if (!errors.isEmpty())
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request returned with errors");
				}
				// Report any errors we have discovered back to the original form
				saveErrors(request, errors);
	 	    	saveToken(request);
	 	    	return (new ActionForward(mapping.getInput()));
			}
			else if (testSetParametersSubmitButton != null)
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"params_set\"");
				}
				return (mapping.findForward("params_set"));
			}
			else
			{
				if(log.isDebugEnabled())
				{
					log.debug(this.getClass() + " request completed.  Forwarding to \"defaults_restored\"");
				}
				return (mapping.findForward("defaults_restored"));
			}

	    }
	    else  //Request was cancelled
	    {
			if(log.isDebugEnabled())
			{
				log.debug(this.getClass() + " request was cancelled");
			}
			return (mapping.findForward("cancelled"));
		}

	}
}
