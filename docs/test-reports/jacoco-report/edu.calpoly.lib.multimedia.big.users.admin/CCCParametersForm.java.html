    private String upperPercentJobMarkupString = null;

	/**
	 * Ensures that all values are non-null, that all number strings contain only numeric characters,
     * that all numbers are greater than or equal to zero, and that for each period the mins are less than or equal to the max.
	 */
	public ActionErrors validate(ActionMapping mapping, HttpServletRequest request)
	{
        ActionErrors errors = new ActionErrors();

        boolean nullParams = false;
        boolean validNumbers = true;
        boolean positiveNumbers = true;
        boolean minsValid = true;

		int minNumberCCCs = 0;
		int maxNumberCCCs = 0;
		int minCCCBidsPerJob = 0;
		int maxCCCBidsPerJob = 0;
		int cccTurnover = 0;
		int lowerPercent = 0;
		int upperPercent = 0;


		if((minNumberCCCsString == null) || (minNumberCCCsString.trim().length() < 1))
			nullParams = true;
		if((maxNumberCCCsString == null) || (maxNumberCCCsString.trim().length() < 1))
		    nullParams = true;
		if((minCCCBidsPerJobString == null) || (minCCCBidsPerJobString.trim().length() < 1))
			nullParams = true;
		if((maxCCCBidsPerJobString == null) || (maxCCCBidsPerJobString.trim().length() < 1))
			nullParams = true;
		if((cccTurnoverString == null) || (cccTurnoverString.trim().length() < 1))
			nullParams = true;
		if((lowerPercentJobMarkupString == null) || (lowerPercentJobMarkupString.trim().length() < 1))
			nullParams = true;
		if((upperPercentJobMarkupString == null) || (upperPercentJobMarkupString.trim().length() < 1))
			nullParams = true;

		if (nullParams) //nullParams == true
		{
			errors.add("nullParameters", new ActionMessage("error.parameters.null"));
		}
		else           //nullParams == false
		{
			//Next make sure all of the number Strings contain only numeric characters.

			//Check that the strings are valid numbers by trying to convert them to ints and checking for a NumberFormatException
			try
			{
				minNumberCCCs = Integer.parseInt(minNumberCCCsString);
				maxNumberCCCs = Integer.parseInt(maxNumberCCCsString);
				minCCCBidsPerJob = Integer.parseInt(minCCCBidsPerJobString);
				maxCCCBidsPerJob = Integer.parseInt(maxCCCBidsPerJobString);
				cccTurnover = Integer.parseInt(cccTurnoverString);
				lowerPercent = Integer.parseInt(lowerPercentJobMarkupString);
				upperPercent = Integer.parseInt(upperPercentJobMarkupString);
			}
			catch(NumberFormatException nfe)
			{
				validNumbers = false;
			}
		}

		if(!validNumbers && !nullParams) //validNumbers == false AND nullParams == false
		{
			errors.add("numberFormat", new ActionMessage("error.parameters.number.format"));
		}
		else if(!nullParams)				//validNumbers == true AND nullParams == false
		{
			//Check that all the numbers are positive or zero
			if(minNumberCCCs < 1)
				positiveNumbers = false;
			if(maxNumberCCCs < 0)
				positiveNumbers = false;
			if(minCCCBidsPerJob < 1)
				positiveNumbers = false;
			if(maxCCCBidsPerJob < 0)
				positiveNumbers = false;
			if(cccTurnover < 0)
				positiveNumbers = false;
			if(lowerPercent < 0)
				positiveNumbers = false;
			if(upperPercent < 0)
				positiveNumbers = false;
		}

		if(!positiveNumbers && validNumbers && !nullParams) //positiveNumbers = false AND nullParams == false AND validNumbers == true
		{
			errors.add("negativeValue", new ActionMessage("error.parameters.number.negative"));
		}

		//Check that all min values are greater than or equal to their corresponding max value
		if(validNumbers && !nullParams)  //validNumbers == true AND nullParams == false
		{
			if(minNumberCCCs > maxNumberCCCs)
				minsValid = false;

			if(minCCCBidsPerJob > maxCCCBidsPerJob)
			    minsValid = false;

			if(lowerPercent > upperPercent)
			    minsValid = false;
		}

		if(!minsValid && validNumbers && !nullParams)  //minsValid == false AND validNumbers == true AND nullParams == false
		{
			errors.add("minmax", new ActionMessage("error.parameters.minmax2"));
		}

		if(minNumberCCCs < minCCCBidsPerJob)
		{
			errors.add("min", new ActionMessage("error.parameters.ccc"));
		}

		return errors;
	}
